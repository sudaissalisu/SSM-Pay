   540→/**
   541→ * Calculate the mean of an array of numbers
   542→ * @param values - Array of numeric values
   543→ * @returns The arithmetic mean
   544→ * @throws {AppError} If array is empty
   545→ */
   546→function calculateMean(values: number[]): number {
   547→  if (values.length === 0) {
   548→    throw new AppError('Cannot calculate mean of empty array', ErrorCode.VALIDATION_ERROR);
   549→  }
   550→  return values.reduce((sum, val) => sum + val, 0) / values.length;
   551→}
   552→
   553→/**
   554→ * Calculate the standard deviation of an array of numbers
   555→ * @param values - Array of numeric values
   556→ * @param usePopulation - Whether to use population (N) vs sample (N-1) standard deviation
   557→ * @returns The standard deviation
   558→ * @throws {AppError} If array has fewer than 2 elements
   559→ */
   560→function calculateStandardDeviation(values: number[], usePopulation: boolean = false): number {
   561→  if (values.length < 2) {
   562→    throw new AppError('Cannot calculate standard deviation with fewer than 2 values', ErrorCode.VALIDATION_ERROR);
   563→  }
   564→  const mean = calculateMean(values);
   565→  const squaredDifferences = values.map(val => Math.pow(val - mean, 2));
   566→  const variance = squaredDifferences.reduce((sum, val) => sum + val, 0) / (usePopulation ? values.length : values.length - 1);
   567→  return Math.sqrt(variance);
   568→}
   569→
   570→/**
   571→ * Calculate the median of an array of numbers
   572→ * @param values - Array of numeric values
   573→ * @returns The median value
   574→ * @throws {AppError} If array is empty
   575→ */
   576→function calculateMedian(values: number[]): number {
   577→  if (values.length === 0) {
   578→    throw new AppError('Cannot calculate median of empty array', ErrorCode.VALIDATION_ERROR);
   579→  }
   580→  const sorted = [...values].sort((a, b) => a - b);
   581→  const mid = Math.floor(sorted.length / 2);
   582→  return sorted.length % 2 !== 0 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
   583→}
   584→
   585→/**
   586→ * Calculate the Median Absolute Deviation (MAD)
   587→ * @param values - Array of numeric values
   588→ * @returns The MAD value
   589→ * @throws {AppError} If array is empty
   590→ */
   591→function calculateMAD(values: number[]): number {
   592→  const median = calculateMedian(values);
   593→  const deviations = values.map(val => Math.abs(val - median));
   594→  return calculateMedian(deviations);
   595→}

   597→/**
   598→ * Calculate quartiles Q1 and Q3
   599→ * @param values - Array of numeric values
   600→ * @returns Object containing Q1 and Q3
   601→ * @throws {AppError} If array has fewer than 4 elements
   602→ */
   603→function calculateQuartiles(values: number[]): { q1: number; q3: number } {
   604→  if (values.length < 4) {
   605→    throw new AppError('Cannot calculate quartiles with fewer than 4 values', ErrorCode.VALIDATION_ERROR);
   606→  }
   607→  const sorted = [...values].sort((a, b) => a - b);
   608→  const q1Index = Math.floor(sorted.length * 0.25);
   609→  const q3Index = Math.floor(sorted.length * 0.75);
   610→  return {
   611→    q1: sorted[q1Index],
   612→    q3: sorted[q3Index],
   613→  };
   614→}

   616→/**
   617→ * Calculate the Haversine distance between two geographic points
   618→ * @param lat1 - Latitude of first point
   619→ * @param lon1 - Longitude of first point
   620→ * @param lat2 - Latitude of second point
   621→ * @param lon2 - Longitude of second point
   622→ * @returns Distance in kilometers
   623→ */
   624→function calculateHaversineDistance(
   625→  lat1: number,
   626→  lon1: number,
   627→  lat2: number,
   628→  lon2: number
   629→): number {
   630→  const dLat = toRadians(lat2 - lat1);
   631→  const dLon = toRadians(lon2 - lon1);
   632→  
   633→  const a =
   634→    Math.sin(dLat / 2) * Math.sin(dLat / 2) +
   635→    Math.cos(toRadians(lat1)) * Math.cos(toRadians(lat2)) *
   636→    Math.sin(dLon / 2) * Math.sin(dLon / 2);
   637→    
   638→  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
   639→  
   640→  return EARTH_RADIUS_KM * c;
   641→}

   652→/**
   653→ * Clamp a value between minimum and maximum bounds
   654→ * @param value - Value to clamp
   655→ * @param min - Minimum bound
   656→ * @param max - Maximum bound
   657→ * @returns Clamped value

   659→function clamp(value: number, min: number, max: number): number {

   663→/**
   664→ * Normalize a value to 0-1 range using min-max normalization
   665→ * @param value - Value to normalize
   666→ * @param min - Minimum possible value
   667→ * @param max - Maximum possible value
   668→ * @returns Normalized value between 0 and 1
   669→ */
   670→function normalize(value: number, min: number, max: number): number {
   671→  if (max === min) return 0.5;
   672→  return clamp((value - min) / (max - min), 0, 1);
   673→}

   675→/**
   676→ * Calculate percentile rank of a value in a dataset
   677→ * @param value - Value to find percentile for
   678→ * @param dataset - Dataset to compare against
   679→ * @returns Percentile rank (0-100)
   680→ */
   681→function calculatePercentileRank(value: number, dataset: number[]): number {
   682→  if (dataset.length === 0) return 50;
   683→  const sorted = [...dataset].sort((a, b) => a - b);
   684→  let belowCount = 0;
   685→  for (const val of sorted) {
   686→    if (val < value) belowCount++;
   687→  }
   688→  return (belowCount / sorted.length) * 100;
   689→}

   843→   * @template T - Type of items in the buffer
   844→   * @param maxSize - Maximum number of items
   845→   * @param windowMs - Window duration in milliseconds
   846→   * @returns Sliding window buffer instance
   847→   */
   848→  private createSlidingWindow<T>(maxSize: number, windowMs: number): SlidingWindowBuffer<T> {
   849→    return {
   850→      buffer: [],
   851→      maxSize,
   852→      windowMs,
   853→      add(item: T): void {
   854→        this.buffer.push(item);
   855→        if (this.buffer.length > this.maxSize) {
   856→          this.buffer.shift();
   857→        }
   858→      },
   859→      getAll(): T[] {
   860→        return [...this.buffer];
   861→      },
   862→      size(): number {
   863→        return this.buffer.length;
   864→      },
   865→      clear(): void {
   866→        this.buffer = [];
   867→      },
   868→      getInWindow(now: Date): T[] {
   869→        const windowStart = new Date(now.getTime() - this.windowMs);
   870→        // Type assertion for timestamp access
   871→        return this.buffer.filter((item) => {
   872→          const typedItem = item as unknown as { timestamp: Date };
   873→          return typedItem.timestamp >= windowStart;
   874→        });
   875→      },
   876→    };
   877→  }

  1045→  detectStatisticalOutliers(

  1059→    switch (method) {
  1060→      case StatisticalMethod.Z_SCORE:

  1063→      case StatisticalMethod.IQR:

  1072→        this.applyIQRDetection(values, results);
  1073→        this.applyModifiedZScoreDetection(values, results);
  1074→        // Mark as outlier if any method flags it
  1075→        results.forEach(r => {
  1076→          r.isOutlier = r.score > 1;
  1077→        });
  1078→        break;
  1079→    }
  1080→
  1081→    return results;

  1084→  /**
  1085→   * Detect anomalies in time series data
  1086→   * 
  1087→   * @param data - Time series data points
  1088→   * @param options - Analysis options
  1089→   * @returns Array of anomalous points
  1090→   */
  1091→  detectTimeSeriesAnomalies(
  1092→    data: TimeSeriesPoint[],
  1093→    options: {
  1094→      seasonalityPeriod?: number;
  1095→      trendSensitivity?: number;
  1096→    } = {}
  1097→  ): TimeSeriesPoint[] {
  1098→    if (!Array.isArray(data) || data.length < this.config.minSamplesRequired) {
  1099→      logger.warn('Insufficient time series data for analysis');
  1100→      return [];
  1101→    }
  1102→
  1103→    const seasonalityPeriod = options.seasonalityPeriod || 24; // Default: hourly seasonality
  1104→    const trendSensitivity = options.trendSensitivity || 2.0;
  1105→
  1106→    const values = data.map(d => d.value);
  1107→    const outliers = this.detectStatisticalOutliers(values, StatisticalMethod.HYBRID);
  1108→    
  1109→    const anomalousIndices = new Set(
  1110→      outliers.filter(o => o.isOutlier).map(o => o.index)
  1111→    );
  1112→
  1113→    // Additional time-series specific checks
  1114→    const anomalousPoints: TimeSeriesPoint[] = [];
  1115→    
  1116→    for (let i = 0; i < data.length; i++) {
  1117→      if (anomalousIndices.has(i)) {
  1118→        anomalousPoints.push(data[i]);
  1119→        continue;
  1120→      }
  1121→
  1122→      // Check for sudden changes (point-to-point difference)
  1123→      if (i > 0) {
  1124→        const prevValue = data[i - 1].value;
  1125→        const currValue = data[i].value;
  1126→        const changePercent = prevValue !== 0 
  1127→          ? Math.abs((currValue - prevValue) / prevValue) * 100 
  1128→          : 0;
  1129→        
  1130→        if (changePercent > 100 * trendSensitivity) {
  1131→          anomalousPoints.push(data[i]);
  1132→        }
  1133→      }
  1134→    }
  1135→
  1136→    return anomalousPoints;
  1137→  }

  1191→  async processStreamEvent(transaction: TransactionData): Promise<AnalysisResult> {
  1192→    const startTime = performance.now();
  1193→    
  1194→    try {
  1195→      // Add to sliding window
  1196→      this.slidingWindow.add(transaction);
  1197→      
  1198→      // Update streaming stats
  1199→      this.streamingStats.totalEventsProcessed++;
  1200→      this.streamingStats.eventsInWindow = this.slidingWindow.size();
  1201→      
  1202→      // Get recent transactions for context
  1203→      const recentTransactions = this.slidingWindow.getAll();
  1204→      
  1205→      // Run quick analysis optimized for streaming
  1206→      const result = await this.analyzeTransaction(transaction);
  1207→      
  1208→      // Track processing time
  1209→      const processingTime = performance.now() - startTime;
  1210→      this.recentProcessingTimes.push(processingTime);
  1211→      if (this.recentProcessingTimes.length > 100) {
  1212→        this.recentProcessingTimes.shift();
  1213→      }
  1214→      this.streamingStats.avgProcessingTimeMs = 
  1215→        this.recentProcessingTimes.reduce((a, b) => a + b, 0) / this.recentProcessingTimes.length;
  1216→      
  1217→      // Update anomaly rate
  1218→      this.streamingStats.currentAnomalyRate = 
  1219→        this.streamingStats.totalEventsProcessed > 0
  1220→          ? this.streamingStats.totalAnomaliesDetected / this.streamingStats.totalEventsProcessed
  1221→          : 0;
  1222→      
  1223→      // Update uptime
  1224→      if (this.streamingStartTime) {
  1225→        this.streamingStats.uptimeSeconds = 
  1226→          (Date.now() - this.streamingStartTime.getTime()) / 1000;
  1227→      }
  1228→      
  1229→      // Emit data event
  1230→      this.emitEvent({ type: 'data', payload: transaction, timestamp: new Date() });
  1231→      
  1232→      return result;
  1233→    } catch (error) {
  1234→      this.emitEvent({ 
  1235→        type: 'error', 
  1236→        payload: error instanceof Error ? error : new Error(String(error)), 
  1237→        timestamp: new Date() 
  1238→      });
  1239→      throw error;
  1240→    }
  1241→  }

  1281→  /**
  1282→   * Update user profile with transaction data
  1283→   * 
  1284→   * @param transaction - Transaction to incorporate into profile
  1285→   */
  1286→  updateUserProfile(transaction: TransactionData): void {
  1287→    const existingProfile = this.userProfiles.get(transaction.customerId);
  1288→    
  1289→    if (existingProfile) {
  1290→      // Update existing profile
  1291→      const n = existingProfile.totalTransactions;
  1292→      const newAvgAmount = (existingProfile.avgAmount * n + transaction.amount) / (n + 1);
  1293→      
  1294→      // Update standard deviation using online algorithm (Welford's method)
  1295→      // For simplicity, we'll use a approximation based on the difference from mean
  1296→      const diffFromMean = Math.abs(transaction.amount - existingProfile.avgAmount);
  1297→      const newStdAmount = Math.sqrt(
  1298→        (existingProfile.stdAmount * existingProfile.stdAmount * (n - 1) + diffFromMean * diffFromMean) / n
  1299→      );
  1300→      
  1301→      // Update typical hours
  1302→      const hour = transaction.timestamp.getHours();
  1303→      if (!existingProfile.typicalHours.includes(hour)) {
  1304→        existingProfile.typicalHours.push(hour);
  1305→      }
  1306→      
  1307→      // Update typical days
  1308→      const day = transaction.timestamp.getDay();
  1309→      if (!existingProfile.typicalDays.includes(day)) {
  1310→        existingProfile.typicalDays.push(day);
  1311→      }
  1312→      
  1313→      // Update known devices
  1314→      existingProfile.knownDevices.add(transaction.deviceFingerprint);
  1315→      
  1316→      // Update known locations
  1317→      existingProfile.knownLocations.add(transaction.countryCode);
  1318→      
  1319→      // Update preferred methods
  1320→      existingProfile.preferredMethods.add(transaction.paymentMethod);
  1321→      
  1322→      // Update timing
  1323→      const timeDiff = transaction.timestamp.getTime() - existingProfile.lastTransactionDate.getTime();
  1324→      existingProfile.avgTimeBetweenTxnMinutes = 
  1325→        (existingProfile.avgTimeBetweenTxnMinutes * n + timeDiff / 60000) / (n + 1);
  1326→      
  1327→      existingProfile.avgAmount = newAvgAmount;
  1328→      existingProfile.stdAmount = newStdAmount || 0; // Ensure it's at least 0
  1329→      // Set minimum std to avoid division by zero issues if all amounts are same
  1330→      if (existingProfile.stdAmount === 0 && n > 1) {
  1331→        existingProfile.stdAmount = existingProfile.avgAmount * 0.1; // Assume 10% variation
  1332→      }
  1333→      existingProfile.totalTransactions = n + 1;
  1334→      existingProfile.lastTransactionDate = transaction.timestamp;
  1335→      existingProfile.dailyVelocity = existingProfile.totalTransactions / Math.max(existingProfile.accountAgeDays, 1);
  1336→    } else {
  1337→      // Create new profile
  1338→      const newProfile: UserProfile = {
  1339→        customerId: transaction.customerId,
  1340→        avgAmount: transaction.amount,
  1341→        stdAmount: 0,
  1342→        medianAmount: transaction.amount,
  1343→        typicalHours: [transaction.timestamp.getHours()],
  1344→        typicalDays: [transaction.timestamp.getDay()],
  1345→        knownDevices: new Set([transaction.deviceFingerprint]),
  1346→        knownLocations: new Set([transaction.countryCode]),
  1347→        knownIpAddresses: new Set([transaction.ipAddress]),
  1348→        preferredMethods: new Set([transaction.paymentMethod]),
  1349→        accountAgeDays: 1,
  1350→        totalTransactions: 1,
  1351→        lastTransactionDate: transaction.timestamp,
  1352→        avgTimeBetweenTxnMinutes: 0,
  1353→        dailyVelocity: 1,
  1354→      };
  1355→      
  1356→      this.userProfiles.set(transaction.customerId, newProfile);
  1357→    }
  1358→    
  1359→    logger.debug('User profile updated', {
  1360→      event: 'analyzer.profile.updated',
  1361→      metadata: { customerId: transaction.customerId },
  1362→    });
  1363→  }

  1486→  /**
  1487→   * Run statistical outlier detection on transaction
  1488→   * @private
  1489→   * @param transaction - Transaction to analyze
  1490→   * @returns Array of detection results
  1491→   */
  1492→  private runStatisticalAnalysis(transaction: TransactionData): DetectionResult[] {
  1493→    const results: DetectionResult[] = [];
  1494→    const customerId = transaction.customerId;
  1495→    
  1496→    // Check for extremely large amounts (absolute threshold, no history needed)
  1497→    // This catches obvious anomalies like system max values or unreasonably large transactions
  1498→    const EXTREME_AMOUNT_THRESHOLD = 10000000; // 10 million in smallest currency unit
  1499→    if (transaction.amount > EXTREME_AMOUNT_THRESHOLD) {
  1500→      results.push({
  1501→        isAnomalous: true,
  1502→        category: AnomalyCategory.STATISTICAL,
  1503→        severity: AnomalySeverity.CRITICAL,
  1504→        confidence: Math.min(transaction.amount / EXTREME_AMOUNT_THRESHOLD / 10, 1),
  1505→        score: transaction.amount / EXTREME_AMOUNT_THRESHOLD,
  1506→        threshold: EXTREME_AMOUNT_THRESHOLD,
  1507→        actualValue: transaction.amount,
  1508→        description: `Extremely large transaction amount detected`,
  1509→        details: { 
  1510→          amount: transaction.amount,
  1511→          threshold: EXTREME_AMOUNT_THRESHOLD,
  1512→          method: 'extreme_amount_check',
  1513→        },
  1514→        detectedAt: new Date(),
  1515→      });
  1516→    }
  1517→    
  1518→    // Get historical amounts for this user
  1519→    const userHistoryKey = `amounts:${customerId}`;
  1520→    let historicalAmounts = this.timeSeriesData.get(userHistoryKey)?.map(p => p.value) || [];
  1521→    
  1522→    // Add current transaction for analysis
  1523→    const allAmounts = [...historicalAmounts, transaction.amount];
  1524→    
  1525→    if (allAmounts.length >= this.config.minSamplesRequired) {
  1526→      // Z-Score Analysis
  1527→      try {
  1528→        const mean = calculateMean(historicalAmounts);
  1529→        const std = calculateStandardDeviation(historicalAmounts);
  1530→        
  1531→        if (std > 0) {
  1532→          const zScore = Math.abs((transaction.amount - mean) / std);
  1533→          
  1534→          if (zScore > this.thresholds.zScoreThreshold) {
  1535→            results.push({
  1536→              isAnomalous: true,
  1537→              category: AnomalyCategory.STATISTICAL,
  1538→              severity: this.scoreToSeverity(zScore / this.thresholds.zScoreThreshold),
  1539→              confidence: clamp(zScore / (this.thresholds.zScoreThreshold * 2), 0, 1),
  1540→              score: zScore,
  1541→              threshold: this.thresholds.zScoreThreshold,
  1542→              actualValue: transaction.amount,
  1543→              description: `Transaction amount Z-score (${zScore.toFixed(2)}) exceeds threshold`,
  1544→              details: { mean, stdDev: std, method: 'z_score' },
  1545→              detectedAt: new Date(),
  1546→            });
  1547→          }
  1548→        }
  1549→      } catch (error) {
  1550→        logger.warn('Z-score calculation failed', { event: 'analyzer.zscore.error', error: error as Error });
  1551→      }
  1552→      
  1553→      // IQR Analysis
  1554→      try {
  1555→        const { q1, q3 } = calculateQuartiles(allAmounts);
  1556→        const iqr = q3 - q1;
  1557→        const lowerBound = q1 - this.thresholds.iqrMultiplier * iqr;
  1558→        const upperBound = q3 + this.thresholds.iqrMultiplier * iqr;
  1559→        
  1560→        if (transaction.amount < lowerBound || transaction.amount > upperBound) {
  1561→          const deviation = transaction.amount > upperBound 
  1562→            ? transaction.amount - upperBound 
  1563→            : lowerBound - transaction.amount;
  1564→            
  1565→          results.push({
  1566→            isAnomalous: true,
  1567→            category: AnomalyCategory.STATISTICAL,
  1568→            severity: deviation > iqr ? AnomalySeverity.HIGH : AnomalySeverity.MEDIUM,
  1569→            confidence: clamp(deviation / (iqr * 2), 0, 1),
  1570→            score: deviation / iqr,
  1571→            threshold: this.thresholds.iqrMultiplier,
  1572→            actualValue: transaction.amount,
  1573→            description: `Transaction amount outside IQR bounds [${lowerBound}, ${upperBound}]`,
  1574→            details: { q1, q3, iqr, lowerBound, upperBound, method: 'iqr' },
  1575→            detectedAt: new Date(),
  1576→          });
  1577→        }
  1578→      } catch (error) {
  1579→        logger.warn('IQR calculation failed', { event: 'analyzer.iqr.error', error: error as Error });
  1580→      }
  1581→      
  1582→      // Modified Z-Score (using MAD)
  1583→      try {
  1584→        const mad = calculateMAD(historicalAmounts);
  1585→        const median = calculateMedian(historicalAmounts);
  1586→        
  1587→        if (mad > 0) {
  1588→          const modifiedZScore = (0.6745 * (transaction.amount - median)) / mad;
  1589→          const absModifiedZScore = Math.abs(modifiedZScore);
  1590→          
  1591→          if (absModifiedZScore > this.thresholds.modifiedZScoreThreshold) {
  1592→            results.push({
  1593→              isAnomalous: true,
  1594→              category: AnomalyCategory.STATISTICAL,
  1595→              severity: this.scoreToSeverity(absModifiedZScore / this.thresholds.modifiedZScoreThreshold),
  1596→              confidence: clamp(absModifiedZScore / (this.thresholds.modifiedZScoreThreshold * 2), 0, 1),
  1597→              score: absModifiedZScore,
  1598→              threshold: this.thresholds.modifiedZScoreThreshold,
  1599→              actualValue: transaction.amount,
  1600→              description: `Modified Z-score (${absModifiedZScore.toFixed(2)}) exceeds threshold`,
  1601→              details: { median, mad, method: 'modified_z_score' },
  1602→              detectedAt: new Date(),
  1603→            });
  1604→          }
  1605→        }
  1606→      } catch (error) {
  1607→        logger.warn('Modified Z-score calculation failed', { 
  1608→          event: 'analyzer.modified_zscore.error', 
  1609→          error: error as Error 
  1610→        });
  1611→      }
  1612→    }
  1613→    
  1614→    return results;
  1615→  }

  1661→      results.push({

  1673→    }

  1675→    // Check time since last transaction

  1677→    const timeSinceLastMinutes = timeSinceLastTxn / 60000;

  1681→      
  1682→      // Flag if transaction happens much sooner than expected (potential automated attack)

  1689→          score: 1 / Math.max(timeRatio, 0.01),
  1690→          threshold: 0.1,
  1691→          actualValue: timeSinceLastMinutes,
  1692→          description: 'Rapid successive transactions detected',
  1693→          details: { 
  1694→            timeSinceLastMinutes: timeSinceLastMinutes.toFixed(2),
  1695→            averageInterval: profile.avgTimeBetweenTxnMinutes,
  1696→          },
  1697→          detectedAt: new Date(),
  1698→        });
  1699→      }
  1700→    }

  1702→    return results;
  1703→  }

  1783→  /**

  1801→      const riskScore = this.calculateLocationRisk(transaction.countryCode, profile);

  1810→          threshold: this.thresholds.newLocationRiskThreshold,
  1811→          actualValue: riskScore,
  1812→          description: `Transaction from new/unusual location: ${transaction.countryCode}`,
  1813→          details: {
  1814→            newCountry: transaction.countryCode,
  1815→            knownCountries: Array.from(profile.knownLocations),
  1816→            riskScore,
  1817→          },
  1818→          detectedAt: new Date(),
  1819→        });
  1820→      }
  1821→    }
  1822→    
  1823→    // Distance/velocity analysis if coordinates available
  1824→    if (transaction.latitude && transaction.longitude) {
  1825→      const userGeoHistory = this.geoHistory.get(transaction.customerId) || [];
  1826→      
  1827→      if (userGeoHistory.length > 0) {
  1828→        const lastLocation = userGeoHistory[userGeoHistory.length - 1];
  1829→        const distance = calculateHaversineDistance(
  1830→          lastLocation.latitude,
  1831→          lastLocation.longitude,
  1832→          transaction.latitude,
  1833→          transaction.longitude
  1834→        );
  1835→        
  1836→        const timeDiffMinutes = (transaction.timestamp.getTime() - lastLocation.timestamp.getTime()) / 60000;
  1837→        
  1838→        // Check for impossible travel
  1839→        if (distance > this.thresholds.maxGeoDistanceKm) {
  1840→          if (timeDiffMinutes < this.thresholds.minTimeForDistanceMinutes) {
  1841→            const requiredTime = distance / 900; // Assuming max 900 km/h travel speed
  1842→            
  1843→            results.push({
  1844→              isAnomalous: true,
  1845→              category: AnomalyCategory.GEOGRAPHIC,
  1846→              severity: AnomalySeverity.CRITICAL,
  1847→              confidence: 0.95,
  1848→              score: distance,
  1849→              threshold: this.thresholds.maxGeoDistanceKm,
  1850→              actualValue: distance,
