   363→// ============== Audit Log Service Implementation ==============
   364→
   365→/**
   366→ * Enterprise-grade Audit Logging Service
   367→ * 
   368→ * Features:
   369→ * - Comprehensive event logging with structured data
   370→ * - Tamper-evident hash chain for integrity verification
   371→ * - Flexible querying and search capabilities
   372→ * - Multiple export formats for compliance
   373→ * - Configurable retention policies
   374→ * - Full data change tracking
   375→ * 
   376→ * @example
   377→ * ```typescript
   378→ * const auditLog = new AuditLogService();
   379→ * 
   380→ * await auditLog.log({
   381→ *   eventType: AuditEventType.TRANSACTION,
   382→ *   action: AuditAction.PAYMENT_INITIATED,
   383→ *   userId: 'user-123',
   384→ *   username: 'john.doe',
   385→ *   description: 'Payment initiated for order #ORD-456',
   386→ *   resource: { type: 'transaction', id: 'txn-789' },
   387→ *   changes: [{ field: 'status', previousValue: null, newValue: 'pending', changeType: 'created' }],
   388→ * });
   389→ * ```
   390→ */
   391→export class AuditLogService {
   392→  /** In-memory event store */
   393→  private events: Map<string, AuditEvent> = new Map();
   394→  
   395→  /** Ordered event IDs for sequence tracking */
   396→  private orderedEventIds: string[] = [];
   397→  
   398→  /** Current hash chain state */
   399→  private lastHash: string = '';
   400→  
   401→  /** Current sequence number */
   402→  private sequenceNumber: number = 0;
   403→  
   404→  /** Active retention policies */
   405→  private retentionPolicies: RetentionPolicy[] = [];
   406→  
   407→  /** Statistics cache */
   408→  private statsCache: AuditStatistics | null = null;
   409→  
   410→  /** Whether service is initialized */
   411→  private initialized: boolean = false;
   412→
   413→  /**
   414→   * Create a new AuditLogService instance
   415→   * @param config Optional configuration overrides
   416→   */
   417→  constructor(config?: {
   418→    defaultRetentionDays?: number;
   419→    maxInMemoryEvents?: number;
   420→    enableAutoCleanup?: boolean;
   421→  }) {
   422→    // Initialize default retention policy
   423→    this.retentionPolicies.push({
   424→      name: 'default',
   425→      description: 'Default retention policy for all events',
   426→      retentionDays: config?.defaultRetentionDays ?? DEFAULT_RETENTION_DAYS,
   427→      active: true,
   428→      createdAt: new Date(),
   429→      updatedAt: new Date(),
   430→    });
   431→
   432→    logger.info('AuditLogService initialized', {
   433→      event: 'audit.init',
   434→      metadata: { 
   435→        defaultRetentionDays: config?.defaultRetentionDays ?? DEFAULT_RETENTION_DAYS,
   436→        maxInMemoryEvents: config?.maxInMemoryEvents ?? MAX_IN_MEMORY_EVENTS,
   437→      },
   438→    });
   439→
   440→    this.initialized = true;
   441→  }
   442→
   443→  // ============== Core Logging Methods ==============
   444→
   445→  /**
   446→   * Log a new audit event
   447→   * 
   448→   * @param event The audit event data (without system-managed fields)
   449→   * @returns The complete audit event with generated fields
   450→   * @throws {AppError} If validation fails or service not initialized
   451→   * 
   452→   * @example
   453→   * ```typescript
   454→   * const event = await auditLog.log({
   455→   *   eventType: AuditEventType.AUTHENTICATION,
   456→   *   action: AuditAction.LOGIN,
   457→   *   userId: 'user-123',
   458→   *   username: 'john.doe',
   459→   *   description: 'User logged in successfully',
   460→   *   source: { ipAddress: '192.168.1.1', userAgent: 'Mozilla/5.0...' },
   461→   * });
   462→   * ```
   463→   */
   464→  async log(event: Omit<AuditEvent, 'id' | 'timestamp' | 'previousHash' | 'hash' | 'sequenceNumber'>): Promise<AuditEvent> {
   465→    if (!this.initialized) {
   466→      throw new AppError('AuditLogService not initialized', ErrorCode.UNKNOWN_ERROR);
   467→    }
   468→
   469→    // Validate required fields
   470→    this.validateEventInput(event);
   471→
   472→    // Generate unique ID
   473→    const id = this.generateEventId();
   474→
   475→    // Create timestamp
   476→    const timestamp = new Date();
   477→
   478→    // Build complete event with hash chain
   479→    const completeEvent: AuditEvent = {
   480→      ...event,
   481→      id,
   482→      timestamp,
   483→      previousHash: this.lastHash,
   484→      sequenceNumber: ++this.sequenceNumber,
   485→    };
   486→
   487→    // Calculate and set hash
   488→    completeEvent.hash = this.calculateEventHash(completeEvent);
   489→
   490→    // Store the event
   491→    this.events.set(id, completeEvent);
   492→    this.orderedEventIds.push(id);
   493→
   494→    // Update last hash for chain
   495→    this.lastHash = completeEvent.hash;
   496→
   497→    // Invalidate stats cache
   498→    this.statsCache = null;
   499→
   500→    // Log to application logger
   501→    logger.info(`Audit event logged: ${event.action}`, {
   502→      event: `audit.${event.eventType.toLowerCase()}.${event.action.toLowerCase()}`,
   503→      metadata: {
   504→        eventId: id,
   505→        userId: event.userId,
   506→        action: event.action,
   507→        severity: event.severity,
   508→        outcome: event.outcome,
   509→      },
   510→    });
   511→
   512→    return completeEvent;
   513→  }
   514→
   515→  /**
   516→   * Log an authentication event (convenience method)
   517→   * 
   518→   * @param action The authentication action
   519→   * @param userId User identifier
   520→   * @param options Additional options
   521→   * @returns The created audit event
   522→   */
   523→  async logAuthentication(
   524→    action: AuditAction.LOGIN | AuditAction.LOGOUT | AuditAction.LOGIN_FAILED | AuditAction.PASSWORD_CHANGE | AuditAction.MFA_ENABLED | AuditAction.MFA_DISABLED,
   525→    userId: string,
   526→    options: {
   527→      username?: string;
   528→      outcome?: AuditOutcome;
   529→      description?: string;
   530→      source?: AuditSource;
   531→      metadata?: Record<string, unknown>;
   532→      correlationId?: string;
   533→    } = {}
   534→  ): Promise<AuditEvent> {
   535→    const descriptions: Record<string, string> = {
   536→      [AuditAction.LOGIN]: `User ${options.username || userId} logged in`,
   537→      [AuditAction.LOGOUT]: `User ${options.username || userId} logged out`,
   538→      [AuditAction.LOGIN_FAILED]: `Failed login attempt for ${options.username || userId}`,
   539→      [AuditAction.PASSWORD_CHANGE]: `Password changed for user ${options.username || userId}`,
   540→      [AuditAction.MFA_ENABLED]: `MFA enabled for user ${options.username || userId}`,
   541→      [AuditAction.MFA_DISABLED]: `MFA disabled for user ${options.username || userId}`,
   542→    };
   543→
   544→    return this.log({
   545→      eventType: AuditEventType.AUTHENTICATION,
   546→      action,
   547→      userId,
   548→      username: options.username,
   549→      severity: action === AuditAction.LOGIN_FAILED ? AuditSeverity.WARNING : AuditSeverity.INFO,
   550→      outcome: options.outcome ?? (action === AuditAction.LOGIN_FAILED ? AuditOutcome.FAILURE : AuditOutcome.SUCCESS),
   551→      description: options.description ?? descriptions[action],
   552→      source: options.source,
   553→      metadata: options.metadata,
   554→      correlationId: options.correlationId,
   555→    });
   556→  }
   557→
   558→  /**
   559→   * Log a transaction event (convenience method)
   560→   * 
   561→   * @param action The transaction action
   562→   * @param userId User identifier
   563→   * @param transactionId Transaction identifier
   564→   * @param options Additional options
   565→   * @returns The created audit event
   566→   */
   567→  async logTransaction(
   568→    action: AuditAction.PAYMENT_INITIATED | AuditAction.PAYMENT_COMPLETED | AuditAction.PAYMENT_FAILED | AuditAction.REFUND_INITIATED | AuditAction.REFUND_COMPLETED,
   569→    userId: string,
   570→    transactionId: string,
   571→    options: {
   572→      username?: string;
   573→      amount?: number;
   574→      currency?: string;
   575→      outcome?: AuditOutcome;
   576→      description?: string;
   577→      source?: AuditSource;
   578→      changes?: DataChange[];
   579→      metadata?: Record<string, unknown>;
   580→      correlationId?: string;
   581→    } = {}
   582→  ): Promise<AuditEvent> {
   583→    const severityMap: Record<string, AuditSeverity> = {
   584→      [AuditAction.PAYMENT_FAILED]: AuditSeverity.WARNING,
   585→      [AuditAction.REFUND_INITIATED]: AuditSeverity.WARNING,
   586→      [AuditAction.PAYMENT_INITIATED]: AuditSeverity.INFO,
   587→      [AuditAction.PAYMENT_COMPLETED]: AuditSeverity.INFO,
   588→      [AuditAction.REFUND_COMPLETED]: AuditSeverity.INFO,
   589→    };
   590→
   591→    const descriptions: Record<string, string> = {
   592→      [AuditAction.PAYMENT_INITIATED]: `Payment initiated: ${options.amount ? `${options.amount} ${options.currency}` : transactionId}`,
   593→      [AuditAction.PAYMENT_COMPLETED]: `Payment completed: ${options.amount ? `${options.amount} ${options.currency}` : transactionId}`,
   594→      [AuditAction.PAYMENT_FAILED]: `Payment failed: ${options.amount ? `${options.amount} ${options.currency}` : transactionId}`,
   595→      [AuditAction.REFUND_INITIATED]: `Refund initiated: ${transactionId}`,
   596→      [AuditAction.REFUND_COMPLETED]: `Refund completed: ${transactionId}`,
   597→    };
   598→
   599→    return this.log({
   600→      eventType: AuditEventType.TRANSACTION,
   601→      action,
   602→      userId,
   603→      username: options.username,
   604→      severity: severityMap[action] ?? AuditSeverity.INFO,
   605→      outcome: options.outcome ?? (action === AuditAction.PAYMENT_FAILED ? AuditOutcome.FAILURE : AuditOutcome.SUCCESS),
   606→      description: options.description ?? descriptions[action],
   607→      source: options.source,
   608→      resource: {
   609→        type: 'transaction',
   610→        id: transactionId,
   611→        metadata: options.amount ? { amount: options.amount, currency: options.currency } : undefined,
   612→      },
   613→      changes: options.changes,
   614→      metadata: options.metadata,
   615→      correlationId: options.correlationId,
   616→    });
   617→  }

   627→  async logConfigurationChange(
   628→    configKey: string,
   629→    userId: string,
   630→    options: {
   631→      username?: string;
   632→      previousValue?: unknown;
   633→      newValue?: unknown;
   634→      outcome?: AuditOutcome;
   635→      description?: string;
   636→      source?: AuditSource;
   637→      metadata?: Record<string, unknown>;
   638→      correlationId?: string;
   639→    } = {}
   640→  ): Promise<AuditEvent> {
   641→    const changes: DataChange[] = [];
   642→    
   643→    if (options.previousValue !== undefined || options.newValue !== undefined) {
   644→      changes.push({
   645→        field: configKey,
   646→        previousValue: options.previousValue ?? null,
   647→        newValue: options.newValue ?? null,
   648→        changeType: options.previousValue === undefined ? 'created' : 'updated',
   649→      });
   650→    }
   651→
   652→    return this.log({
   653→      eventType: AuditEventType.CONFIGURATION,
   654→      action: AuditAction.CONFIG_UPDATED,
   655→      userId,
   656→      username: options.username,
   657→      severity: AuditSeverity.WARNING,
   658→      outcome: options.outcome ?? AuditOutcome.SUCCESS,
   659→      description: options.description ?? `Configuration updated: ${configKey}`,
   660→      source: options.source,
   661→      resource: {
   662→        type: 'config',
   663→        id: configKey,
   664→      },
   665→      changes: changes.length > 0 ? changes : undefined,
   666→      metadata: options.metadata,
   667→      correlationId: options.correlationId,
   668→    });
   669→  }

   719→  /**
   720→   * Search audit logs with flexible filtering
   721→   * 
   722→   * @param params Query parameters for filtering
   723→   * @returns Array of matching audit events
   724→   * 
   725→   * @example
   726→   * ```typescript
   727→   * // Find all failed login attempts in the last 24 hours
   728→   * const results = await auditLog.query({
   729→   *   eventTypes: [AuditEventType.AUTHENTICATION],
   730→   *   actions: [AuditAction.LOGIN_FAILED],
   731→   *   startDate: new Date(Date.now() - 24 * 60 * 60 * 1000),
   732→   *   sortOrder: 'desc',
   733→   * });
   734→   * ```
   735→   */
   736→  async query(params: AuditQueryParams = {}): Promise<AuditEvent[]> {
   737→    let results = Array.from(this.events.values());
   738→
   739→    // Apply filters
   740→    if (params.eventTypes && params.eventTypes.length > 0) {
   741→      results = results.filter(e => params.eventTypes!.includes(e.eventType));
   742→    }
   743→
   744→    if (params.actions && params.actions.length > 0) {
   745→      results = results.filter(e => params.actions!.includes(e.action));
   746→    }
   747→
   748→    if (params.userIds && params.userIds.length > 0) {
   749→      results = results.filter(e => params.userIds!.includes(e.userId));
   750→    }
   751→
   752→    if (params.severities && params.severities.length > 0) {
   753→      results = results.filter(e => params.severities!.includes(e.severity));
   754→    }
   755→
   756→    if (params.outcomes && params.outcomes.length > 0) {
   757→      results = results.filter(e => params.outcomes!.includes(e.outcome));
   758→    }
   759→
   760→    if (params.startDate) {
   761→      results = results.filter(e => e.timestamp >= params.startDate!);
   762→    }
   763→
   764→    if (params.endDate) {
   765→      results = results.filter(e => e.timestamp <= params.endDate!);
   766→    }
   767→
   768→    if (params.searchTerm) {
   769→      const term = params.searchTerm.toLowerCase();
   770→      results = results.filter(e => 
   771→        e.description.toLowerCase().includes(term) ||
   772→        e.userId.toLowerCase().includes(term) ||
   773→        (e.username && e.username.toLowerCase().includes(term)) ||
   774→        (e.metadata && JSON.stringify(e.metadata).toLowerCase().includes(term))
   775→      );
   776→    }
   777→
   778→    if (params.resourceType) {
   779→      results = results.filter(e => e.resource?.type === params.resourceType);
   780→    }
   781→
   782→    if (params.resourceId) {
   783→      results = results.filter(e => e.resource?.id === params.resourceId);
   784→    }
   785→
   786→    if (params.correlationId) {
   787→      results = results.filter(e => e.correlationId === params.correlationId);
   788→    }
   789→
   790→    // Sort results
   791→    const sortBy = params.sortBy ?? 'timestamp';
   792→    const sortOrder = params.sortOrder ?? 'desc';
   793→
   794→    results.sort((a, b) => {
   795→      let comparison = 0;
   796→      
   797→      switch (sortBy) {
   798→        case 'timestamp':
   799→          comparison = a.timestamp.getTime() - b.timestamp.getTime();
   800→          break;
   801→        case 'severity':
   802→          const severityOrder = { [AuditSeverity.EMERGENCY]: 4, [AuditSeverity.CRITICAL]: 3, [AuditSeverity.WARNING]: 2, [AuditSeverity.INFO]: 1 };
   803→          comparison = severityOrder[a.severity] - severityOrder[b.severity];
   804→          break;
   805→        case 'eventType':
   806→          comparison = a.eventType.localeCompare(b.eventType);
   807→          break;
   808→      }
   809→      
   810→      return sortOrder === 'desc' ? -comparison : comparison;
   811→    });
   812→
   813→    // Apply pagination
   814→    const offset = params.offset ?? 0;
   815→    const limit = params.limit ?? 100;
   816→    
   817→    results = results.slice(offset, offset + limit);
   818→
   819→    logger.debug('Audit query executed', {
   820→      event: 'audit.query',
   821→      metadata: { params, resultCount: results.length },
   822→    });
   823→
   824→    return results;
   825→  }

   871→  /**
   872→   * Get comprehensive audit statistics
   873→   * 
   874→   * @returns Statistics summary
   875→   */
   876→  async getStatistics(): Promise<AuditStatistics> {
   877→    if (this.statsCache) {
   878→      return this.statsCache;
   879→    }
   880→
   881→    const allEvents = Array.from(this.events.values());
   882→    const totalEvents = allEvents.length;
   883→
   884→    // Initialize counters
   885→    const eventsByType = {} as Record<AuditEventType, number>;
   886→    const eventsBySeverity = {} as Record<AuditSeverity, number>;
   887→    const eventsByOutcome = {} as Record<AuditOutcome, number>;
   888→
   889→    // Initialize all enum values
   890→    Object.values(AuditEventType).forEach(type => { eventsByType[type] = 0; });
   891→    Object.values(AuditSeverity).forEach(severity => { eventsBySeverity[severity] = 0; });
   892→    Object.values(AuditOutcome).forEach(outcome => { eventsByOutcome[outcome] = 0; });
   893→
   894→    // Count events
   895→    let oldestEvent: Date | undefined;
   896→    let newestEvent: Date | undefined;
   897→    let recentEvents = 0;
   898→    const oneHourAgo = new Date(Date.now() - 60 * 60 * 1000);
   899→
   900→    for (const event of allEvents) {
   901→      eventsByType[event.eventType]++;
   902→      eventsBySeverity[event.severity]++;
   903→      eventsByOutcome[event.outcome]++;
   904→
   905→      if (!oldestEvent || event.timestamp < oldestEvent) {
   906→        oldestEvent = event.timestamp;
   907→      }
   908→      if (!newestEvent || event.timestamp > newestEvent) {
   909→        newestEvent = event.timestamp;
   910→      }
   911→      if (event.timestamp >= oneHourAgo) {
   912→        recentEvents++;
   913→      }
   914→    }
   915→
   916→    // Verify chain integrity
   917→    const integrityResult = await this.verifyIntegrity();
   918→
   919→    this.statsCache = {
   920→      totalEvents,
   921→      eventsByType,
   922→      eventsBySeverity,
   923→      eventsByOutcome,
   924→      recentEvents,
   925→      oldestEvent,
   926→      newestEvent,
   927→      chainIntegrity: integrityResult.valid,
   928→      chainBreakPoint: integrityResult.invalidIndex,
   929→    };
   930→
   931→    return this.statsCache;
   932→  }

   993→  // ============== Integrity & Security Methods ==============
   994→
   995→  /**
   996→   * Verify the integrity of the hash chain
   997→   * 
   998→   * Detects any tampering with audit logs by verifying
   999→   * that each event's hash correctly chains to the next.
  1000→   * 
  1001→   * @returns Integrity verification result
  1002→   */
  1003→  async verifyIntegrity(): Promise<IntegrityResult> {
  1004→    const events = this.orderedEventIds.map(id => this.events.get(id)!).filter(Boolean);
  1005→
  1006→    if (events.length === 0) {
  1007→      return {
  1008→        valid: true,
  1009→        totalEvents: 0,
  1010→        verifiedAt: new Date(),
  1011→      };
  1012→    }
  1013→
  1014→    // Verify chain from beginning
  1015→    let expectedPreviousHash = '';
  1016→    
  1017→    for (let i = 0; i < events.length; i++) {
  1018→      const event = events[i];
  1019→
  1020→      // Check previous hash linkage
  1021→      if (event.previousHash !== expectedPreviousHash) {
  1022→        return {
  1023→          valid: false,
  1024→          totalEvents: events.length,
  1025→          firstInvalidEvent: event,
  1026→          invalidIndex: i,
  1027→          mismatchDetails: {
  1028→            eventId: event.id,
  1029→            expectedHash: expectedPreviousHash,
  1030→            actualHash: event.previousHash ?? '',
  1031→          },
  1032→          verifiedAt: new Date(),
  1033→        };
  1034→      }
  1035→
  1036→      // Recalculate and verify event hash
  1037→      const calculatedHash = this.calculateEventHash({ ...event, hash: undefined });
  1038→      if (event.hash !== calculatedHash) {
  1039→        return {
  1040→          valid: false,
  1041→          totalEvents: events.length,
  1042→          firstInvalidEvent: event,
  1043→          invalidIndex: i,
  1044→          mismatchDetails: {
  1045→            eventId: event.id,
  1046→            expectedHash: calculatedHash,
  1047→            actualHash: event.hash ?? '',
  1048→          },
  1049→          verifiedAt: new Date(),
  1050→        };
  1051→      }
  1052→
  1053→      expectedPreviousHash = event.hash;
  1054→    }
  1055→
  1056→    return {
  1057→      valid: true,
  1058→      totalEvents: events.length,
  1059→      verifiedAt: new Date(),
  1060→    };
  1061→  }

  1140→  /**
  1141→   * Apply retention policies to purge old events
  1142→   * 
  1143→   * @returns Number of events purged
  1144→   */
  1145→  async applyRetentionPolicies(): Promise<number> {
  1146→    const now = new Date();
  1147→    let totalPurged = 0;
  1148→
  1149→    for (const policy of this.retentionPolicies.filter(p => p.active)) {
  1150→      const cutoffDate = new Date(now.getTime() - policy.retentionDays * 24 * 60 * 60 * 1000);
  1151→      
  1152→      const eventsToPurge: string[] = [];
  1153→      
  1154→      for (const [id, event] of this.events.entries()) {
  1155→        // Check if policy applies to this event type
  1156→        if (policy.appliesToEventTypes && !policy.appliesToEventTypes.includes(event.eventType)) {
  1157→          continue;
  1158→        }
  1159→
  1160→        // Check minimum severity
  1161→        if (policy.minSeverity) {
  1162→          const severityLevels = [AuditSeverity.INFO, AuditSeverity.WARNING, AuditSeverity.CRITICAL, AuditSeverity.EMERGENCY];
  1163→          const eventLevel = severityLevels.indexOf(event.severity);
  1164→          const minLevel = severityLevels.indexOf(policy.minSeverity);
  1165→          if (eventLevel < minLevel) {
  1166→            continue;
  1167→          }
  1168→        }
  1169→
  1170→        // Check age
  1171→        if (event.timestamp < cutoffDate) {
  1172→          eventsToPurge.push(id);
  1173→        }
  1174→      }
  1175→
  1176→      // Purge events
  1177→      for (const id of eventsToPurge) {
  1178→        this.events.delete(id);
  1179→        const idx = this.orderedEventIds.indexOf(id);
  1180→        if (idx !== -1) {
  1181→          this.orderedEventIds.splice(idx, 1);
  1182→        }
  1183→        totalPurged++;
  1184→      }
  1185→    }
  1186→
  1187→    if (totalPurged > 0) {
  1188→      this.statsCache = null;
  1189→      
  1190→      logger.info('Retention policies applied', {
  1191→        event: 'audit.retention.applied',
  1192→        metadata: { eventsPurged: totalPurged, remainingEvents: this.events.size },
  1193→      });
  1194→    }
  1195→
  1196→    return totalPurged;
  1197→  }

  1253→   * 
  1254→   * @returns Unique identifier string
  1255→   */
  1256→  private generateEventId(): string {
  1257→    const timestamp = Date.now().toString(36);
  1258→    const randomPart = Math.random().toString(36).substring(2, 10);
  1259→    const seqPart = this.sequenceNumber.toString(36).padStart(6, '0');
  1260→    return `AUD-${timestamp}-${randomPart}-${seqPart}`;
  1261→  }

  1263→  /**
  1264→   * Calculate SHA-256 hash of an event for chain integrity
  1265→   * 
  1266→   * @param event The event to hash
  1267→   * @returns Hex-encoded hash string
  1268→   */
  1269→  private calculateEventHash(event: Partial<AuditEvent>): string {
  1270→    // Create canonical representation for hashing
  1271→    const canonical = [
  1272→      event.id,
  1273→      event.eventType,
  1274→      event.action,
  1275→      event.userId,
  1276→      event.timestamp?.toISOString(),
  1277→      event.severity,
  1278→      event.outcome,
  1279→      event.description,
  1280→      event.previousHash,
  1281→      event.sequenceNumber,
  1282→      JSON.stringify(event.changes ?? {}),
  1283→      JSON.stringify(event.metadata ?? {}),
  1284→    ].join('|');
  1285→
  1286→    // Simple hash implementation (in production, use crypto.subtle)
  1287→    return this.simpleHash(canonical);
  1288→  }
  1289→
  1290→  /**
  1291→   * Simple hash function for demo purposes
  1292→   * In production, replace with crypto.subtle.digest
  1293→   * 
  1294→   * @param input String to hash
  1295→   * @returns Hash string
  1296→   */
  1297→  private simpleHash(input: string): string {
  1298→    let hash = 0;
  1299→    for (let i = 0; i < input.length; i++) {
  1300→      const char = input.charCodeAt(i);
  1301→      hash = ((hash << 5) - hash) + char;
  1302→      hash = hash & hash; // Convert to 32-bit integer
  1303→    }
  1304→    
  1305→    // Convert to hex string
  1306→    const hashStr = Math.abs(hash).toString(16).padStart(8, '0');
  1307→    
  1308→    // Add some entropy based on content length and timestamp
  1309→    const extraEntropy = (input.length * 31 + Date.now()).toString(16);
  1310→    
  1311→    return `${hashStr}-${extraEntropy.substring(0, 8)}-${btoa(input.substring(0, 32)).substring(0, 8)}`;
  1312→  }

  1314→  /**
  1315→   * Validate event input data
  1316→   * 
  1317→   * @param event Event data to validate
  1318→   * @throws {AppError} If validation fails
  1319→   */
  1320→  private validateEventInput(event: Omit<AuditEvent, 'id' | 'timestamp' | 'previousHash' | 'hash' | 'sequenceNumber'>): void {
  1321→    if (!event.eventType || !Object.values(AuditEventType).includes(event.eventType)) {
  1322→      throw new AppError('Invalid or missing eventType', ErrorCode.VALIDATION_ERROR, { context: { eventType: event.eventType } });
  1323→    }
  1324→
  1325→    if (!event.action || !Object.values(AuditAction).includes(event.action)) {
  1326→      throw new AppError('Invalid or missing action', ErrorCode.VALIDATION_ERROR, { context: { action: event.action } });
  1327→    }
  1328→
  1329→    if (!event.userId || typeof event.userId !== 'string' || event.userId.trim().length === 0) {
  1330→      throw new AppError('userId is required and must be a non-empty string', ErrorCode.VALIDATION_ERROR);
  1331→    }
  1332→
  1333→    if (!event.severity || !Object.values(AuditSeverity).includes(event.severity)) {
  1334→      throw new AppError('Invalid or missing severity', ErrorCode.VALIDATION_ERROR, { context: { severity: event.severity } });
  1335→    }
  1336→
  1337→    if (!event.outcome || !Object.values(AuditOutcome).includes(event.outcome)) {
  1338→      throw new AppError('Invalid or missing outcome', ErrorCode.VALIDATION_ERROR, { context: { outcome: event.outcome } });
  1339→    }
  1340→
  1341→    if (!event.description || typeof event.description !== 'string' || event.description.trim().length === 0) {
  1342→      throw new AppError('description is required and must be a non-empty string', ErrorCode.VALIDATION_ERROR);
  1343→    }
  1344→  }

  1346→  /**
  1347→   * Export events to CSV format
  1348→   * 
  1349→   * @param events Events to export
  1350→   * @param options Export options
  1351→   * @returns CSV formatted string
  1352→   */
  1353→  private exportToCSV(events: AuditEvent[], options: ExportOptions): string {
  1354→    const headers = [
  1355→      'id',
  1356→      'timestamp',
  1357→      'eventType',
  1358→      'action',
  1359→      'userId',
  1360→      'username',
  1361→      'severity',
  1362→      'outcome',
  1363→      'description',
  1364→      'resourceType',
  1365→      'resourceId',
  1366→      'sourceIpAddress',
  1367→      'correlationId',
  1368→      ...(options.includeHashChain ? ['sequenceNumber', 'previousHash', 'hash'] : []),
  1369→      ...(options.includeMetadata ? ['metadata'] : []),
  1370→    ];
  1371→
  1372→    const rows = events.map(event => [
  1373→      this.csvEscape(event.id),
  1374→      event.timestamp.toISOString(),
  1375→      event.eventType,
  1376→      event.action,
  1377→      this.csvEscape(event.userId),
  1378→      this.csvEscape(event.username ?? ''),
  1379→      event.severity,
  1380→      event.outcome,
  1381→      this.csvEscape(event.description),
  1382→      event.resource?.type ?? '',
  1383→      event.resource?.id ?? '',
  1384→      event.source?.ipAddress ?? '',
  1385→      event.correlationId ?? '',
  1386→      ...(options.includeHashChain ? [event.sequenceNumber ?? '', event.previousHash ?? '', event.hash ?? ''] : []),
  1387→      ...(options.includeMetadata ? [this.csvEscape(JSON.stringify(event.metadata ?? {}))] : []),
  1388→    ]);
  1389→
  1390→    return [headers.join(','), ...rows.map(row => row.join(','))].join('\n');
  1391→  }

  1394→   * Export events to JSON format
  1395→   * 
  1396→   * @param events Events to export
  1397→   * @param options Export options
  1398→   * @returns JSON formatted string
  1399→   */
  1400→  private exportToJSON(events: AuditEvent[], options: ExportOptions): string {
  1401→    let processedEvents = events;
  1402→
  1403→    if (!options.includeMetadata) {
  1404→      processedEvents = events.map(({ metadata, ...rest }) => rest);
  1405→    }
  1406→
  1407→    if (!options.includeHashChain) {
  1408→      processedEvents = processedEvents.map(({ previousHash, hash, sequenceNumber, ...rest }) => rest);
  1409→    }
  1410→
  1411→    return JSON.stringify({
  1412→      exportedAt: new Date().toISOString(),
  1413→      totalRecords: processedEvents.length,
  1414→      events: processedEvents,
  1415→    }, null, 2);
  1416→  }

  1418→  /**
  1419→   * Escape a value for CSV output
  1420→   * 
  1421→   * @param value Value to escape
  1422→   * @returns Escaped CSV value
  1423→   */
  1424→  private csvEscape(value: string): string {
  1425→    if (value.includes(',') || value.includes('"') || value.includes('\n')) {
  1426→      return `"${value.replace(/"/g, '""')}"`;
  1427→    }
  1428→    return value;
  1429→  }

  1431→  /**
  1432→   * Redact sensitive fields from events
  1433→   * 
  1434→   * @param events Events to process
  1435→   * @param fields Fields to redact
  1436→   * @returns Redacted events
  1437→   */
  1438→  private redactEvents(events: AuditEvent[], fields: string[]): AuditEvent[] {
  1439→    return events.map(event => {
  1440→      const redacted = { ...event };
  1441→      
  1442→      // Redact from metadata
  1443→      if (redacted.metadata) {
  1444→        redacted.metadata = { ...redacted.metadata };
  1445→        for (const field of fields) {
  1446→          if (field in redacted.metadata) {
  1447→            (redacted.metadata as Record<string, unknown>)[field] = '[REDACTED]';
  1448→          }
  1449→        }
  1450→      }
  1451→
  1452→      // Redact from changes
  1453→      if (redacted.changes) {
  1454→        redacted.changes = redacted.changes.map(change => {
  1455→          if (fields.includes(change.field)) {
  1456→            return {
  1457→              ...change,
  1458→              previousValue: '[REDACTED]',
  1459→              newValue: '[REDACTED]',
  1460→            };
  1461→          }
  1462→          return change;
  1463→        });
  1464→      }
  1465→
  1466→      // Redact from source
  1467→      if (redacted.source) {
  1468→        redacted.source = { ...redacted.source };
  1469→        for (const field of fields) {
  1470→          if (field in redacted.source) {
  1471→            (redacted.source as Record<string, unknown>)[field] = '[REDACTED]';
  1472→          }
  1473→        }
  1474→      }
  1475→
  1476→      return redacted;
  1477→    });
  1478→  }
