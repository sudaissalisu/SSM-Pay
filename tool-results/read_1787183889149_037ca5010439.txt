   485→// ============== Statistical Utility Functions ==============

   487→/**
   488→ * Calculate mean of numeric array
   489→ * @param values - Array of numbers
   490→ * @returns Mean value
   491→ */
   492→export function calculateMean(values: number[]): number {
   493→  if (values.length === 0) return 0;
   494→  return values.reduce((sum, val) => sum + val, 0) / values.length;
   495→}
   496→
   497→/**
   498→ * Calculate standard deviation
   499→ * @param values - Array of numbers
   500→ * @returns Standard deviation (population)
   501→ */
   502→export function calculateStdDev(values: number[]): number {
   503→  if (values.length === 0) return 0;
   504→  const mean = calculateMean(values);
   505→  const squaredDiffs = values.map(val => Math.pow(val - mean, 2));
   506→  return Math.sqrt(squaredDiffs.reduce((sum, val) => sum + val, 0) / values.length);
   507→}
   508→
   509→/**
   510→ * Calculate variance
   511→ * @param values - Array of numbers
   512→ * @returns Variance
   513→ */
   514→export function calculateVariance(values: number[]): number {
   515→  const stdDev = calculateStdDev(values);
   516→  return stdDev * stdDev;
   517→}
   518→
   519→/**
   520→ * Calculate median value
   521→ * @param values - Array of numbers
   522→ * @returns Median value
   523→ */
   524→export function calculateMedian(values: number[]): number {
   525→  if (values.length === 0) return 0;
   526→  const sorted = [...values].sort((a, b) => a - b);
   527→  const mid = Math.floor(sorted.length / 2);
   528→  return sorted.length % 2 !== 0 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
   529→}
   530→
   531→/**
   532→ * Calculate percentile
   533→ * @param values - Array of numbers
   534→ * @param percentile - Percentile to calculate (0-100)
   535→ * @returns Value at given percentile
   536→ */
   537→export function calculatePercentile(values: number[], percentile: number): number {
   538→  if (values.length === 0) return 0;
   539→  const sorted = [...values].sort((a, b) => a - b);
   540→  const index = (percentile / 100) * (sorted.length - 1);
   541→  const lower = Math.floor(index);
   542→  const upper = Math.ceil(index);
   543→  if (lower === upper) return sorted[lower];
   544→  return sorted[lower] * (upper - index) + sorted[upper] * (index - lower);
   545→}

   547→/**
   548→ * Calculate correlation coefficient (Pearson)
   549→ * @param x - First array of numbers
   550→ * @param y - Second array of numbers
   551→ * @returns Correlation coefficient (-1 to 1)
   552→ */
   553→export function calculateCorrelation(x: number[], y: number[]): number {
   554→  if (x.length !== y.length || x.length === 0) return 0;
   555→  
   556→  const n = x.length;
   557→  const meanX = calculateMean(x);
   558→  const meanY = calculateMean(y);
   559→  
   560→  let numerator = 0;
   561→  let denomX = 0;
   562→  let denomY = 0;
   563→  
   564→  for (let i = 0; i < n; i++) {
   565→    const dx = x[i] - meanX;
   566→    const dy = y[i] - meanY;
   567→    numerator += dx * dy;
   568→    denomX += dx * dx;
   569→    denomY += dy * dy;
   570→  }
   571→  
   572→  const denominator = Math.sqrt(denomX * denomY);
   573→  return denominator === 0 ? 0 : numerator / denominator;
   574→}

   576→/**
   577→ * Calculate coefficient of determination (R-squared)
   578→ * @param actual - Actual values
   579→ * @param predicted - Predicted values
   580→ * @returns R-squared value (0-1)
   581→ */
   582→export function calculateRSquared(actual: number[], predicted: number[]): number {
   583→  if (actual.length !== predicted.length || actual.length === 0) return 0;
   584→  
   585→  const meanActual = calculateMean(actual);
   586→  const ssTot = actual.reduce((sum, val) => sum + Math.pow(val - meanActual, 2), 0);
   587→  const ssRes = actual.reduce((sum, val, i) => sum + Math.pow(val - predicted[i], 2), 0);
   588→  
   589→  return ssTot === 0 ? 1 : 1 - (ssRes / ssTot);
   590→}

   592→/**
   593→ * Normalize values to 0-1 range using min-max scaling
   594→ * @param values - Array of numbers
   595→ * @returns Normalized array and scaling parameters
   596→ */
   597→export function normalizeMinMax(
   598→  values: number[]
   599→): { normalized: number[]; min: number; max: number; scale: number } {
   600→  if (values.length === 0) return { normalized: [], min: 0, max: 0, scale: 1 };
   601→  
   602→  const min = Math.min(...values);
   603→  const max = Math.max(...values);
   604→  const scale = max - min || 1;
   605→  
   606→  return {
   607→    normalized: values.map(val => (val - min) / scale),
   608→    min,
   609→    max,
   610→    scale,
   611→  };
   612→}

   614→/**
   615→ * Z-score normalization (standardization)
   616→ * @param values - Array of numbers
   617→ * @returns Standardized array and statistics
   618→ */
   619→export function standardizeZScore(
   620→  values: number[]
   621→): { standardized: number[]; mean: number; stdDev: number } {
   622→  if (values.length === 0) return { standardized: [], mean: 0, stdDev: 1 };
   623→  
   624→  const mean = calculateMean(values);
   625→  const stdDev = calculateStdDev(values) || 1;
   626→  
   627→  return {
   628→    standardized: values.map(val => (val - mean) / stdDev),
   629→    mean,
   630→    stdDev,
   631→  };
   632→}

   634→/**
   635→ * Apply sigmoid activation function
   636→ * @param x - Input value
   637→ * @returns Sigmoid output (0-1)

   639→export function sigmoid(x: number): number {
   640→  return 1 / (1 + Math.exp(-Math.max(-500, Math.min(500, x))));
   641→}

   652→/**
   653→ * Calculate simple moving average
   654→ * @param values - Time series data
   655→ * @param window - Window size
   656→ * @returns Moving averages array
   657→ */
   658→export function simpleMovingAverage(values: number[], window: number): number[] {
   659→  if (window <= 0 || values.length === 0) return [];
   660→  if (window >= values.length) return [calculateMean(values)];
   661→  
   662→  const result: number[] = [];
   663→  for (let i = window - 1; i < values.length; i++) {
   664→    const windowSlice = values.slice(i - window + 1, i + 1);
   665→    result.push(calculateMean(windowSlice));
   666→  }
   667→  
   668→  return result;
   669→}

   671→/**
   672→ * Calculate weighted moving average
   673→ * @param values - Time series data
   674→ * @param weights - Weights for each position (most recent first or last based on order)
   675→ * @returns Weighted moving averages array
   676→ */
   677→export function weightedMovingAverage(values: number[], weights: number[]): number[] {
   678→  if (weights.length === 0 || values.length === 0) return [];
   679→  const window = weights.length;
   680→  if (values.length < window) return [calculateMean(values)];
   681→  
   682→  const weightSum = weights.reduce((a, b) => a + b, 0);
   683→  const result: number[] = [];
   684→  
   685→  for (let i = window - 1; i < values.length; i++) {
   686→    const windowSlice = values.slice(i - window + 1, i + 1);
   687→    let weightedSum = 0;
   688→    for (let j = 0; j < window; j++) {
   689→      weightedSum += windowSlice[j] * weights[j];
   690→    }
   691→    result.push(weightedSum / weightSum);
   692→  }
   693→  
   694→  return result;
   695→}
   696→
   697→/**
   698→ * Calculate exponential moving average
   699→ * @param values - Time series data
   700→ * @param alpha - Smoothing factor (0-1), higher = more responsive
   701→ * @returns Exponential moving averages array
   702→ */
   703→export function exponentialMovingAverage(values: number[], alpha: number = 0.3): number[] {
   704→  if (values.length === 0) return [];
   705→  if (alpha <= 0 || alpha > 1) alpha = 0.3;
   706→  
   707→  const ema: number[] = [values[0]];
   708→  for (let i = 1; i < values.length; i++) {
   709→    ema.push(alpha * values[i] + (1 - alpha) * ema[i - 1]);
   710→  }
   711→  
   712→  return ema;
   713→}

   715→/**
   716→ * Apply double exponential smoothing (Holt's method)
   717→ * @param values - Time series data
   718→ * @param alpha - Level smoothing factor
   719→ * @param beta - Trend smoothing factor
   720→ * @returns Smoothed values and trend components
   721→ */
   722→export function doubleExponentialSmoothing(
   723→  values: number[],
   724→  alpha: number = 0.3,
   725→  beta: number = 0.1
   726→): { smoothed: number[]; trend: number[] } {
   727→  if (values.length < 2) {
   728→    return { smoothed: values.slice(), trend: new Array(values.length).fill(0) };
   729→  }
   730→  
   731→  const smoothed: number[] = [values[0]];
   732→  const trend: number[] = [values[1] - values[0]];
   733→  const level: number[] = [values[0]];
   734→  
   735→  for (let i = 1; i < values.length; i++) {
   736→    const prevLevel = level[i - 1];
   737→    const prevTrend = trend[i - 1];
   738→    
   739→    const newLevel = alpha * values[i] + (1 - alpha) * (prevLevel + prevTrend);
   740→    const newTrend = beta * (newLevel - prevLevel) + (1 - beta) * prevTrend;
   741→    
   742→    level.push(newLevel);
   743→    trend.push(newTrend);
   744→    smoothed.push(newLevel + newTrend);
   745→  }
   746→  
   747→  return { smoothed, trend };
   748→}
   749→
   750→/**
   751→ * Apply triple exponential smoothing (Holt-Winters)
   752→ * @param values - Time series data
   753→ * @param seasonLength - Length of seasonal cycle
   754→ * @param alpha - Level smoothing factor
   755→ * @param beta - Trend smoothing factor
   756→ * @param gamma - Seasonal smoothing factor
   757→ * @returns Smoothed values with seasonal components
   758→ */
   759→export function holtWintersSmoothing(
   760→  values: number[],
   761→  seasonLength: number = 7,
   762→  alpha: number = 0.3,
   763→  beta: number = 0.1,
   764→  gamma: number = 0.1
   765→): {
   766→  smoothed: number[];
   767→  level: number[];
   768→  trend: number[];
   769→  seasonal: number[];
   770→} {
   771→  if (values.length < seasonLength * 2) {
   772→    logger.warn('Insufficient data for Holt-Winters smoothing', {
   773→      event: 'ml.holt-warnings',
   774→      metadata: { dataLength: values.length, requiredLength: seasonLength * 2 },
   775→    });
   776→    return {
   777→      smoothed: values.slice(),
   778→      level: values.slice(),
   779→      trend: new Array(values.length).fill(0),
   780→      seasonal: new Array(values.length).fill(0),
   781→    };
   782→  }
   783→  
   784→  // Initialize seasonal indices
   785→  const seasonalIndices: number[] = [];
   786→  for (let s = 0; s < seasonLength; s++) {
   787→    const seasonValues: number[] = [];
   788→    for (let i = s; i < values.length; i += seasonLength) {
   789→      seasonValues.push(values[i]);
   790→    }
   791→    const seasonAvg = calculateMean(seasonValues);
   792→    const globalAvg = calculateMean(values);
   793→    seasonalIndices.push(globalAvg > 0 ? seasonAvg / globalAvg : 1);
   794→  }
   795→  
   796→  // Initialize level and trend using first season
   797→  let level = values[0] / seasonalIndices[0];
   798→  const initialTrend =
   799→    (calculateMean(values.slice(0, seasonLength)) -
   800→      calculateMean(values.slice(seasonLength, seasonLength * 2))) /
   801→    seasonLength;
   802→  let trend = initialTrend;
   803→  
   804→  const levels: number[] = [level];
   805→  const trends: number[] = [trend];
   806→  const seasonal: number[] = seasonalIndices.slice();
   807→  const smoothed: number[] = [];
   808→  
   809→  for (let i = 0; i < values.length; i++) {
   810→    const s = i % seasonLength;
   811→    
   812→    if (i > 0) {
   813→      // Update level, trend, and seasonal
   814→      const newLevel =
   815→        alpha * (values[i] / seasonal[s]) + (1 - alpha) * (level + trend);
   816→      const newTrend =
   817→        beta * (newLevel - level) + (1 - beta) * trend;
   818→      const newSeasonal =
   819→        gamma * (values[i] / newLevel) + (1 - gamma) * seasonal[s];
   820→      
   821→      level = newLevel;
   822→      trend = newTrend;
   823→      seasonal[s] = newSeasonal;
   824→      
   825→      levels.push(level);
   826→      trends.push(trend);
   827→    }
   828→    
   829→    smoothed.push((level + trend) * seasonal[s]);
   830→  }
   831→  
   832→  return { smoothed, level: levels, trend: trends, seasonal };
   833→}
   834→
   835→/**
   836→ * Perform linear regression
   837→ * @param x - Independent variable values
   838→ * @param y - Dependent variable values
   839→ * @returns Regression coefficients and statistics
   840→ */
   841→export function linearRegression(x: number[], y: number[]): {
   842→  slope: number;
   843→  intercept: number;
   844→  rSquared: number;
   845→  standardError: number;
   846→  predict: (x: number) => number;
   847→} {
   848→  if (x.length !== y.length || x.length < 2) {
   849→    return {
   850→      slope: 0,
   851→      intercept: 0,
   852→      rSquared: 0,
   853→      standardError: 0,
   854→      predict: (val: number) => 0,
   855→    };
   856→  }
   857→  
   858→  const n = x.length;
   859→  const meanX = calculateMean(x);
   860→  const meanY = calculateMean(y);
   861→  
   862→  let sumXY = 0;
   863→  let sumXX = 0;
   864→  
   865→  for (let i = 0; i < n; i++) {
   866→    sumXY += (x[i] - meanX) * (y[i] - meanY);
   867→    sumXX += (x[i] - meanX) * (x[i] - meanX);
   868→  }
   869→  
   870→  const slope = sumXX === 0 ? 0 : sumXY / sumXX;
   871→  const intercept = meanY - slope * meanX;
   872→  
   873→  // Calculate R-squared
   874→  const predicted = x.map(xi => slope * xi + intercept);
   875→  const rSquared = calculateRSquared(y, predicted);
   876→  
   877→  // Calculate standard error
   878→  const residuals = y.map((yi, i) => yi - predicted[i]);
   879→  const mse = residuals.reduce((sum, r) => sum + r * r, 0) / (n - 2);
   880→  const standardError = Math.sqrt(mse);
   881→  
   882→  return {
   883→    slope,
   884→    intercept,
   885→    rSquared,
   886→    standardError,
   887→    predict: (val: number) => slope * val + intercept,
   888→  };
   889→}
   890→
   891→/**
   892→ * Perform polynomial regression
   893→ * @param x - Independent variable values
   894→ * @param y - Dependent variable values
   895→ * @param degree - Polynomial degree
   896→ * @returns Coefficients and prediction function
   897→ */
   898→export function polynomialRegression(
   899→  x: number[],
   900→  y: number[],
   901→  degree: number = 2
   902→): {
   903→  coefficients: number[];
   904→  rSquared: number;
   905→  predict: (x: number) => number;
   906→} {
   907→  if (x.length !== y.length || x.length <= degree) {
   908→    return {
   909→      coefficients: new Array(degree + 1).fill(0),
   910→      rSquared: 0,
   911→      predict: (_val: number) => 0,
   912→    };
   913→  }
   914→  
   915→  // Build Vandermonde matrix and solve using normal equations
   916→  const n = x.length;
   917→  const m = degree + 1;
   918→  
   919→  // Create design matrix X
   920→  const X: number[][] = [];
   921→  for (let i = 0; i < n; i++) {
   922→    const row: number[] = [];
   923→    for (let j = 0; j < m; j++) {
   924→      row.push(Math.pow(x[i], j));
   925→    }
   926→    X.push(row);
   927→  }
   928→  
   929→  // Compute X^T * X
   930→  const XtX: number[][] = Array(m).fill(null).map(() => Array(m).fill(0));
   931→  for (let i = 0; i < m; i++) {
   932→    for (let j = 0; j < m; j++) {
   933→      for (let k = 0; k < n; k++) {
   934→        XtX[i][j] += X[k][i] * X[k][j];
   935→      }
   936→    }
   937→  }
   938→  
   939→  // Compute X^T * y
   940→  const Xty: number[] = Array(m).fill(0);
   941→  for (let i = 0; i < m; i++) {
   942→    for (let k = 0; k < n; k++) {
   943→      Xty[i] += X[k][i] * y[k];
   944→    }
   945→  }
   946→  
   947→  // Solve system using Gaussian elimination (simplified)
   948→  const coefficients = gaussianElimination(XtX, Xty);
   949→  
   950→  // Calculate R-squared
   951→  const predicted = x.map(xi =>
   952→    coefficients.reduce((sum, coef, j) => sum + coef * Math.pow(xi, j), 0)
   953→  );
   954→  const rSquared = calculateRSquared(y, predicted);
   955→  
   956→  return {
   957→    coefficients,
   958→    rSquared,
   959→    predict: (val: number) =>
   960→      coefficients.reduce((sum, coef, j) => sum + coef * Math.pow(val, j), 0),
   961→  };
   962→}
   963→
   964→/**
   965→ * Gaussian elimination for solving linear systems
   966→ * @param matrix - Coefficient matrix
   967→ * @param vector - Right-hand side vector
   968→ * @returns Solution vector
   969→ */
   970→function gaussianElimination(matrix: number[][], vector: number[]): number[] {
   971→  const n = vector.length;
   972→  const aug = matrix.map((row, i) => [...row, vector[i]]);
   973→  
   974→  // Forward elimination
   975→  for (let col = 0; col < n; col++) {
   976→    // Find pivot
   977→    let maxRow = col;
   978→    for (let row = col + 1; row < n; row++) {
   979→      if (Math.abs(aug[row][col]) > Math.abs(aug[maxRow][col])) {
   980→        maxRow = row;
   981→      }
   982→    }
   983→    
   984→    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
   985→    
   986→    if (Math.abs(aug[col][col]) < 1e-10) continue;
   987→    
   988→    for (let row = col + 1; row < n; row++) {
   989→      const factor = aug[row][col] / aug[col][col];
   990→      for (let j = col; j <= n; j++) {
   991→        aug[row][j] -= factor * aug[col][j];
   992→      }
   993→    }
   994→  }
   995→  
   996→  // Back substitution
   997→  const solution = new Array(n).fill(0);
   998→  for (let row = n - 1; row >= 0; row--) {
   999→    if (Math.abs(aug[row][row]) < 1e-10) {
  1000→      solution[row] = 0;
  1001→      continue;
  1002→    }
  1003→    solution[row] = aug[row][n];
  1004→    for (let col = row + 1; col < n; col++) {
  1005→      solution[row] -= aug[row][col] * solution[col];
  1006→    }
  1007→    solution[row] /= aug[row][row];
  1008→  }
  1009→  
  1010→  return solution;
  1011→}

  1013→/**
  1014→ * Detect anomalies using z-score method
  1015→ * @param values - Data series
  1016→ * @param threshold - Number of standard deviations
  1017→ * @returns Array of anomaly indices and scores
  1018→ */
  1019→export function detectAnomaliesZScore(
  1020→  values: number[],
  1021→  threshold: number = 2.5
  1022→): { index: number; value: number; zScore: number }[] {
  1023→  const { standardized, mean, stdDev } = standardizeZScore(values);
  1024→  const anomalies: { index: number; value: number; zScore: number }[] = [];
  1025→  
  1026→  for (let i = 0; i < standardized.length; i++) {
  1027→    if (Math.abs(standardized[i]) > threshold) {
  1028→      anomalies.push({
  1029→        index: i,
  1030→        value: values[i],
  1031→        zScore: standardized[i],
  1032→      });
  1033→    }
  1034→  }

  1036→  return anomalies;
  1037→}

  1039→/**
  1040→ * Detect anomalies using IQR method
  1041→ * @param values - Data series
  1042→ * @param multiplier - IQR multiplier (typically 1.5)
  1043→ * @returns Anomaly indices and values
  1044→ */
  1045→export function detectAnomaliesIQR(
  1046→  values: number[],
  1047→  multiplier: number = 1.5
  1048→): { index: number; value: number; type: 'outlier_high' | 'outlier_low' }[] {
  1049→  if (values.length === 0) return [];
  1050→  
  1051→  const q1 = calculatePercentile(values, 25);
  1052→  const q3 = calculatePercentile(values, 75);
  1053→  const iqr = q3 - q1;
  1054→  const lowerBound = q1 - multiplier * iqr;
  1055→  const upperBound = q3 + multiplier * iqr;
  1056→  
  1057→  const anomalies: { index: number; value: number; type: 'outlier_high' | 'outlier_low' }[] = [];
  1058→  
  1059→  for (let i = 0; i < values.length; i++) {
  1060→    if (values[i] < lowerBound) {
  1061→      anomalies.push({ index: i, value: values[i], type: 'outlier_low' });
  1062→    } else if (values[i] > upperBound) {
  1063→      anomalies.push({ index: i, value: values[i], type: 'outlier_high' });
  1064→    }
  1065→  }
  1066→  
  1067→  return anomalies;
  1068→}

  1309→      
  1310→      // Calculate prediction interval width based on historical error
  1311→      const residuals = counts.map((c, i) => c - (hwCounts.smoothed[i] || c));
  1312→      const residualStd = calculateStdDev(residuals);
  1313→      const zValue = this.getZValue(this.config.confidenceLevel);
  1314→      
  1315→      for (let i = 1; i <= horizon; i++) {
  1316→        const forecastDate = new Date(lastDate);
  1317→        forecastDate.setDate(forecastDate.getDate() + i);
  1318→        
  1319→        const seasonalIndex = seasonalIndices[(this.trainingData.length + i - 1) % seasonalIndices.length];
  1320→        const trendComponent = lastTrend * i;
  1321→        const baseForecast = (lastLevel + trendComponent) * seasonalIndex;
  1322→        
  1323→        // Widen confidence intervals further into the future
  1324→        const uncertaintyMultiplier = Math.sqrt(i);
  1325→        const intervalWidth = zValue * residualStd * uncertaintyMultiplier;
  1326→        
  1327→        predictions.push({
  1328→          date: forecastDate,
  1329→          predictedCount: Math.round(Math.max(0, baseForecast)),
  1330→          lowerBound: Math.round(Math.max(0, baseForecast - intervalWidth)),
  1331→          upperBound: Math.round(baseForecast + intervalWidth),
  1332→          pointEstimate: Math.round(Math.max(0, baseForecast)),
  1333→          seasonalComponent: seasonalIndex,
  1334→          trendComponent: trendComponent,
  1335→        });
  1336→      }
  1337→      
  1338→      // Calculate model metrics
  1339→      const fittedValues = hwCounts.smoothed.slice(-counts.length);
  1340→      const metrics = this.calculateMetrics(counts, fittedValues);
  1341→      
  1342→      const result: PredictionResult<VolumePrediction[]> = {
  1343→        predictions,
  1344→        modelVersion: this.modelVersion.version,
  1345→        predictedAt: new Date(),
  1346→        confidence: this.config.confidenceLevel,
  1347→        featuresUsed: ['count', 'volume', 'seasonal_indices', 'trend'],
  1348→        processingTimeMs: Date.now() - startTime,
  1349→        metrics,
  1350→      };
  1351→      
  1352→      this.setCache(cacheKey, result);
  1353→      
  1354→      logger.info('Volume prediction completed', {
  1355→        event: 'ml.predictor.volume-complete',
  1356→        metadata: {
  1357→          horizon,
  1358→          processingTimeMs: result.processingTimeMs,
  1359→          firstPrediction: predictions[0]?.predictedCount,
  1360→          lastPrediction: predictions[predictions.length - 1]?.predictedCount,
  1361→        },
  1362→      });
  1363→      
  1364→      return result;
  1365→    } catch (error) {
  1366→      logger.error('Volume prediction failed', {
  1367→        event: 'ml.predictor.volume-error',
  1368→        error: error instanceof Error ? error : new Error(String(error)),
  1369→      });
  1370→      throw error instanceof AppError ? error : new AppError(
  1371→        'Failed to generate volume prediction',
  1372→        ErrorCode.UNKNOWN_ERROR,
  1373→        { cause: error instanceof Error ? error : undefined }
  1374→      );
  1375→    }
  1376→  }

  1378→  /**
  1379→   * Calculate seasonal indices from historical data
  1380→   * @private
  1381→   */
  1382→  private calculateSeasonalIndices(values: number[]): number[] {
  1383→    const period = this.config.seasonalityPeriod;
  1384→    const indices: number[] = [];
  1385→    const overallMean = calculateMean(values);
  1386→    
  1387→    if (overallMean === 0) return new Array(period).fill(1);
  1388→    
  1389→    for (let s = 0; s < period; s++) {
  1390→      const seasonValues: number[] = [];
  1391→      for (let i = s; i < values.length; i += period) {
  1392→        seasonValues.push(values[i]);
  1393→      }
  1394→      indices.push(calculateMean(seasonValues) / overallMean);
  1395→    }
  1396→    
  1397→    return indices;
  1398→  }

  1400→  /**
  1401→   * Get Z-value for confidence level
  1402→   * @private
  1403→   */
  1404→  private getZValue(confidence: number): number {
  1405→    const zScores: Record<number, number> = {
  1406→      0.90: 1.645,
  1407→      0.95: 1.96,
  1408→      0.99: 2.576,
  1409→    };
  1410→    // Find closest match or interpolate
  1411→    const keys = Object.keys(zScores).map(Number).sort((a, b) => a - b);
  1412→    for (const key of keys) {
  1413→      if (confidence <= key) return zScores[key];
  1414→    }
  1415→    return 1.96; // Default to 95%
  1416→  }

  1565→  /**
  1566→   * Calculate logistic-style churn score
  1567→   * @private
  1568→   */
  1569→  private calculateLogisticScore(factors: ChurnFactor[]): number {
  1570→    let weightedSum = 0;
  1571→    let totalWeight = 0;
  1572→    
  1573→    for (const factor of factors) {
  1574→      // Normalize factor value to 0-1 range based on whether concerning
  1575→      const normalizedValue = factor.isConcerning ? 1 : 0.3;
  1576→      weightedSum += factor.weight * normalizedValue;
  1577→      totalWeight += factor.weight;
  1578→    }
  1579→    
  1580→    const rawScore = totalWeight > 0 ? weightedSum / totalWeight : 0;
  1581→    
  1582→    // Apply sigmoid to get probability
  1583→    // Shift to make moderate scores map to reasonable probabilities
  1584→    return sigmoid((rawScore - 0.4) * 10);
  1585→  }

  1870→  /**
  1871→   * Calculate time-based risk
  1872→   * @private
  1873→   */
  1874→  private calculateTimeRisk(hourOfDay: number, dayOfWeek: number): RiskIndicator {
  1875→    // Business hours are typically safer
  1876→    const isBusinessHours = hourOfDay >= 8 && hourOfDay <= 18;
  1877→    const isWeekend = dayOfWeek === 0 || dayOfWeek === 6;
  1878→    
  1879→    let risk = 0.05; // Base low risk during business hours on weekdays
  1880→    
  1881→    if (!isBusinessHours) risk += 0.08;
  1882→    if (isWeekend) risk += 0.05;
  1883→    
  1884→    // Late night transactions have higher risk
  1885→    if (hourOfDay >= 0 && hourOfDay < 6) risk += 0.1;
  1886→    
  1887→    const severity = risk > 0.15 ? 'warning' : 'info';
  1888→    
  1889→    return {
  1890→      name: 'temporal_risk',
  1891→      value: hourOfDay,
