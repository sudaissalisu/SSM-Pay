   529→// ============== Main Risk Engine Class ==============
   530→
   531→/**
   532→ * Enterprise Risk Scoring Engine
   533→ * 
   534→ * Provides comprehensive multi-factor risk assessment for payment transactions.
   535→ * Supports dynamic threshold adjustment, historical tracking, and regulatory compliance.
   536→ * 
   537→ * @class RiskEngine
   538→ * @example
   539→ * ```typescript
   540→ * const engine = new RiskEngine({
   541→ *   profile: 'conservative',
   542→ *   enableAML: true,
   543→ * });
   544→ * 
   545→ * const result = await engine.assessTransactionRisk(transaction, customer, history);
   546→ * ```
   547→ */
   548→export class RiskEngine {
   549→  /** Current configuration */
   550→  private config: RiskEngineConfig;
   551→  
   552→  /** Active threshold configuration */
   553→  private thresholds: RiskThresholds;
   554→  
   555→  /** Active factor weights */
   556→  private weights: RiskFactorWeights;
   557→  
   558→  /** Historical assessment records */
   559→  private history: Map<string, RiskHistoryRecord[]> = new Map();
   560→  
   561→  /** Assessment cache for performance */
   562→  private assessmentCache: Map<string, { result: RiskAssessmentResult; expiry: Date }> = new Map();
   563→  
   564→  /** Running statistics */
   565→  private stats: RiskStatistics = this.initializeStatistics();
   566→
   567→  /**
   568→   * Creates a new RiskEngine instance
   569→   * 
   570→   * @param config - Configuration options for the risk engine
   571→   * @throws {AppError} If configuration is invalid
   572→   */
   573→  constructor(config: Partial<RiskEngineConfig> = {}) {
   574→    const profile: RiskProfile = config.profile || 'moderate';
   575→    
   576→    // Validate profile
   577→    if (!['conservative', 'moderate', 'aggressive'].includes(profile)) {
   578→      throw new AppError(
   579→        `Invalid risk profile: ${profile}`,
   580→        ErrorCode.INVALID_CONFIG,
   581→        { severity: 'error', context: { provided: profile } }
   582→      );
   583→    }
   584→
   585→    // Build complete configuration
   586→    this.config = {
   587→      ...DEFAULT_CONFIG,
   588→      ...config,
   589→      profile,
   590→    };
   591→
   592→    // Initialize thresholds with defaults and overrides
   593→    this.thresholds = {
   594→      ...DEFAULT_THRESHOLDS[profile],
   595→      ...(this.config.thresholds || {}),
   596→    };
   597→
   598→    // Initialize weights with defaults and overrides
   599→    this.weights = {
   600→      ...DEFAULT_WEIGHTS[profile],
   601→      ...(this.config.weights || {}),
   602→    };
   603→
   604→    // Validate weights sum to approximately 1
   605→    const weightSum = Object.values(this.weights).reduce((sum, w) => sum + w, 0);
   606→    if (Math.abs(weightSum - 1.0) > 0.01) {
   607→      logger.warn('Risk factor weights do not sum to 1.0', {
   608→        event: 'risk_engine.init',
   609→        metadata: { weightSum, weights: this.weights },
   610→      });
   611→    }
   612→
   613→    logger.info('Risk Engine initialized', {
   614→      event: 'risk_engine.init',
   615→      metadata: {
   616→        profile: this.config.profile,
   617→        thresholds: this.thresholds,
   618→        version: this.config.version,
   619→      },
   620→    });
   621→  }
   622→
   623→  // ============== Public API Methods ==============
   624→
   625→  /**
   626→   * Perform comprehensive risk assessment on a transaction
   627→   * 
   628→   * @param transaction - The transaction to assess
   629→   * @param customer - Customer profile data
   630→   * @param history - Transaction history for velocity analysis
   631→   * @param device - Optional device intelligence data
   632→   * @param geoLocation - Optional geolocation data
   633→   * @returns Complete risk assessment result
   634→   * 
   635→   * @example
   636→   * ```typescript
   637→   * const result = await engine.assessTransactionRisk(
   638→   *   transactionData,
   639→   *   customerProfile,
   640→   *   transactionHistory,
   641→   *   deviceInfo,
   642→   *   geoData
   643→   * );
   644→   * 
   645→   * if (result.shouldBlock) {
   646→   *   return error('Transaction blocked due to high risk');
   647→   * }
   648→   * ```
   649→   */
   650→  async assessTransactionRisk(
   651→    transaction: TransactionInput,
   652→    customer: CustomerProfile,
   653→    history: TransactionHistory,
   654→    device?: DeviceIntelligence,
   655→    geoLocation?: GeoLocationData
   656→  ): Promise<RiskAssessmentResult> {
   657→    const startTime = Date.now();
   658→
   659→    try {
   660→      // Check cache first
   661→      const cacheKey = this.generateCacheKey(transaction);
   662→      const cached = this.getCachedAssessment(cacheKey);
   663→      if (cached) {
   664→        logger.debug('Returning cached risk assessment', {
   665→          event: 'risk_engine.cache_hit',
   666→          metadata: { transactionId: transaction.transactionId },
   667→        });
   668→        return cached;
   669→      }
   670→
   671→      // Validate inputs
   672→      this.validateInputs(transaction, customer, history);
   673→
   674→      // Calculate individual factor scores
   675→      const factors: RiskFactorContribution[] = [];
   676→
   677→      // 1. Amount-based risk factors
   678→      factors.push(this.calculateAmountRisk(transaction, history));
   679→
   680→      // 2. Velocity/rate-based risk factors
   681→      factors.push(this.calculateVelocityRisk(transaction, history));
   682→
   683→      // 3. Device-based risk factors
   684→      factors.push(this.calculateDeviceRisk(transaction, device));
   685→
   686→      // 4. Geographic risk factors
   687→      factors.push(this.calculateGeographicRisk(transaction, geoLocation, customer));
   688→
   689→      // 5. Behavioral/pattern risk factors
   690→      factors.push(this.calculateBehavioralRisk(transaction, history, customer));
   691→
   692→      // 6. Historical risk factors
   693→      factors.push(this.calculateHistoricalRisk(customer));
   694→
   695→      // 7. Compliance/AML risk factors
   696→      const amlAlerts = this.config.enableAML 
   697→        ? this.performAMLChecks(transaction, history)
   698→        : [];
   699→
   700→      factors.push(this.calculateComplianceRisk(amlAlerts, customer));
   701→
   702→      // 8. Customer profile risk factors
   703→      factors.push(this.calculateCustomerProfileRisk(transaction, customer));
   704→
   705→      // Calculate weighted composite score
   706→      const riskScore = this.calculateCompositeScore(factors);
   707→
   708→      // Determine risk level
   709→      const riskLevel = this.classifyRiskLevel(riskScore);
   710→
   711→      // Determine authentication requirements
   712→      const requiredAuth = this.determineAuthRequirement(riskLevel, factors, customer);
   713→
   714→      // Determine blocking/flagging
   715→      const shouldBlock = this.shouldBlockTransaction(riskLevel, amlAlerts, customer);
   716→      const shouldFlag = !shouldBlock && (riskLevel >= RiskLevel.HIGH || amlAlerts.length > 0);
   717→
   718→      // Generate recommendations
   719→      const recommendations = this.generateRecommendations(riskLevel, factors, amlAlerts);
   720→
   721→      // Calculate confidence based on data completeness
   722→      const confidence = this.calculateConfidence(factors, device, geoLocation);
   723→
   724→      // Build final result
   725→      const result: RiskAssessmentResult = {
   726→        riskScore: Math.round(riskScore * 100) / 100,
   727→        riskLevel,
   728→        requiredAuth,
   729→        shouldBlock,
   730→        shouldFlag,
   731→        factors,
   732→        amlAlerts,
   733→        assessedAt: new Date(),
   734→        assessmentVersion: this.config.version,
   735→        recommendations,
   736→        confidence,
   737→        processingTimeMs: Date.now() - startTime,
   738→      };
   739→
   740→      // Cache the result
   741→      this.cacheAssessment(cacheKey, result);
   742→
   743→      // Update statistics
   744→      this.updateStatistics(result);
   745→
   746→      // Log the assessment
   747→      logger.info('Risk assessment completed', {
   748→        event: 'risk_engine.assessment',
   749→        metadata: {
   750→          transactionId: transaction.transactionId,
   751→          customerId: customer.customerId,
   752→          riskScore: result.riskScore,
   753→          riskLevel: result.riskLevel,
   754→          shouldBlock: result.shouldBlock,
   755→          authRequired: result.requiredAuth,
   756→          amlAlerts: amlAlerts.length,
   757→          processingTimeMs: result.processingTimeMs,
   758→        },
   759→      });
   760→
   761→      return result;
   762→
   763→    } catch (error) {
   764→      const appError = error instanceof AppError 
   765→        ? error 
   766→        : new AppError('Risk assessment failed', ErrorCode.UNKNOWN_ERROR, { cause: error as Error });
   767→
   768→      logger.error('Risk assessment error', {
   769→        event: 'risk_engine.error',
   770→        metadata: { transactionId: transaction.transactionId },
   771→        error: appError,
   772→      });
   773→
   774→      throw appError;
   775→    }
   776→  }

   964→  /**
   965→   * Calculate amount-related risk factors
   966→   * Considers absolute amount, deviation from average, and round amounts
   967→   */
   968→  private calculateAmountRisk(
   969→    transaction: TransactionInput,
   970→    history: TransactionHistory
   971→  ): RiskFactorContribution {
   972→    let score = 0;
   973→    const indicators: string[] = [];
   974→    const metadata: Record<string, unknown> = {};
   975→
   976→    // Factor 1: Absolute amount risk (logarithmic scale)
   977→    const amountInMajor = transaction.amount / 100; // Convert from minor units
   978→    if (amountInMajor > 10000) {
   979→      score += 40;
   980→      indicators.push('very_high_amount');
   981→    } else if (amountInMajor > 5000) {
   982→      score += 25;
   983→      indicators.push('high_amount');
   984→    } else if (amountInMajor > 1000) {
   985→      score += 15;
   986→      indicators.push('elevated_amount');
   987→    } else if (amountInMajor > 500) {
   988→      score += 8;
   989→      indicators.push('moderate_amount');
   990→    }
   991→    metadata.amountInMajor = amountInMajor;
   992→
   993→    // Factor 2: Deviation from user's average
   994→    if (history.avgAmount30d > 0) {
   995→      const deviation = Math.abs(amountInMajor - history.avgAmount30d / 100) / (history.stdAmount30d / 100 || 1);
   996→      if (deviation > 5) {
   997→        score += 35;
   998→        indicators.push('extreme_deviation');
   999→      } else if (deviation > 3) {
  1000→        score += 20;
  1001→        indicators.push('high_deviation');
  1002→      } else if (deviation > 2) {
  1003→        score += 10;
  1004→        indicators.push('moderate_deviation');
  1005→      }
  1006→      metadata.deviation = Math.round(deviation * 100) / 100;
  1007→    }
  1008→
  1009→    // Factor 3: Exceeds maximum historical
  1010→    if (history.maxAmount30d > 0 && transaction.amount > history.maxAmount30d) {
  1011→      score += 15;
  1012→      indicators.push('exceeds_historical_max');
  1013→      metadata.previousMax = history.maxAmount30d;
  1014→    }
  1015→
  1016→    // Factor 4: Round amount detection (potential structuring)
  1017→    if (amountInMajor > 1000) {
  1018→      const roundedToThousand = amountInMajor % 1000 === 0;
  1019→      const roundedToHundred = amountInMajor % 100 === 0;
  1020→      
  1021→      if (roundedToThousand) {
  1022→        score += 15;
  1023→        indicators.push('round_thousand_amount');
  1024→      } else if (roundedToHundred) {
  1025→        score += 8;
  1026→        indicators.push('round_hundred_amount');
  1027→      }
  1028→    }
  1029→
  1030→    // Factor 5: Just below reporting threshold (potential structuring)
  1031→    const structuringRange = [900000, 950000, 990000]; // In cents ($9,000-$9,900)
  1032→    if (structuringRange.some(threshold => 
  1033→      Math.abs(transaction.amount - threshold) < 5000
  1034→    )) {
  1035→      score += 25;
  1036→      indicators.push('near_structuring_threshold');
  1037→    }
  1038→
  1039→    // Cap at 100
  1040→    score = Math.min(score, 100);
  1041→
  1042→    return {
  1043→      factorId: 'amount_risk',
  1044→      factorName: 'Amount Risk Analysis',
  1045→      score,
  1046→      maxScore: 100,
  1047→      rawValue: { amount: transaction.amount, amountInMajor },
  1048→      indicators,
  1049→      metadata,
  1050→    };
  1051→  }

  1053→  /**
  1054→   * Calculate velocity/rate-based risk factors
  1055→   * Analyzes transaction frequency patterns
  1056→   */
  1057→  private calculateVelocityRisk(
  1058→    transaction: TransactionInput,
  1059→    history: TransactionHistory
  1060→  ): RiskFactorContribution {
  1061→    let score = 0;
  1062→    const indicators: string[] = [];
  1063→    const metadata: Record<string, unknown> = {};
  1064→
  1065→    // Factor 1: Hourly velocity
  1066→    if (history.lastHourCount > SUSPICIOUS_PATTERNS.MAX_VELOCITY_PER_HOUR) {
  1067→      score += 35;
  1068→      indicators.push('excessive_hourly_velocity');
  1069→    } else if (history.lastHourCount > SUSPICIOUS_PATTERNS.MAX_VELOCITY_PER_HOUR * 0.6) {
  1070→      score += 15;
  1071→      indicators.push('elevated_hourly_velocity');
  1072→    }
  1073→    metadata.hourlyCount = history.lastHourCount;
  1074→
  1075→    // Factor 2: Daily velocity
  1076→    if (history.last24hCount > SUSPICIOUS_PATTERNS.MAX_VELOCITY_PER_DAY) {
  1077→      score += 35;
  1078→      indicators.push('excessive_daily_velocity');
  1079→    } else if (history.last24hCount > SUSPICIOUS_PATTERNS.MAX_VELOCITY_PER_DAY * 0.6) {
  1080→      score += 15;
  1081→      indicators.push('elevated_daily_velocity');
  1082→    }
  1083→    metadata.dailyCount = history.last24hCount;
  1084→
  1085→    // Factor 3: Failed transaction ratio
  1086→    const totalAttempts = history.last24hCount + history.failedTxnCount24h + history.declinedTxnCount24h;
  1087→    if (totalAttempts > 5) {
  1088→      const failRate = (history.failedTxnCount24h + history.declinedTxnCount24h) / totalAttempts;
  1089→      if (failRate > 0.5) {
  1090→        score += 25;
  1091→        indicators.push('high_failure_rate');
  1092→      } else if (failRate > 0.3) {
  1093→        score += 12;
  1094→        indicators.push('elevated_failure_rate');
  1095→      }
  1096→      metadata.failureRate = Math.round(failRate * 100) / 100;
  1097→    }
  1098→
  1099→    // Factor 4: Rapid successive transactions
  1100→    if (history.minutesSinceLastTxn < 1 && history.lastHourCount > 3) {
  1101→      score += 20;
  1102→      indicators.push('rapid_succession');
  1103→    } else if (history.minutesSinceLastTxn < 2 && history.lastHourCount > 5) {
  1104→      score += 10;
  1105→      indicators.push('quick_succession');
  1106→    }
  1107→    metadata.minutesSinceLastTxn = history.minutesSinceLastTxn;
  1108→
  1109→    // Factor 5: New recipient ratio
  1110→    if (history.last24hCount > 3) {
  1111→      const newRecipientRatio = history.newRecipientCount24h / history.last24hCount;
  1112→      if (newRecipientRatio > SUSPICIOUS_PATTERNS.NEW_RECIPIENT_RATIO) {
  1113→        score += 15;
  1114→        indicators.push('many_new_recipients');
  1115→      }
  1116→      metadata.newRecipientRatio = Math.round(newRecipientRatio * 100) / 100;
  1117→    }
  1118→
  1119→    score = Math.min(score, 100);
  1120→
  1121→    return {
  1122→      factorId: 'velocity_risk',
  1123→      factorName: 'Velocity Analysis',
  1124→      score,
  1125→      maxScore: 100,
  1126→      rawValue: { 
  1127→        hourlyCount: history.lastHourCount, 
  1128→        dailyCount: history.last24hCount 
  1129→      },
  1130→      indicators,
  1131→      metadata,
  1132→    };
  1133→  }

  1135→  /**
  1136→   * Calculate device-based risk factors
  1137→   * Analyzes device fingerprint and integrity signals
  1138→   */
  1139→  private calculateDeviceRisk(
  1140→    transaction: TransactionInput,
  1141→    device?: DeviceIntelligence
  1142→  ): RiskFactorContribution {
  1143→    let score = 0;
  1144→    const indicators: string[] = [];
  1145→    const metadata: Record<string, unknown> = {};
  1146→
  1147→    // If no device data, apply moderate risk
  1148→    if (!device) {
  1149→      return {
  1150→        factorId: 'device_risk',
  1151→        factorName: 'Device Analysis',
  1152→        score: 30, // Default moderate risk without data
  1153→        maxScore: 100,
  1154→        rawValue: null,
  1155→        indicators: ['no_device_data'],
  1156→        metadata: { reason: 'No device intelligence provided' },
  1157→      };
  1158→    }
  1159→
  1160→    // Factor 1: Known device check
  1161→    if (!device.isKnownDevice) {
  1162→      score += 25;
  1163→      indicators.push('unknown_device');
  1164→    }
  1165→    metadata.isKnownDevice = device.isKnownDevice;
  1166→
  1167→    // Factor 2: Device trust score (inverse relationship)
  1168→    score += (100 - device.trustScore) * 0.3;
  1169→    if (device.trustScore < 30) {
  1170→      indicators.push('low_device_trust');
  1171→    } else if (device.trustScore < 60) {
  1172→      indicators.push('moderate_device_trust');
  1173→    }
  1174→    metadata.deviceTrustScore = device.trustScore;
  1175→
  1176→    // Factor 3: New device (age check)
  1177→    if (device.deviceAgeDays < 1) {
  1178→      score += 20;
  1179→      indicators.push('brand_new_device');
  1180→    } else if (device.deviceAgeDays < 7) {
  1181→      score += 10;
  1182→      indicators.push('recent_device');
  1183→    }
  1184→    metadata.deviceAgeDays = device.deviceAgeDays;
  1185→
  1186→    // Factor 4: Emulator/virtual environment
  1187→    if (device.isEmulator) {
  1188→      score += 40;
  1189→      indicators.push('emulator_detected');
  1190→    }
  1191→    metadata.isEmulator = device.isEmulator;
  1192→
  1193→    // Factor 5: Root/jailbreak detection
  1194→    if (device.isRooted) {
  1195→      score += 35;
  1196→      indicators.push('root_detected');
  1197→    }
  1198→    metadata.isRooted = device.isRooted;
  1199→
  1200→    // Factor 6: VPN/Proxy detection
  1201→    if (device.isVpnOrProxy) {
  1202→      score += 15;
  1203→      indicators.push('vpn_proxy_detected');
  1204→    }
  1205→    metadata.isVpnOrProxy = device.isVpnOrProxy;
  1206→
  1207→    // Factor 7: Tor network
  1208→    if (device.isTor) {
  1209→      score += 50;
  1210→      indicators.push('tor_network');
  1211→    }
  1212→    metadata.isTor = device.isTor;
  1213→
  1214→    // Factor 8: Integrity score
  1215→    score += (100 - device.integrityScore) * 0.2;
  1216→    if (device.integrityScore < 50) {
  1217→      indicators.push('low_integrity_score');
  1218→    }
  1219→    metadata.integrityScore = device.integrityScore;
  1220→
  1221→    score = Math.min(score, 100);
  1222→
  1223→    return {
  1224→      factorId: 'device_risk',
  1225→      factorName: 'Device Analysis',
  1226→      score: Math.round(score),
  1227→      maxScore: 100,
  1228→      rawValue: device,
  1229→      indicators,
  1230→      metadata,
  1231→    };
  1232→  }

  1234→  /**
  1235→   * Calculate geographic/location-based risk factors
  1236→   * Analyzes IP geolocation and cross-border patterns
  1237→   */
  1238→  private calculateGeographicRisk(
  1239→    transaction: TransactionInput,
  1240→    geoLocation?: GeoLocationData,
  1241→    customer?: CustomerProfile
  1242→  ): RiskFactorContribution {
  1243→    let score = 0;
  1244→    const indicators: string[] = [];
  1245→    const metadata: Record<string, unknown> = {};
  1246→
  1247→    // If no geo data, use basic country from transaction
  1248→    const effectiveGeo: GeoLocationData = geoLocation || {
  1249→      countryCode: transaction.countryCode,
  1250→      matchesAccountCountry: true,
  1251→      isHighRiskCountry: HIGH_RISK_COUNTRIES.has(transaction.countryCode),
  1252→      isSanctionedCountry: SANCTIONED_COUNTRIES.has(transaction.countryCode),
  1253→      countryRiskScore: HIGH_RISK_COUNTRIES.has(transaction.countryCode) ? 80 : 20,
  1254→      timezoneConsistent: true,
  1255→      connectionType: 'unknown',
  1256→    };
  1257→
  1258→    // Factor 1: High-risk country
  1259→    if (effectiveGeo.isHighRiskCountry) {
  1260→      score += 35;
  1261→      indicators.push('high_risk_country');
  1262→    }
  1263→    metadata.isHighRiskCountry = effectiveGeo.isHighRiskCountry;
  1264→
  1265→    // Factor 2: Sanctioned country (should be caught in pre-check too)
  1266→    if (effectiveGeo.isSanctionedCountry) {
  1267→      score += 100;
  1268→      indicators.push('sanctioned_country');
  1269→    }
  1270→    metadata.isSanctionedCountry = effectiveGeo.isSanctionedCountry;
  1271→
  1272→    // Factor 3: Country mismatch
  1273→    if (!effectiveGeo.matchesAccountCountry) {
  1274→      score += 25;
  1275→      indicators.push('country_mismatch');
  1276→    }
  1277→    metadata.matchesAccountCountry = effectiveGeo.matchesAccountCountry;
  1278→
  1279→    // Factor 4: Country risk score
  1280→    score += effectiveGeo.countryRiskScore * 0.3;
  1281→    metadata.countryRiskScore = effectiveGeo.countryRiskScore;
  1282→
  1283→    // Factor 5: Distance from home location
  1284→    if (effectiveGeo.distanceFromHome !== undefined) {
  1285→      if (effectiveGeo.distanceFromHome > 5000) {
  1286→        // > 5000 km
  1287→        score += 25;

  1319→
  1320→    score = Math.min(score, 100);

  1333→  /**
  1334→   * Calculate behavioral/pattern-based risk factors
  1335→   * Analyzes transaction patterns and anomalies
  1336→   */
  1337→  private calculateBehavioralRisk(
  1338→    transaction: TransactionInput,
  1339→    history: TransactionHistory,
  1340→    customer: CustomerProfile
  1341→  ): RiskFactorContribution {
  1342→    let score = 0;
  1343→    const indicators: string[] = [];
  1344→    const metadata: Record<string, unknown> = {};
  1345→
  1346→    // Factor 1: Time-based anomaly (transactions at unusual hours)
  1347→    const hour = transaction.timestamp.getHours();
  1348→    const isUnusualHour = hour >= 0 && hour < 5; // Midnight to 5 AM
  1349→    if (isUnusualHour && transaction.amount > 50000) {
  1350→      score += 15;
  1351→      indicators.push('unusual_hour_large_amount');
  1352→    }
  1353→    metadata.transactionHour = hour;
  1354→
  1355→    // Factor 2: Weekend/holiday large transaction
  1356→    const dayOfWeek = transaction.timestamp.getDay();
  1357→    const isWeekend = dayOfWeek === 0 || dayOfWeek === 6;
  1358→    if (isWeekend && transaction.amount > 100000) {
  1359→      score += 10;
  1360→      indicators.push('weekend_large_transaction');
  1361→    }
  1362→    metadata.dayOfWeek = dayOfWeek;
  1363→
  1364→    // Factor 3: First transaction after long silence
  1365→    if (history.minutesSinceLastTxn > 1440) { // > 24 hours
  1366→      if (customer.totalTransactions > 10) {
  1367→        // Not a new customer
  1368→        score += 10;
  1369→        indicators.push('return_after_silence');
  1370→      }
  1371→    } else if (history.minutesSinceLastTxn > 10080) { // > 7 days
  1372→      score += 15;
  1373→      indicators.push('return_after_long_absence');
  1374→    }
  1375→    metadata.silenceMinutes = history.minutesSinceLastTxn;
  1376→
  1377→    // Factor 4: Spending pattern break
  1378→    if (history.avgAmount30d > 0 && history.last30dCount >= 5) {
  1379→      const ratio = (transaction.amount / 100) / (history.avgAmount30d / 100);
  1380→      if (ratio > 10) {
  1381→        score += 25;
  1382→        indicators.push('spending_pattern_break');
  1383→      } else if (ratio > 5) {
  1384→        score += 12;
  1385→        indicators.push('elevated_spending');
  1386→      }
  1387→      metadata.spendingRatio = Math.round(ratio * 100) / 100;
  1388→    }
  1389→
  1390→    // Factor 5: New account rapid activity
  1391→    const accountAgeDays = Math.floor(
  1392→      (Date.now() - customer.accountCreatedDate.getTime()) / (1000 * 60 * 60 * 24)
  1393→    );
  1394→    if (accountAgeDays < 7 && history.last24hCount > 5) {
  1395→      score += 25;
  1396→      indicators.push('new_account_rapid_activity');
  1397→    } else if (accountAgeDays < 30 && history.last24hCount > 20) {
  1398→      score += 15;
  1399→      indicators.push('young_account_high_activity');
  1400→    }
  1401→    metadata.accountAgeDays = accountAgeDays;
  1402→
  1403→    // Factor 6: High-risk merchant category
  1404→    if (transaction.merchantCategoryCode && HIGH_RISK_MCCS.has(transaction.merchantCategoryCode)) {
  1405→      score += 20;
  1406→      indicators.push('high_risk_mcc');
  1407→    }
  1408→    metadata.mcc = transaction.merchantCategoryCode;
  1409→
  1410→    // Factor 7: Self-transfer detection (source == destination)
  1411→    if (transaction.sourceAccountId === transaction.destinationAccountId) {
  1412→      score += 30;
  1413→      indicators.push('self_transfer_suspicious');
  1414→    }
  1415→
  1416→    score = Math.min(score, 100);
  1417→
  1418→    return {
  1419→      factorId: 'behavioral_risk',
  1420→      factorName: 'Behavioral Pattern Analysis',
  1421→      score: Math.round(score),
  1422→      maxScore: 100,
  1423→      rawValue: { hour, dayOfWeek, accountAgeDays },
  1424→      indicators,
  1425→      metadata,
  1426→    };
  1427→  }

  1665→  /**
  1666→   * Perform Anti-Money Laundering checks on transaction
  1667→   * 
  1668→   * @param transaction - Transaction to check
  1669→   * @param history - Transaction history for pattern analysis
  1670→   * @returns Array of triggered AML alerts
  1671→   */
  1672→  performAMLChecks(
  1673→    transaction: TransactionInput,
  1674→    history: TransactionHistory
  1675→  ): AMLAlert[] {
  1676→    const alerts: AMLAlert[] = [];
  1677→    const now = new Date();
  1678→
  1679→    // Check 1: Large transaction reporting threshold (e.g., $10,000)
  1680→    const largeTxnThreshold = 1000000; // $10,000 in cents
  1681→    if (transaction.amount >= largeTxnThreshold) {
  1682→      alerts.push({
  1683→        type: AMLAlertType.LARGE_TRANSACTION,
  1684→        severity: transaction.amount >= largeTxnThreshold * 2 ? 4 : 3,
  1685→        description: `Transaction amount ${transaction.amount / 100} exceeds reporting threshold`,
  1686→        threshold: largeTxnThreshold,
  1687→        actualValue: transaction.amount,
  1688→        sarRecommended: transaction.amount >= largeTxnThreshold * 1.5,
  1689→        timestamp: now,
  1690→      });
  1691→    }
  1692→
  1693→    // Check 2: Structuring detection (multiple transactions just below threshold)
  1694→    if (history.last24hVolume > SUSPICIOUS_PATTERNS.STRUCTURING_THRESHOLD * 2 &&
  1695→        history.last24hCount >= 3) {
  1696→      const avg24h = history.last24hVolume / history.last24hCount;
  1697→      if (avg24h < SUSPICIOUS_PATTERNS.STRUCTURING_THRESHOLD &&
  1698→          history.last24hVolume > SUSPICIOUS_PATTERNS.STRUCTURING_THRESHOLD) {
  1699→        alerts.push({
  1700→          type: AMLAlertType.STRUCTURING,
  1701→          severity: 4,
  1702→          description: 'Potential structuring pattern detected - multiple transactions below reporting threshold',
  1703→          threshold: SUSPICIOUS_PATTERNS.STRUCTURING_THRESHOLD,
  1704→          actualValue: history.last24hVolume,
  1705→          sarRecommended: true,
  1706→          timestamp: now,
  1707→        });
  1708→      }
  1709→    }
  1710→
  1711→    // Check 3: High-risk country
  1712→    if (HIGH_RISK_COUNTRIES.has(transaction.countryCode)) {
  1713→      alerts.push({
  1714→        type: AMLAlertType.HIGH_RISK_COUNTRY,
  1715→        severity: SANCTIONED_COUNTRIES.has(transaction.countryCode) ? 5 : 3,
  1716→        description: `Transaction involving high-risk jurisdiction: ${transaction.countryCode}`,
  1717→        sarRecommended: SANCTIONED_COUNTRIES.has(transaction.countryCode),
  1718→        timestamp: now,
  1719→      });
  1720→    }
  1721→
  1722→    // Check 4: Velocity breach
  1723→    if (history.lastHourCount > SUSPICIOUS_PATTERNS.MAX_VELOCITY_PER_HOUR * 2) {
  1724→      alerts.push({
  1725→        type: AMLAlertType.VELOCITY_BREACH,
  1726→        severity: 3,
  1727→        description: `Excessive transaction velocity: ${history.lastHourCount} transactions per hour`,
  1728→        threshold: SUSPICIOUS_PATTERNS.MAX_VELOCITY_PER_HOUR * 2,
  1729→        actualValue: history.lastHourCount,
  1730→        sarRecommended: false,
  1731→        timestamp: now,
  1732→      });
  1733→    }
  1734→
  1735→    // Check 5: Unusual activity pattern
  1736→    const volumeRatio = history.last24hVolume / Math.max(history.last7dVolume, 1);
  1737→    if (volumeRatio > 0.8 && history.last7dCount > 10) {
  1738→      // Most of weekly volume in one day
  1739→      alerts.push({
  1740→        type: AMLAlertType.UNUSUAL_ACTIVITY,
  1741→        severity: 2,
  1742→        description: 'Unusual concentration of transaction volume within 24-hour period',
  1743→        actualValue: volumeRatio,
  1744→        sarRecommended: false,
  1745→        timestamp: now,
  1746→      });
  1747→    }
  1748→
  1749→    if (alerts.length > 0) {

  1752→        metadata: {
  1753→          transactionId: transaction.transactionId,
  1754→          alertCount: alerts.length,
  1755→          alertTypes: alerts.map(a => a.type),
  1756→        },
  1757→      });
  1758→    }
  1759→
  1760→    return alerts;
  1761→  }

  1765→  /**
  1766→   * Calculate weighted composite score from all factors
  1767→   */
  1768→  private calculateCompositeScore(factors: RiskFactorContribution[]): number {
  1769→    const weightEntries: Array<[keyof RiskFactorWeights, string]> = [
  1770→      ['amountWeight', 'amount_risk'],
  1771→      ['velocityWeight', 'velocity_risk'],
  1772→      ['deviceWeight', 'device_risk'],
  1773→      ['geographicWeight', 'geographic_risk'],
  1774→      ['behavioralWeight', 'behavioral_risk'],
  1775→      ['historicalWeight', 'historical_risk'],
  1776→      ['complianceWeight', 'compliance_risk'],
  1777→      ['customerProfileWeight', 'customer_profile_risk'],
  1778→    ];
  1779→
  1780→    let compositeScore = 0;
  1781→
  1782→    for (const [weightKey, factorId] of weightEntries) {
  1783→      const factor = factors.find(f => f.factorId === factorId);
  1784→      if (factor) {
  1785→        const weight = this.weights[weightKey];
  1786→        compositeScore += factor.score * weight;
  1787→      }
  1788→    }
  1789→
  1790→    return Math.min(100, Math.max(0, compositeScore));
  1791→  }

  1796→  private classifyRiskLevel(score: number): RiskLevel {
  1797→    if (score <= this.thresholds.lowMax) return RiskLevel.LOW;

  1803→  /**
  1804→   * Determine authentication requirement based on risk level
  1805→   */
  1806→  private determineAuthRequirement(
  1807→    riskLevel: RiskLevel,
  1808→    factors: RiskFactorContribution[],
  1809→    customer: CustomerProfile
  1810→  ): AuthRequirement {
  1811→    // Critical always blocks
  1812→    if (riskLevel === RiskLevel.CRITICAL) {
  1813→      return AuthRequirement.BLOCKED;
  1814→    }
  1815→
  1816→    // High risk requires strong auth
  1817→    if (riskLevel === RiskLevel.HIGH) {
  1818→      // Check for device-related risks
  1819→      const hasDeviceRisk = factors
  1820→        .find(f => f.factorId === 'device_risk' && f.score > 50);
  1821→      
  1822→      if (hasDeviceRisk) {
  1823→        return AuthRequirement.STEP_UP;
  1824→      }
  1825→      return AuthRequirement.BIOMETRIC;
  1826→    }
  1827→
  1828→    // Medium risk requires OTP
  1829→    if (riskLevel === RiskLevel.MEDIUM) {
  1830→      // New device or location adds friction
  1831→      const isNewDevice = factors
  1832→        .find(f => f.factorId === 'device_risk' && f.indicators.includes('unknown_device'));
  1833→      const isNewLocation = factors
  1834→        .find(f => f.factorId === 'geographic_risk' && f.indicators.includes('country_mismatch'));
  1835→      
  1836→      if (isNewDevice || isNewLocation) {
  1837→        return AuthRequirement.STEP_UP;
  1838→      }
  1839→      return AuthRequirement.OTP;
  1840→    }
  1841→
  1842→    // Low risk - check KYC for unverified users
  1843→    if (customer.kycStatus !== KYCStatus.VERIFIED && customer.kycStatus !== KYCStatus.PARTIAL) {
  1844→      return AuthRequirement.OTP;
  1845→    }
  1846→
  1847→    return AuthRequirement.NONE;
  1848→  }

  1850→  /**

  1853→  private shouldBlockTransaction(
  1854→    riskLevel: RiskLevel,
  1855→    amlAlerts: AMLAlert[],
  1856→    customer: CustomerProfile
