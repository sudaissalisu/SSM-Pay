   352→// ============== Statistical Analysis Functions ==============
   353→
   354→/**
   355→ * Calculate Z-score for amount anomaly detection
   356→ * Z-score measures how many standard deviations a value is from the mean
   357→ */
   358→export function calculateZScore(
   359→  value: number,
   360→  mean: number,
   361→  stdDev: number
   362→): number {
   363→  if (stdDev === 0) return 0;
   364→  return (value - mean) / stdDev;
   365→}
   366→
   367→/**
   368→ * Detect anomalies using modified Z-score method
   369→ * More robust to outliers than standard Z-score
   370→ */
   371→export function detectAnomaliesModifiedZScore(
   372→  values: number[],
   373→  threshold: number = 3.5
   374→): { index: number; value: number; zScore: number }[] {
   375→  if (values.length < 3) return [];
   376→
   377→  const median = values.slice().sort((a, b) => a - b)[Math.floor(values.length / 2)];
   378→  
   379→  const mad = values
   380→    .map(v => Math.abs(v - median))
   381→    .sort((a, b) => a - b)[Math.floor(values.length / 2)];
   382→
   383→  if (mad === 0) return [];
   384→
   385→  return values
   386→    .map((value, index) => ({
   387→      index,
   388→      value,
   389→      zScore: (0.6745 * (value - median)) / mad,
   390→    }))
   391→    .filter(({ zScore }) => Math.abs(zScore) > threshold);
   392→}

   394→/**
   395→ * Calculate percentile rank for a value in a distribution
   396→ */
   397→export function calculatePercentile(
   398→  value: number,
   399→  values: number[]
   400→): number {
   401→  if (values.length === 0) return 50;
   402→  const sorted = values.slice().sort((a, b) => a - b);
   403→  const rank = sorted.filter(v => v <= value).length;
   404→  return (rank / sorted.length) * 100;
   405→}

   407→/**
   408→ * Exponential moving average for real-time calculations
   409→ */
   410→export function calculateEMA(
   411→  currentValue: number,
   412→  previousEMA: number,
   413→  alpha: number = 0.3
   414→): number {
   415→  if (previousEMA === 0) return currentValue;
   416→  return alpha * currentValue + (1 - alpha) * previousEMA;
   417→}
   418→
   419→/**
   420→ * Calculate entropy of a dataset (measure of randomness)
   421→ */
   422→export function calculateEntropy(values: number[]): number {
   423→  if (values.length === 0) return 0;
   424→
   425→  const total = values.reduce((sum, v) => sum + Math.abs(v), 0);
   426→  if (total === 0) return 0;
   427→
   428→  const probabilities = values.map(v => Math.abs(v) / total);
   429→  return -probabilities.reduce((entropy, p) => {
   430→    if (p === 0) return entropy;
   431→    return entropy + p * Math.log2(p);
   432→  }, 0);
   433→}

   437→/**
   438→ * Enterprise Fraud Detection Engine
   439→ * 
   440→ * Provides real-time transaction risk scoring using:
   441→ * - Rule-based scoring engine
   442→ * - Statistical anomaly detection
   443→ * - Behavioral analysis

   485→  /**
   486→   * Analyze a transaction for fraud risk
   487→   * Main entry point for fraud detection
   488→   */
   489→  async analyzeTransaction(features: TransactionFeatures): Promise<FraudScore> {
   490→    const startTime = performance.now();
   491→    const decisionId = this.generateDecisionId();
   492→
   493→    logger.info('Analyzing transaction for fraud', {
   494→      event: 'fraud.analyze.start',
   495→      metadata: { decisionId, amount: features.amount, currency: features.currency },
   496→    });
   497→
   498→    try {
   499→      // Check cache first
   500→      const cacheKey = this.generateCacheKey(features);
   501→      const cached = this.cache.get(cacheKey);
   502→      if (cached) {
   503→        logger.debug('Returning cached fraud score', {
   504→          event: 'fraud.analyze.cache_hit',
   505→          metadata: { decisionId: cached.decisionId },
   506→        });
   507→        return cached;
   508→      }
   509→
   510→      // Run all analyses
   511→      const [ruleScores, anomalyResult] = await Promise.all([
   512→        this.evaluateRules(features),
   513→        this.detectAnomalies(features),
   514→      ]);
   515→
   516→      // Combine scores
   517→      const combinedFactors = this.combineFactors(ruleScores, anomalyResult, features);
   518→      const finalScore = this.calculateFinalScore(combinedFactors);
   519→
   520→      // Determine action
   521→      const result: FraudScore = {
   522→        score: finalScore,
   523→        level: this.classifyRiskLevel(finalScore),
   524→        shouldBlock: finalScore >= BLOCK_THRESHOLD || combinedFactors.some(f => 
   525→          FRAUD_RULES.find(r => r.id === f.triggeredRules[0])?.isBlocking
   526→        ),
   527→        requiresReview: finalScore >= REVIEW_THRESHOLD && finalScore < BLOCK_THRESHOLD,
   528→        factors: combinedFactors,
   529→        processingTimeMs: performance.now() - startTime,
   530→        modelVersion: MODEL_VERSION,
   531→        decisionId,
   532→      };
   533→
   534→      // Cache result
   535→      this.cache.set(cacheKey, result);
   536→      
   537→      // Schedule cache cleanup
   538→      setTimeout(() => this.cache.delete(cacheKey), this.cacheTTL);
   539→
   540→      // Send to analytics if callback provided
   541→      this.analyticsCallback?.(result);
   542→
   543→      logger.info('Fraud analysis complete', {
   544→        event: 'fraud.analyze.complete',
   545→        metadata: {
   546→          decisionId,
   547→          score: result.score,
   548→          level: result.level,
   549→          shouldBlock: result.shouldBlock,
   550→          processingTimeMs: result.processingTimeMs,
   551→        },
   552→      });
   553→
   554→      return result;
   555→    } catch (error) {
   556→      logger.appError(error instanceof Error ? error : new Error(String(error)), {
   557→        action: 'analyzeTransaction',
   558→        decisionId,
   559→      });
   560→      
   561→      // Return safe default on error
   562→      return {
   563→        score: MEDIUM_RISK_THRESHOLD,
   564→        level: 'medium',
   565→        shouldBlock: false,
   566→        requiresReview: true,
   567→        factors: [],
   568→        processingTimeMs: performance.now() - startTime,
   569→        modelVersion: MODEL_VERSION,
   570→        decisionId,
   571→      };
   572→    }
   573→  }

   575→  /**
   576→   * Evaluate all fraud rules against transaction features
   577→   */
   578→  private async evaluateRules(features: TransactionFeatures): Promise<RiskFactor[]> {
   579→    const rules = this.getAllRules();
   580→    const triggeredFactors: RiskFactor[] = [];
   581→
   582→    for (const rule of rules) {
   583→      try {
   584→        const isTriggered = rule.evaluate(features);
   585→        
   586→        if (isTriggered) {
   587→          triggeredFactors.push({
   588→            name: rule.name,
   589→            description: rule.description,
   590→            scoreContribution: rule.scoreImpact,
   591→            weight: rule.weight,
   592→            triggeredRules: [rule.id],
   593→          });
   594→
   595→          logger.debug(`Rule triggered: ${rule.id}`, {
   596→            event: 'fraud.rule.triggered',
   597→            metadata: { ruleId: rule.id, scoreImpact: rule.scoreImpact },
   598→          });
   599→        }
   600→      } catch (error) {
   601→        logger.warn(`Error evaluating rule ${rule.id}`, {
   602→          event: 'fraud.rule.error',
   603→          metadata: { ruleId: rule.id, error: String(error) },
   604→        });
   605→      }
   606→    }
   607→
   608→    return triggeredFactors;
   609→  }

   611→  /**
   612→   * Detect statistical anomalies in transaction
   613→   */
   614→  private async detectAnomalies(features: TransactionFeatures): Promise<AnomalyDetectionResult> {
   615→    const zScore = calculateZScore(
   616→      features.amount,
   617→      features.avgAmount30d,
   618→      features.stdAmount30d
   619→    );
   620→
   621→    const isAnomalous = Math.abs(zScore) > 2.5;
   622→    
   623→    let anomalyType: AnomalyDetectionResult['anomalyType'] = null;
   624→    let confidence = Math.min(Math.abs(zScore) / 5, 1);
   625→
   626→    if (isAnomalous) {
   627→      if (zScore > 2.5) {
   628→        anomalyType = 'amount_spike';
   629→      } else if (zScore < -2.5) {
   630→        anomalyType = 'amount_spike'; // Unusually small
   631→      }
   632→
   633→      if (features.isNewDevice) {
   634→        anomalyType = 'new_device';
   635→        confidence = Math.max(confidence, 0.8);
   636→      }
   637→
   638→      if (features.isNewCountry) {
   639→        anomalyType = 'new_location';
   640→        confidence = Math.max(confidence, 0.75);
   641→      }
   642→
   643→      if (features.txnCount1h > 5) {
   644→        anomalyType = 'velocity_burst';
   645→        confidence = Math.max(confidence, 0.9);
   646→      }
   647→    }
   648→
   649→    return {
   650→      amountZScore: zScore,
   651→      isAnomalous,
   652→      anomalyType,
   653→      confidence,
   654→    };
   655→  }
   656→
   657→  /**
   658→   * Combine rule-based scores with anomaly detection results
   659→   */
   660→  private combineFactors(
   661→    ruleFactors: RiskFactor[],
   662→    anomaly: AnomalyDetectionResult,
   663→    features: TransactionFeatures
   664→  ): RiskFactor[] {
   665→    const combined = [...ruleFactors];
   666→
   667→    // Add anomaly as a factor if significant
   668→    if (anomaly.isAnomalous) {
   669→      const anomalyScore = Math.min(Math.abs(anomaly.amountZScore) * 10, 50);
   670→      combined.push({
   671→        name: 'Statistical Anomaly',
   672→        description: `Transaction shows ${anomaly.anomalyType} pattern (z-score: ${anomaly.amountZScore.toFixed(2)})`,
   673→        scoreContribution: anomalyScore,
   674→        weight: 0.15,
   675→        triggeredRules: [`ANOMALY_${anomaly.anomalyType?.toUpperCase()}`],
   676→      });
   677→    }
   678→
   679→    // Add velocity factor if high
   680→    if (features.systemVelocity > 50) {
   681→      combined.push({
   682→        name: 'System Velocity Alert',
   683→        description: `Elevated system-wide transaction velocity (${features.systemVelocity} txn/min)`,
   684→        scoreContribution: Math.min(features.systemVelocity, 40),
   685→        weight: CATEGORY_WEIGHTS.velocity,
   686→        triggeredRules: ['SYS_VEL_001'],
   687→      });
   688→    }
   689→
   690→    return combined;
   691→  }
   692→
   693→  /**
   694→   * Calculate final weighted score from all factors
   695→   */
   696→  private calculateFinalScore(factors: RiskFactor[]): number {
   697→    if (factors.length === 0) return 5; // Base low risk score
   698→
   699→    const weightedSum = factors.reduce((sum, factor) => {
   700→      return sum + (factor.scoreContribution * factor.weight);
   701→    }, 0);
   702→
   703→    const maxPossibleScore = factors.reduce((sum, factor) => {
   704→      return sum + (100 * factor.weight);
   705→    }, 0);
   706→
   707→    // Normalize to 0-100 scale
   708→    const normalizedScore = maxPossibleScore > 0 
   709→      ? (weightedSum / maxPossibleScore) * 100 
   710→      : 0;
   711→
   712→    return Math.min(Math.round(normalizedScore), 100);
   713→  }

   715→  /**
   716→   * Classify risk level from numeric score
   717→   */
   718→  private classifyRiskLevel(score: number): FraudScore['level'] {
   719→    if (score >= BLOCK_THRESHOLD) return 'critical';
   720→    if (score >= REVIEW_THRESHOLD) return 'high';
   721→    if (score >= MEDIUM_RISK_THRESHOLD) return 'medium';
   722→    return 'low';
   723→  }

   811→  /**
   812→   * Generate cache key from features
   813→   */
   814→  private generateCacheKey(features: TransactionFeatures): string {
   815→    const keyData = `${features.amount}:${features.currency}:${features.customerHash}:${features.deviceId}:${features.timestamp.getTime()}`;
   816→    
   817→    // Simple hash function for cache key
   818→    let hash = 0;
   819→    for (let i = 0; i < keyData.length; i++) {
   820→      const char = keyData.charCodeAt(i);
   821→      hash = ((hash << 5) - hash) + char;
   822→      hash = hash & hash; // Convert to 32-bit integer
   823→    }
   824→    
   825→    return `fc_${Math.abs(hash).toString(36)}`;
   826→  }

   828→  /**
   829→   * Build or update user behavior profile
   830→   */
   831→  static buildUserProfile(
   832→    transactionHistory: { amount: number; timestamp: Date; countryCode: string; deviceId: string }[]
   833→  ): UserBehaviorProfile {
   834→    if (transactionHistory.length === 0) {
   835→      throw new AppError('Cannot build profile from empty history', ErrorCode.INVALID_INPUT);
   836→    }
   837→
   838→    const amounts = transactionHistory.map(t => t.amount).sort((a, b) => a - b);
   839→    const countries = [...new Set(transactionHistory.map(t => t.countryCode))];
   840→    const devices = [...new Set(transactionHistory.map(t => t.deviceId))];
   841→    
   842→    const now = new Date();
   843→    const accountAgeMs = now.getTime() - transactionHistory[0].timestamp.getTime();
   844→    const accountAgeDays = accountAgeMs / (1000 * 60 * 60 * 24);
   845→
   846→    // Calculate percentiles
   847→    const getPercentile = (p: number) => amounts[Math.floor(amounts.length * p / 100)] || 0;
   848→
   849→    return {
   850→      userId: 'derived_from_history',
   851→      amountPercentiles: {
   852→        p25: getPercentile(25),
   853→        p50: getPercentile(50),
   854→        p75: getPercentile(75),
   855→        p90: getPercentile(90),
   856→        p99: getPercentile(99),
   857→      },
   858→      activeHours: this.calculateActiveHours(transactionHistory),
   859→      typicalCountries: countries,
   860→      knownDevices: devices,
   861→      accountAgeDays,
   862→      historicalFraudFlags: 0, // Would come from DB
   863→      trustScore: this.calculateTrustScore(accountAgeDays, devices.length, countries.length),
   864→    };
   865→  }
   866→
   867→  /**
   868→   * Calculate typical active hours from transaction history
   869→   */
   870→  private static calculateActiveHours(
   871→    history: { timestamp: Date }[]
   872→  ): number[] {
   873→    const hourCounts = new Array(24).fill(0);
   874→    
   875→    for (const tx of history) {
   876→      hourCounts[tx.timestamp.getHours()]++;
   877→    }
   878→
   879→    const avgCount = hourCounts.reduce((a, b) => a + b, 0) / 24;
   880→    
   881→    return hourCounts
   882→      .map((count, hour) => ({ hour, count }))
   883→      .filter(({ count }) => count > avgCount)
   884→      .map(({ hour }) => hour);
   885→  }
   886→
   887→  /**
   888→   * Calculate trust score based on account characteristics
   889→   */
   890→  private static calculateTrustScore(
   891→    accountAgeDays: number,
   892→    deviceCount: number,
   893→    countryCount: number
   894→  ): number {
   895→    let score = 50; // Base score
   896→
   897→    // Age bonus (up to 25 points)
   898→    score += Math.min(accountAgeDays / 30 * 5, 25);
   899→
   900→    // Device diversity penalty (too many devices is suspicious)
   901→    if (deviceCount > 5) {
   902→      score -= (deviceCount - 5) * 5;
   903→    }
   904→
   905→    // Country diversity penalty
   906→    if (countryCount > 3) {
   907→      score -= (countryCount - 3) * 10;
   908→    }
   909→
   910→    return Math.max(0, Math.min(100, score));
   911→  }
