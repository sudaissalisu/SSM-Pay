   523→// ============== Main Analytics Class ==============
   524→
   525→/**
   526→ * BehavioralAnalyticsEngine - Core class for analyzing user behavior patterns
   527→ *
   528→ * @example
   529→ * ```typescript
   530→ * const engine = new BehavioralAnalyticsEngine();
   531→ * await engine.initialize();
   532→ *
   533→ * // Track an event
   534→ * await engine.trackEvent({
   535→ *   id: 'evt_123',
   536→ *   userId: 'user_456',
   537→ *   sessionId: 'sess_789',
   538→ *   eventType: 'page_view',
   539→ *   page: '/dashboard',
   540→ *   timestamp: new Date(),
   541→ *   metadata: {},
   542→ *   deviceInfo: { ... }
   543→ * });
   544→ *
   545→ * // Get user profile
   546→ * const profile = await engine.getUserProfile('user_456');
   547→ * ```
   548→ */
   549→export class BehavioralAnalyticsEngine {
   550→  private config: AnalyticsConfig;
   551→  private eventStore: Map<string, BehaviorEvent[]>;
   552→  private sessionStore: Map<string, SessionData>;
   553→  private profileCache: Map<string, { profile: UserProfile; cachedAt: Date }>;
   554→  private initialized: boolean;
   555→
   556→  constructor(config: Partial<AnalyticsConfig> = {}) {
   557→    this.config = { ...DEFAULT_CONFIG, ...config };
   558→    this.eventStore = new Map();
   559→    this.sessionStore = new Map();
   560→    this.profileCache = new Map();
   561→    this.initialized = false;
   562→
   563→    logger.info('BehavioralAnalyticsEngine instance created', {
   564→      event: 'behavioral_analytics.init',
   565→      metadata: { config: this.config },
   566→    });
   567→  }
   568→
   569→  /**
   570→   * Initialize the analytics engine
   571→   * Sets up internal stores and validates configuration
   572→   */
   573→  async initialize(): Promise<void> {
   574→    try {
   575→      // Validate configuration
   576→      this.validateConfig();
   577→
   578→      // Initialize stores
   579→      this.eventStore = new Map();
   580→      this.sessionStore = new Map();
   581→      this.profileCache = new Map();
   582→
   583→      this.initialized = true;
   584→
   585→      logger.info('BehavioralAnalyticsEngine initialized successfully', {
   586→        event: 'behavioral_analytics.ready',
   587→        metadata: { config: this.config },
   588→      });
   589→    } catch (error) {
   590→      const appError = error instanceof AppError ? error : new AppError(
   591→        'Failed to initialize BehavioralAnalyticsEngine',
   592→        ErrorCode.UNKNOWN_ERROR,
   593→        { cause: error instanceof Error ? error : undefined }
   594→      );
   595→      logger.error('BehavioralAnalyticsEngine initialization failed', {
   596→        event: 'behavioral_analytics.init_error',
   597→        error: appError,
   598→      });
   599→      throw appError;
   600→    }
   601→  }
   602→
   603→  /**
   604→   * Validate engine configuration
   605→   * @throws AppError if configuration is invalid
   606→   */
   607→  private validateConfig(): void {
   608→    const { engagementWeights, segmentationThresholds } = this.config;
   609→
   610→    // Validate engagement weights sum to reasonable range
   611→    const weightSum = Object.values(engagementWeights).reduce((a, b) => a + b, 0);
   612→    if (weightSum <= 0 || weightSum > 20) {
   613→      throw new AppError(
   614→        'Invalid engagement weights configuration',
   615→        ErrorCode.INVALID_CONFIG,
   616→        { context: { weightSum } }
   617→      );
   618→    }
   619→
   620→    // Validate segmentation thresholds are logical
   621→    if (segmentationThresholds.atRiskDaysInactive >= segmentationThresholds.churnedDaysInactive) {
   622→      throw new AppError(
   623→        'At-risk threshold must be less than churned threshold',
   624→        ErrorCode.INVALID_CONFIG,
   625→        { context: segmentationThresholds }
   626→      );
   627→    }
   628→  }
   629→
   630→  /**
   631→   * Track a new behavior event
   632→   * @param event - The behavior event to track
   633→   * @throws AppError if engine not initialized or event invalid
   634→   */
   635→  async trackEvent(event: BehaviorEvent): Promise<void> {
   636→    this.ensureInitialized();
   637→    this.validateEvent(event);
   638→
   639→    try {
   640→      // Store event by user
   641→      const userEvents = this.eventStore.get(event.userId) || [];
   642→      userEvents.push(event);
   643→      this.eventStore.set(event.userId, userEvents);
   644→
   645→      // Update or create session
   646→      this.updateSession(event);
   647→
   648→      // Invalidate profile cache for this user
   649→      this.profileCache.delete(event.userId);
   650→
   651→      logger.debug('Event tracked successfully', {
   652→        event: 'behavioral_analytics.event_tracked',
   653→        metadata: {
   654→          eventId: event.id,
   655→          userId: event.userId,
   656→          eventType: event.eventType,
   657→          page: event.page,
   658→        },
   659→      });
   660→    } catch (error) {
   661→      const appError = error instanceof AppError ? error : new AppError(
   662→        'Failed to track event',
   663→        ErrorCode.UNKNOWN_ERROR,
   664→        { cause: error instanceof Error ? error : undefined }
   665→      );
   666→      logger.error('Event tracking failed', {
   667→        event: 'behavioral_analytics.track_error',
   668→        error: appError,
   669→        metadata: { eventId: event.id },
   670→      });
   671→      throw appError;
   672→    }
   673→  }
   674→
   675→  /**
   676→   * Track multiple events in batch
   677→   * @param events - Array of behavior events to track
   678→   */
   679→  async trackEventsBatch(events: BehaviorEvent[]): Promise<void> {
   680→    this.ensureInitialized();
   681→
   682→    if (!events.length) {
   683→      return;
   684→    }
   685→
   686→    for (const event of events) {
   687→      await this.trackEvent(event);
   688→    }
   689→
   690→    logger.info(`Batch tracking completed: ${events.length} events`, {
   691→      event: 'behavioral_analytics.batch_tracked',
   692→      metadata: { eventCount: events.length },
   693→    });
   694→  }
   695→
   696→  /**
   697→   * Validate event structure and required fields
   698→   * @param event - Event to validate
   699→   * @throws AppError if validation fails
   700→   */
   701→  private validateEvent(event: BehaviorEvent): void {
   702→    if (!event.id || typeof event.id !== 'string') {
   703→      throw new AppError('Event ID is required and must be a string', ErrorCode.VALIDATION_ERROR);
   704→    }
   705→    if (!event.userId || typeof event.userId !== 'string') {
   706→      throw new AppError('User ID is required and must be a string', ErrorCode.VALIDATION_ERROR);
   707→    }
   708→    if (!event.sessionId || typeof event.sessionId !== 'string') {
   709→      throw new AppError('Session ID is required and must be a string', ErrorCode.VALIDATION_ERROR);
   710→    }
   711→    if (!event.eventType || !this.isValidEventType(event.eventType)) {
   712→      throw new AppError(`Invalid event type: ${event.eventType}`, ErrorCode.VALIDATION_ERROR);
   713→    }
   714→    if (!event.page || typeof event.page !== 'string') {
   715→      throw new AppError('Page is required and must be a string', ErrorCode.VALIDATION_ERROR);
   716→    }
   717→    if (!(event.timestamp instanceof Date) || isNaN(event.timestamp.getTime())) {
   718→      throw new AppError('Timestamp must be a valid Date object', ErrorCode.VALIDATION_ERROR);
   719→    }
   720→    if (!event.deviceInfo || !event.deviceInfo.deviceId) {
   721→      throw new AppError('Device info with device ID is required', ErrorCode.VALIDATION_ERROR);
   722→    }
   723→  }

   732→      'payment_init', 'payment_complete', 'payment_failed', 'navigation', 'search',

   739→  /**
   740→   * Update session data with new event
   741→   * @param event - New event to incorporate into session
   742→   */
   743→  private updateSession(event: BehaviorEvent): void {
   744→    let session = this.sessionStore.get(event.sessionId);
   745→
   746→    if (!session) {
   747→      // Create new session
   748→      session = this.createSession(event);
   749→      this.sessionStore.set(event.sessionId, session);
   750→    } else {
   751→      // Update existing session
   752→      session.events.push(event);
   753→      
   754→      if (!session.pagesVisited.includes(event.page)) {
   755→        session.pagesVisited.push(event.page);
   756→      }
   757→
   758→      // Count interactions (non-page_view events)
   759→      if (event.eventType !== 'page_view') {
   760→        session.interactionCount++;
   761→      } else {
   762→        session.pageViewCount++;
   763→      }
   764→
   765→      // Check for conversion events
   766→      if (this.isConversionEvent(event.eventType)) {
   767→        session.converted = true;
   768→        session.conversionType = this.getConversionType(event.eventType);
   769→      }
   770→
   771→      // Update end time
   772→      session.endTime = event.timestamp;
   773→      session.durationSeconds = Math.floor(
   774→        (event.timestamp.getTime() - session.startTime.getTime()) / 1000
   775→      );
   776→
   777→      // Update bounce status
   778→      session.isBounce = session.pageViewCount <= 1 && session.interactionCount === 0;
   779→
   780→      // Update exit page
   781→      session.exitPage = event.page;
   782→    }
   783→  }

  1006→  private calculatePageFrequencies(events: BehaviorEvent[]): PageFrequency[] {
  1007→    const pageCounts = new Map<string, number>();
  1008→    
  1009→    for (const event of events) {
  1010→      if (event.eventType === 'page_view') {
  1011→        pageCounts.set(event.page, (pageCounts.get(event.page) || 0) + 1);
  1012→      }
  1013→    }
  1014→
  1015→    const totalPages = Array.from(pageCounts.values()).reduce((a, b) => a + b, 0);
  1016→    
  1017→    return Array.from(pageCounts.entries())
  1018→      .map(([page, count]) => ({
  1019→        page,
  1020→        count,
  1021→        percentage: (count / totalPages) * 100,
  1022→      }))
  1023→      .sort((a, b) => b.count - a.count)
  1024→      .slice(0, 10); // Top 10 pages
  1025→  }

  1085→  /**
  1086→   * Calculate composite behavioral scores
  1087→   * @param events - Sorted array of events
  1088→   * @param sessions - User sessions
  1089→   * @returns Object containing all behavioral scores
  1090→   */
  1091→  private calculateBehaviorScores(
  1092→    events: BehaviorEvent[],
  1093→    sessions: SessionData[]
  1094→  ): BehaviorScores {
  1095→    const weights = this.config.engagementWeights;
  1096→
  1097→    // Engagement Score Calculation
  1098→    let engagementScore = 0;
  1099→    const eventCounts = new Map<EventType, number>();
  1100→    
  1101→    for (const event of events) {
  1102→      eventCounts.set(event.eventType, (eventCounts.get(event.eventType) || 0) + 1);
  1103→    }
  1104→
  1105→    engagementScore += (eventCounts.get('page_view') || 0) * weights.pageViewWeight;
  1106→    engagementScore += (eventCounts.get('click') || 0) * weights.clickWeight;
  1107→    engagementScore += ((eventCounts.get('form_submit') || 0) + (eventCounts.get('form_start') || 0)) * weights.formWeight;
  1108→    engagementScore += ((eventCounts.get('payment_init') || 0) + (eventCounts.get('payment_complete') || 0)) * weights.paymentWeight;
  1109→    engagementScore += (eventCounts.get('navigation') || 0) * weights.navigationWeight;
  1110→    engagementScore += (eventCounts.get('search') || 0) * weights.searchWeight;
  1111→    engagementScore += (eventCounts.get('item_select') || 0) * weights.featureAdoptionWeight;
  1112→
  1113→    // Normalize to 0-100 scale (assuming ~50 weighted events as baseline for 100)
  1114→    engagementScore = Math.min(100, Math.round((engagementScore / 50) * 100));
  1115→
  1116→    // Consistency Score - measures regularity of usage patterns
  1117→    const consistencyScore = this.calculateConsistencyScore(sessions);
  1118→
  1119→    // Conversion Propensity - based on past conversion behavior
  1120→    const conversionPropensity = this.calculateConversionPropensity(sessions);
  1121→
  1122→    // Exploration Score - variety of features/pages explored
  1123→    const uniquePages = new Set(events.filter(e => e.eventType === 'page_view').map(e => e.page)).size;
  1124→    const explorationScore = Math.min(100, Math.round((uniquePages / 15) * 100)); // 15+ pages = full score
  1125→
  1126→    // Loyalty Score - based on retention and frequency
  1127→    const loyaltyScore = this.calculateLoyaltyScore(events, sessions);
  1128→
  1129→    // Trust Score - based on behavior consistency and risk indicators
  1130→    const trustScore = this.calculateTrustScore(events, consistencyScore);
  1131→
  1132→    return {
  1133→      engagementScore,
  1134→      consistencyScore,
  1135→      conversionPropensity,
  1136→      explorationScore,
  1137→      loyaltyScore,
  1138→      trustScore,
  1139→    };
  1140→  }

  1142→  /**
  1143→   * Calculate consistency score based on session regularity
  1144→   * @param sessions - User sessions
  1145→   * @returns Consistency score 0-100
  1146→   */
  1147→  private calculateConsistencyScore(sessions: SessionData[]): number {
  1148→    if (sessions.length < 2) {
  1149→      return sessions.length === 1 ? 50 : 0;
  1150→    }
  1151→
  1152→    // Calculate variance in session durations
  1153→    const durations = sessions.map(s => s.durationSeconds).filter(d => d > 0);
  1154→    if (durations.length < 2) return 50;
  1155→
  1156→    const meanDuration = durations.reduce((a, b) => a + b, 0) / durations.length;
  1157→    const variance = durations.reduce((sum, d) => sum + Math.pow(d - meanDuration, 2), 0) / durations.length;
  1158→    const stdDev = Math.sqrt(variance);
  1159→
  1160→    // Lower variance = higher consistency (inverse relationship)
  1161→    // Normalize assuming stdDev of 300+ seconds is low consistency
  1162→    const consistencyScore = Math.max(0, Math.min(100, Math.round(100 - (stdDev / 3))));
  1163→    
  1164→    return consistencyScore;
  1165→  }

  1167→  /**
  1168→   * Calculate conversion propensity based on historical data
  1169→   * @param sessions - User sessions
  1170→   * @returns Conversion propensity score 0-100
  1171→   */
  1172→  private calculateConversionPropensity(sessions: SessionData[]): number {
  1173→    if (sessions.length === 0) return 0;
  1174→
  1175→    const convertedSessions = sessions.filter(s => s.converted).length;
  1176→    const baseRate = (convertedSessions / sessions.length) * 100;
  1177→
  1178→    // Factor in recent conversion velocity
  1179→    const recentSessions = sessions.slice(-10); // Last 10 sessions
  1180→    const recentConversionRate = recentSessions.length > 0
  1181→      ? (recentSessions.filter(s => s.converted).length / recentSessions.length) * 100
  1182→      : 0;
  1183→
  1184→    // Weight recent conversions more heavily
  1185→    const weightedScore = Math.round(baseRate * 0.4 + recentConversionRate * 0.6);
  1186→    
  1187→    return Math.min(100, weightedScore);
  1188→  }

  1190→  /**
  1191→   * Calculate loyalty score based on usage patterns
  1192→   * @param events - User events
  1193→   * @param sessions - User sessions
  1194→   * @returns Loyalty score 0-100
  1195→   */
  1196→  private calculateLoyaltyScore(events: BehaviorEvent[], sessions: SessionData[]): number {
  1197→    if (events.length === 0) return 0;
  1198→
  1199→    const firstEvent = events[0].timestamp;
  1200→    const lastEvent = events[events.length - 1].timestamp;
  1201→    const daysActive = (lastEvent.getTime() - firstEvent.getTime()) / (1000 * 60 * 60 * 24);
  1202→
  1203→    if (daysActive < 1) return 30; // New user baseline
  1204→
  1205→    // Frequency component
  1206→    const frequencyScore = Math.min(100, (sessions.length / daysActive) * 20); // 5+ sessions/day = 100
  1207→
  1208→    // Recency component (more recent activity = higher loyalty indicator)
  1209→    const daysSinceLastActivity = (Date.now() - lastEvent.getTime()) / (1000 * 60 * 60 * 24);
  1210→    const recencyScore = Math.max(0, Math.min(100, 100 - (daysSinceLastActivity * 3))); // Lose 3 pts/day
  1211→
  1212→    // Duration component (longer tenure = higher loyalty)
  1213→    const tenureScore = Math.min(100, daysActive * 2); // 50+ days = 100
  1214→
  1215→    return Math.round(frequencyScore * 0.35 + recencyScore * 0.35 + tenureScore * 0.3);
  1216→  }

  1218→  /**
  1219→   * Calculate trust score based on behavior patterns
  1220→   * @param events - User events
  1221→   * @param consistencyScore - Pre-calculated consistency score
  1222→   * @returns Trust score 0-100
  1223→   */
  1224→  private calculateTrustScore(events: BehaviorEvent[], consistencyScore: number): number {
  1225→    let trustScore = 50; // Start at neutral
  1226→
  1227→    // Consistency contributes positively
  1228→    trustScore += (consistencyScore - 50) * 0.3;
  1229→
  1230→    // Check for suspicious patterns
  1231→    const rapidActions = this.detectRapidActionPattern(events);
  1232→    if (rapidActions.detected) {
  1233→      trustScore -= 20;
  1234→    }
  1235→
  1236→    // Check for unusual timing patterns
  1237→    const unusualTiming = this.detectUnusualTiming(events);
  1238→    if (unusualTiming) {
  1239→      trustScore -= 10;
  1240→    }
  1241→
  1242→    // Positive signals
  1243→    const hasVerificationEvents = events.some(e => 
  1244→      e.eventType === 'payment_complete' || e.eventType === 'kyc_submitted' as EventType
  1245→    );
  1246→    if (hasVerificationEvents) {
  1247→      trustScore += 15;
  1248→    }
  1249→
  1250→    // Regular login pattern
  1251→    const loginEvents = events.filter(e => e.eventType === 'login');
  1252→    if (loginEvents.length >= 3) {
  1253→      trustScore += 10;
  1254→    }
  1255→
  1256→    return Math.max(0, Math.min(100, Math.round(trustScore)));
  1257→  }

  1259→  /**
  1260→   * Detect rapid/suspicious action patterns
  1261→   * @param events - Events to analyze
  1262→   * @returns Detection result
  1263→   */
  1264→  private detectRapidActionPattern(events: BehaviorEvent[]): { detected: boolean; details?: string } {
  1265→    if (events.length < 5) return { detected: false };
  1266→
  1267→    // Check for multiple rapid-fire actions (< 1 second apart)
  1268→    let rapidCount = 0;
  1269→    for (let i = 1; i < events.length; i++) {
  1270→      const timeDiff = events[i].timestamp.getTime() - events[i - 1].timestamp.getTime();
  1271→      if (timeDiff < 1000) { // Less than 1 second
  1272→        rapidCount++;
  1273→      }
  1274→    }
  1275→

  1372→  /**
  1373→   * Calculate risk score for fraud/security purposes
  1374→   * @param events - User events
  1375→   * @param scores - Behavioral scores
  1376→   * @returns Risk score 0-100 (higher = more risky)
  1377→   */
  1378→  private calculateRiskScore(events: BehaviorEvent[], scores: BehaviorScores): number {
  1379→    let riskScore = 0;
  1380→
  1381→    // Low trust increases risk
  1382→    riskScore += (100 - scores.trustScore) * 0.4;
  1383→
  1384→    // Inconsistent behavior increases risk
  1385→    riskScore += (100 - scores.consistencyScore) * 0.2;
  1386→
  1387→    // Rapid action detection
  1388→    const rapidActions = this.detectRapidActionPattern(events);
  1389→    if (rapidActions.detected) {
  1390→      riskScore += 25;
  1391→    }
  1392→
  1393→    // Multiple failed payments
  1394→    const failedPayments = events.filter(e => e.eventType === 'payment_failed').length;
  1395→    riskScore += Math.min(20, failedPayments * 5);
  1396→
  1397→    // Device diversity (using many devices can be risky)
  1398→    const devices = new Set(events.map(e => e.deviceInfo.deviceId));
  1399→    if (devices.size > 5) {
  1400→      riskScore += 15;
  1401→    }
  1402→
  1403→    // Geographic anomalies (if geo data available)
  1404→    const geoAnomaly = this.detectGeoAnomalies(events);
  1405→    if (geoAnomaly) {
  1406→      riskScore += 20;
  1407→    }
  1408→
  1409→    return Math.min(100, Math.round(riskScore));
  1410→  }

  1441→  /**
  1442→   * Calculate churn probability using behavioral indicators
  1443→   * @param events - User events
  1444→   * @param scores - Behavioral scores
  1445→   * @param sessions - User sessions
  1446→   * @returns Churn probability 0-1
  1447→   */
  1448→  private calculateChurnProbability(
  1449→    events: BehaviorEvent[],
  1450→    scores: BehaviorScores,
  1451→    sessions: SessionData[]
  1452→  ): number {
  1453→    let churnProbability = 0;
  1454→
  1455→    // Recency factor (most important)
  1456→    const lastEvent = events[events.length - 1];
  1457→    const daysSinceLastActivity = (Date.now() - lastEvent.timestamp.getTime()) / (1000 * 60 * 60 * 24);
  1458→    
  1459→    if (daysSinceLastActivity > 30) {
  1460→      churnProbability += 0.5;
  1461→    } else if (daysSinceLastActivity > 14) {
  1462→      churnProbability += 0.3;
  1463→    } else if (daysSinceLastActivity > 7) {
  1464→      churnProbability += 0.15;
  1465→    }
  1466→
  1467→    // Engagement decline factor
  1468→    if (scores.engagementScore < 30) {
  1469→      churnProbability += 0.2;
  1470→    } else if (scores.engagementScore < 50) {
  1471→      churnProbability += 0.1;
  1472→    }
  1473→
  1474→    // Session frequency decline
  1475→    if (sessions.length >= 3) {
  1476→      const recentSessions = sessions.slice(-3);
  1477→      const olderSessions = sessions.slice(0, 3);
  1478→      const recentAvgDuration = recentSessions.reduce((s, sess) => s + sess.durationSeconds, 0) / 3;
  1479→      const olderAvgDuration = olderSessions.reduce((s, sess) => s + sess.durationSeconds, 0) / 3;
  1480→      
  1481→      if (olderAvgDuration > 0 && recentAvgDuration < olderAvgDuration * 0.5) {
  1482→        churnProbability += 0.15;
  1483→      }
  1484→    }
  1485→
  1486→    // Loyalty factor (inverse)
  1487→    churnProbability += (100 - scores.loyaltyScore) * 0.002;
  1488→
  1489→    return Math.min(1, Math.round(churnProbability * 100) / 100);
  1490→  }

  1493→   * Predict customer lifetime value
  1494→   * @param sessions - User sessions
  1495→   * @param scores - Behavioral scores
  1496→   * @returns Predicted LTV in base currency units
  1497→   */
  1498→  private predictLifetimeValue(sessions: SessionData[], scores: BehaviorScores): number {
  1499→    // Base LTV calculation using simplified model
  1500→    const avgTransactionValue = 150; // Base assumption
  1501→    
  1502→    // Adjust based on engagement and loyalty
  1503→    const engagementMultiplier = 0.5 + (scores.engagementScore / 100);
  1504→    const loyaltyMultiplier = 0.5 + (scores.loyaltyScore / 100);
  1505→    
  1506→    // Estimate transactions per month based on session frequency
  1507→    const monthlySessions = sessions.length / Math.max(1, this.getMonthsFromFirstSession(sessions));
  1508→    const estimatedMonthlyTxns = Math.min(monthlySessions * 0.3, 10); // Cap at 10/month
  1509→    
  1510→    // Project over 12 months
  1511→    const projectedMonths = 12 * loyaltyMultiplier;
  1512→    
  1513→    return Math.round(avgTransactionValue * estimatedMonthlyTxns * engagementMultiplier * projectedMonths);
  1514→  }

  1612→  private identifyNavigationPatterns(sessions: SessionData[]): NavigationPattern[] {
  1613→    const patternMap = new Map<string, {
  1614→      sequences: string[][];
  1615→      durations: number[];
  1616→      conversions: number;
  1617→      totalCount: number;
  1618→    }>();
  1619→
  1620→    for (const session of sessions) {
  1621→      if (session.pagesVisited.length < 2) continue;
  1622→
  1623→      // Extract page sequence (limit to 5 pages for pattern matching)
  1624→      const sequence = session.pagesVisited.slice(0, 5);
  1625→      const patternKey = sequence.join(' -> ');
  1626→
  1627→      const existing = patternMap.get(patternKey) || {
  1628→        sequences: [],
  1629→        durations: [],
  1630→        conversions: 0,
  1631→        totalCount: 0,
  1632→      };
  1633→
  1634→      existing.sequences.push(sequence);
  1635→      existing.durations.push(session.durationSeconds);
  1636→      if (session.converted) existing.conversions++;

  1638→

  1641→
  1642→    // Convert to NavigationPattern array

  1652→      // Generate pattern name from key
  1653→      const name = this.generatePatternName(key);
  1654→

  1657→        name,
  1658→        pageSequence: data.sequences[0],
  1659→        frequency: data.totalCount,
  1660→        avgDuration: Math.round(avgDuration),
  1661→        conversionRate: Math.round(conversionRate * 100) / 100,
  1662→        confidence: Math.min(1, data.totalCount / 10), // More occurrences = higher confidence
  1663→      });
  1664→    }

  1666→    // Sort by frequency and return top patterns

  1750→  /**
  1751→   * Analyze a conversion funnel
  1752→   * @param funnel - Funnel definition to analyze
  1753→   * @param options - Analysis options
  1754→   * @returns Funnel analysis results
  1755→   */
  1756→  async analyzeFunnel(
  1757→    funnel: ConversionFunnel,
  1758→    options?: { startDate?: Date; endDate?: Date; userId?: string }
  1759→  ): Promise<FunnelAnalysis> {
  1760→    this.ensureInitialized();
  1761→
  1762→    const startDate = options?.startDate || new Date(Date.now() - 30 * 24 * 60 * 60 * 1000);
  1763→    const endDate = options?.endDate || new Date();
  1764→
  1765→    // Collect all sessions in date range
  1766→    let relevantSessions: SessionData[] = [];
  1767→    
  1768→    Array.from(this.sessionStore.values()).forEach((session) => {
  1769→      if (session.startTime >= startDate && session.startTime <= endDate) {
  1770→        if (!options?.userId || session.userId === options.userId) {
  1771→          relevantSessions.push(session);
  1772→        }
  1773→      }
  1774→    });
  1775→
  1776→    // Initialize step counts
  1777→    const stepCounts = new Array(funnel.steps.length).fill(0);
  1778→    const totalTimePerStep = new Array(funnel.steps.length).fill(0);
  1779→    const usersAtEachStep = new Map<number, Set<string>>();
  1780→
  1781→    // Process each session through the funnel
  1782→    for (const session of relevantSessions) {
  1783→      let maxStepReached = -1;
  1784→      const stepTimes: number[] = [];
  1785→
  1786→      for (let stepIndex = 0; stepIndex < funnel.steps.length; stepIndex++) {
  1787→        const step = funnel.steps[stepIndex];
  1788→        
  1789→        // Find matching event in session
  1790→        const matchingEvent = session.events.find(e => 
  1791→          e.eventType === step.expectedEventType &&
  1792→          (e.page === step.pageOrAction || e.element === step.pageOrAction)
  1793→        );
  1794→
  1795→        if (matchingEvent) {
  1796→          maxStepReached = stepIndex;
  1797→          stepTimes[stepIndex] = matchingEvent.timestamp.getTime();
  1798→          
  1799→          if (!usersAtEachStep.has(stepIndex)) {
  1800→            usersAtEachStep.set(stepIndex, new Set());
  1801→          }
  1802→          usersAtEachStep.get(stepIndex)!.add(session.userId);
  1803→        } else {
  1804→          break; // User didn't complete this step
  1805→        }
  1806→      }
  1807→
  1808→      // Update counts for reached steps
  1809→      for (let i = 0; i <= maxStepReached; i++) {
  1810→        stepCounts[i]++;
  1811→      }
  1812→
  1813→      // Calculate time spent at each step
  1814→      for (let i = 0; i < stepTimes.length - 1; i++) {
  1815→        if (stepTimes[i] && stepTimes[i + 1]) {
  1816→          totalTimePerStep[i] += (stepTimes[i + 1] - stepTimes[i]) / 1000;
  1817→        }
  1818→      }
  1819→    }
  1820→
  1821→    // Calculate conversion rates between steps
  1822→    const stepConversionRates: number[] = [];
  1823→    for (let i = 0; i < stepCounts.length; i++) {
  1824→      if (i === 0) {
  1825→        stepConversionRates.push(100); // Entry step is 100%
  1826→      } else if (stepCounts[i - 1] > 0) {
  1827→        stepConversionRates.push(Math.round((stepCounts[i] / stepCounts[i - 1]) * 10000) / 100);
