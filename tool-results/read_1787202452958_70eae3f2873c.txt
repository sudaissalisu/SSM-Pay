    86→/**
    87→ * Base prediction result interface

    89→export interface BasePredictionResult {
    90→  /** Unique prediction request identifier */
    91→  predictionId: string;
    92→  /** Model that generated this prediction */
    93→  modelId: string;
    94→  /** Timestamp when prediction was made */
    95→  timestamp: Date;

    97→  processingTimeMs: number;

   103→  confidence: number;
   104→  /** Additional metadata about the prediction */
   105→  metadata?: Record<string, unknown>;
   106→}
