   541→   541→ *   profile: 'conservative',
   542→   542→ *   enableAML: true,
   543→   543→ * });
   544→   544→ * 
   545→   545→ * const result = await engine.assessTransactionRisk(transaction, customer, history);
   546→   546→ * ```
   547→   547→ */
   548→   548→export class RiskEngine {
   549→   549→  /** Current configuration */
   550→   550→  private config: RiskEngineConfig;
   551→   551→  
   552→   552→  /** Active threshold configuration */
   553→   553→  private thresholds: RiskThresholds;
   554→   554→

   556→   556→  private weights: RiskFactorWeights;

   560→   560→

   562→   562→  private assessmentCache: Map<string, { result: RiskAssessmentResult; expiry: Date }> = new Map();
   563→   563→  
   564→   564→  /** Running statistics */
   565→   565→  private stats: RiskStatistics = this.initializeStatistics();
   566→   566→
   567→   567→  /**
   568→   568→   * Creates a new RiskEngine instance
   569→   569→   * 
   570→   570→   * @param config - Configuration options for the risk engine
   571→   571→   * @throws {AppError} If configuration is invalid
   572→   572→   */
   573→   573→  constructor(config: Partial<RiskEngineConfig> = {}) {
   574→   574→    const profile: RiskProfile = config.profile || 'moderate';
   575→   575→    
   576→   576→    // Validate profile
   577→   577→    if (!['conservative', 'moderate', 'aggressive'].includes(profile)) {
   578→   578→      throw new AppError(
   579→   579→        `Invalid risk profile: ${profile}`,
   580→   580→        ErrorCode.INVALID_CONFIG,
   581→   581→        { severity: 'error', context: { provided: profile } }
   582→   582→      );
   583→   583→    }
   584→   584→
   585→   585→    // Build complete configuration
   586→   586→    this.config = {
   587→   587→      ...DEFAULT_CONFIG,
   588→   588→      ...config,
   589→   589→      profile,
   590→   590→    };
   591→   591→
   592→   592→    // Initialize thresholds with defaults and overrides
   593→   593→    this.thresholds = {
   594→   594→      ...DEFAULT_THRESHOLDS[profile],
   595→   595→      ...(this.config.thresholds || {}),
   596→   596→    };
   597→   597→
   598→   598→    // Initialize weights with defaults and overrides
   599→   599→    this.weights = {
   600→   600→      ...DEFAULT_WEIGHTS[profile],
   601→   601→      ...(this.config.weights || {}),
   602→   602→    };
   603→   603→
   604→   604→    // Validate weights sum to approximately 1
   605→   605→    const weightSum = Object.values(this.weights).reduce((sum, w) => sum + w, 0);
   606→   606→    if (Math.abs(weightSum - 1.0) > 0.01) {
   607→   607→      logger.warn('Risk factor weights do not sum to 1.0', {
   608→   608→        event: 'risk_engine.init',
   609→   609→        metadata: { weightSum, weights: this.weights },
   610→   610→      });
   611→   611→    }
   612→   612→
   613→   613→    logger.info('Risk Engine initialized', {
   614→   614→      event: 'risk_engine.init',
   615→   615→      metadata: {
   616→   616→        profile: this.config.profile,
   617→   617→        thresholds: this.thresholds,
   618→   618→        version: this.config.version,
   619→   619→      },
   620→   620→    });
   621→   621→  }

   623→   623→  // ============== Public API Methods ==============
   624→   624→
   625→   625→  /**
   626→   626→   * Perform comprehensive risk assessment on a transaction
   627→   627→   * 
   628→   628→   * @param transaction - The transaction to assess
   629→   629→   * @param customer - Customer profile data
   630→   630→   * @param history - Transaction history for velocity analysis
   631→   631→   * @param device - Optional device intelligence data
   632→   632→   * @param geoLocation - Optional geolocation data
   633→   633→   * @returns Complete risk assessment result
   634→   634→   * 
   635→   635→   * @example
   636→   636→   * ```typescript
   637→   637→   * const result = await engine.assessTransactionRisk(
   638→   638→   *   transactionData,
   639→   639→   *   customerProfile,
   640→   640→   *   transactionHistory,
   641→   641→   *   deviceInfo,
   642→   642→   *   geoData
   643→   643→   * );
   644→   644→   * 
   645→   645→   * if (result.shouldBlock) {
   646→   646→   *   return error('Transaction blocked due to high risk');
   647→   647→   * }
   648→   648→   * ```
   649→   649→   */
   650→   650→  async assessTransactionRisk(
   651→   651→    transaction: TransactionInput,
   652→   652→    customer: CustomerProfile,
   653→   653→    history: TransactionHistory,
   654→   654→    device?: DeviceIntelligence,
   655→   655→    geoLocation?: GeoLocationData
   656→   656→  ): Promise<RiskAssessmentResult> {
   657→   657→    const startTime = Date.now();
   658→   658→
   659→   659→    try {
   660→   660→      // Check cache first
   661→   661→      const cacheKey = this.generateCacheKey(transaction);
   662→   662→      const cached = this.getCachedAssessment(cacheKey);
   663→   663→      if (cached) {
   664→   664→        logger.debug('Returning cached risk assessment', {
   665→   665→          event: 'risk_engine.cache_hit',
   666→   666→          metadata: { transactionId: transaction.transactionId },
   667→   667→        });
   668→   668→        return cached;
   669→   669→      }
   670→   670→
   671→   671→      // Validate inputs
   672→   672→      this.validateInputs(transaction, customer, history);
   673→   673→
   674→   674→      // Calculate individual factor scores
   675→   675→      const factors: RiskFactorContribution[] = [];
   676→   676→
   677→   677→      // 1. Amount-based risk factors
   678→   678→      factors.push(this.calculateAmountRisk(transaction, history));
   679→   679→
   680→   680→      // 2. Velocity/rate-based risk factors
   681→   681→      factors.push(this.calculateVelocityRisk(transaction, history));
   682→   682→
   683→   683→      // 3. Device-based risk factors
   684→   684→      factors.push(this.calculateDeviceRisk(transaction, device));
   685→   685→
   686→   686→      // 4. Geographic risk factors
   687→   687→      factors.push(this.calculateGeographicRisk(transaction, geoLocation, customer));
   688→   688→
   689→   689→      // 5. Behavioral/pattern risk factors
   690→   690→      factors.push(this.calculateBehavioralRisk(transaction, history, customer));
   691→   691→
   692→   692→      // 6. Historical risk factors
   693→   693→      factors.push(this.calculateHistoricalRisk(customer));
   694→   694→
   695→   695→      // 7. Compliance/AML risk factors
   696→   696→      const amlAlerts = this.config.enableAML 
   697→   697→        ? this.performAMLChecks(transaction, history)
   698→   698→        : [];
   699→   699→
   700→   700→      factors.push(this.calculateComplianceRisk(amlAlerts, customer));
   701→   701→
   702→   702→      // 8. Customer profile risk factors
   703→   703→      factors.push(this.calculateCustomerProfileRisk(transaction, customer));
   704→   704→
   705→   705→      // Calculate weighted composite score
   706→   706→      const riskScore = this.calculateCompositeScore(factors);
   707→   707→
   708→   708→      // Determine risk level
   709→   709→      const riskLevel = this.classifyRiskLevel(riskScore);
   710→   710→
   711→   711→      // Determine authentication requirements
   712→   712→      const requiredAuth = this.determineAuthRequirement(riskLevel, factors, customer);
   713→   713→
   714→   714→      // Determine blocking/flagging
   715→   715→      const shouldBlock = this.shouldBlockTransaction(riskLevel, amlAlerts, customer);
   716→   716→      const shouldFlag = !shouldBlock && (riskLevel >= RiskLevel.HIGH || amlAlerts.length > 0);
   717→   717→
   718→   718→      // Generate recommendations
   719→   719→      const recommendations = this.generateRecommendations(riskLevel, factors, amlAlerts);
   720→   720→
   721→   721→      // Calculate confidence based on data completeness
   722→   722→      const confidence = this.calculateConfidence(factors, device, geoLocation);
   723→   723→
   724→   724→      // Build final result
   725→   725→      const result: RiskAssessmentResult = {
   726→   726→        riskScore: Math.round(riskScore * 100) / 100,
   727→   727→        riskLevel,
   728→   728→        requiredAuth,
   729→   729→        shouldBlock,
   730→   730→        shouldFlag,
   731→   731→        factors,
   732→   732→        amlAlerts,
   733→   733→        assessedAt: new Date(),
   734→   734→        assessmentVersion: this.config.version,
   735→   735→        recommendations,
   736→   736→        confidence,
   737→   737→        processingTimeMs: Date.now() - startTime,
   738→   738→      };
   739→   739→
   740→   740→      // Cache the result
   741→   741→      this.cacheAssessment(cacheKey, result);
   742→   742→
   743→   743→      // Update statistics
   744→   744→      this.updateStatistics(result);
   745→   745→
   746→   746→      // Log the assessment
   747→   747→      logger.info('Risk assessment completed', {
   748→   748→        event: 'risk_engine.assessment',
   749→   749→        metadata: {
   750→   750→          transactionId: transaction.transactionId,
   751→   751→          customerId: customer.customerId,
   752→   752→          riskScore: result.riskScore,
   753→   753→          riskLevel: result.riskLevel,
   754→   754→          shouldBlock: result.shouldBlock,
   755→   755→          authRequired: result.requiredAuth,
   756→   756→          amlAlerts: amlAlerts.length,
   757→   757→          processingTimeMs: result.processingTimeMs,
   758→   758→        },
   759→   759→      });
   760→   760→
   761→   761→      return result;
   762→   762→
   763→   763→    } catch (error) {
   764→   764→      const appError = error instanceof AppError 
   765→   765→        ? error 
   766→   766→        : new AppError('Risk assessment failed', ErrorCode.UNKNOWN_ERROR, { cause: error as Error });
   767→   767→
   768→   768→      logger.error('Risk assessment error', {
   769→   769→        event: 'risk_engine.error',
   770→   770→        metadata: { transactionId: transaction.transactionId },
   771→   771→        error: appError,
   772→   772→      });
   773→   773→
   774→   774→      throw appError;
   775→   775→    }
   776→   776→  }

   778→   778→  /**
   779→   779→   * Quick risk pre-check before full assessment

   791→   791→      };
   792→   792→    }

   794→   794→    // Check for obviously suspicious amounts
   795→   795→    if (transaction.amount <= 0) {
   796→   796→      return {
   797→   797→        passed: false,
   798→   798→        reason: 'Invalid transaction amount',
   799→   799→      };
   800→   800→    }
   801→   801→

   804→   804→      return {
   805→   805→        passed: false,
   806→   806→        reason: 'Future transaction timestamp',

   808→   808→    }

   829→   829→    const record: RiskHistoryRecord = {

   835→   835→      decisionMaker,
   836→   836→      notes,

   839→   839→
   840→   840→    // Store in history (using transactionId as key for simplicity)
   841→   841→    const existing = this.history.get(transactionId) || [];
   842→   842→    existing.push(record);
   843→   843→    
   844→   844→    // Trim to max size

   855→   855→
   856→   856→    // Adaptive mode: adjust thresholds based on outcomes
   857→   857→    if (this.config.adaptiveMode && decisionMaker === 'analyst') {
   858→   858→      this.adaptFromFeedback(record);
   859→   859→    }
   860→   860→  }

   962→   962→  // ============== Private Calculation Methods ==============

   964→   964→  /**
   965→   965→   * Calculate amount-related risk factors
   966→   966→   * Considers absolute amount, deviation from average, and round amounts
   967→   967→   */
   968→   968→  private calculateAmountRisk(
   969→   969→    transaction: TransactionInput,
   970→   970→    history: TransactionHistory
   971→   971→  ): RiskFactorContribution {
   972→   972→    let score = 0;
   973→   973→    const indicators: string[] = [];
   974→   974→    const metadata: Record<string, unknown> = {};
   975→   975→
   976→   976→    // Factor 1: Absolute amount risk (logarithmic scale)
   977→   977→    const amountInMajor = transaction.amount / 100; // Convert from minor units
   978→   978→    if (amountInMajor > 10000) {
   979→   979→      score += 40;
   980→   980→      indicators.push('very_high_amount');
   981→   981→    } else if (amountInMajor > 5000) {
   982→   982→      score += 25;
   983→   983→      indicators.push('high_amount');
   984→   984→    } else if (amountInMajor > 1000) {
   985→   985→      score += 15;
   986→   986→      indicators.push('elevated_amount');
   987→   987→    } else if (amountInMajor > 500) {
   988→   988→      score += 8;
   989→   989→      indicators.push('moderate_amount');
   990→   990→    }
   991→   991→    metadata.amountInMajor = amountInMajor;
   992→   992→
   993→   993→    // Factor 2: Deviation from user's average
   994→   994→    if (history.avgAmount30d > 0) {
   995→   995→      const deviation = Math.abs(amountInMajor - history.avgAmount30d / 100) / (history.stdAmount30d / 100 || 1);
   996→   996→      if (deviation > 5) {
   997→   997→        score += 35;
   998→   998→        indicators.push('extreme_deviation');
   999→   999→      } else if (deviation > 3) {
  1000→  1000→        score += 20;
  1001→  1001→        indicators.push('high_deviation');
  1002→  1002→      } else if (deviation > 2) {
  1003→  1003→        score += 10;
  1004→  1004→        indicators.push('moderate_deviation');
  1005→  1005→      }
  1006→  1006→      metadata.deviation = Math.round(deviation * 100) / 100;
  1007→  1007→    }
  1008→  1008→
  1009→  1009→    // Factor 3: Exceeds maximum historical
  1010→  1010→    if (history.maxAmount30d > 0 && transaction.amount > history.maxAmount30d) {
  1011→  1011→      score += 15;
  1012→  1012→      indicators.push('exceeds_historical_max');
  1013→  1013→      metadata.previousMax = history.maxAmount30d;
  1014→  1014→    }
  1015→  1015→
  1016→  1016→    // Factor 4: Round amount detection (potential structuring)
  1017→  1017→    if (amountInMajor > 1000) {
  1018→  1018→      const roundedToThousand = amountInMajor % 1000 === 0;
  1019→  1019→      const roundedToHundred = amountInMajor % 100 === 0;
  1020→  1020→      
  1021→  1021→      if (roundedToThousand) {
  1022→  1022→        score += 15;
  1023→  1023→        indicators.push('round_thousand_amount');
  1024→  1024→      } else if (roundedToHundred) {
  1025→  1025→        score += 8;
  1026→  1026→        indicators.push('round_hundred_amount');
  1027→  1027→      }
  1028→  1028→    }
  1029→  1029→
  1030→  1030→    // Factor 5: Just below reporting threshold (potential structuring)
  1031→  1031→    const structuringRange = [900000, 950000, 990000]; // In cents ($9,000-$9,900)
  1032→  1032→    if (structuringRange.some(threshold => 
  1033→  1033→      Math.abs(transaction.amount - threshold) < 5000
  1034→  1034→    )) {
  1035→  1035→      score += 25;
  1036→  1036→      indicators.push('near_structuring_threshold');
  1037→  1037→    }
  1038→  1038→
  1039→  1039→    // Cap at 100
  1040→  1040→    score = Math.min(score, 100);
  1041→  1041→
  1042→  1042→    return {
  1043→  1043→      factorId: 'amount_risk',
  1044→  1044→      factorName: 'Amount Risk Analysis',
  1045→  1045→      score,
  1046→  1046→      maxScore: 100,
  1047→  1047→      rawValue: { amount: transaction.amount, amountInMajor },
  1048→  1048→      indicators,
  1049→  1049→      metadata,
  1050→  1050→    };
  1051→  1051→  }
  1052→  1052→
  1053→  1053→  /**
  1054→  1054→   * Calculate velocity/rate-based risk factors
  1055→  1055→   * Analyzes transaction frequency patterns
  1056→  1056→   */
  1057→  1057→  private calculateVelocityRisk(
  1058→  1058→    transaction: TransactionInput,
