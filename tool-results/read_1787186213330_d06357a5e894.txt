   363→   363→// ============== Audit Log Service Implementation ==============
   364→   364→
   365→   365→/**
   366→   366→ * Enterprise-grade Audit Logging Service
   367→   367→ * 
   368→   368→ * Features:
   369→   369→ * - Comprehensive event logging with structured data
   370→   370→ * - Tamper-evident hash chain for integrity verification
   371→   371→ * - Flexible querying and search capabilities
   372→   372→ * - Multiple export formats for compliance
   373→   373→ * - Configurable retention policies
   374→   374→ * - Full data change tracking
   375→   375→ * 
   376→   376→ * @example
   377→   377→ * ```typescript
   378→   378→ * const auditLog = new AuditLogService();
   379→   379→ * 
   380→   380→ * await auditLog.log({
   381→   381→ *   eventType: AuditEventType.TRANSACTION,
   382→   382→ *   action: AuditAction.PAYMENT_INITIATED,
   383→   383→ *   userId: 'user-123',
   384→   384→ *   username: 'john.doe',
   385→   385→ *   description: 'Payment initiated for order #ORD-456',
   386→   386→ *   resource: { type: 'transaction', id: 'txn-789' },
   387→   387→ *   changes: [{ field: 'status', previousValue: null, newValue: 'pending', changeType: 'created' }],
   388→   388→ * });
   389→   389→ * ```
   390→   390→ */
   391→   391→export class AuditLogService {
   392→   392→  /** In-memory event store */
   393→   393→  private events: Map<string, AuditEvent> = new Map();
   394→   394→  
   395→   395→  /** Ordered event IDs for sequence tracking */
   396→   396→  private orderedEventIds: string[] = [];
   397→   397→  
   398→   398→  /** Current hash chain state */
   399→   399→  private lastHash: string = '';
   400→   400→  
   401→   401→  /** Current sequence number */
   402→   402→  private sequenceNumber: number = 0;
   403→   403→  
   404→   404→  /** Active retention policies */
   405→   405→  private retentionPolicies: RetentionPolicy[] = [];
   406→   406→  
   407→   407→  /** Statistics cache */
   408→   408→  private statsCache: AuditStatistics | null = null;
   409→   409→  
   410→   410→  /** Whether service is initialized */
   411→   411→  private initialized: boolean = false;
   412→   412→
   413→   413→  /**
   414→   414→   * Create a new AuditLogService instance
   415→   415→   * @param config Optional configuration overrides
   416→   416→   */
   417→   417→  constructor(config?: {
   418→   418→    defaultRetentionDays?: number;
   419→   419→    maxInMemoryEvents?: number;
   420→   420→    enableAutoCleanup?: boolean;
   421→   421→  }) {
   422→   422→    // Initialize default retention policy
   423→   423→    this.retentionPolicies.push({
   424→   424→      name: 'default',
   425→   425→      description: 'Default retention policy for all events',
   426→   426→      retentionDays: config?.defaultRetentionDays ?? DEFAULT_RETENTION_DAYS,
   427→   427→      active: true,
   428→   428→      createdAt: new Date(),
   429→   429→      updatedAt: new Date(),
   430→   430→    });
   431→   431→
   432→   432→    logger.info('AuditLogService initialized', {
   433→   433→      event: 'audit.init',
   434→   434→      metadata: { 
   435→   435→        defaultRetentionDays: config?.defaultRetentionDays ?? DEFAULT_RETENTION_DAYS,
   436→   436→        maxInMemoryEvents: config?.maxInMemoryEvents ?? MAX_IN_MEMORY_EVENTS,
   437→   437→      },
   438→   438→    });
   439→   439→
   440→   440→    this.initialized = true;
   441→   441→  }
   442→   442→
   443→   443→  // ============== Core Logging Methods ==============
   444→   444→
   445→   445→  /**
   446→   446→   * Log a new audit event
   447→   447→   * 
   448→   448→   * @param event The audit event data (without system-managed fields)
   449→   449→   * @returns The complete audit event with generated fields
   450→   450→   * @throws {AppError} If validation fails or service not initialized
   451→   451→   * 
   452→   452→   * @example
   453→   453→   * ```typescript
   454→   454→   * const event = await auditLog.log({
   455→   455→   *   eventType: AuditEventType.AUTHENTICATION,
   456→   456→   *   action: AuditAction.LOGIN,
   457→   457→   *   userId: 'user-123',
   458→   458→   *   username: 'john.doe',
   459→   459→   *   description: 'User logged in successfully',
   460→   460→   *   source: { ipAddress: '192.168.1.1', userAgent: 'Mozilla/5.0...' },
   461→   461→   * });
   462→   462→   * ```
   463→   463→   */
   464→   464→  async log(event: Omit<AuditEvent, 'id' | 'timestamp' | 'previousHash' | 'hash' | 'sequenceNumber'>): Promise<AuditEvent> {
   465→   465→    if (!this.initialized) {
   466→   466→      throw new AppError('AuditLogService not initialized', ErrorCode.UNKNOWN_ERROR);
   467→   467→    }
   468→   468→
   469→   469→    // Validate required fields
   470→   470→    this.validateEventInput(event);
   471→   471→
   472→   472→    // Generate unique ID
   473→   473→    const id = this.generateEventId();
   474→   474→
   475→   475→    // Create timestamp
   476→   476→    const timestamp = new Date();
   477→   477→
   478→   478→    // Build complete event with hash chain
   479→   479→    const completeEvent: AuditEvent = {
   480→   480→      ...event,
   481→   481→      id,
   482→   482→      timestamp,
   483→   483→      previousHash: this.lastHash,
   484→   484→      sequenceNumber: ++this.sequenceNumber,
   485→   485→    };
   486→   486→
   487→   487→    // Calculate and set hash
   488→   488→    completeEvent.hash = this.calculateEventHash(completeEvent);
   489→   489→
   490→   490→    // Store the event
   491→   491→    this.events.set(id, completeEvent);
   492→   492→    this.orderedEventIds.push(id);
   493→   493→
   494→   494→    // Update last hash for chain
   495→   495→    this.lastHash = completeEvent.hash;
   496→   496→
   497→   497→    // Invalidate stats cache
   498→   498→    this.statsCache = null;
   499→   499→
   500→   500→    // Log to application logger
   501→   501→    logger.info(`Audit event logged: ${event.action}`, {
   502→   502→      event: `audit.${event.eventType.toLowerCase()}.${event.action.toLowerCase()}`,
   503→   503→      metadata: {
   504→   504→        eventId: id,
   505→   505→        userId: event.userId,
   506→   506→        action: event.action,
   507→   507→        severity: event.severity,
   508→   508→        outcome: event.outcome,
   509→   509→      },
   510→   510→    });
   511→   511→
   512→   512→    return completeEvent;
   513→   513→  }
   514→   514→
   515→   515→  /**
   516→   516→   * Log an authentication event (convenience method)
   517→   517→   * 
   518→   518→   * @param action The authentication action
   519→   519→   * @param userId User identifier
   520→   520→   * @param options Additional options
   521→   521→   * @returns The created audit event
   522→   522→   */
   523→   523→  async logAuthentication(
   524→   524→    action: AuditAction.LOGIN | AuditAction.LOGOUT | AuditAction.LOGIN_FAILED | AuditAction.PASSWORD_CHANGE | AuditAction.MFA_ENABLED | AuditAction.MFA_DISABLED,
   525→   525→    userId: string,
   526→   526→    options: {
   527→   527→      username?: string;
   528→   528→      outcome?: AuditOutcome;
   529→   529→      description?: string;
   530→   530→      source?: AuditSource;
   531→   531→      metadata?: Record<string, unknown>;
   532→   532→      correlationId?: string;
   533→   533→    } = {}
   534→   534→  ): Promise<AuditEvent> {

   719→   719→  /**
   720→   720→   * Search audit logs with flexible filtering
   721→   721→   * 
   722→   722→   * @param params Query parameters for filtering
   723→   723→   * @returns Array of matching audit events
   724→   724→   * 
   725→   725→   * @example
   726→   726→   * ```typescript
   727→   727→   * // Find all failed login attempts in the last 24 hours
   728→   728→   * const results = await auditLog.query({
   729→   729→   *   eventTypes: [AuditEventType.AUTHENTICATION],
   730→   730→   *   actions: [AuditAction.LOGIN_FAILED],
   731→   731→   *   startDate: new Date(Date.now() - 24 * 60 * 60 * 1000),
   732→   732→   *   sortOrder: 'desc',
   733→   733→   * });
   734→   734→   * ```
   735→   735→   */
   736→   736→  async query(params: AuditQueryParams = {}): Promise<AuditEvent[]> {
   737→   737→    let results = Array.from(this.events.values());
   738→   738→
   739→   739→    // Apply filters
   740→   740→    if (params.eventTypes && params.eventTypes.length > 0) {
   741→   741→      results = results.filter(e => params.eventTypes!.includes(e.eventType));
   742→   742→    }
   743→   743→
   744→   744→    if (params.actions && params.actions.length > 0) {
   745→   745→      results = results.filter(e => params.actions!.includes(e.action));
   746→   746→    }
   747→   747→
   748→   748→    if (params.userIds && params.userIds.length > 0) {
   749→   749→      results = results.filter(e => params.userIds!.includes(e.userId));
   750→   750→    }
   751→   751→
   752→   752→    if (params.severities && params.severities.length > 0) {
   753→   753→      results = results.filter(e => params.severities!.includes(e.severity));
   754→   754→    }
   755→   755→
   756→   756→    if (params.outcomes && params.outcomes.length > 0) {
   757→   757→      results = results.filter(e => params.outcomes!.includes(e.outcome));
   758→   758→    }
   759→   759→
   760→   760→    if (params.startDate) {
   761→   761→      results = results.filter(e => e.timestamp >= params.startDate!);
   762→   762→    }
   763→   763→
   764→   764→    if (params.endDate) {
   765→   765→      results = results.filter(e => e.timestamp <= params.endDate!);
   766→   766→    }
   767→   767→
   768→   768→    if (params.searchTerm) {
   769→   769→      const term = params.searchTerm.toLowerCase();
   770→   770→      results = results.filter(e => 
   771→   771→        e.description.toLowerCase().includes(term) ||
   772→   772→        e.userId.toLowerCase().includes(term) ||
   773→   773→        (e.username && e.username.toLowerCase().includes(term)) ||
   774→   774→        (e.metadata && JSON.stringify(e.metadata).toLowerCase().includes(term))
   775→   775→      );
   776→   776→    }
   777→   777→
   778→   778→    if (params.resourceType) {
   779→   779→      results = results.filter(e => e.resource?.type === params.resourceType);
   780→   780→    }
   781→   781→
   782→   782→    if (params.resourceId) {
   783→   783→      results = results.filter(e => e.resource?.id === params.resourceId);
   784→   784→    }
   785→   785→
   786→   786→    if (params.correlationId) {
   787→   787→      results = results.filter(e => e.correlationId === params.correlationId);
   788→   788→    }
   789→   789→
   790→   790→    // Sort results
   791→   791→    const sortBy = params.sortBy ?? 'timestamp';
   792→   792→    const sortOrder = params.sortOrder ?? 'desc';
   793→   793→
   794→   794→    results.sort((a, b) => {
   795→   795→      let comparison = 0;
   796→   796→      
   797→   797→      switch (sortBy) {
   798→   798→        case 'timestamp':
   799→   799→          comparison = a.timestamp.getTime() - b.timestamp.getTime();
   800→   800→          break;
   801→   801→        case 'severity':
   802→   802→          const severityOrder = { [AuditSeverity.EMERGENCY]: 4, [AuditSeverity.CRITICAL]: 3, [AuditSeverity.WARNING]: 2, [AuditSeverity.INFO]: 1 };
   803→   803→          comparison = severityOrder[a.severity] - severityOrder[b.severity];
   804→   804→          break;
   805→   805→        case 'eventType':
   806→   806→          comparison = a.eventType.localeCompare(b.eventType);
   807→   807→          break;
   808→   808→      }
   809→   809→      
   810→   810→      return sortOrder === 'desc' ? -comparison : comparison;
   811→   811→    });
   812→   812→
   813→   813→    // Apply pagination
   814→   814→    const offset = params.offset ?? 0;
   815→   815→    const limit = params.limit ?? 100;
   816→   816→    
   817→   817→    results = results.slice(offset, offset + limit);
   818→   818→
   819→   819→    logger.debug('Audit query executed', {
   820→   820→      event: 'audit.query',
   821→   821→      metadata: { params, resultCount: results.length },
   822→   822→    });
   823→   823→
   824→   824→    return results;
   825→   825→  }

   871→   871→  /**
   872→   872→   * Get comprehensive audit statistics
   873→   873→   * 
   874→   874→   * @returns Statistics summary
   875→   875→   */
   876→   876→  async getStatistics(): Promise<AuditStatistics> {
   877→   877→    if (this.statsCache) {
   878→   878→      return this.statsCache;
   879→   879→    }
   880→   880→
   881→   881→    const allEvents = Array.from(this.events.values());
   882→   882→    const totalEvents = allEvents.length;
   883→   883→
   884→   884→    // Initialize counters
   885→   885→    const eventsByType = {} as Record<AuditEventType, number>;
   886→   886→    const eventsBySeverity = {} as Record<AuditSeverity, number>;
   887→   887→    const eventsByOutcome = {} as Record<AuditOutcome, number>;
   888→   888→
   889→   889→    // Initialize all enum values
   890→   890→    Object.values(AuditEventType).forEach(type => { eventsByType[type] = 0; });
   891→   891→    Object.values(AuditSeverity).forEach(severity => { eventsBySeverity[severity] = 0; });
   892→   892→    Object.values(AuditOutcome).forEach(outcome => { eventsByOutcome[outcome] = 0; });
   893→   893→
   894→   894→    // Count events
   895→   895→    let oldestEvent: Date | undefined;
   896→   896→    let newestEvent: Date | undefined;
   897→   897→    let recentEvents = 0;
   898→   898→    const oneHourAgo = new Date(Date.now() - 60 * 60 * 1000);
   899→   899→
   900→   900→    for (const event of allEvents) {
   901→   901→      eventsByType[event.eventType]++;
   902→   902→      eventsBySeverity[event.severity]++;
   903→   903→      eventsByOutcome[event.outcome]++;
   904→   904→
   905→   905→      if (!oldestEvent || event.timestamp < oldestEvent) {
   906→   906→        oldestEvent = event.timestamp;
   907→   907→      }
   908→   908→      if (!newestEvent || event.timestamp > newestEvent) {
   909→   909→        newestEvent = event.timestamp;
   910→   910→      }
   911→   911→      if (event.timestamp >= oneHourAgo) {
   912→   912→        recentEvents++;
   913→   913→      }
   914→   914→    }
   915→   915→
   916→   916→    // Verify chain integrity
   917→   917→    const integrityResult = await this.verifyIntegrity();
   918→   918→
   919→   919→    this.statsCache = {
   920→   920→      totalEvents,
   921→   921→      eventsByType,
   922→   922→      eventsBySeverity,
   923→   923→      eventsByOutcome,
   924→   924→      recentEvents,
   925→   925→      oldestEvent,
   926→   926→      newestEvent,
   927→   927→      chainIntegrity: integrityResult.valid,
   928→   928→      chainBreakPoint: integrityResult.invalidIndex,
   929→   929→    };
   930→   930→
   931→   931→    return this.statsCache;
   932→   932→  }

   941→   941→   * 
   942→   942→   * @example

   969→   969→      default:

   993→   993→  // ============== Integrity & Security Methods ==============
   994→   994→
   995→   995→  /**
   996→   996→   * Verify the integrity of the hash chain
   997→   997→   * 
   998→   998→   * Detects any tampering with audit logs by verifying
   999→   999→   * that each event's hash correctly chains to the next.
  1000→  1000→   *

  1290→  1290→  /**
  1291→  1291→   * Simple hash function for demo purposes
  1292→  1292→   * In production, replace with crypto.subtle.digest
  1293→  1293→   * 
  1294→  1294→   * @param input String to hash
  1295→  1295→   * @returns Hash string
  1296→  1296→   */
  1297→  1297→  private simpleHash(input: string): string {
  1298→  1298→    let hash = 0;
  1299→  1299→    for (let i = 0; i < input.length; i++) {
  1300→  1300→      const char = input.charCodeAt(i);
  1301→  1301→      hash = ((hash << 5) - hash) + char;
  1302→  1302→      hash = hash & hash; // Convert to 32-bit integer
  1303→  1303→    }
  1304→  1304→    
  1305→  1305→    // Convert to hex string
  1306→  1306→    const hashStr = Math.abs(hash).toString(16).padStart(8, '0');
  1307→  1307→    
  1308→  1308→    // Add some entropy based on content length and timestamp
  1309→  1309→    const extraEntropy = (input.length * 31 + Date.now()).toString(16);
  1310→  1310→    
  1311→  1311→    return `${hashStr}-${extraEntropy.substring(0, 8)}-${btoa(input.substring(0, 32)).substring(0, 8)}`;
  1312→  1312→  }

  1448→  1448→          }

  1451→  1451→

  1465→  1465→

  1467→  1467→      if (redacted.source) {
