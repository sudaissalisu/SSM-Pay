   562→   562→   * Generate HMAC-SHA256 signature for a payload
   563→   563→   * 
   564→   564→   * @param payload - The payload object to sign
   565→   565→   * @param secret - The secret key for signing
   566→   566→   * @param timestamp - Optional timestamp for inclusion in signature
   567→   567→   * @returns Hex-encoded signature string
   568→   568→   * 
   569→   569→   * @example
   570→   570→   * ```typescript
   571→   571→   * const signature = webhookService.generateSignature(
   572→   572→   *   { id: 'evt_123', type: 'payment.completed', data: {} },
   573→   573→   *   'my-secret-key'
   574→   574→   * );
   575→   575→   * ```
   576→   576→   */
   577→   577→  generateSignature(
   578→   578→    payload: Record<string, unknown>,
   579→   579→    secret: string,
   580→   580→    timestamp?: number
   581→   581→  ): string {
   582→   582→    const ts = timestamp || Date.now();
   583→   583→    const payloadString = JSON.stringify(payload);
   584→   584→    const signedPayload = `${ts}.${payloadString}`;
   585→   585→    
   586→   586→    const signature = createHmac('sha256', secret)
   587→   587→      .update(signedPayload)
   588→   588→      .digest('hex');
   589→   589→    
   590→   590→    return `t=${ts},v1=${signature}`;
   591→   591→  }
   592→   592→
   593→   593→  /**
   594→   594→   * Verify HMAC-SHA256 signature for an incoming webhook
   595→   595→   * 
   596→   596→   * Uses timing-safe comparison to prevent timing attacks.
   597→   597→   * Supports both timestamped and plain signatures.
   598→   598→   * 
   599→   599→   * @param payload - The raw payload to verify
   600→   600→   * @param signature - The signature to verify against
   601→   601→   * @param secret - The secret key used for verification
   602→   602→   * @param maxAge - Maximum age in ms for timestamped signatures (default: 5 minutes)
   603→   603→   * @returns True if signature is valid, false otherwise
   604→   604→   * 
   605→   605→   * @example
   606→   606→   * ```typescript
   607→   607→   * const isValid = await webhookService.verifySignature(
   608→   608→   *   requestBody,
   609→   609→   *   requestHeaders['x-ssm-pay-signature'],
   610→   610→   *   endpointSecret
   611→   611→   * );
   612→   612→   * ```
   613→   613→   */
   614→   614→  async verifySignature(
   615→   615→    payload: unknown,
   616→   616→    signature: string,
   617→   617→    secret: string,
   618→   618→    maxAge: number = 5 * 60 * 1000 // 5 minutes
   619→   619→  ): Promise<boolean> {
   620→   620→    try {
   621→   621→      // Parse signature format: t=timestamp,v1=signature
   622→   622→      const signatureMatch = signature.match(/t=(\d+),v1=([a-f0-9]+)/);
   623→   623→      
   624→   624→      if (!signatureMatch) {
   625→   625→        this.stats.signaturesFailed++;
   626→   626→        logger.warn('Invalid signature format', {
   627→   627→          event: 'webhook.signature.invalid_format',
   628→   628→          metadata: { signaturePrefix: signature.substring(0, 20) },
   629→   629→        });
   630→   630→        return false;
   631→   631→      }
   632→   632→
   633→   633→      const [, timestampStr, providedSignature] = signatureMatch;
   634→   634→      const timestamp = parseInt(timestampStr, 10);
   635→   635→      
   636→   636→      // Check timestamp freshness
   637→   637→      const now = Date.now();
   638→   638→      const timestampAge = now - timestamp;
   639→   639→      
   640→   640→      if (timestampAge < 0) {
   641→   641→        this.stats.signaturesFailed++;
   642→   642→        logger.warn('Signature timestamp is in the future', {
   643→   643→          event: 'webhook.signature.future_timestamp',
   644→   644→          metadata: { timestampAge: Math.abs(timestampAge) },
   645→   645→        });
   646→   646→        return false;
   647→   647→      }
   648→   648→      
   649→   649→      if (timestampAge > maxAge) {
   650→   650→        this.stats.signaturesFailed++;
   651→   651→        logger.warn('Signature timestamp too old', {
   652→   652→          event: 'webhook.signature.expired',
   653→   653→          metadata: { timestampAge, maxAge },
   654→   654→        });
   655→   655→        return false;
   656→   656→      }
   657→   657→
   658→   658→      // Generate expected signature
   659→   659→      const payloadString = JSON.stringify(payload);
   660→   660→      const signedPayload = `${timestamp}.${payloadString}`;
   661→   661→      
   662→   662→      const expectedSignature = createHmac('sha256', secret)
   663→   663→        .update(signedPayload)
   664→   664→        .digest('hex');
   665→   665→
   666→   666→      // Use timing-safe comparison
   667→   667→      const isValid = timingSafeEqual(
   668→   668→        Buffer.from(providedSignature),
   669→   669→        Buffer.from(expectedSignature)
   670→   670→      );
   671→   671→
   672→   672→      if (isValid) {
   673→   673→        this.stats.signaturesVerified++;
   674→   674→        logger.debug('Signature verified successfully', {
   675→   675→          event: 'webhook.signature.verified',
   676→   676→        });
   677→   677→      } else {
   678→   678→        this.stats.signaturesFailed++;
   679→   679→        logger.warn('Signature verification failed', {
   680→   680→          event: 'webhook.signature.mismatch',
   681→   681→        });
   682→   682→      }
   683→   683→
   684→   684→      return isValid;
   685→   685→    } catch (error) {
   686→   686→      this.stats.signaturesFailed++;
   687→   687→      logger.error('Error during signature verification', {
   688→   688→        event: 'webhook.signature.error',
   689→   689→        error: error instanceof Error ? error : new Error(String(error)),
   690→   690→      });
   691→   691→      return false;
   692→   692→    }
   693→   693→  }
   694→   694→
   695→   695→  // ============== Incoming Webhook Processing ==============
   696→   696→
   697→   697→  /**
   698→   698→   * Process an incoming webhook request
   699→   699→   * 
   700→   700→   * Handles the complete workflow:
   701→   701→   * 1. Validate payload structure
   702→   702→   * 2. Verify signature
   703→   703→   * 3. Check idempotency
   704→   704→   * 4. Parse and normalize event
   705→   705→   * 5. Route to handlers/subscribers
   706→   706→   * 
   707→   707→   * @param payload - The raw webhook payload
   708→   708→   * @param headers - HTTP headers containing signature
   709→   709→   * @returns Processing result with status and optional event
   710→   710→   * 
   711→   711→   * @example
   712→   712→   * ```typescript
   713→   713→   * const result = await webhookService.processIncomingWebhook(
   714→   714→   *   req.body,
   715→   715→   *   req.headers
   716→   716→   * );
   717→   717→   * 
   718→   718→   * if (result.valid && result.event) {
   719→   719→   *   console.log('Processed event:', result.event.type);
   720→   720→   * }
   721→   721→   * ```
   722→   722→   */
   723→   723→  async processIncomingWebhook(
   724→   724→    payload: unknown,
   725→   725→    headers: Record<string, string>
   726→   726→  ): Promise<{
   727→   727→    valid: boolean;
   728→   728→    event?: WebhookEvent;
   729→   729→    error?: string;
   730→   730→    errorCode?: ErrorCode;
   731→   731→  }> {
   732→   732→    this.stats.totalEventsReceived++;
   733→   733→    
   734→   734→    const startTime = Date.now();
   735→   735→    
   736→   736→    try {
   737→   737→      // Step 1: Validate payload exists
   738→   738→      if (!payload || typeof payload !== 'object') {
   739→   739→        throw new AppError('Invalid payload: must be an object', ErrorCode.VALIDATION_ERROR);
   740→   740→      }
   741→   741→
   742→   742→      const typedPayload = payload as Record<string, unknown>;
   743→   743→
   744→   744→      // Step 2: Check payload size
   745→   745→      const payloadSize = JSON.stringify(payload).length;
   746→   746→      if (payloadSize > this.config.maxPayloadSizeBytes) {
   747→   747→        throw new AppError(
   748→   748→          `Payload exceeds maximum size of ${this.config.maxPayloadSizeBytes} bytes`,
   749→   749→          ErrorCode.VALIDATION_ERROR
   750→   750→        );
   751→   751→      }
   752→   752→
   753→   753→      // Step 3: Extract and validate required fields
   754→   754→      const eventId = typedPayload.id as string;
   755→   755→      const eventType = typedPayload.type as string;
   756→   756→      const eventCreated = typedPayload.created as string;
   757→   757→      const eventData = typedPayload.data as Record<string, unknown>;
   758→   758→
   759→   759→      if (!eventId || !eventType || !eventData) {
   760→   760→        throw new AppError(
   761→   761→          'Missing required fields: id, type, or data',
   762→   762→          ErrorCode.VALIDATION_ERROR,
   763→   763→          { context: { hasId: !!eventId, hasType: !!eventType, hasData: !!eventData } }
   764→   764→        );
   765→   765→      }
   766→   766→
   767→   767→      // Step 4: Verify signature
   768→   768→      const signatureHeader = headers[WEBHOOK_HEADERS.SIGNATURE.toLowerCase()] ||
   769→   769→                             headers[WEBHOOK_HEADERS.SIGNATURE];
   770→   770→      
   771→   771→      if (!signatureHeader) {
   772→   772→        throw new AppError(
   773→   773→          'Missing signature header',
   774→   774→          ErrorCode.VALIDATION_ERROR,
   775→   775→          { context: { headerName: WEBHOOK_HEADERS.SIGNATURE } }
   776→   776→        );
   777→   777→      }
   778→   778→
   779→   779→      const isValid = await this.verifySignature(
   780→   780→        payload,
   781→   781→        signatureHeader,
   782→   782→        this.config.defaultSigningSecret
   783→   783→      );
   784→   784→
   785→   785→      if (!isValid) {
   786→   786→        throw new AppError(
   787→   787→          'Signature verification failed',
   788→   788→          ErrorCode.PAYMENT_VERIFICATION_FAILED
   789→   789→        );
   790→   790→      }
   791→   791→
   792→   792→      // Step 5: Check idempotency
   793→   793→      if (this.isDuplicateEvent(eventId)) {
   794→   794→        this.stats.duplicateEventsRejected++;
   795→   795→        logger.info('Duplicate webhook event rejected', {
   796→   796→          event: 'webhook.duplicate_rejected',
   797→   797→          metadata: { eventId, eventType },
   798→   798→        });
   799→   799→        return { valid: true }; // Return success but don't reprocess
   800→   800→      }
   801→   801→
   802→   802→      this.markEventAsProcessed(eventId);
   803→   803→
   804→   804→      // Step 6: Construct normalized event
   805→   805→      const event: WebhookEvent = {
   806→   806→        id: eventId,
   807→   807→        type: eventType,
   808→   808→        data: eventData,
   809→   809→        createdAt: eventCreated || new Date().toISOString(),
   810→   810→        version: EVENT_SCHEMA_VERSION,
   811→   811→        source: 'external',
   812→   812→        correlationId: typedPayload.correlationId as string | undefined,
   813→   813→        parentEventId: typedPayload.parentEventId as string | undefined,
   814→   814→      };
   815→   815→
   816→   816→      // Step 7: Route event to subscribers
   817→   817→      await this.routeEventToSubscribers(event);
   818→   818→
   819→   819→      this.stats.totalEventsProcessed++;
   820→   820→      this.stats.uniqueEventsProcessed++;
   821→   821→      this.stats.lastProcessedAt = new Date();
   822→   822→
   823→   823→      const processingTime = Date.now() - startTime;
   824→   824→      logger.info('Incoming webhook processed successfully', {
   825→   825→        event: 'webhook.incoming.processed',
   826→   826→        metadata: {
   827→   827→          eventId: event.id,
   828→   828→          eventType: event.type,
   829→   829→          processingTimeMs: processingTime,
   830→   830→        },
   831→   831→      });
   832→   832→
   833→   833→      return { valid: true, event };
   834→   834→    } catch (error) {
   835→   835→      this.stats.totalEventsFailed++;
   836→   836→      
   837→   837→      const appError = error instanceof AppError 
   838→   838→        ? error 
   839→   839→        : new AppError(
   840→   840→            error instanceof Error ? error.message : 'Unknown error processing webhook',
   841→   841→            ErrorCode.PAYMENT_CALLBACK_FAILED,
   842→   842→            { cause: error instanceof Error ? error : undefined }
   843→   843→          );
   844→   844→
   845→   845→      logger.error('Failed to process incoming webhook', {
   846→   846→        event: 'webhook.incoming.failed',
   847→   847→        error: appError,
   848→   848→        metadata: { processingTimeMs: Date.now() - startTime },
   849→   849→      });
   850→   850→
   851→   851→      return {
   852→   852→        valid: false,
   853→   853→        error: appError.message,
   854→   854→        errorCode: appError.code,
   855→   855→      };
   856→   856→    }
   857→   857→  }
   858→   858→
   859→   859→  // ============== Endpoint Management ==============
   860→   860→
   861→   861→  /**
   862→   862→   * Register a new webhook endpoint
   863→   863→   * 
   864→   864→   * @param config - Endpoint configuration
   865→   865→   * @returns The registered endpoint
   866→   866→   * @throws AppError if configuration is invalid
   867→   867→   * 
   868→   868→   * @example
   869→   869→   * ```typescript
   870→   870→   * const endpoint = await webhookService.registerEndpoint({
   871→   871→   *   url: 'https://api.example.com/webhooks',
   872→   872→   *   secret: 'whsec_my_secret_key',
   873→   873→   *   subscribedEvents: ['payment.*', 'transaction.*'],
   874→   874→   *   description: 'Primary payment webhook handler'
   875→   875→   * });
   876→   876→   * ```
   877→   877→   */
   878→   878→  registerEndpoint(config: {
   879→   879→    url: string;
   880→   880→    secret: string;
   881→   881→    subscribedEvents: string[];
   882→   882→    description?: string;
   883→   883→    metadata?: Record<string, unknown>;
   884→   884→    maxRetries?: number;
   885→   885→    retryConfig?: Partial<RetryConfiguration>;
   886→   886→  }): WebhookEndpoint {
   887→   887→    // Validate URL
   888→   888→    try {
   889→   889→      new URL(config.url);
   890→   890→    } catch {
   891→   891→      throw new AppError(
   892→   892→        `Invalid URL: ${config.url}`,
   893→   893→        ErrorCode.INVALID_CONFIG,
   894→   894→        { context: { url: config.url } }
   895→   895→      );
   896→   896→    }
   897→   897→
   898→   898→    // Validate secret
   899→   899→    if (!config.secret || config.secret.length < 16) {
   900→   900→      throw new AppError(
   901→   901→        'Secret must be at least 16 characters',
   902→   902→        ErrorCode.VALIDATION_ERROR,
   903→   903→        { context: { secretLength: config.secret?.length || 0 } }
   904→   904→      );
   905→   905→    }
   906→   906→
   907→   907→    // Validate subscribed events
   908→   908→    if (!config.subscribedEvents || config.subscribedEvents.length === 0) {
   909→   909→      throw new AppError(
   910→   910→        'At least one event subscription is required',
   911→   911→        ErrorCode.VALIDATION_ERROR
   912→   912→      );
   913→   913→    }
   914→   914→
   915→   915→    const endpointId = `ep_${Date.now()}_${randomUUID().slice(0, 8)}`;
   916→   916→    const now = new Date();
   917→   917→
   918→   918→    const endpoint: WebhookEndpoint = {
   919→   919→      id: endpointId,
   920→   920→      url: config.url,
   921→   921→      secret: config.secret,
   922→   922→      subscribedEvents: config.subscribedEvents,
   923→   923→      isActive: true,
   924→   924→      createdAt: now,
   925→   925→      updatedAt: now,
   926→   926→      consecutiveFailures: 0,
   927→   927→      totalDeliveries: 0,
   928→   928→      successfulDeliveries: 0,
   929→   929→      maxRetries: config.maxRetries ?? this.config.defaultRetryConfig.maxAttempts,
   930→   930→      retryConfig: { ...this.config.defaultRetryConfig, ...config.retryConfig },
   931→   931→      metadata: config.metadata,
   932→   932→      description: config.description,
   933→   933→    };
   934→   934→
   935→   935→    this.endpoints.set(endpointId, endpoint);
   936→   936→    this.updateEndpointStats();
   937→   937→
   938→   938→    logger.info('Webhook endpoint registered', {
   939→   939→      event: 'webhook.endpoint.registered',
   940→   940→      metadata: {
   941→   941→        endpointId,
   942→   942→        url: config.url,
   943→   943→        eventCount: config.subscribedEvents.length,
   944→   944→      },
   945→   945→    });
   946→   946→
   947→   947→    return endpoint;
   948→   948→  }
   949→   949→
   950→   950→  /**
   951→   951→   * Update an existing webhook endpoint
   952→   952→   * 
   953→   953→   * @param endpointId - ID of endpoint to update
   954→   954→   * @param updates - Fields to update
   955→   955→   * @returns Updated endpoint
   956→   956→   * @throws AppError if endpoint not found
   957→   957→   */
   958→   958→  updateEndpoint(
   959→   959→    endpointId: string,
   960→   960→    updates: Partial<{
   961→   961→      url: string;
   962→   962→      secret: string;
   963→   963→      subscribedEvents: string[];
   964→   964→      isActive: boolean;
   965→   965→      description: string;
   966→   966→      metadata: Record<string, unknown>;
   967→   967→      maxRetries: number;
   968→   968→      retryConfig: Partial<RetryConfiguration>;
   969→   969→    }>
   970→   970→  ): WebhookEndpoint {
   971→   971→    const endpoint = this.endpoints.get(endpointId);
   972→   972→    
   973→   973→    if (!endpoint) {
   974→   974→      throw new AppError(
   975→   975→        `Endpoint not found: ${endpointId}`,
   976→   976→        ErrorCode.NOT_FOUND,

   979→   979→    }
   980→   980→
   981→   981→    // Apply updates
   982→   982→    Object.assign(endpoint, updates, { updatedAt: new Date() });
   983→   983→

   986→   986→    }
   987→   987→
   988→   988→    this.endpoints.set(endpointId, endpoint);
   989→   989→    this.updateEndpointStats();
   990→   990→
   991→   991→    logger.info('Webhook endpoint updated', {
   992→   992→      event: 'webhook.endpoint.updated',
   993→   993→      metadata: { endpointId, updatedFields: Object.keys(updates) },
   994→   994→    });
   995→   995→
   996→   996→    return endpoint;
   997→   997→  }
   998→   998→
   999→   999→  /**
  1000→  1000→   * Remove a webhook endpoint
  1001→  1001→   * 
  1002→  1002→   * @param endpointId - ID of endpoint to remove
  1003→  1003→   * @returns True if removed, false if not found
  1004→  1004→   */
  1005→  1005→  removeEndpoint(endpointId: string): boolean {
  1006→  1006→    const endpoint = this.endpoints.get(endpointId);
  1007→  1007→    
  1008→  1008→    if (!endpoint) {
  1009→  1009→      return false;
  1010→  1010→    }
  1011→  1011→
  1012→  1012→    // Move pending deliveries to dead letter queue

  1020→  1020→      metadata: { endpointId, url: endpoint.url },

  1024→  1024→  }
  1025→  1025→
  1026→  1026→  /**
  1027→  1027→   * Get an endpoint by ID
  1028→  1028→   * 
  1029→  1029→   * @param endpointId - Endpoint ID
  1030→  1030→   * @returns Endpoint or undefined if not found

  1032→  1032→  getEndpoint(endpointId: string): WebhookEndpoint | undefined {

  1034→  1034→  }
  1035→  1035→
  1036→  1036→  /**
  1037→  1037→   * Get all registered endpoints
  1038→  1038→   * 
  1039→  1039→   * @param options - Filter options
  1040→  1040→   * @returns Array of endpoints
  1041→  1041→   */
  1042→  1042→  getEndpoints(options?: {
  1043→  1043→    activeOnly?: boolean;
  1044→  1044→    eventTypes?: string[];
  1045→  1045→  }): WebhookEndpoint[] {
  1046→  1046→    let endpoints = Array.from(this.endpoints.values());
  1047→  1047→
  1048→  1048→    if (options?.activeOnly) {
  1049→  1049→      endpoints = endpoints.filter(ep => ep.isActive);
  1050→  1050→    }
  1051→  1051→
  1052→  1052→    if (options?.eventTypes?.length) {
  1053→  1053→      endpoints = endpoints.filter(ep =>
  1054→  1054→        ep.subscribedEvents.some(pattern =>
  1055→  1055→          options.eventTypes!.some(type => this.matchEventPattern(type, pattern))
  1056→  1056→        )
  1057→  1057→      );
  1058→  1058→    }
  1059→  1059→
  1060→  1060→    return endpoints;
  1061→  1061→  }
  1062→  1062→
  1063→  1063→  /**
  1064→  1064→   * Disable an endpoint (soft delete)
  1065→  1065→   * 
  1066→  1066→   * @param endpointId - Endpoint ID to disable
  1067→  1067→   * @returns Updated endpoint
  1068→  1068→   */
  1069→  1069→  disableEndpoint(endpointId: string): WebhookEndpoint {
  1070→  1070→    return this.updateEndpoint(endpointId, { isActive: false });
  1071→  1071→  }
  1072→  1072→
  1073→  1073→  /**
  1074→  1074→   * Re-enable a disabled endpoint
  1075→  1075→   * 
  1076→  1076→   * @param endpointId - Endpoint ID to enable
  1077→  1077→   * @returns Updated endpoint
  1078→  1078→   */
  1079→  1079→  enableEndpoint(endpointId: string): WebhookEndpoint {
  1080→  1080→    const endpoint = this.updateEndpoint(endpointId, { isActive: true });
  1081→  1081→    // Reset consecutive failures when re-enabling
  1082→  1082→    endpoint.consecutiveFailures = 0;
  1083→  1083→    return endpoint;
  1084→  1084→  }
  1085→  1085→
  1086→  1086→  // ============== Event Dispatch & Routing ==============
  1087→  1087→
  1088→  1088→  /**
  1089→  1089→   * Create and dispatch a new webhook event
  1090→  1090→   * 
  1091→  1091→   * @param type - Event type
  1092→  1092→   * @param data - Event payload data
  1093→  1093→   * @param options - Additional event options
  1094→  1094→   * @returns Created event and delivery promises
  1095→  1095→   * 
  1096→  1096→   * @example
  1097→  1097→   * ```typescript

  1099→  1099→   *   'payment.completed',

  1105→  1105→  async dispatchEvent(

  1107→  1107→    data: Record<string, unknown>,
  1108→  1108→    options?: {
  1109→  1109→      source?: string;
  1110→  1110→      correlationId?: string;
  1111→  1111→      parentEventId?: string;
  1112→  1112→      priority?: number;
  1113→  1113→    }
  1114→  1114→  ): Promise<{
  1115→  1115→    event: WebhookEvent;
  1116→  1116→    deliveries: Promise<WebhookDelivery[]>[];
  1117→  1117→  }> {
  1118→  1118→    const event: WebhookEvent = {
  1119→  1119→      id: `evt_${Date.now()}_${randomUUID().slice(0, 8)}`,
  1120→  1120→      type,
  1121→  1121→      data,
  1122→  1122→      createdAt: new Date().toISOString(),
  1123→  1123→      version: EVENT_SCHEMA_VERSION,
  1124→  1124→      source: options?.source || 'ssm-pay',
  1125→  1125→      correlationId: options?.correlationId,
  1126→  1126→      parentEventId: options?.parentEventId,
  1127→  1127→    };
  1128→  1128→
  1129→  1129→    this.stats.totalEventsReceived++;
  1130→  1130→

  1132→  1132→      event: 'webhook.event.dispatched',

  1145→  1145→    }

  1149→  1149→
  1150→  1150→    this.stats.totalEventsProcessed++;

  1153→  1153→  }
  1154→  1154→
  1155→  1155→  /**
  1156→  1156→   * Find endpoints subscribed to a specific event type
  1157→  1157→   * 
  1158→  1158→   * @param eventType - Event type to match
  1159→  1159→   * @returns Matching endpoints
  1160→  1160→   * @private
  1161→  1161→   */
  1162→  1162→  private findEndpointsForEventType(eventType: string): WebhookEndpoint[] {
  1163→  1163→    const matchingEndpoints: WebhookEndpoint[] = [];
  1164→  1164→
  1165→  1165→    for (const endpoint of this.endpoints.values()) {
  1166→  1166→      if (!endpoint.isActive) continue;
  1167→  1167→
  1168→  1168→      const isMatch = endpoint.subscribedEvents.some(pattern =>
  1169→  1169→        this.matchEventPattern(eventType, pattern)
  1170→  1170→      );
  1171→  1171→
  1172→  1172→      if (isMatch) {
  1173→  1173→        matchingEndpoints.push(endpoint);
  1174→  1174→      }
  1175→  1175→    }
  1176→  1176→
  1177→  1177→    return matchingEndpoints;
  1178→  1178→  }
  1179→  1179→
  1180→  1180→  /**
  1181→  1181→   * Match event type against a pattern (supports wildcards)
  1182→  1182→   * 
  1183→  1183→   * @param eventType - Actual event type
  1184→  1184→   * @param pattern - Pattern to match against
  1185→  1185→   * @returns True if matches
  1186→  1186→   * @private
  1187→  1187→   */
  1188→  1188→  private matchEventPattern(eventType: string, pattern: string): boolean {
  1189→  1189→    // Exact match
  1190→  1190→    if (eventType === pattern) return true;
  1191→  1191→
  1192→  1192→    // Wildcard: payment.* matches payment.completed, payment.failed, etc.
  1193→  1193→    if (pattern.endsWith('*')) {
  1194→  1194→      const prefix = pattern.slice(0, -1); // Remove *
  1195→  1195→      // Support both "event.*" and "event*" patterns
  1196→  1196→      if (pattern.endsWith('.*')) {
  1197→  1197→        return eventType.startsWith(prefix);
  1198→  1198→      }
  1199→  1199→      return eventType.startsWith(prefix);
  1200→  1200→    }
  1201→  1201→
  1202→  1202→    // Regex pattern support
  1203→  1203→    if (pattern.startsWith('^') || pattern.endsWith('$')) {
  1204→  1204→      try {
  1205→  1205→        const regex = new RegExp(pattern);
  1206→  1206→        return regex.test(eventType);
  1207→  1207→      } catch {
  1208→  1208→        return false;
  1209→  1209→      }
  1210→  1210→    }
  1211→  1211→
  1212→  1212→    return false;
  1213→  1213→  }
  1214→  1214→
  1215→  1215→  // ============== Queue Management ==============
  1216→  1216→
  1217→  1217→  /**
  1218→  1218→   * Add a delivery to the processing queue
  1219→  1219→   * 
  1220→  1220→   * @param event - Event to deliver
  1221→  1221→   * @param endpoint - Target endpoint
  1222→  1222→   * @param priority - Queue priority (lower = higher)
  1223→  1223→   * @returns Queue item
  1224→  1224→   * @private
  1225→  1225→   */
  1226→  1226→  private enqueueDelivery(
  1227→  1227→    event: WebhookEvent,
  1228→  1228→    endpoint: WebhookEndpoint,
  1229→  1229→    priority: number = 5
  1230→  1230→  ): WebhookQueueItem {
  1231→  1231→    // Check queue size limit
  1232→  1232→    if (this.deliveryQueue.length >= this.config.maxQueueSize) {
  1233→  1233→      logger.warn('Webhook queue full, rejecting delivery', {
  1234→  1234→        event: 'webhook.queue.full',
  1235→  1235→        metadata: {
  1236→  1236→          currentSize: this.deliveryQueue.length,
  1237→  1237→          maxSize: this.config.maxQueueSize,
  1238→  1238→          eventId: event.id,
  1239→  1239→        },
  1240→  1240→      });
  1241→  1241→      throw new AppError(
  1242→  1242→        'Webhook queue is full',
  1243→  1243→        ErrorCode.API_REQUEST_FAILED,
  1244→  1244→        { context: { currentSize: this.deliveryQueue.length, maxSize: this.config.maxQueueSize } }
  1245→  1245→      );
  1246→  1246→    }
  1247→  1247→
  1248→  1248→    const queueItem: WebhookQueueItem = {
  1249→  1249→      id: `qi_${Date.now()}_${randomUUID().slice(0, 8)}`,
  1250→  1250→      event,
  1251→  1251→      endpointId: endpoint.id,
  1252→  1252→      priority,
  1253→  1253→      queuedAt: new Date(),
  1254→  1254→      scheduledFor: new Date(), // Schedule immediately
  1255→  1255→      attempts: 0,
  1256→  1256→      status: 'waiting',
  1257→  1257→    };
  1258→  1258→
  1259→  1259→    // Insert maintaining priority order
  1260→  1260→    const insertIndex = this.deliveryQueue.findIndex(
  1261→  1261→      item => item.priority > priority
  1262→  1262→    );
  1263→  1263→    
  1264→  1264→    if (insertIndex === -1) {
  1265→  1265→      this.deliveryQueue.push(queueItem);
  1266→  1266→    } else {
  1267→  1267→      this.deliveryQueue.splice(insertIndex, 0, queueItem);
  1268→  1268→    }
  1269→  1269→
  1270→  1270→    this.updateQueueStats();
  1271→  1271→
  1272→  1272→    logger.debug('Delivery enqueued', {
  1273→  1273→      event: 'webhook.queue.enqueued',
  1274→  1274→      metadata: {
  1275→  1275→        queueItemId: queueItem.id,
  1276→  1276→        eventId: event.id,
  1277→  1277→        endpointId: endpoint.id,
  1278→  1278→        position: this.deliveryQueue.indexOf(queueItem),
  1279→  1279→      },
  1280→  1280→    });
  1281→  1281→
  1282→  1282→    return queueItem;
  1283→  1283→  }
  1284→  1284→
  1285→  1285→  /**
  1286→  1286→   * Process a queue item (deliver webhook)
  1287→  1287→   * 
  1288→  1288→   * @param queueItem - Item to process
  1289→  1289→   * @returns Array of delivery records
  1290→  1290→   * @private
  1291→  1291→   */
  1292→  1292→  private async processQueueItem(queueItem: WebhookQueueItem): Promise<WebhookDelivery[]> {
  1293→  1293→    // Wait until scheduled time
  1294→  1294→    const now = new Date();
  1295→  1295→    if (queueItem.scheduledFor > now) {
  1296→  1296→      const delay = queueItem.scheduledFor.getTime() - now.getTime();
  1297→  1297→      await new Promise(resolve => setTimeout(resolve, delay));
  1298→  1298→    }
  1299→  1299→
  1300→  1300→    // Check concurrency limit
  1301→  1301→    if (this.currentProcessingCount >= this.config.maxConcurrency) {
  1302→  1302→      // Requeue with slight delay
  1303→  1303→      queueItem.scheduledFor = new Date(Date.now() + 100);
  1304→  1304→      return this.processQueueItem(queueItem);
  1305→  1305→    }
  1306→  1306→
  1307→  1307→    queueItem.status = 'processing';
  1308→  1308→    this.currentProcessingCount++;
  1309→  1309→
  1310→  1310→    try {
  1311→  1311→      const endpoint = this.endpoints.get(queueItem.endpointId);
  1312→  1312→      
  1313→  1313→      if (!endpoint) {
  1314→  1314→        throw new Error(`Endpoint not found: ${queueItem.endpointId}`);
  1315→  1315→      }
  1316→  1316→
  1317→  1317→      const delivery = await this.executeDelivery(queueItem.event, endpoint);
  1318→  1318→      queueItem.status = 'completed';
  1319→  1319→      
  1320→  1320→      return [delivery];
  1321→  1321→    } catch (error) {
  1322→  1322→      queueItem.status = 'failed';
  1323→  1323→      throw error;
  1324→  1324→    } finally {
  1325→  1325→      this.currentProcessingCount--;
  1326→  1326→      this.updateQueueStats();
  1327→  1327→    }
  1328→  1328→  }
  1329→  1329→
  1330→  1330→  /**
  1331→  1331→   * Execute actual HTTP delivery to endpoint
  1332→  1332→   * 
  1333→  1333→   * @param event - Event to deliver
  1334→  1334→   * @param endpoint - Target endpoint
  1335→  1335→   * @returns Delivery record
  1336→  1336→   * @private
  1337→  1337→   */
  1338→  1338→  private async executeDelivery(
  1339→  1339→    event: WebhookEvent,
  1340→  1340→    endpoint: WebhookEndpoint
  1341→  1341→  ): Promise<WebhookDelivery> {
  1342→  1342→    const deliveryId = `del_${Date.now()}_${randomUUID().slice(0, 8)}`;
  1343→  1343→    const startTime = Date.now();
  1344→  1344→
  1345→  1345→    // Get or create delivery record
  1346→  1346→    let delivery = this.deliveries.get(deliveryId);
  1347→  1347→    
  1348→  1348→    if (!delivery) {
  1349→  1349→      delivery = {
  1350→  1350→        id: deliveryId,
  1351→  1351→        event,
  1352→  1352→        endpointId: endpoint.id,
  1353→  1353→        status: 'processing',
  1354→  1354→        attemptNumber: 1,
  1355→  1355→        maxAttempts: endpoint.maxRetries,
  1356→  1356→        createdAt: new Date(),
  1357→  1357→        updatedAt: new Date(),
  1358→  1358→      };
  1359→  1359→      this.deliveries.set(deliveryId, delivery);
  1360→  1360→    } else {
  1361→  1361→      delivery.attemptNumber++;
  1362→  1362→      delivery.updatedAt = new Date();
  1363→  1363→      delivery.status = 'processing';
  1364→  1364→    }
  1365→  1365→
  1366→  1366→    this.stats.totalDeliveriesAttempted++;
  1367→  1367→    endpoint.totalDeliveries++;
  1368→  1368→
  1369→  1369→    try {
  1370→  1370→      // Prepare payload
  1371→  1371→      const payload: IncomingWebhookPayload = {
  1372→  1372→        id: event.id,
  1373→  1373→        type: event.type,
  1374→  1374→        created: event.createdAt,
  1375→  1375→        data: event.data,
  1376→  1376→      };
  1377→  1377→
  1378→  1378→      // Generate signature
  1379→  1379→      const signature = this.generateSignature(payload, endpoint.secret);
  1380→  1380→      delivery.signature = signature;
  1381→  1381→
  1382→  1382→      // Prepare headers
  1383→  1383→      const headers: Record<string, string> = {
  1384→  1384→        'Content-Type': 'application/json',
  1385→  1385→        [WEBHOOK_HEADERS.SIGNATURE]: signature,
  1386→  1386→        [WEBHOOK_HEADERS.TIMESTAMP]: Date.now().toString(),
  1387→  1387→        [WEBHOOK_HEADERS.EVENT_ID]: event.id,
  1388→  1388→        [WEBHOOK_HEADERS.EVENT_TYPE]: event.type,
  1389→  1389→        [WEBHOOK_HEADERS.DELIVERY]: deliveryId,
  1390→  1390→        [WEBHOOK_HEADERS.ATTEMPT]: delivery.attemptNumber.toString(),
  1391→  1391→        'User-Agent': 'SSM-Pay-Webhook/2.0',
  1392→  1392→      };
  1393→  1393→
  1394→  1394→      delivery.requestHeaders = headers;
  1395→  1395→
  1396→  1396→      // Execute HTTP request with timeout
  1397→  1397→      const controller = new AbortController();
  1398→  1398→      const timeout = setTimeout(() => controller.abort(), 30000); // 30s timeout
  1399→  1399→
  1400→  1400→      const response = await fetch(endpoint.url, {
  1401→  1401→        method: 'POST',
  1402→  1402→        headers,
  1403→  1403→        body: JSON.stringify(payload),
  1404→  1404→        signal: controller.signal,
  1405→  1405→      });
  1406→  1406→
  1407→  1407→      clearTimeout(timeout);
  1408→  1408→
  1409→  1409→      const durationMs = Date.now() - startTime;
  1410→  1410→      delivery.durationMs = durationMs;
  1411→  1411→      delivery.responseStatusCode = response.status;
  1412→  1412→      delivery.responseBody = await response.text().catch(() => '');
  1413→  1413→
  1414→  1414→      if (response.ok) {
  1415→  1415→        // Success
  1416→  1416→        delivery.status = 'delivered';
  1417→  1417→        delivery.deliveredAt = new Date();
  1418→  1418→        endpoint.lastSuccessfulDeliveryAt = new Date();
  1419→  1419→        endpoint.successfulDeliveries++;
  1420→  1420→        endpoint.consecutiveFailures = 0;
  1421→  1421→
  1422→  1422→        this.stats.totalDeliveriesSucceeded++;
  1423→  1423→        this.updateAverageDeliveryTime(durationMs);
  1424→  1424→
  1425→  1425→        logger.info('Webhook delivered successfully', {
  1426→  1426→          event: 'webhook.delivery.success',
  1427→  1427→          metadata: {
  1428→  1428→            deliveryId,
  1429→  1429→            endpointId: endpoint.id,
  1430→  1430→            eventId: event.id,
  1431→  1431→            statusCode: response.status,
  1432→  1432→            attempt: delivery.attemptNumber,
  1433→  1433→            durationMs,
  1434→  1434→          },
  1435→  1435→        });
  1436→  1436→      } else if (endpoint.retryConfig.retryableStatusCodes.includes(response.status)) {
  1437→  1437→        // Retryable error
  1438→  1438→        throw new AppError(
  1439→  1439→          `HTTP ${response.status}: ${delivery.responseBody}`,
  1440→  1440→          ErrorCode.API_RESPONSE_ERROR,
  1441→  1441→          { context: { statusCode: response.status, endpointUrl: endpoint.url } }
  1442→  1442→        );
  1443→  1443→      } else {
  1444→  1444→        // Non-retryable error
  1445→  1445→        delivery.status = 'failed';
  1446→  1446→        delivery.errorMessage = `HTTP ${response.status}: ${delivery.responseBody}`;
  1447→  1447→        
  1448→  1448→        this.stats.totalDeliveriesFailed++;
  1449→  1449→        endpoint.consecutiveFailures++;
  1450→  1450→        endpoint.lastFailureAt = new Date();
  1451→  1451→
  1452→  1452→        // Move to dead letter if appropriate
  1453→  1453→        if (delivery.attemptNumber >= delivery.maxAttempts) {
  1454→  1454→          this.moveToDeadLetterQueue(delivery, endpoint, 'max_retries_exceeded');
  1455→  1455→        }
  1456→  1456→
  1457→  1457→        logger.warn('Webhook delivery failed (non-retryable)', {
  1458→  1458→          event: 'webhook.delivery.failed',
  1459→  1459→          metadata: {
  1460→  1460→            deliveryId,
  1461→  1461→            endpointId: endpoint.id,
  1462→  1462→            statusCode: response.status,
  1463→  1463→            attempt: delivery.attemptNumber,
  1464→  1464→          },
  1465→  1465→        });
  1466→  1466→      }
  1467→  1467→    } catch (error) {
  1468→  1468→      const durationMs = Date.now() - startTime;
  1469→  1469→      delivery.durationMs = durationMs;
  1470→  1470→      delivery.errorMessage = error instanceof Error ? error.message : String(error);
  1471→  1471→      delivery.errorCode = error instanceof AppError ? error.code : 'UNKNOWN';
  1472→  1472→
  1473→  1473→      this.stats.totalDeliveriesFailed++;
  1474→  1474→      endpoint.consecutiveFailures++;
  1475→  1475→      endpoint.lastFailureAt = new Date();
  1476→  1476→
  1477→  1477→      // Determine if we should retry
  1478→  1478→      if (delivery.attemptNumber < delivery.maxAttempts) {
  1479→  1479→        delivery.status = 'retrying';
  1480→  1480→        delivery.nextRetryAt = this.calculateRetryTime(delivery, endpoint);
  1481→  1481→        
  1482→  1482→        this.stats.totalDeliveriesRetried++;
  1483→  1483→
  1484→  1484→        logger.warn('Scheduling webhook retry', {
  1485→  1485→          event: 'webhook.delivery.retry_scheduled',
  1486→  1486→          metadata: {
  1487→  1487→            deliveryId,
  1488→  1488→            endpointId: endpoint.id,
  1489→  1489→            attempt: delivery.attemptNumber,
  1490→  1490→            nextRetry: delivery.nextRetryAt?.toISOString(),
  1491→  1491→          },
  1492→  1492→        });
  1493→  1493→
  1494→  1494→        // Schedule retry
  1495→  1495→        const retryDelay = delivery.nextRetryAt.getTime() - Date.now();
  1496→  1496→        setTimeout(async () => {
  1497→  1497→          await this.executeDelivery(event, endpoint);
  1498→  1498→        }, retryDelay);
  1499→  1499→      } else {
  1500→  1500→        // Max retries exceeded - move to DLQ
  1501→  1501→        delivery.status = 'failed';
  1502→  1502→        this.moveToDeadLetterQueue(delivery, endpoint, 'max_retries_exceeded');
  1503→  1503→
  1504→  1504→        logger.error('Webhook moved to dead letter queue', {
  1505→  1505→          event: 'webhook.delivery.dead_lettered',
  1506→  1506→          metadata: {
  1507→  1507→            deliveryId,
  1508→  1508→            endpointId: endpoint.id,
  1509→  1509→            totalAttempts: delivery.attemptNumber,
  1510→  1510→            lastError: delivery.errorMessage,
  1511→  1511→          },
  1512→  1512→        });
  1513→  1513→      }
  1514→  1514→    }
  1515→  1515→
  1516→  1516→    delivery.updatedAt = new Date();
  1517→  1517→    this.deliveries.set(deliveryId, delivery);
  1518→  1518→
  1519→  1519→    return delivery;
  1520→  1520→  }
  1521→  1521→
  1522→  1522→  /**
  1523→  1523→   * Calculate next retry time with exponential backoff and jitter
  1524→  1524→   * 
  1525→  1525→   * @param delivery - Current delivery record
  1526→  1526→   * @param endpoint - Target endpoint
  1527→  1527→   * @returns Next retry timestamp
  1528→  1528→   * @private
  1529→  1529→   */
  1530→  1530→  private calculateRetryTime(
  1531→  1531→    delivery: WebhookDelivery,
  1532→  1532→    endpoint: WebhookEndpoint
  1533→  1533→  ): Date {
  1534→  1534→    const { retryConfig } = endpoint;
  1535→  1535→    
  1536→  1536→    // Calculate base delay with exponential backoff
  1537→  1537→    const baseDelay = retryConfig.initialDelayMs * 
  1538→  1538→      Math.pow(retryConfig.backoffMultiplier, delivery.attemptNumber - 1);
  1539→  1539→    
  1540→  1540→    // Cap at maximum delay
  1541→  1541→    const cappedDelay = Math.min(baseDelay, retryConfig.maxDelayMs);
  1542→  1542→
