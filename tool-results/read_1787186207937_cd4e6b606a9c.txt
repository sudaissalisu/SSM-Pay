   356→// ============== Main Monitoring Class ==============
   357→
   358→/**
   359→ * Enterprise-grade Monitoring & Observability Service
   360→ * 
   361→ * Provides comprehensive metrics collection, health checking,
   362→ * performance monitoring, alerting, and dashboard generation.
   363→ * 
   364→ * @example
   365→ * ```typescript
   366→ * import { monitoringService } from '@/services/monitoring';
   367→ * 
   368→ * // Record a counter increment
   369→ * monitoringService.incrementCounter('payment.requests.total', { method: 'initiate' });
   370→ * 
   371→ * // Track performance
   372→ * const stopTimer = monitoringService.startTimer('api.payment.init');
   373→ * // ... do work ...
   374→ * stopTimer({ success: true });
   375→ * 
   376→ * // Check system health
   377→ * const health = await monitoringService.getSystemHealth();
   378→ * ```
   379→ */
   380→class MonitoringService {
   381→  // ============== Private State ==============
   382→  
   383→  /** Storage for all registered metrics */
   384→  private metrics: Map<string, Metric> = new Map();
   385→  
   386→  /** Performance records for time-windowed analysis */
   387→  private performanceRecords: Map<string, PerformanceRecord[]> = new Map();
   388→  
   389→  /** Error tracking counts */
   390→  private errorCounts: Map<string, { count: number; timestamps: Date[] }> = new Map();
   391→  
   392→  /** Registered alert thresholds */
   393→  private alerts: Map<string, AlertThreshold> = new Map();
   394→  
   395→  /** Currently active alerts */
   396→  private activeAlerts: Map<string, ActiveAlert> = new Map();
   397→  
   398→  /** Health check registry */
   399→  private healthChecks: Map<string, () => Promise<HealthCheckResult>> = new Map();
   400→  
   401→  /** Historical data points for dashboard visualization */
   402→  private historicalData: Map<string, DashboardDataPoint[]> = new Map();
   403→  
   404→  /** Cleanup interval handle */
   405→  private cleanupInterval: NodeJS.Timeout | null = null;
   406→  
   407→  /** Whether the service has been initialized */
   408→  private initialized: boolean = false;
   409→
   410→  // ============== Initialization ==============
   411→
   412→  /**
   413→   * Initialize the monitoring service
   414→   * Sets up built-in metrics, starts background tasks
   415→   */
   416→  initialize(): void {
   417→    if (this.initialized) {
   418→      logger.warn('Monitoring service already initialized', { event: 'monitoring.init' });
   419→      return;
   420→    }
   421→
   422→    logger.info('Initializing monitoring service', { event: 'monitoring.init' });
   423→
   424→    // Register built-in metrics
   425→    this.registerBuiltInMetrics();
   426→    
   427→    // Start cleanup interval
   428→    this.startCleanupInterval();
   429→    
   430→    this.initialized = true;
   431→    logger.info('Monitoring service initialized successfully', { event: 'monitoring.ready' });
   432→  }
   433→
   434→  /**
   435→   * Register built-in application metrics
   436→   * @private
   437→   */
   438→  private registerBuiltInMetrics(): void {
   439→    // Request counters

   441→    this.createCounter('http_requests_success_total', 'Total successful HTTP requests');
   442→    this.createCounter('http_requests_error_total', 'Total failed HTTP requests');
   443→    
   444→    // Payment-specific counters
   445→    this.createCounter('payments_initiated_total', 'Total payment initiations');
   446→    this.createCounter('payments_completed_total', 'Total completed payments');
   447→    this.createCounter('payments_failed_total', 'Total failed payments');
   448→    this.createCounter('payments_refunded_total', 'Total refunded payments');
   449→    
   450→    // Latency histograms
   451→    this.createHistogram('http_request_duration_ms', 'HTTP request duration in milliseconds');
   452→    this.createHistogram('payment_processing_duration_ms', 'Payment processing duration in milliseconds');
   453→    this.createHistogram('external_api_call_duration_ms', 'External API call duration in milliseconds');
   454→    
   455→    // Gauges for current state

   457→    this.createGauge('pending_payments', 'Payments awaiting completion');
   458→    this.createGauge('queue_depth', 'Current processing queue depth');
   459→    
   460→    // Business metrics

   462→    this.createGauge('average_transaction_value', 'Average transaction value');
   463→  }
   464→
   465→  /**
   466→   * Start the background cleanup interval
   467→   * @private
   468→   */
   469→  private startCleanupInterval(): void {
   470→    // Run cleanup every 5 minutes
   471→    this.cleanupInterval = setInterval(() => {
   472→      try {
   473→        this.cleanupOldData();
   474→        this.evaluateAllAlerts();
   475→      } catch (error) {
   476→        logger.error('Error in monitoring cleanup cycle', {
   477→          event: 'monitoring.error',
   478→          error: error instanceof Error ? error : new AppError(String(error)),
   479→        });
   480→      }
   481→    }, 5 * 60 * 1000);
   482→    
   483→    // Don't prevent process exit
   484→    if (this.cleanupInterval.unref) {
   485→      this.cleanupInterval.unref();
   486→    }
   487→  }
   488→
   489→  /**
   490→   * Clean up old data beyond retention period
   491→   * @private
   492→   */
   493→  private cleanupOldData(): void {
   494→    const cutoffTime = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000); // 7 days ago
   495→    
   496→    // Clean up performance records
   497→    for (const [key, records] of Array.from(this.performanceRecords.entries())) {
   498→      const filtered = records.filter(r => r.timestamp > cutoffTime);
   499→      if (filtered.length !== records.length) {
   500→        this.performanceRecords.set(key, filtered);
   501→      }
   502→    }
   503→    
   504→    // Clean up historical data
   505→    for (const [key, dataPoints] of Array.from(this.historicalData.entries())) {
   506→      const filtered = dataPoints.filter(d => d.timestamp > cutoffTime);
   507→      if (filtered.length !== dataPoints.length) {
   508→        this.historicalData.set(key, filtered.slice(-MAX_DATA_POINTS_PER_METRIC));
   509→      }
   510→    }
   511→    
   512→    // Clean up error timestamps
   513→    for (const [key, data] of Array.from(this.errorCounts.entries())) {
   514→      const filteredTimestamps = data.timestamps.filter(t => t > cutoffTime);
   515→      if (filteredTimestamps.length !== data.timestamps.length) {
   516→        this.errorCounts.set(key, {
   517→          count: filteredTimestamps.length,
   518→          timestamps: filteredTimestamps,
   519→        });
   520→      }
   521→    }
   522→
   523→    logger.debug('Completed monitoring data cleanup', {
   524→      event: 'monitoring.cleanup',
   525→      metadata: { cutoffTime: cutoffTime.toISOString() },
   526→    });
   527→  }

   529→  // ============== Counter Metrics ==============
   530→
   531→  /**
   532→   * Create a new counter metric
   533→   * @param name - Unique metric name
   534→   * @param description - Human-readable description
   535→   * @param labels - Initial labels
   536→   * @returns The created counter metric
   537→   * @throws {AppError} If metric already exists
   538→   */
   539→  createCounter(
   540→    name: string,
   541→    description: string = '',
   542→    labels: MetricLabels = {}
   543→  ): CounterMetric {
   544→    if (this.metrics.has(name)) {
   545→      throw new AppError(
   546→        `Metric '${name}' already exists`,
   547→        ErrorCode.VALIDATION_ERROR,
   548→        { severity: 'warning' }
   549→      );
   550→    }
   551→
   552→    const now = new Date();
   553→    const counter: CounterMetric = {
   554→      name,
   555→      description,
   556→      type: 'counter',
   557→      labels: { ...labels },
   558→      value: 0,
   559→      totalIncrements: 0,
   560→      createdAt: now,
   561→      updatedAt: now,
   562→    };
   563→
   564→    this.metrics.set(name, counter);
   565→    logger.debug(`Created counter metric: ${name}`, { event: 'metric.created', metadata: { name, type: 'counter' } });
   566→    
   567→    return counter;
   568→  }
   569→
   570→  /**
   571→   * Increment a counter metric
   572→   * @param name - Metric name to increment
   573→   * @param amount - Amount to add (default: 1)
   574→   * @param labels - Additional labels for this increment
   575→   * @throws {AppError} If metric doesn't exist or is not a counter
   576→   */
   577→  incrementCounter(
   578→    name: string,
   579→    amount: number = 1,
   580→    labels: MetricLabels = {}
   581→  ): void {
   582→    const metric = this.metrics.get(name);
   583→    
   584→    if (!metric) {
   585→      // Auto-create if doesn't exist
   586→      this.createCounter(name);
   587→      return this.incrementCounter(name, amount, labels);
   588→    }
   589→    
   590→    if (metric.type !== 'counter') {
   591→      throw new AppError(
   592→        `Metric '${name}' is not a counter`,
   593→        ErrorCode.VALIDATION_ERROR,
   594→        { severity: 'warning' }
   595→      );
   596→    }
   597→
   598→    metric.value += amount;
   599→    metric.totalIncrements++;
   600→    metric.updatedAt = new Date();
   601→    
   602→    // Merge labels if provided
   603→    if (Object.keys(labels).length > 0) {
   604→      metric.labels = { ...metric.labels, ...labels };
   605→    }
   606→
   607→    // Record historical data point
   608→    this.recordHistoricalDataPoint(name, metric.value);
   609→  }

   611→  /**
   612→   * Get current counter value
   613→   * @param name - Counter metric name
   614→   * @returns Current counter value
   615→   */
   616→  getCounterValue(name: string): number {
   617→    const metric = this.metrics.get(name);
   618→    
   619→    if (!metric || metric.type !== 'counter') {
   620→      return 0;
   621→    }
   622→    
   623→    return metric.value;
   624→  }

   626→  // ============== Gauge Metrics ==============
   627→
   628→  /**
   629→   * Create a new gauge metric
   630→   * @param name - Unique metric name
   631→   * @param description - Human-readable description
   632→   * @param labels - Initial labels
   633→   * @returns The created gauge metric
   634→   */
   635→  createGauge(
   636→    name: string,
   637→    description: string = '',
   638→    labels: MetricLabels = {}
   639→  ): GaugeMetric {
   640→    if (this.metrics.has(name)) {
   641→      throw new AppError(
   642→        `Metric '${name}' already exists`,
   643→        ErrorCode.VALIDATION_ERROR,
   644→        { severity: 'warning' }
   645→      );
   646→    }
   647→
   648→    const now = new Date();
   649→    const gauge: GaugeMetric = {
   650→      name,
   651→      description,
   652→      type: 'gauge',
   653→      labels: { ...labels },
   654→      value: 0,
   655→      min: Infinity,
   656→      max: -Infinity,
   657→      createdAt: now,
   658→      updatedAt: now,
   659→    };
   660→
   661→    this.metrics.set(name, gauge);
   662→    logger.debug(`Created gauge metric: ${name}`, { event: 'metric.created', metadata: { name, type: 'gauge' } });
   663→    
   664→    return gauge;
   665→  }
   666→
   667→  /**
   668→   * Set a gauge metric value
   669→   * @param name - Gauge metric name
   670→   * @param value - Value to set
   671→   * @param labels - Additional labels
   672→   */
   673→  setGauge(name: string, value: number, labels: MetricLabels = {}): void {
   674→    let metric = this.metrics.get(name) as GaugeMetric | undefined;
   675→    
   676→    if (!metric) {
   677→      this.createGauge(name);
   678→      metric = this.metrics.get(name) as GaugeMetric;
   679→    }
   680→    
   681→    if (metric!.type !== 'gauge') {
   682→      throw new AppError(
   683→        `Metric '${name}' is not a gauge`,
   684→        ErrorCode.VALIDATION_ERROR,
   685→        { severity: 'warning' }
   686→      );
   687→    }
   688→
   689→    metric!.value = value;
   690→    metric!.min = Math.min(metric!.min, value);
   691→    metric!.max = Math.max(metric!.max, value);
   692→    metric!.updatedAt = new Date();
   693→    
   694→    if (Object.keys(labels).length > 0) {
   695→      metric!.labels = { ...metric!.labels, ...labels };
   696→    }
   697→
   698→    this.recordHistoricalDataPoint(name, value);
   699→  }

   701→  /**
   702→   * Increment a gauge metric
   703→   * @param name - Gauge metric name

   738→  // ============== Histogram Metrics ==============
   739→
   740→  /**
   741→   * Create a new histogram metric
   742→   * @param name - Unique metric name
   743→   * @param description - Human-readable description
   744→   * @param buckets - Bucket boundaries
   745→   * @param labels - Initial labels
   746→   * @returns The created histogram metric
   747→   */
   748→  createHistogram(
   749→    name: string,
   750→    description: string = '',
   751→    buckets: number[] = DEFAULT_LATENCY_BUCKETS,
   752→    labels: MetricLabels = {}
   753→  ): HistogramMetric {
   754→    if (this.metrics.has(name)) {
   755→      throw new AppError(
   756→        `Metric '${name}' already exists`,
   757→        ErrorCode.VALIDATION_ERROR,
   758→        { severity: 'warning' }
   759→      );
   760→    }
   761→
   762→    // Sort buckets and ensure they're positive
   763→    const sortedBuckets = [...buckets].filter(b => b > 0).sort((a, b) => a - b);
   764→
   765→    const now = new Date();
   766→    const histogram: HistogramMetric = {
   767→      name,
   768→      description,
   769→      type: 'histogram',
   770→      labels: { ...labels },
   771→      count: 0,
   772→      sum: 0,
   773→      buckets: sortedBuckets.map(upperBound => ({
   774→        upperBound,
   775→        count: 0,
   776→      })),
   777→      createdAt: now,
   778→      updatedAt: now,
   779→    };
   780→
   781→    this.metrics.set(name, histogram);
   782→    logger.debug(`Created histogram metric: ${name}`, { event: 'metric.created', metadata: { name, type: 'histogram' } });
   783→    
   784→    return histogram;
   785→  }

   787→  /**
   788→   * Observe a value in a histogram
   789→   * @param name - Histogram metric name
   790→   * @param value - Value to observe
   791→   * @param labels - Additional labels
   792→   */
   793→  observeHistogram(name: string, value: number, labels: MetricLabels = {}): void {
   794→    let metric = this.metrics.get(name) as HistogramMetric | undefined;
   795→    
   796→    if (!metric) {
   797→      this.createHistogram(name);
   798→      metric = this.metrics.get(name) as HistogramMetric;
   799→    }
   800→    
   801→    if (metric!.type !== 'histogram') {
   802→      throw new AppError(
   803→        `Metric '${name}' is not a histogram`,
   804→        ErrorCode.VALIDATION_ERROR,
   805→        { severity: 'warning' }
   806→      );
   807→    }
   808→
   809→    // Update basic stats
   810→    metric!.count++;
   811→    metric!.sum += value;
   812→    metric!.updatedAt = new Date();
   813→
   814→    // Find appropriate bucket
   815→    for (const bucket of metric!.buckets) {
   816→      if (value <= bucket.upperBound) {
   817→        bucket.count++;
   818→      }
   819→    }
   820→
   821→    // Merge labels
   822→    if (Object.keys(labels).length > 0) {
   823→      metric!.labels = { ...metric!.labels, ...labels };
   824→    }
   825→
   826→    // Record raw value for percentile calculations
   827→    this.recordHistoricalDataPoint(`${name}_raw`, value);
   828→  }

   830→  /**
   831→   * Get histogram statistics including percentiles
   832→   * @param name - Histogram metric name
   833→   * @returns Histogram with calculated percentiles
   834→   */
   835→  getHistogramStats(name: string): HistogramMetric | null {
   836→    const metric = this.metrics.get(name);
   837→    
   838→    if (!metric || metric.type !== 'histogram') {
   839→      return null;
   840→    }
   841→
   842→    // Get all observed values for percentile calculation
   843→    const rawData = this.historicalData.get(`${name}_raw`) || [];
   844→    const values = rawData.map(d => d.value).sort((a, b) => a - b);
   845→
   846→    if (values.length > 0) {
   847→      metric.percentiles = {
   848→        50: this.calculatePercentile(values, 50),
   849→        90: this.calculatePercentile(values, 90),
   850→        95: this.calculatePercentile(values, 95),
   851→        99: this.calculatePercentile(values, 99),
   852→      };
   853→    }
   854→
   855→    return metric;
   856→  }
   857→
   858→  /**
   859→   * Calculate percentile from sorted values
   860→   * @private
   861→   */
   862→  private calculatePercentile(sortedValues: number[], percentile: number): number {
   863→    if (sortedValues.length === 0) return 0;
   864→    
   865→    const index = (percentile / 100) * (sortedValues.length - 1);
   866→    const lower = Math.floor(index);
   867→    const upper = Math.ceil(index);
   868→    
   869→    if (lower === upper) {
   870→      return sortedValues[lower];
   871→    }
   872→    
   873→    return sortedValues[lower] + (sortedValues[upper] - sortedValues[lower]) * (index - lower);
   874→  }

   878→  /**
   879→   * Start a timer for performance measurement
   880→   * @param operation - Name of the operation being timed
   881→   * @returns Function to call when operation completes
   882→   * 
   883→   * @example
   884→   * ```typescript
   885→   * const endTimer = monitoringService.startTimer('database.query');
   886→   * // ... perform operation ...
   887→   * endTimer({ success: true });
   888→   * ```
   889→   */
   890→  startTimer(operation: string): (result?: { success: boolean; error?: string; labels?: MetricLabels }) => number {
   891→    const startTime = Date.now();
   892→    
   893→    return (result?: { success: boolean; error?: string; labels?: MetricLabels }): number => {
   894→      const endTime = Date.now();
   895→      const durationMs = endTime - startTime;
   896→      
   897→      const record: PerformanceRecord = {
   898→        operation,
   899→        durationMs,
   900→        timestamp: new Date(startTime),
   901→        success: result?.success ?? true,
   902→        error: result?.error,
   903→        labels: result?.labels,
   904→      };
   905→      
   906→      // Store performance record
   907→      const records = this.performanceRecords.get(operation) || [];
   908→      records.push(record);
   909→      this.performanceRecords.set(operation, records);
   910→      
   911→      // Update corresponding histogram if exists
   912→      const histogramName = `${operation}_duration_ms`;
   913→      if (this.metrics.has(histogramName)) {
   914→        this.observeHistogram(histogramName, durationMs, result?.labels);
   915→      }
   916→      
   917→      return durationMs;
   918→    };
   919→  }

   921→  /**
   922→   * Get aggregated performance statistics for an operation
   923→   * @param operation - Operation name
   924→   * @param query - Time window query options
   925→   * @returns Aggregated performance statistics
   926→   */
   927→  getPerformanceStats(operation: string, query?: TimeWindowQuery): PerformanceStats {
   928→    const records = this.getFilteredPerformanceRecords(operation, query);
   929→    
   930→    if (records.length === 0) {
   931→      return this.emptyPerformanceStats(operation);
   932→    }
   933→    
   934→    const durations = records.map(r => r.durationMs).sort((a, b) => a - b);
   935→    const successRecords = records.filter(r => r.success);
   936→    const errorRecords = records.filter(r => !r.success);
   937→    
   938→    // Calculate time window for throughput
   939→    const timeSpanMs = this.getTimeSpanMs(records, query);
   940→    
   941→    return {
   942→      operation,
   943→      totalCount: records.length,
   944→      successCount: successRecords.length,
   945→      errorCount: errorRecords.length,
   946→      avgDurationMs: durations.reduce((a, b) => a + b, 0) / durations.length,
   947→      minDurationMs: durations[0],
   948→      maxDurationMs: durations[durations.length - 1],
   949→      p50Ms: this.calculatePercentile(durations, 50),
   950→      p95Ms: this.calculatePercentile(durations, 95),
   951→      p99Ms: this.calculatePercentile(durations, 99),
   952→      errorRate: (errorRecords.length / records.length) * 100,
   953→      throughput: timeSpanMs > 0 ? (records.length / (timeSpanMs / 1000)) : 0,
   954→    };
   955→  }

   958→   * Get performance stats for all operations

   960→   * @returns Map of operation names to their stats

   970→  }

   993→  /**
   994→   * Get filtered performance records based on time window
   995→   * @private
   996→   */
   997→  private getFilteredPerformanceRecords(operation: string, query?: TimeWindowQuery): PerformanceRecord[] {
   998→    let records = this.performanceRecords.get(operation) || [];
   999→    
  1000→    if (query) {
  1001→      const { startTime, endTime } = this.getTimeBounds(query.window, query.startTime, query.endTime);
  1002→      records = records.filter(r => r.timestamp >= startTime && r.timestamp <= endTime);
  1003→      
  1004→      // Apply label filters
  1005→      if (query.labelFilters) {
  1006→        records = records.filter(record => {
  1007→          if (!record.labels) return false;
  1008→          return Object.entries(query.labelFilters!).every(
  1009→            ([key, value]) => record.labels![key] === value
  1010→          );
  1011→        });
  1012→      }
  1013→    }
  1014→    
  1015→    return records;
  1016→  }

  1018→  /**
  1019→   * Calculate time span in milliseconds from records
  1020→   * @private
  1021→   */
  1022→  private getTimeSpanMs(records: PerformanceRecord[], query?: TimeWindowQuery): number {
  1023→    if (query?.startTime && query?.endTime) {
  1024→      return query.endTime.getTime() - query.startTime.getTime();
  1025→    }
  1026→    
  1027→    if (records.length < 2) {
  1028→      return records.length === 1 ? 60000 : 0; // Assume 1 minute for single record
  1029→    }
  1030→    
  1031→    return records[records.length - 1].timestamp.getTime() - records[0].timestamp.getTime();
  1032→  }

  1060→  /**
  1061→   * Get error rates for all tracked error codes
  1062→   * @param window - Time window for rate calculation
  1063→   * @returns Array of error rate entries
  1064→   */
  1065→  getErrorRates(window: TimeWindow = '1h'): ErrorRateEntry[] {
  1066→    const { startTime } = this.getTimeBounds(window);
  1067→    const { startTime: prevStartTime } = this.getTimeBounds(window, undefined, startTime);
  1068→    
  1069→    const entries: ErrorRateEntry[] = [];
  1070→    
  1071→    for (const [errorCode, data] of Array.from(this.errorCounts.entries())) {
  1072→      const currentCount = data.timestamps.filter(t => t >= startTime).length;
  1073→      const previousCount = data.timestamps.filter(t => t >= prevStartTime && t < startTime).length;
  1074→      
  1075→      // Calculate rate (errors per minute)
  1076→      const windowMinutes = this.getWindowMinutes(window);
  1077→      const rate = currentCount / windowMinutes;
  1078→      
  1079→      // Determine trend
  1080→      let trend: 'up' | 'down' | 'stable' = 'stable';
  1081→      if (previousCount > 0) {
  1082→        const changeRatio = currentCount / previousCount;
  1083→        if (changeRatio > 1.2) trend = 'up';
  1084→        else if (changeRatio < 0.8) trend = 'down';
  1085→      }
  1086→      
  1087→      entries.push({
  1088→        errorCode,
  1089→        count: currentCount,
  1090→        rate,
  1091→        trend,
  1092→        previousCount,
  1093→      });
  1094→    }
  1095→    
  1096→    // Sort by count descending
  1097→    entries.sort((a, b) => b.count - a.count);
  1098→    
  1099→    return entries;
  1100→  }

  1192→    return {

  1203→   * Perform deep health check (all components including dependencies)
  1204→   * @returns Complete system health status
  1205→   */
  1206→  async checkDeepHealth(): Promise<SystemHealth> {
  1207→    const startTime = Date.now();
  1208→    const checks: HealthCheckResult[] = [];
  1209→    
  1210→    // Run liveness check
  1211→    checks.push(await this.checkLiveness());

  1213→    // Run readiness check
  1214→    const readiness = await this.checkReadiness();

  1218→    for (const [component, checkFn] of Array.from(this.healthChecks.entries())) {
  1219→      try {
  1220→        // Add timeout to prevent hanging checks
  1221→        const result = await Promise.race([
  1222→          checkFn(),
  1223→          new Promise<HealthCheckResult>((resolve) =>
  1224→            setTimeout(() => resolve({
  1225→              component,
  1226→              status: 'unhealthy' as HealthStatus,
  1227→              message: 'Health check timeout',
  1228→              timestamp: new Date(),
  1229→            }), HEALTH_CHECK_TIMEOUT_MS)
  1230→          ),
  1231→        ]);
  1232→        checks.push(result);
  1233→      } catch (error) {
  1234→        checks.push({
  1235→          component,
  1236→          status: 'unhealthy',
  1237→          message: 'Health check threw error',
  1238→          error: error instanceof Error ? error.message : String(error),
  1239→          timestamp: new Date(),
  1240→        });
  1241→      }
  1242→    }
  1243→    
  1244→    // Determine overall status
  1245→    const statuses = checks.map(c => c.status);
  1246→    let overallStatus: HealthStatus = 'healthy';
  1247→    
  1248→    if (statuses.includes('unhealthy')) {
  1249→      overallStatus = 'unhealthy';
  1250→    } else if (statuses.includes('degraded')) {
  1251→      overallStatus = 'degraded';
  1252→    } else if (statuses.includes('unknown')) {
  1253→      overallStatus = 'unknown';
  1254→    }
  1255→    
  1256→    const uptimeSeconds = (Date.now() - APPLICATION_START_TIME) / 1000;
  1257→    
  1258→    return {
  1259→      status: overallStatus,
  1260→      checks,
  1261→      uptimeSeconds,
  1262→      timestamp: new Date(),
  1263→      version: APP_VERSION,
  1264→    };
  1265→  }

  1320→      timeWindow: options.timeWindow || '5m',

  1322→      onTrigger: options.onTrigger,
  1323→      createdAt: new Date(),
  1324→      triggerCount: 0,
  1325→    };
  1326→
  1327→    this.alerts.set(alert.id, alert);
  1328→    logger.info(`Created alert: ${options.id}`, { event: 'alert.created', metadata: { alertId: options.id } });
  1329→    
  1330→    return alert;
  1331→  }

  1333→  /**
  1334→   * Remove an alert threshold
  1335→   * @param id - Alert ID to remove
  1336→   */
  1337→  removeAlert(id: string): boolean {
  1338→    const deleted = this.alerts.delete(id);
  1339→    this.activeAlerts.delete(id);
  1340→    
  1341→    if (deleted) {
  1342→      logger.info(`Removed alert: ${id}`, { event: 'alert.removed', metadata: { alertId: id } });
  1343→    }
  1344→    
  1345→    return deleted;
  1346→  }

  1348→  /**
  1349→   * Evaluate a specific alert against current metric values
  1350→   * @param alertId - Alert ID to evaluate
  1351→   * @returns Whether the alert is currently triggering
  1352→   */
  1353→  evaluateAlert(alertId: string): boolean {
  1354→    const alert = this.alerts.get(alertId);
  1355→    if (!alert) {
  1356→      return false;
  1357→    }
  1358→
  1359→    const metric = this.metrics.get(alert.metricName);
  1360→    if (!metric) {
  1361→      return false;
  1362→    }
  1363→
  1364→    // Get value based on metric type
  1365→    let currentValue: number;
  1366→    if (metric.type === 'histogram') {
  1367→      currentValue = (metric as HistogramMetric).sum / Math.max((metric as HistogramMetric).count, 1);
  1368→    } else {
  1369→      currentValue = (metric as CounterMetric | GaugeMetric).value;
  1370→    }
  1371→    const isTriggered = this.evaluateCondition(currentValue, alert.condition, alert.value);
  1372→
  1373→    if (isTriggered && !alert.isActive) {
  1374→      // Alert is newly triggered
  1375→      alert.isActive = true;
  1376→      alert.lastTriggeredAt = new Date();
  1377→      alert.triggerCount++;
  1378→
  1379→      const activeAlert: ActiveAlert = {
  1380→        ...alert,
  1381→        currentValue,
  1382→        triggeredAt: alert.lastTriggeredAt,
  1383→      };
  1384→
  1385→      this.activeAlerts.set(alertId, activeAlert);
  1386→
  1387→      logger.warn(`Alert triggered: ${alertId}`, {
  1388→        event: 'alert.triggered',
  1389→        metadata: {
  1390→          alertId,
  1391→          metricName: alert.metricName,
  1392→          currentValue,
  1393→          threshold: alert.value,
  1394→          severity: alert.severity,
  1395→        },
  1396→      });
  1397→
  1398→      // Execute callback if defined
  1399→      if (alert.onTrigger) {
  1400→        try {
  1401→          alert.onTrigger(activeAlert);
  1402→        } catch (error) {
  1403→          logger.error('Alert callback execution failed', {
  1404→            event: 'alert.callback_error',
  1405→            error: error instanceof Error ? error : new AppError(String(error)),
  1406→          });
  1407→        }
  1408→      }
  1409→    } else if (!isTriggered && alert.isActive) {
  1410→      // Alert is resolved
  1411→      alert.isActive = false;
  1412→      const activeAlert = this.activeAlerts.get(alertId);
  1413→      if (activeAlert) {
  1414→        activeAlert.resolvedAt = new Date();
  1415→      }
  1416→
  1417→      logger.info(`Alert resolved: ${alertId}`, {
  1418→        event: 'alert.resolved',
  1419→        metadata: { alertId, resolvedAt: new Date().toISOString() },
  1420→      });
  1421→    }
  1422→
  1423→    return isTriggered;
  1424→  }
  1425→
  1426→  /**
  1427→   * Evaluate all registered alerts
  1428→   * @returns Array of newly triggered alerts
  1429→   */
  1430→  evaluateAllAlerts(): ActiveAlert[] {
  1431→    const newlyTriggered: ActiveAlert[] = [];
  1432→
  1433→    for (const alertId of Array.from(this.alerts.keys())) {
  1434→      const wasActive = this.alerts.get(alertId)?.isActive;
  1435→      const isTriggered = this.evaluateAlert(alertId);
  1436→      
  1437→      if (isTriggered && !wasActive) {
  1438→        const active = this.activeAlerts.get(alertId);
  1439→        if (active) {
  1440→          newlyTriggered.push(active);
  1441→        }
  1442→      }
  1443→    }
  1444→
  1445→    return newlyTriggered;
  1446→  }

  1466→   * @private
  1467→   */
  1468→  private evaluateCondition(value: number, condition: string, threshold: number): boolean {
  1469→    switch (condition) {
  1470→      case 'gt': return value > threshold;

  1473→      case 'lte': return value <= threshold;

  1881→/**
  1882→ * Decorator factory for automatic performance tracking
  1883→ * Can be used to wrap methods for automatic timing
  1884→ * 
  1885→ * @example
  1886→ * ```typescript
  1887→ * class PaymentService {
  1888→ *   @trackPerformance('payment.process')
  1889→ *   async processPayment(amount: number) {
  1890→ *     // Method implementation
  1891→ *   }
  1892→ * }
  1893→ * ```
  1894→ */
  1895→export function trackPerformance(operation: string) {
  1896→  return function (
  1897→    target: unknown,
  1898→    propertyKey: string,
  1899→    descriptor: PropertyDescriptor
  1900→  ): PropertyDescriptor {
  1901→    const originalMethod = descriptor.value;
  1902→    
  1903→    descriptor.value = async function (...args: unknown[]) {
  1904→      const endTimer = monitoringService.startTimer(operation);
  1905→      
  1906→      try {
  1907→        const result = await originalMethod.apply(this, args);
  1908→        endTimer({ success: true });
  1909→        return result;
  1910→      } catch (error) {
  1911→        endTimer({
  1912→          success: false,
  1913→          error: error instanceof Error ? error.message : String(error),
  1914→        });
  1915→        throw error;
  1916→      }
  1917→    };
  1918→    
  1919→    return descriptor;
  1920→  };
  1921→}
  1922→
  1923→/**
  1924→ * Higher-order function for wrapping async operations with performance tracking
  1925→ * 
  1926→ * @param operation - Operation name for tracking
  1927→ * @param fn - Async function to wrap
  1928→ * @returns Wrapped function with automatic tracking
  1929→ * 
  1930→ * @example
  1931→ * ```typescript
  1932→ * const trackedCall = trackAsyncOperation('db.query', () => database.query(sql));
  1933→ * const result = await trackedCall();
  1934→ * ```
  1935→ */
  1936→export async function trackAsyncOperation<T>(
  1937→  operation: string,
  1938→  fn: () => Promise<T>
  1939→): Promise<T> {
  1940→  const endTimer = monitoringService.startTimer(operation);
  1941→  
  1942→  try {
  1943→    const result = await fn();
  1944→    endTimer({ success: true });
  1945→    return result;
  1946→  } catch (error) {
  1947→    endTimer({
  1948→      success: false,
  1949→      error: error instanceof Error ? error.message : String(error),
  1950→    });
  1951→    throw error;
  1952→  }
  1953→}
