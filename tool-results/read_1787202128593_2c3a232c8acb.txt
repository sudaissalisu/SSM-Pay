   871→   871→export interface AppError {
   872→   872→  code: string;
   873→   873→  message: string;
   874→   874→  statusCode: number;
   875→   875→  details?: Record<string, unknown>;
   876→   876→  requestId?: string;
   877→   877→  stack?: string;
   878→   878→  cause?: Error;
   879→   879→  suggestions?: string[];
   880→   880→  documentationUrl?: string;
   881→   881→  retryable: boolean;
   882→   882→}
   883→   883→
   884→   884→export class SSMError extends Error implements AppError {
   885→   885→  code: string;
   886→   886→  statusCode: number;
   887→   887→  details?: Record<string, unknown>;
   888→   888→  requestId?: string;
   889→   889→  suggestions?: string[];
   890→   890→  documentationUrl?: string;
   891→   891→  retryable: boolean;
   892→   892→
   893→   893→  constructor(
   894→   894→    message: string,
   895→   895→    options: {
   896→   896→      code: string;
   897→   897→      statusCode?: number;
   898→   898→      details?: Record<string, unknown>;
   899→   899→      cause?: Error;
   900→   900→      suggestions?: string[];
   901→   901→      documentationUrl?: string;
   902→   902→    }
   903→   903→  ) {
   904→   904→    super(message);
   905→   905→    this.name = 'SSMError';
   906→   906→    this.code = options.code;
   907→   907→    this.statusCode = options.statusCode || 500;
   908→   908→    this.message = message;
   909→   909→    this.details = options.details;
   910→   910→    this.cause = options.cause;
   911→   911→    this.suggestions = options.suggestions || [];
   912→   912→    this.documentationUrl = options.documentationUrl;
   913→   913→    this.retryable = false;
   914→   914→
   915→   915→    if (Error.captureStackTrace) {
   916→   916→      Error.captureStackTrace(this, SSMError);
   917→   917→    }
   918→   918→  }
   919→   919→
   920→   920→  toJSON(): AppError {
   921→   921→    return {
   922→   922→      code: this.code,
   923→   923→      message: this.message,
   924→   924→      statusCode: this.statusCode,
   925→   925→      details: this.details,
   926→   926→      requestId: this.requestId,
   927→   927→      stack: this.stack,
   928→   928→      suggestions: this.suggestions,
   929→   929→      documentationUrl: this.documentationUrl,
   930→   930→      retryable: this.retryable,
   931→   931→    };
   932→   932→  }
   933→   933→}
