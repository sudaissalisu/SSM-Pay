   465→   465→  /**
   466→   466→   * Start the background cleanup interval
   467→   467→   * @private
   468→   468→   */
   469→   469→  private startCleanupInterval(): void {
   470→   470→    // Run cleanup every 5 minutes
   471→   471→    this.cleanupInterval = setInterval(() => {
   472→   472→      try {
   473→   473→        this.cleanupOldData();
   474→   474→        this.evaluateAllAlerts();
   475→   475→      } catch (error) {
   476→   476→        logger.error('Error in monitoring cleanup cycle', {
   477→   477→          event: 'monitoring.error',
   478→   478→          error: error instanceof Error ? error : new AppError(String(error)),
   479→   479→        });
   480→   480→      }
   481→   481→    }, 5 * 60 * 1000);
   482→   482→    
   483→   483→    // Don't prevent process exit
   484→   484→    if (this.cleanupInterval.unref) {
   485→   485→      this.cleanupInterval.unref();
   486→   486→    }
   487→   487→  }

   489→   489→  /**
   490→   490→   * Clean up old data beyond retention period
   491→   491→   * @private
   492→   492→   */
   493→   493→  private cleanupOldData(): void {
   494→   494→    const cutoffTime = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000); // 7 days ago
   495→   495→    
   496→   496→    // Clean up performance records
   497→   497→    for (const [key, records] of Array.from(this.performanceRecords.entries())) {
   498→   498→      const filtered = records.filter(r => r.timestamp > cutoffTime);
   499→   499→      if (filtered.length !== records.length) {
   500→   500→        this.performanceRecords.set(key, filtered);
   501→   501→      }
   502→   502→    }
   503→   503→    
   504→   504→    // Clean up historical data
   505→   505→    for (const [key, dataPoints] of Array.from(this.historicalData.entries())) {
   506→   506→      const filtered = dataPoints.filter(d => d.timestamp > cutoffTime);
   507→   507→      if (filtered.length !== dataPoints.length) {
   508→   508→        this.historicalData.set(key, filtered.slice(-MAX_DATA_POINTS_PER_METRIC));
   509→   509→      }
   510→   510→    }
   511→   511→    
   512→   512→    // Clean up error timestamps
   513→   513→    for (const [key, data] of Array.from(this.errorCounts.entries())) {
   514→   514→      const filteredTimestamps = data.timestamps.filter(t => t > cutoffTime);
   515→   515→      if (filteredTimestamps.length !== data.timestamps.length) {
   516→   516→        this.errorCounts.set(key, {
   517→   517→          count: filteredTimestamps.length,
   518→   518→          timestamps: filteredTimestamps,
   519→   519→        });
   520→   520→      }
   521→   521→    }
   522→   522→
   523→   523→    logger.debug('Completed monitoring data cleanup', {
   524→   524→      event: 'monitoring.cleanup',
   525→   525→      metadata: { cutoffTime: cutoffTime.toISOString() },
   526→   526→    });
   527→   527→  }

   531→   531→  /**

   540→   540→    name: string,
   541→   541→    description: string = '',
   542→   542→    labels: MetricLabels = {}
   543→   543→  ): CounterMetric {
   544→   544→    if (this.metrics.has(name)) {
   545→   545→      throw new AppError(
   546→   546→        `Metric '${name}' already exists`,
   547→   547→        ErrorCode.VALIDATION_ERROR,
   548→   548→        { severity: 'warning' }
   549→   549→      );
   550→   550→    }
   551→   551→
   552→   552→    const now = new Date();
   553→   553→    const counter: CounterMetric = {
   554→   554→      name,
   555→   555→      description,
   556→   556→      type: 'counter',
   557→   557→      labels: { ...labels },
   558→   558→      value: 0,
   559→   559→      totalIncrements: 0,
   560→   560→      createdAt: now,
   561→   561→      updatedAt: now,

   748→   748→  createHistogram(
   749→   749→    name: string,
   750→   750→    description: string = '',
   751→   751→    buckets: number[] = DEFAULT_LATENCY_BUCKETS,
   752→   752→    labels: MetricLabels = {}

   754→   754→    if (this.metrics.has(name)) {
   755→   755→      throw new AppError(

   760→   760→    }
   761→   761→
   762→   762→    // Sort buckets and ensure they're positive
   763→   763→    const sortedBuckets = [...buckets].filter(b => b > 0).sort((a, b) => a - b);

   772→   772→      sum: 0,

   774→   774→        upperBound,
   775→   775→        count: 0,
   776→   776→      })),
   777→   777→      createdAt: now,
   778→   778→      updatedAt: now,
   779→   779→    };
   780→   780→
   781→   781→    this.metrics.set(name, histogram);

   788→   788→   * Observe a value in a histogram
   789→   789→   * @param name - Histogram metric name
   790→   790→   * @param value - Value to observe
   791→   791→   * @param labels - Additional labels
   792→   792→   */
   793→   793→  observeHistogram(name: string, value: number, labels: MetricLabels = {}): void {
   794→   794→    let metric = this.metrics.get(name) as HistogramMetric | undefined;
   795→   795→    
   796→   796→    if (!metric) {
   797→   797→      this.createHistogram(name);
   798→   798→      metric = this.metrics.get(name) as HistogramMetric;
   799→   799→    }
   800→   800→    
   801→   801→    if (metric!.type !== 'histogram') {
   802→   802→      throw new AppError(
   803→   803→        `Metric '${name}' is not a histogram`,
   804→   804→        ErrorCode.VALIDATION_ERROR,
   805→   805→        { severity: 'warning' }
   806→   806→      );
   807→   807→    }
   808→   808→
   809→   809→    // Update basic stats
   810→   810→    metric!.count++;
   811→   811→    metric!.sum += value;
   812→   812→    metric!.updatedAt = new Date();
   813→   813→
   814→   814→    // Find appropriate bucket
   815→   815→    for (const bucket of metric!.buckets) {
   816→   816→      if (value <= bucket.upperBound) {
   817→   817→        bucket.count++;
   818→   818→      }
   819→   819→    }
   820→   820→
   821→   821→    // Merge labels
   822→   822→    if (Object.keys(labels).length > 0) {
   823→   823→      metric!.labels = { ...metric!.labels, ...labels };
   824→   824→    }
   825→   825→
   826→   826→    // Record raw value for percentile calculations
   827→   827→    this.recordHistoricalDataPoint(`${name}_raw`, value);
   828→   828→  }

   833→   833→   * @returns Histogram with calculated percentiles
   834→   834→   */
   835→   835→  getHistogramStats(name: string): HistogramMetric | null {
   836→   836→    const metric = this.metrics.get(name);
   837→   837→    
   838→   838→    if (!metric || metric.type !== 'histogram') {
   839→   839→      return null;
   840→   840→    }
   841→   841→
   842→   842→    // Get all observed values for percentile calculation
   843→   843→    const rawData = this.historicalData.get(`${name}_raw`) || [];
   844→   844→    const values = rawData.map(d => d.value).sort((a, b) => a - b);
   845→   845→
   846→   846→    if (values.length > 0) {
   847→   847→      metric.percentiles = {
   848→   848→        50: this.calculatePercentile(values, 50),
   849→   849→        90: this.calculatePercentile(values, 90),
   850→   850→        95: this.calculatePercentile(values, 95),
   851→   851→        99: this.calculatePercentile(values, 99),
   852→   852→      };
   853→   853→    }
   854→   854→
   855→   855→    return metric;
   856→   856→  }

   858→   858→  /**
   859→   859→   * Calculate percentile from sorted values
   860→   860→   * @private
   861→   861→   */
   862→   862→  private calculatePercentile(sortedValues: number[], percentile: number): number {
   863→   863→    if (sortedValues.length === 0) return 0;
   864→   864→    
   865→   865→    const index = (percentile / 100) * (sortedValues.length - 1);
   866→   866→    const lower = Math.floor(index);
   867→   867→    const upper = Math.ceil(index);
   868→   868→    
   869→   869→    if (lower === upper) {
   870→   870→      return sortedValues[lower];
   871→   871→    }
   872→   872→    
   873→   873→    return sortedValues[lower] + (sortedValues[upper] - sortedValues[lower]) * (index - lower);
   874→   874→  }

   878→   878→  /**
   879→   879→   * Start a timer for performance measurement
   880→   880→   * @param operation - Name of the operation being timed
   881→   881→   * @returns Function to call when operation completes
   882→   882→   * 
   883→   883→   * @example
   884→   884→   * ```typescript
   885→   885→   * const endTimer = monitoringService.startTimer('database.query');
   886→   886→   * // ... perform operation ...
   887→   887→   * endTimer({ success: true });
   888→   888→   * ```
   889→   889→   */
   890→   890→  startTimer(operation: string): (result?: { success: boolean; error?: string; labels?: MetricLabels }) => number {
   891→   891→    const startTime = Date.now();
   892→   892→    
   893→   893→    return (result?: { success: boolean; error?: string; labels?: MetricLabels }): number => {
   894→   894→      const endTime = Date.now();
   895→   895→      const durationMs = endTime - startTime;
   896→   896→      
   897→   897→      const record: PerformanceRecord = {
   898→   898→        operation,

   902→   902→        error: result?.error,
   903→   903→        labels: result?.labels,
   904→   904→      };
   905→   905→      
   906→   906→      // Store performance record
   907→   907→      const records = this.performanceRecords.get(operation) || [];
   908→   908→      records.push(record);
   909→   909→      this.performanceRecords.set(operation, records);
   910→   910→      
   911→   911→      // Update corresponding histogram if exists
   912→   912→      const histogramName = `${operation}_duration_ms`;
   913→   913→      if (this.metrics.has(histogramName)) {
   914→   914→        this.observeHistogram(histogramName, durationMs, result?.labels);
   915→   915→      }
   916→   916→      
   917→   917→      return durationMs;
   918→   918→    };
   919→   919→  }

   921→   921→  /**
   922→   922→   * Get aggregated performance statistics for an operation
   923→   923→   * @param operation - Operation name
   924→   924→   * @param query - Time window query options
   925→   925→   * @returns Aggregated performance statistics
   926→   926→   */
   927→   927→  getPerformanceStats(operation: string, query?: TimeWindowQuery): PerformanceStats {
   928→   928→    const records = this.getFilteredPerformanceRecords(operation, query);
   929→   929→    
   930→   930→    if (records.length === 0) {
   931→   931→      return this.emptyPerformanceStats(operation);
   932→   932→    }
   933→   933→    
   934→   934→    const durations = records.map(r => r.durationMs).sort((a, b) => a - b);
   935→   935→    const successRecords = records.filter(r => r.success);
   936→   936→    const errorRecords = records.filter(r => !r.success);
   937→   937→    
   938→   938→    // Calculate time window for throughput
   939→   939→    const timeSpanMs = this.getTimeSpanMs(records, query);
   940→   940→    
   941→   941→    return {
   942→   942→      operation,
   943→   943→      totalCount: records.length,
   944→   944→      successCount: successRecords.length,
   945→   945→      errorCount: errorRecords.length,
   946→   946→      avgDurationMs: durations.reduce((a, b) => a + b, 0) / durations.length,
   947→   947→      minDurationMs: durations[0],
   948→   948→      maxDurationMs: durations[durations.length - 1],
   949→   949→      p50Ms: this.calculatePercentile(durations, 50),
   950→   950→      p95Ms: this.calculatePercentile(durations, 95),
   951→   951→      p99Ms: this.calculatePercentile(durations, 99),
   952→   952→      errorRate: (errorRecords.length / records.length) * 100,
   953→   953→      throughput: timeSpanMs > 0 ? (records.length / (timeSpanMs / 1000)) : 0,
   954→   954→    };
   955→   955→  }

   994→   994→   * Get filtered performance records based on time window
   995→   995→   * @private
   996→   996→   */
   997→   997→  private getFilteredPerformanceRecords(operation: string, query?: TimeWindowQuery): PerformanceRecord[] {
   998→   998→    let records = this.performanceRecords.get(operation) || [];
   999→   999→    
  1000→  1000→    if (query) {
  1001→  1001→      const { startTime, endTime } = this.getTimeBounds(query.window, query.startTime, query.endTime);
  1002→  1002→      records = records.filter(r => r.timestamp >= startTime && r.timestamp <= endTime);
  1003→  1003→      
  1004→  1004→      // Apply label filters
  1005→  1005→      if (query.labelFilters) {
  1006→  1006→        records = records.filter(record => {
  1007→  1007→          if (!record.labels) return false;
  1008→  1008→          return Object.entries(query.labelFilters!).every(
  1009→  1009→            ([key, value]) => record.labels![key] === value
  1010→  1010→          );
  1011→  1011→        });
  1012→  1012→      }
  1013→  1013→    }
  1014→  1014→    
  1015→  1015→    return records;
  1016→  1016→  }

  1020→  1020→   * @private
  1021→  1021→   */
  1022→  1022→  private getTimeSpanMs(records: PerformanceRecord[], query?: TimeWindowQuery): number {
  1023→  1023→    if (query?.startTime && query?.endTime) {
  1024→  1024→      return query.endTime.getTime() - query.startTime.getTime();
  1025→  1025→    }
  1026→  1026→    
  1027→  1027→    if (records.length < 2) {
  1028→  1028→      return records.length === 1 ? 60000 : 0; // Assume 1 minute for single record
  1029→  1029→    }
  1030→  1030→    
  1031→  1031→    return records[records.length - 1].timestamp.getTime() - records[0].timestamp.getTime();
  1032→  1032→  }

  1062→  1062→   * @param window - Time window for rate calculation
  1063→  1063→   * @returns Array of error rate entries
  1064→  1064→   */
  1065→  1065→  getErrorRates(window: TimeWindow = '1h'): ErrorRateEntry[] {
  1066→  1066→    const { startTime } = this.getTimeBounds(window);
  1067→  1067→    const { startTime: prevStartTime } = this.getTimeBounds(window, undefined, startTime);
  1068→  1068→    
  1069→  1069→    const entries: ErrorRateEntry[] = [];
  1070→  1070→

  1092→  1092→        previousCount,
  1093→  1093→      });
  1094→  1094→    }
  1095→  1095→

  1348→  1348→  /**
  1349→  1349→   * Evaluate a specific alert against current metric values

  1355→  1355→    if (!alert) {

  1360→  1360→    if (!metric) {

  1368→  1368→    } else {

  1378→  1378→
  1379→  1379→      const activeAlert: ActiveAlert = {
  1380→  1380→        ...alert,
  1381→  1381→        currentValue,
  1382→  1382→        triggeredAt: alert.lastTriggeredAt,

  1881→  1881→/**
  1882→  1882→ * Decorator factory for automatic performance tracking
  1883→  1883→ * Can be used to wrap methods for automatic timing
  1884→  1884→ * 
  1885→  1885→ * @example
  1886→  1886→ * ```typescript
  1887→  1887→ * class PaymentService {
  1888→  1888→ *   @trackPerformance('payment.process')
  1889→  1889→ *   async processPayment(amount: number) {
  1890→  1890→ *     // Method implementation
  1891→  1891→ *   }
  1892→  1892→ * }
  1893→  1893→ * ```
  1894→  1894→ */
  1895→  1895→export function trackPerformance(operation: string) {
  1896→  1896→  return function (
  1897→  1897→    target: unknown,
  1898→  1898→    propertyKey: string,
  1899→  1899→    descriptor: PropertyDescriptor
  1900→  1900→  ): PropertyDescriptor {
  1901→  1901→    const originalMethod = descriptor.value;
  1902→  1902→    
  1903→  1903→    descriptor.value = async function (...args: unknown[]) {
  1904→  1904→      const endTimer = monitoringService.startTimer(operation);
  1905→  1905→      
  1906→  1906→      try {
  1907→  1907→        const result = await originalMethod.apply(this, args);
  1908→  1908→        endTimer({ success: true });
  1909→  1909→        return result;
  1910→  1910→      } catch (error) {
  1911→  1911→        endTimer({
  1912→  1912→          success: false,
  1913→  1913→          error: error instanceof Error ? error.message : String(error),
  1914→  1914→        });
  1915→  1915→        throw error;
  1916→  1916→      }
  1917→  1917→    };
  1918→  1918→    
  1919→  1919→    return descriptor;
  1920→  1920→  };
  1921→  1921→}
  1922→  1922→
  1923→  1923→/**
  1924→  1924→ * Higher-order function for wrapping async operations with performance tracking
  1925→  1925→ * 
  1926→  1926→ * @param operation - Operation name for tracking
  1927→  1927→ * @param fn - Async function to wrap
  1928→  1928→ * @returns Wrapped function with automatic tracking
  1929→  1929→ * 
  1930→  1930→ * @example
  1931→  1931→ * ```typescript
  1932→  1932→ * const trackedCall = trackAsyncOperation('db.query', () => database.query(sql));
  1933→  1933→ * const result = await trackedCall();
  1934→  1934→ * ```
  1935→  1935→ */
  1936→  1936→export async function trackAsyncOperation<T>(
  1937→  1937→  operation: string,
  1938→  1938→  fn: () => Promise<T>
  1939→  1939→): Promise<T> {
  1940→  1940→  const endTimer = monitoringService.startTimer(operation);
  1941→  1941→  
  1942→  1942→  try {
  1943→  1943→    const result = await fn();
  1944→  1944→    endTimer({ success: true });
  1945→  1945→    return result;
  1946→  1946→  } catch (error) {
  1947→  1947→    endTimer({
  1948→  1948→      success: false,
  1949→  1949→      error: error instanceof Error ? error.message : String(error),
  1950→  1950→    });
  1951→  1951→    throw error;
  1952→  1952→  }
  1953→  1953→}
