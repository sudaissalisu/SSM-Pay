   540→   540→/**
   541→   541→ * Calculate the mean of an array of numbers
   542→   542→ * @param values - Array of numeric values
   543→   543→ * @returns The arithmetic mean
   544→   544→ * @throws {AppError} If array is empty
   545→   545→ */
   546→   546→function calculateMean(values: number[]): number {
   547→   547→  if (values.length === 0) {
   548→   548→    throw new AppError('Cannot calculate mean of empty array', ErrorCode.VALIDATION_ERROR);
   549→   549→  }
   550→   550→  return values.reduce((sum, val) => sum + val, 0) / values.length;
   551→   551→}
   552→   552→
   553→   553→/**
   554→   554→ * Calculate the standard deviation of an array of numbers
   555→   555→ * @param values - Array of numeric values
   556→   556→ * @param usePopulation - Whether to use population (N) vs sample (N-1) standard deviation
   557→   557→ * @returns The standard deviation
   558→   558→ * @throws {AppError} If array has fewer than 2 elements
   559→   559→ */
   560→   560→function calculateStandardDeviation(values: number[], usePopulation: boolean = false): number {
   561→   561→  if (values.length < 2) {
   562→   562→    throw new AppError('Cannot calculate standard deviation with fewer than 2 values', ErrorCode.VALIDATION_ERROR);
   563→   563→  }
   564→   564→  const mean = calculateMean(values);
   565→   565→  const squaredDifferences = values.map(val => Math.pow(val - mean, 2));
   566→   566→  const variance = squaredDifferences.reduce((sum, val) => sum + val, 0) / (usePopulation ? values.length : values.length - 1);
   567→   567→  return Math.sqrt(variance);
   568→   568→}
   569→   569→
   570→   570→/**
   571→   571→ * Calculate the median of an array of numbers
   572→   572→ * @param values - Array of numeric values
   573→   573→ * @returns The median value
   574→   574→ * @throws {AppError} If array is empty
   575→   575→ */
   576→   576→function calculateMedian(values: number[]): number {
   577→   577→  if (values.length === 0) {
   578→   578→    throw new AppError('Cannot calculate median of empty array', ErrorCode.VALIDATION_ERROR);
   579→   579→  }
   580→   580→  const sorted = [...values].sort((a, b) => a - b);
   581→   581→  const mid = Math.floor(sorted.length / 2);
   582→   582→  return sorted.length % 2 !== 0 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
   583→   583→}
   584→   584→
   585→   585→/**
   586→   586→ * Calculate the Median Absolute Deviation (MAD)
   587→   587→ * @param values - Array of numeric values
   588→   588→ * @returns The MAD value
   589→   589→ * @throws {AppError} If array is empty
   590→   590→ */
   591→   591→function calculateMAD(values: number[]): number {
   592→   592→  const median = calculateMedian(values);
   593→   593→  const deviations = values.map(val => Math.abs(val - median));
   594→   594→  return calculateMedian(deviations);
   595→   595→}

   597→   597→/**
   598→   598→ * Calculate quartiles Q1 and Q3
   599→   599→ * @param values - Array of numeric values
   600→   600→ * @returns Object containing Q1 and Q3
   601→   601→ * @throws {AppError} If array has fewer than 4 elements
   602→   602→ */
   603→   603→function calculateQuartiles(values: number[]): { q1: number; q3: number } {
   604→   604→  if (values.length < 4) {
   605→   605→    throw new AppError('Cannot calculate quartiles with fewer than 4 values', ErrorCode.VALIDATION_ERROR);
   606→   606→  }
   607→   607→  const sorted = [...values].sort((a, b) => a - b);
   608→   608→  const q1Index = Math.floor(sorted.length * 0.25);
   609→   609→  const q3Index = Math.floor(sorted.length * 0.75);
   610→   610→  return {
   611→   611→    q1: sorted[q1Index],
   612→   612→    q3: sorted[q3Index],
   613→   613→  };
   614→   614→}

   616→   616→/**
   617→   617→ * Calculate the Haversine distance between two geographic points
   618→   618→ * @param lat1 - Latitude of first point
   619→   619→ * @param lon1 - Longitude of first point
   620→   620→ * @param lat2 - Latitude of second point
   621→   621→ * @param lon2 - Longitude of second point
   622→   622→ * @returns Distance in kilometers
   623→   623→ */
   624→   624→function calculateHaversineDistance(
   625→   625→  lat1: number,
   626→   626→  lon1: number,
   627→   627→  lat2: number,
   628→   628→  lon2: number
   629→   629→): number {
   630→   630→  const dLat = toRadians(lat2 - lat1);
   631→   631→  const dLon = toRadians(lon2 - lon1);
   632→   632→  
   633→   633→  const a =
   634→   634→    Math.sin(dLat / 2) * Math.sin(dLat / 2) +
   635→   635→    Math.cos(toRadians(lat1)) * Math.cos(toRadians(lat2)) *
   636→   636→    Math.sin(dLon / 2) * Math.sin(dLon / 2);
   637→   637→    
   638→   638→  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
   639→   639→  
   640→   640→  return EARTH_RADIUS_KM * c;
   641→   641→}

   643→   643→/**
   644→   644→ * Convert degrees to radians
   645→   645→ * @param degrees - Angle in degrees
   646→   646→ * @returns Angle in radians
   647→   647→ */
   648→   648→function toRadians(degrees: number): number {
   649→   649→  return degrees * (Math.PI / 180);
   650→   650→}

   652→   652→/**
   653→   653→ * Clamp a value between minimum and maximum bounds
   654→   654→ * @param value - Value to clamp
   655→   655→ * @param min - Minimum bound
   656→   656→ * @param max - Maximum bound
   657→   657→ * @returns Clamped value
   658→   658→ */
   659→   659→function clamp(value: number, min: number, max: number): number {
   660→   660→  return Math.min(Math.max(value, min), max);
   661→   661→}
   662→   662→
   663→   663→/**
   664→   664→ * Normalize a value to 0-1 range using min-max normalization
   665→   665→ * @param value - Value to normalize
   666→   666→ * @param min - Minimum possible value
   667→   667→ * @param max - Maximum possible value
   668→   668→ * @returns Normalized value between 0 and 1
   669→   669→ */
   670→   670→function normalize(value: number, min: number, max: number): number {
   671→   671→  if (max === min) return 0.5;
   672→   672→  return clamp((value - min) / (max - min), 0, 1);
   673→   673→}
   674→   674→
   675→   675→/**
   676→   676→ * Calculate percentile rank of a value in a dataset
   677→   677→ * @param value - Value to find percentile for
   678→   678→ * @param dataset - Dataset to compare against
   679→   679→ * @returns Percentile rank (0-100)
   680→   680→ */
   681→   681→function calculatePercentileRank(value: number, dataset: number[]): number {
   682→   682→  if (dataset.length === 0) return 50;
   683→   683→  const sorted = [...dataset].sort((a, b) => a - b);
   684→   684→  let belowCount = 0;
   685→   685→  for (const val of sorted) {
   686→   686→    if (val < value) belowCount++;
   687→   687→  }
   688→   688→  return (belowCount / sorted.length) * 100;
   689→   689→}

   845→   845→   * @param windowMs - Window duration in milliseconds
   846→   846→   * @returns Sliding window buffer instance
   847→   847→   */
   848→   848→  private createSlidingWindow<T>(maxSize: number, windowMs: number): SlidingWindowBuffer<T> {
   849→   849→    return {
   850→   850→      buffer: [],
   851→   851→      maxSize,
   852→   852→      windowMs,
   853→   853→      add(item: T): void {
   854→   854→        this.buffer.push(item);
   855→   855→        if (this.buffer.length > this.maxSize) {
   856→   856→          this.buffer.shift();
   857→   857→        }
   858→   858→      },
   859→   859→      getAll(): T[] {
   860→   860→        return [...this.buffer];
   861→   861→      },
   862→   862→      size(): number {
   863→   863→        return this.buffer.length;
   864→   864→      },
   865→   865→      clear(): void {
   866→   866→        this.buffer = [];
   867→   867→      },
   868→   868→      getInWindow(now: Date): T[] {
   869→   869→        const windowStart = new Date(now.getTime() - this.windowMs);
   870→   870→        // Type assertion for timestamp access
   871→   871→        return this.buffer.filter((item) => {
   872→   872→          const typedItem = item as unknown as { timestamp: Date };
   873→   873→          return typedItem.timestamp >= windowStart;
   874→   874→        });
   875→   875→      },
   876→   876→    };
   877→   877→  }

  1045→  1045→  detectStatisticalOutliers(

  1049→  1049→    if (!Array.isArray(values) || values.length < this.config.minSamplesRequired) {

  1517→  1517→    
  1518→  1518→    // Get historical amounts for this user
  1519→  1519→    const userHistoryKey = `amounts:${customerId}`;

  1529→  1529→        const std = calculateStandardDeviation(historicalAmounts);
  1530→  1530→        
  1531→  1531→        if (std > 0) {
  1532→  1532→          const zScore = Math.abs((transaction.amount - mean) / std);
  1533→  1533→

  1535→  1535→            results.push({
  1536→  1536→              isAnomalous: true,
  1537→  1537→              category: AnomalyCategory.STATISTICAL,
  1538→  1538→              severity: this.scoreToSeverity(zScore / this.thresholds.zScoreThreshold),
  1539→  1539→              confidence: clamp(zScore / (this.thresholds.zScoreThreshold * 2), 0, 1),
  1540→  1540→              score: zScore,
  1541→  1541→              threshold: this.thresholds.zScoreThreshold,
  1542→  1542→              actualValue: transaction.amount,

  1597→  1597→              score: absModifiedZScore,
  1598→  1598→              threshold: this.thresholds.modifiedZScoreThreshold,
  1599→  1599→              actualValue: transaction.amount,
  1600→  1600→              description: `Modified Z-score (${absModifiedZScore.toFixed(2)}) exceeds threshold`,
  1601→  1601→              details: { median, mad, method: 'modified_z_score' },

  1603→  1603→            });
  1604→  1604→          }
