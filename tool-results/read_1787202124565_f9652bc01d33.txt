   181→  constructor(config: PipelineConfig) {
   182→    this.config = this.validateConfig(config);

   188→    
   189→    logger.info(`MLPipeline initialized: ${config.name} (${config.id})`);

   196→  // ============== Configuration ==============
   197→
   198→  /**
   199→   * Validate and normalize pipeline configuration
   200→   */
   201→  private validateConfig(config: PipelineConfig): PipelineConfig {
   202→    // Validate required fields
   203→    if (!config.id || !config.name) {
   204→      throw new Error('Pipeline ID and name are required');
   205→    }
   206→
   207→    if (!config.models || config.models.length === 0) {
   208→      throw new Error('At least one model must be configured');
   209→    }
   210→
   211→    // Validate weights sum to ~1 for weighted methods
   212→    const enabledModels = config.models.filter((m) => m.enabled);
   213→    const totalWeight = enabledModels.reduce((sum, m) => sum + m.weight, 0);
   214→    
   215→    if (
   216→      [AggregationMethod.WEIGHTED_VOTE, AggregationMethod.WEIGHTED_AVERAGE].includes(
   217→        config.aggregation.method
   218→      ) &&
   219→      Math.abs(totalWeight - 1.0) > 0.01
   220→    ) {
   221→      logger.warn(
   222→        `Model weights (${totalWeight}) do not sum to 1.0. Weights will be normalized.`
   223→      );
   224→    }
   225→
   226→    // Set defaults
   227→    return {
   228→      ...config,
   229→      timeout: {
   230→        perModelMs: config.timeout?.perModelMs ?? 5000,
   231→        totalMs: config.timeout?.totalMs ?? 30000,
   232→      },
   233→      retry: {
   234→        enabled: config.retry?.enabled ?? true,
   235→        maxAttempts: config.retry?.maxAttempts ?? 3,
   236→        initialDelayMs: config.retry?.initialDelayMs ?? 100,
   237→        backoffMultiplier: config.retry?.backoffMultiplier ?? 2,
   238→        maxDelayMs: config.retry?.maxDelayMs ?? 5000,
   239→        retryableErrors: config.retry?.retryableErrors ?? ['timeout', 'network_error'],
   240→      },
   241→      caching: {
   242→        enabled: config.caching?.enabled ?? true,
   243→        ttlMs: config.caching?.ttlMs ?? 300000,
   244→        maxSize: config.caching?.maxSize ?? 1000,
   245→        keyStrategy: config.caching?.keyStrategy ?? 'hash',
   246→      },
   247→      logLevel: config.logLevel ?? 'info',
   248→    };
   249→  }
   250→
   251→  /**

   258→  // ============== Model Registration ==============

   260→  /**
   261→   * Register a model with the pipeline
   262→   * 
   263→   * @param model - The ML predictor to register
   264→   * @throws Error if model ID is already registered or not in config
   265→   */
   266→  registerModel(model: MLPredictor): void {
   267→    const modelConfig = this.config.models.find((m) => m.modelId === model.id);
   268→    
   269→    if (!modelConfig) {
   270→      throw new Error(
   271→        `Model '${model.id}' is not in pipeline configuration. Add it to models array first.`
   272→      );
   273→    }
   274→
   275→    if (this.models.has(model.id)) {
   276→      throw new Error(`Model '${model.id}' is already registered`);
   277→    }
   278→
   279→    this.models.set(model.id, model);
   280→    logger.info(`Model registered: ${model.id} (${model.metadata.name})`);
   281→  }

   284→   * Unregister a model from the pipeline
   285→   * 
   286→   * @param modelId - ID of model to unregister
   287→   */
   288→  unregisterModel(modelId: string): void {
   289→    if (this.models.delete(modelId)) {
   290→      logger.info(`Model unregistered: ${modelId}`);
   291→    }
   292→  }

   294→  /**
   295→   * Get registered model by ID

   299→  }

   301→  /**
   302→   * Get list of all registered model IDs

   316→  // ============== Event System ==============
   317→
   318→  /**
   319→   * Subscribe to pipeline events
   320→   * 
   321→   * @param eventType - Type of event to listen for
   322→   * @param listener - Callback function
   323→   * @returns Unsubscribe function
   324→   */
   325→  on<T = unknown>(
   326→    eventType: PipelineEventType,
   327→    listener: PipelineEventListener<T>
   328→  ): () => void {
   329→    if (!this.listeners.has(eventType)) {
   330→      this.listeners.set(eventType, new Set());
   331→    }
   332→    
   333→    this.listeners.get(eventType)!.add(listener as PipelineEventListener);
   334→    
   335→    // Return unsubscribe function
   336→    return () => {
   337→      this.listeners.get(eventType)?.delete(listener as PipelineEventListener);
   338→    };
   339→  }
   340→
   341→  /**
   342→   * Emit an event to all listeners
   343→   */
   344→  private emit<T>(type: PipelineEventType, data: T, executionId?: string): void {
   345→    const event: PipelineEvent<T> = {
   346→      type,
   347→      timestamp: new Date(),
   348→      data,
   349→      pipelineId: this.config.id,
   350→      executionId,
   351→    };
   352→
   353→    const listeners = this.listeners.get(type);
   354→    if (listeners) {
   355→      for (const listener of listeners) {
   356→        try {
   357→          listener(event as PipelineEvent);
   358→        } catch (error) {
   359→          logger.error(`Event listener error for ${type}:`, error);
   360→        }
   361→      }
   362→    }
   363→  }
   364→
   365→  // ============== Main Execution ==============
   366→
   367→  /**
   368→   * Execute the pipeline on input data
   369→   * 
   370→   * This is the main entry point for running predictions through
   371→   * the configured ML models.
   372→   * 
   373→   * @param input - Input data for prediction
   374→   * @param options - Optional execution options override
   375→   * @returns Complete pipeline result with individual and aggregated results
   376→   * 
   377→   * @example
   378→   * ```typescript
   379→   * const result = await pipeline.execute({
   380→   *   amount: 1000,
   381→   *   userId: 'user_123',
   382→   *   // ... transaction features
   383→   * });
   384→   * 
   385→   * console.log(result.aggregatedResult.finalPrediction);
   386→   * console.log(result.aggregatedResult.confidence);
   387→   * ```
   388→   */
   389→  async execute(
   390→    input: unknown,
   391→    options?: Partial<PipelineConfig>
   392→  ): Promise<PipelineResult> {
   393→    const executionId = this.generateExecutionId();
   394→    const startTime = Date.now();
   395→
   396→    // Merge options with base config if provided
   397→    const effectiveConfig = options
   398→      ? { ...this.config, ...options }
   399→      : this.config;
   400→
   401→    this.emit(PipelineEventType.EXECUTION_STARTED, { input }, executionId);
   402→
   403→    // Check cache first
   404→    if (effectiveConfig.caching.enabled) {
   405→      const cacheKey = this.generateCacheKey(input);
   406→      const cached = this.cache.get(cacheKey);
   407→      
   408→      if (cached) {
   409→        this.emit(PipelineEventType.CACHE_HIT, { cacheKey }, executionId);
   410→        
   411→        return this.createCachedResult(
   412→          cached,
   413→          executionId,
   414→          startTime,
   415→          cacheKey
   416→        );
   417→      }
   418→      
   419→      this.emit(PipelineEventType.CACHE_MISS, { cacheKey }, executionId);
   420→    }
   421→
   422→    try {
   423→      // Handle A/B testing if enabled
   424→      if (effectiveConfig.abTesting?.enabled && this.abTestState) {
   425→        return await this.executeWithABTesting(
   426→          input,
   427→          effectiveConfig,
   428→          executionId,
   429→          startTime
   430→        );
   431→      }
   432→
   433→      // Execute based on mode
   434→      let modelResults: Map<string, BasePredictionResult>;
   435→
   436→      switch (effectiveConfig.mode) {
   437→        case PipelineMode.SEQUENTIAL:
   438→          modelResults = await this.executeSequential(
   439→            input,
   440→            effectiveConfig,
   441→            executionId
   442→          );
   443→          break;
   444→        case PipelineMode.PARALLEL:
   445→          modelResults = await this.executeParallel(
   446→            input,
   447→            effectiveConfig,
   448→            executionId
   449→          );
   450→          break;
   451→        case PipelineMode.DAG:
   452→          modelResults = await this.executeDAG(
   453→            input,
   454→            effectiveConfig,
   455→            executionId
   456→          );
   457→          break;
   458→        case PipelineMode.FALLBACK_CHAIN:
   459→          modelResults = await this.executeFallbackChain(
   460→            input,
   461→            effectiveConfig,
   462→            executionId
   463→          );
   464→          break;
   465→        default:
   466→          throw new Error(`Unknown pipeline mode: ${effectiveConfig.mode}`);
   467→      }
   468→
   469→      // Aggregate results
   470→      const aggregatedResult = this.aggregateResults(
   471→        modelResults,
   472→        effectiveConfig.aggregation
   473→      );
   474→
   475→      // Update metrics
   476→      this.metrics.recordExecution(Date.now() - startTime, true);
   477→
   478→      // Cache result
   479→      if (effectiveConfig.caching.enabled) {
   480→        const cacheKey = this.generateCacheKey(input);
   481→        this.cache.set(cacheKey, aggregatedResult);
   482→      }
   483→
   484→      const result: PipelineResult = {
   485→        executionId,
   486→        pipelineId: effectiveConfig.id,
   487→        startedAt: new Date(startTime),
   488→        completedAt: new Date(),
   489→        totalDurationMs: Date.now() - startTime,
   490→        success: true,
   491→        modelResults,
   492→        aggregatedResult,
   493→        metadata: this.buildMetadata(executionId, modelResults, false, effectiveConfig.mode),
   494→      };
   495→
   496→      this.emit(PipelineEventType.EXECUTION_COMPLETED, result, executionId);
   497→
   498→      return result;
   499→    } catch (error) {
   500→      this.metrics.recordExecution(Date.now() - startTime, false);
   501→      
   502→      const errorResult: PipelineResult = {
   503→        executionId,
   504→        pipelineId: effectiveConfig.id,
   505→        startedAt: new Date(startTime),
   506→        completedAt: new Date(),
   507→        totalDurationMs: Date.now() - startTime,
   508→        success: false,
   509→        modelResults: new Map(),
   510→        aggregatedResult: this.createErrorAggregatedResult(error),
   511→        metadata: this.buildMetadata(executionId, new Map(), true, effectiveConfig.mode),
   512→      };
   513→
   514→      this.emit(PipelineEventType.EXECUTION_FAILED, { error }, executionId);
   515→      
   516→      logger.error(`Pipeline execution failed: ${executionId}`, error);
   517→      
   518→      throw error;
   519→    }
   520→  }

   522→  // ============== Execution Strategies ==============
   523→
   524→  /**
   525→   * Execute models sequentially in priority order
   526→   */
   527→  private async executeSequential(
   528→    input: unknown,
   529→    config: PipelineConfig,
   530→    executionId: string
   531→  ): Promise<Map<string, BasePredictionResult>> {
   532→    const results = new Map<string, BasePredictionResult>();
   533→    const sortedModels = this.getSortedModels(config);
   534→
   535→    for (const modelConfig of sortedModels) {
   536→      if (!modelConfig.enabled) continue;
   537→
   538→      const result = await this.executeModelWithRetry(
   539→        modelConfig.modelId,
   540→        input,
   541→        config,
   542→        executionId
   543→      );
   544→
   545→      results.set(modelConfig.modelId, result);
   546→
   547→      // In sequential mode, check if we should stop early
   548→      // e.g., if fraud detected with high confidence
   549→      if (this.shouldStopEarly(result, config)) {
   550→        logger.debug(
   551→          `Early stopping after ${modelConfig.modelId} due to high-confidence result`
   552→        );
   553→        break;
   554→      }
   555→    }
   556→
   557→    return results;
   558→  }
   559→
   560→  /**
   561→   * Execute all models in parallel
   562→   */
   563→  private async executeParallel(
   564→    input: unknown,
   565→    config: PipelineConfig,
   566→    executionId: string
   567→  ): Promise<Map<string, BasePredictionResult>> {
   568→    const results = new Map<string, BasePredictionResult>();
   569→    const enabledModels = config.models.filter((m) => m.enabled);
   570→
   571→    // Create promises for all models
   572→    const promises = enabledModels.map(async (modelConfig) => {
   573→      const result = await this.executeModelWithRetry(
   574→        modelConfig.modelId,
   575→        input,
   576→        config,
   577→        executionId
   578→      );
   579→      return { modelId: modelConfig.modelId, result };
   580→    });
   581→
   582→    // Execute with overall timeout
   583→    const timeoutPromise = new Promise<never>((_, reject) => {
   584→      setTimeout(() => reject(new Error('Pipeline execution timeout')), config.timeout.totalMs);
   585→    });
   586→
   587→    const settledResults = await Promise.race([
   588→      Promise.allSettled(promises),
   589→      timeoutPromise,
   590→    ]);
   591→
   592→    // Process results
   593→    if (Array.isArray(settledResults)) {
   594→      for (const settled of settledResults) {
   595→        if (settled.status === 'fulfilled') {
   596→          results.set(settled.value.modelId, settled.value.result);
   597→        } else {
   598→          logger.error(`Model execution failed: ${settled.reason}`);
   599→        }
   600→      }
   601→    }
   602→
   603→    return results;
   604→  }
   605→
   606→  /**
   607→   * Execute models respecting dependency graph (DAG mode)
   608→   */
   609→  private async executeDAG(
   610→    input: unknown,
   611→    config: PipelineConfig,
   612→    executionId: string
   613→  ): Promise<Map<string, BasePredictionResult>> {
   614→    const results = new Map<string, BasePredictionResult>();
   615→    const executed = new Set<string>();
   616→    const executing = new Set<string>();
   617→
   618→    const executeWithDependencies = async (
   619→      modelId: string
   620→    ): Promise<void> => {
   621→      if (executed.has(modelId)) return;
   622→      if (executing.has(modelId)) return; // Prevent cycles
   623→
   624→      const modelConfig = config.models.find((m) => m.modelId === modelId);
   625→      if (!modelConfig || !modelConfig.enabled) {
   626→        executed.add(modelId);
   627→        return;
   628→      }
   629→
   630→      executing.add(modelId);
   631→
   632→      // Execute dependencies first
   633→      if (modelConfig.dependencies) {
   634→        for (const dep of modelConfig.dependencies) {
   635→          await executeWithDependencies(dep);
   636→        }
   637→      }
   638→
   639→      // Execute this model
   640→      const result = await this.executeModelWithRetry(
   641→        modelId,
   642→        input,
   643→        config,
   644→        executionId
   645→      );
   646→
   647→      results.set(modelId, result);
   648→      executed.add(modelId);
   649→      executing.delete(modelId);
   650→    };
   651→
   652→    // Execute all enabled models (dependencies will be resolved automatically)
   653→    for (const modelConfig of config.models) {
   654→      if (modelConfig.enabled) {
   655→        await executeWithDependencies(modelConfig.modelId);
   656→      }
   657→    }
   658→
   659→    return results;
   660→  }

   662→  /**
   663→   * Execute models in fallback chain (try each until one succeeds)
   664→   */
   665→  private async executeFallbackChain(
   666→    input: unknown,
   667→    config: PipelineConfig,
   668→    executionId: string
   669→  ): Promise<Map<string, BasePredictionResult>> {
   670→    const results = new Map<string, BasePredictionResult>();
   671→    const sortedModels = this.getSortedModels(config);
   672→
   673→    for (const modelConfig of sortedModels) {
   674→      if (!modelConfig.enabled) continue;
   675→
   676→      try {
   677→        const result = await this.executeModelWithRetry(
   678→          modelConfig.modelId,
   679→          input,
   680→          config,
   681→          executionId
   682→        );
   683→
   684→        results.set(modelConfig.modelId, result);
   685→
   686→        // If successful, stop the chain
   687→        if (result.success && result.confidence >= 0.5) {
   688→          logger.debug(
   689→            `Fallback chain stopped at ${modelConfig.modelId} (successful prediction)`
   690→          );
   691→          break;
   692→        }
   693→      } catch (error) {
   694→        logger.warn(
   695→          `Model ${modelConfig.modelId} failed in fallback chain, trying next...`,
   696→          error
   697→        );
   698→
   699→        // Try fallback model if specified
   700→        if (modelConfig.fallbackModelId) {
   701→          const fallbackConfig = config.models.find(
   702→            (m) => m.modelId === modelConfig.fallbackModelId
   703→          );
   704→          
   705→          if (fallbackConfig && fallbackConfig.enabled) {
   706→            const fallbackResult = await this.executeModelWithRetry(
   707→              fallbackConfig.modelId,
   708→              input,
   709→              config,
   710→              executionId
   711→            );
   712→            
   713→            results.set(fallbackConfig.modelId, fallbackResult);
   714→            
   715→            if (fallbackResult.success) {
   716→              break;
   717→            }
   718→          }
   719→        }
   720→      }
   721→    }
   722→
   723→    return results;
   724→  }

   727→
   728→  /**
   729→   * Execute a single model with retry logic
   730→   */
   731→  private async executeModelWithRetry(
   732→    modelId: string,
   733→    input: unknown,
   734→    config: PipelineConfig,
   735→    executionId: string
   736→  ): Promise<BasePredictionResult> {
   737→    const model = this.models.get(modelId);
   738→    
   739→    if (!model) {
   740→      return this.createModelError(
   741→        modelId,
   742→        `Model '${modelId}' is not registered`,
   743→        'MODEL_NOT_FOUND'
   744→      );
   745→    }
   746→
   747→    if (!model.isAvailable()) {
   748→      return this.createModelError(
   749→        modelId,
   750→        `Model '${modelId}' is not available`,
   751→        'MODEL_UNAVAILABLE'
   752→      );
   753→    }
   754→
   755→    this.emit(PipelineEventType.MODEL_STARTED, { modelId }, executionId);
   756→
   757→    const retryConfig = config.retry;
   758→    let lastError: Error | null = null;
   759→    let attempt = 0;
   760→
   761→    while (attempt <= (retryConfig.enabled ? retryConfig.maxAttempts : 0)) {
   762→      try {
   763→        // Apply timeout
   764→        const result = await Promise.race([
   765→          model.predict(input),
   766→          new Promise<BasePredictionResult>((_, reject) =>
   767→            setTimeout(
   768→              () => reject(new Error('Model prediction timeout')),
   769→              config.timeout.perModelMs
   770→            )
   771→          ),
   772→        ]);
   773→
   774→        this.emit(PipelineEventType.MODEL_COMPLETED, { modelId, result }, executionId);
   775→        this.metrics.recordModelExecution(modelId, Date.now(), true);
   776→
   777→        return result;
   778→      } catch (error) {
   779→        lastError = error as Error;
   780→        attempt++;
   781→
   782→        if (attempt <= retryConfig.maxAttempts && retryConfig.enabled) {
   783→          const delay = Math.min(
   784→            retryConfig.initialDelayMs *
   785→              Math.pow(retryConfig.backoffMultiplier, attempt - 1),
   786→            retryConfig.maxDelayMs
   787→          );
   788→
   789→          this.emit(
   790→            PipelineEventType.MODEL_RETRY,
   791→            { modelId, attempt, delay, error: lastError.message },
   792→            executionId
   793→          );
   794→
   795→          logger.warn(
   796→            `Model ${modelId} attempt ${attempt} failed, retrying in ${delay}ms...`
   797→          );
   798→
   799→          await this.sleep(delay);
   800→        }
   801→      }
   802→    }
   803→
   804→    // All retries exhausted
   805→    this.emit(
   806→      PipelineEventType.MODEL_FAILED,
   807→      { modelId, error: lastError?.message, attempts: attempt },
   808→      executionId
   809→    );
   810→
   811→    this.metrics.recordModelExecution(modelId, Date.now(), false);
   812→
   813→    return this.createModelError(
   814→      modelId,
   815→      lastError?.message || 'Unknown error',
   816→      'MAX_RETRIES_EXCEEDED'
   817→    );
   818→  }

   823→   * Aggregate results from multiple models into a single prediction
   824→   */
   825→  aggregateResults(
   826→    modelResults: Map<string, BasePredictionResult>,
   827→    aggregationConfig: AggregationConfig
   828→  ): AggregatedResult {
   829→    const resultsArray = Array.from(modelResults.values());
   830→    const successfulResults = resultsArray.filter((r) => r.success);
   831→
   832→    if (successfulResults.length === 0) {
   833→      return {
   834→        finalPrediction: null,
   835→        confidence: 0,
   836→        agreement: 0,
   837→        agreeingModels: 0,
   838→        totalModels: resultsArray.length,
   839→        disagreements: [],
   840→        method: aggregationConfig.method,
   841→      };
   842→    }
   843→
   844→    switch (aggregationConfig.method) {
   845→      case AggregationMethod.MAJORITY_VOTE:
   846→        return this.majorityVoteAggregation(successfulResults, modelResults);
   847→      case AggregationMethod.WEIGHTED_VOTE:
   848→        return this.weightedVoteAggregation(successfulResults, modelResults, aggregationConfig);
   849→      case AggregationMethod.AVERAGE:
   850→        return this.averageAggregation(successfulResults);
   851→      case AggregationMethod.WEIGHTED_AVERAGE:
   852→        return this.weightedAverageAggregation(successfulResults, aggregationConfig);
   853→      case AggregationMethod.MAX:
   854→        return this.maxAggregation(successfulResults);
   855→      case AggregationMethod.MIN:
   856→        return this.minAggregation(successfulResults);
   857→      default:
   858→        return this.averageAggregation(successfulResults);
   859→    }
   860→  }

   862→  /**
   863→   * Majority vote aggregation for classification tasks
   864→   */
   865→  private majorityVoteAggregation(
   866→    results: BasePredictionResult[],
   867→    allResults: Map<string, BasePredictionResult>
   868→  ): AggregatedResult {
   869→    // Extract predictions (assuming they have predictedClass or value)
   870→    const votes = new Map<string, number>();
   871→    
   872→    for (const result of results) {
   873→      const prediction = this.extractPredictionValue(result);
   874→      const key = String(prediction);
   875→      votes.set(key, (votes.get(key) || 0) + 1);
   876→    }
   877→
   878→    // Find majority
   879→    let maxVotes = 0;
   880→    let majorityPrediction: string = '';
   881→    
   882→    for (const [prediction, count] of votes.entries()) {
   883→      if (count > maxVotes) {
   884→        maxVotes = count;
   885→        majorityPrediction = prediction;
   886→      }
   887→    }
   888→
   889→    const confidence = maxVotes / results.length;
   890→    const disagreements = this.findDisagreements(results, majorityPrediction);
   891→
   892→    return {
   893→      finalPrediction: majorityPrediction,
   894→      confidence,
   895→      agreement: confidence,
   896→      agreeingModels: maxVotes,
   897→      totalModels: results.length,
   898→      disagreements: disagreements.length > 0 ? disagreements : undefined,
   899→      method: AggregationMethod.MAJORITY_VOTE,
   900→    };
   901→  }

   904→   * Weighted vote aggregation
   905→   */
   906→  private weightedVoteAggregation(
   907→    results: BasePredictionResult[],
   908→    allResults: Map<string, BasePredictionResult>,
   909→    config: AggregationConfig
   910→  ): AggregatedResult {
   911→    const votes = new Map<string, number>();
   912→    const weights = this.getModelWeights();
   913→
   914→    for (const result of results) {
   915→      const weight = weights.get(result.modelId) || 1 / results.length;
   916→      const prediction = String(this.extractPredictionValue(result));
   917→      votes.set(prediction, (votes.get(prediction) || 0) + weight);
   918→    }
   919→
   920→    let maxScore = 0;
   921→    let winningPrediction = '';
   922→
   923→    for (const [prediction, score] of votes.entries()) {
   924→      if (score > maxScore) {
   925→        maxScore = score;
   926→        winningPrediction = prediction;
   927→      }
   928→    }
   929→
   930→    const disagreements = this.findDisagreements(results, winningPrediction);
   931→
   932→    return {
   933→      finalPrediction: winningPrediction,
   934→      confidence: Math.min(maxScore, 1),
   935→      agreement: this.calculateAgreement(results),
   936→      agreeingModels: this.countAgreeingModels(results, winningPrediction),
   937→      totalModels: results.length,
   938→      disagreements: disagreements.length > 0 ? disagreements : undefined,
   939→      method: AggregationMethod.WEIGHTED_VOTE,
   940→    };
   941→  }

   943→  /**
   944→   * Simple average aggregation for numeric outputs
   945→   */
   946→  private averageAggregation(results: BasePredictionResult[]): AggregatedResult {
   947→    const values = results.map((r) => this.extractNumericValue(r));
   948→    const avg = values.reduce((a, b) => a + b, 0) / values.length;
   949→    const std = Math.sqrt(
   950→      values.reduce((sum, v) => sum + (v - avg) ** 2, 0) / values.length
   951→    );
   952→
   953→    return {
   954→      finalPrediction: avg,
   955→      confidence: 1 / (1 + std), // Higher confidence when std is lower
   956→      agreement: this.calculateAgreement(results),
   957→      agreeingModels: results.length,
   958→      totalModels: results.length,
   959→      method: AggregationMethod.AVERAGE,
   960→    };
   961→  }

   966→  private weightedAverageAggregation(

   981→    const avg = totalWeight > 0 ? weightedSum / totalWeight : 0;

   983→    return {
   984→      finalPrediction: avg,
   985→      confidence: this.averageConfidence(results),
   986→      agreement: this.calculateAgreement(results),

  1295→  /**
  1296→   * Calculate agreement level among models (0-1)
  1297→   */
  1298→  private calculateAgreement(results: BasePredictionResult[]): number {
  1299→    if (results.length <= 1) return 1;
  1300→
  1301→    const predictions = results.map((r) => String(this.extractPredictionValue(r)));
  1302→    const uniquePredictions = new Set(predictions);
  1303→
  1304→    // Perfect agreement if all same
  1305→    if (uniquePredictions.size === 1) return 1;
  1306→
  1307→    // No agreement if all different
  1308→    if (uniquePredictions.size === results.length) return 0;
  1309→
  1310→    // Partial agreement based on majority proportion
  1311→    const counts = new Map<string, number>();
  1312→    for (const pred of predictions) {
  1313→      counts.set(pred, (counts.get(pred) || 0) + 1);
  1314→    }
  1315→
  1316→    const maxCount = Math.max(...counts.values());
  1317→    return maxCount / results.length;
  1318→  }

  1333→   * Average confidence across results
  1334→   */
  1335→  private averageConfidence(results: BasePredictionResult[]): number {
  1336→    if (results.length === 0) return 0;
  1337→    return results.reduce((sum, r) => sum + r.confidence, 0) / results.length;
  1338→  }
  1339→
  1340→  /**
  1341→   * Check if we should stop early in sequential mode
  1342→   */
  1343→  private shouldStopEarly(
  1344→    result: BasePredictionResult,
  1345→    config: PipelineConfig
  1346→  ): boolean {
  1347→    // Stop if high confidence detection (e.g., clear fraud)
  1348→    const resultAny = result as Record<string, unknown>;
  1349→    
  1350→    if (resultAny.isAnomalous === true && result.confidence > 0.95) {
  1351→      return true;
  1352→    }
  1353→
  1354→    if (
  1355→      typeof resultAny.riskScore === 'number' &&
  1356→      (resultAny.riskScore as number) > 90
  1357→    ) {
  1358→      return true;
  1359→    }
  1360→
  1361→    return false;
  1362→  }

  1534→  /**
  1535→   * Health check for the entire pipeline
  1536→   */
  1537→  async healthCheck(): Promise<PipelineHealthStatus> {
  1538→    const modelHealths = new Map<string, ModelHealthStatus>();
  1539→
  1540→    for (const [modelId, model] of this.models) {
  1541→      modelHealths.set(modelId, model.getHealth());
  1542→    }
  1543→
  1544→    const healthyModels = Array.from(modelHealths.values()).filter(
  1545→      (h) => h.isHealthy
  1546→    ).length;
  1547→    const totalModels = modelHealths.size;
  1548→
  1549→    const isHealthy = healthyModels === totalModels && totalModels > 0;
  1550→
  1551→    return {
  1552→      isHealthy,
  1553→      pipelineId: this.config.id,
  1554→      totalModels,
  1555→      healthyModels,
  1556→      unhealthyModels: totalModels - healthyModels,
  1557→      modelHealths,
  1558→      uptime: process.uptime(),
  1559→      lastExecutionAt: this.metrics.getLastExecutionTime(),
  1560→    };
  1561→  }

  1574→// ============== Metrics Collector ==============
  1575→
  1576→/**
  1577→ * Internal metrics collector for pipeline monitoring
  1578→ */
  1579→class PipelineMetrics {
  1580→  private totalExecutions = 0;
  1581→  private successfulExecutions = 0;
  1582→  private failedExecutions = 0;
  1583→  private totalExecutionTimeMs = 0;
  1584→  private modelExecutions = new Map<string, { count: number; successes: number }>();
  1585→  private totalRetries = 0;
  1586→  private lastExecutionTime: Date | null = null;
  1587→
  1588→  recordExecution(durationMs: number, success: boolean): void {
  1589→    this.totalExecutions++;
  1590→    this.totalExecutionTimeMs += durationMs;
  1591→    this.lastExecutionTime = new Date();
  1592→
  1593→    if (success) {
  1594→      this.successfulExecutions++;
  1595→    } else {
  1596→      this.failedExecutions++;
  1597→    }
  1598→  }
  1599→
  1600→  recordModelExecution(modelId: string, _timestamp: number, success: void | boolean): void {
  1601→    const current = this.modelExecutions.get(modelId) || { count: 0, successes: 0 };
  1602→    current.count++;
  1603→    if (success) current.successes++;
  1604→    this.modelExecutions.set(modelId, current);
  1605→  }
  1606→
  1607→  recordRetry(): void {
  1608→    this.totalRetries++;
  1609→  }
  1610→
  1611→  getTotalRetries(): number {
  1612→    return this.totalRetries;
  1613→  }
  1614→
  1615→  getLastExecutionTime(): Date | null {
  1616→    return this.lastExecutionTime;
  1617→  }
  1618→
  1619→  getSnapshot(): PipelineMetricsSnapshot {
  1620→    return {
  1621→      totalExecutions: this.totalExecutions,
  1622→      successfulExecutions: this.successfulExecutions,
  1623→      failedExecutions: this.failedExecutions,
  1624→      successRate:
  1625→        this.totalExecutions > 0
  1626→          ? this.successfulExecutions / this.totalExecutions
  1627→          : 0,
  1628→      avgExecutionTimeMs:
  1629→        this.totalExecutions > 0
  1630→          ? this.totalExecutionTimeMs / this.totalExecutions
  1631→          : 0,
  1632→      totalRetries: this.totalRetries,
  1633→      modelExecutions: Object.fromEntries(this.modelExecutions),
  1634→      lastExecutionTime: this.lastExecutionTime,
  1635→    };
  1636→  }
  1637→
  1638→  reset(): void {
  1639→    this.totalExecutions = 0;
  1640→    this.successfulExecutions = 0;
  1641→    this.failedExecutions = 0;
  1642→    this.totalExecutionTimeMs = 0;
  1643→    this.modelExecutions.clear();
  1644→    this.totalRetries = 0;
  1645→  }
  1646→}
