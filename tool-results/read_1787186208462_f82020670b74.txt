   559→  // ============== Signature Verification ==============
   560→
   561→  /**
   562→   * Generate HMAC-SHA256 signature for a payload
   563→   * 
   564→   * @param payload - The payload object to sign
   565→   * @param secret - The secret key for signing
   566→   * @param timestamp - Optional timestamp for inclusion in signature
   567→   * @returns Hex-encoded signature string
   568→   * 
   569→   * @example
   570→   * ```typescript
   571→   * const signature = webhookService.generateSignature(
   572→   *   { id: 'evt_123', type: 'payment.completed', data: {} },
   573→   *   'my-secret-key'
   574→   * );
   575→   * ```
   576→   */
   577→  generateSignature(
   578→    payload: Record<string, unknown>,
   579→    secret: string,
   580→    timestamp?: number
   581→  ): string {
   582→    const ts = timestamp || Date.now();
   583→    const payloadString = JSON.stringify(payload);
   584→    const signedPayload = `${ts}.${payloadString}`;
   585→    
   586→    const signature = createHmac('sha256', secret)
   587→      .update(signedPayload)
   588→      .digest('hex');
   589→    
   590→    return `t=${ts},v1=${signature}`;
   591→  }
   592→
   593→  /**
   594→   * Verify HMAC-SHA256 signature for an incoming webhook
   595→   * 
   596→   * Uses timing-safe comparison to prevent timing attacks.
   597→   * Supports both timestamped and plain signatures.
   598→   * 
   599→   * @param payload - The raw payload to verify
   600→   * @param signature - The signature to verify against
   601→   * @param secret - The secret key used for verification
   602→   * @param maxAge - Maximum age in ms for timestamped signatures (default: 5 minutes)
   603→   * @returns True if signature is valid, false otherwise
   604→   * 
   605→   * @example
   606→   * ```typescript
   607→   * const isValid = await webhookService.verifySignature(
   608→   *   requestBody,
   609→   *   requestHeaders['x-ssm-pay-signature'],
   610→   *   endpointSecret
   611→   * );
   612→   * ```
   613→   */
   614→  async verifySignature(
   615→    payload: unknown,
   616→    signature: string,
   617→    secret: string,
   618→    maxAge: number = 5 * 60 * 1000 // 5 minutes
   619→  ): Promise<boolean> {
   620→    try {
   621→      // Parse signature format: t=timestamp,v1=signature
   622→      const signatureMatch = signature.match(/t=(\d+),v1=([a-f0-9]+)/);
   623→      
   624→      if (!signatureMatch) {
   625→        this.stats.signaturesFailed++;
   626→        logger.warn('Invalid signature format', {
   627→          event: 'webhook.signature.invalid_format',
   628→          metadata: { signaturePrefix: signature.substring(0, 20) },
   629→        });
   630→        return false;
   631→      }
   632→
   633→      const [, timestampStr, providedSignature] = signatureMatch;
   634→      const timestamp = parseInt(timestampStr, 10);
   635→      
   636→      // Check timestamp freshness
   637→      const now = Date.now();
   638→      const timestampAge = now - timestamp;
   639→      
   640→      if (timestampAge < 0) {
   641→        this.stats.signaturesFailed++;
   642→        logger.warn('Signature timestamp is in the future', {
   643→          event: 'webhook.signature.future_timestamp',
   644→          metadata: { timestampAge: Math.abs(timestampAge) },
   645→        });
   646→        return false;
   647→      }
   648→      
   649→      if (timestampAge > maxAge) {
   650→        this.stats.signaturesFailed++;
   651→        logger.warn('Signature timestamp too old', {
   652→          event: 'webhook.signature.expired',
   653→          metadata: { timestampAge, maxAge },
   654→        });
   655→        return false;
   656→      }
   657→
   658→      // Generate expected signature
   659→      const payloadString = JSON.stringify(payload);
   660→      const signedPayload = `${timestamp}.${payloadString}`;
   661→      
   662→      const expectedSignature = createHmac('sha256', secret)
   663→        .update(signedPayload)
   664→        .digest('hex');
   665→
   666→      // Use timing-safe comparison
   667→      const isValid = timingSafeEqual(
   668→        Buffer.from(providedSignature),
   669→        Buffer.from(expectedSignature)
   670→      );
   671→
   672→      if (isValid) {
   673→        this.stats.signaturesVerified++;
   674→        logger.debug('Signature verified successfully', {
   675→          event: 'webhook.signature.verified',
   676→        });
   677→      } else {
   678→        this.stats.signaturesFailed++;
   679→        logger.warn('Signature verification failed', {
   680→          event: 'webhook.signature.mismatch',
   681→        });
   682→      }
   683→
   684→      return isValid;
   685→    } catch (error) {
   686→      this.stats.signaturesFailed++;
   687→      logger.error('Error during signature verification', {
   688→        event: 'webhook.signature.error',
   689→        error: error instanceof Error ? error : new Error(String(error)),
   690→      });
   691→      return false;
   692→    }
   693→  }

   699→   * 
   700→   * Handles the complete workflow:
   701→   * 1. Validate payload structure
   702→   * 2. Verify signature
   703→   * 3. Check idempotency
   704→   * 4. Parse and normalize event
   705→   * 5. Route to handlers/subscribers
   706→   * 
   707→   * @param payload - The raw webhook payload
   708→   * @param headers - HTTP headers containing signature
   709→   * @returns Processing result with status and optional event
   710→   * 
   711→   * @example
   712→   * ```typescript
   713→   * const result = await webhookService.processIncomingWebhook(
   714→   *   req.body,
   715→   *   req.headers
   716→   * );
   717→   * 
   718→   * if (result.valid && result.event) {
   719→   *   console.log('Processed event:', result.event.type);
   720→   * }
   721→   * ```
   722→   */
   723→  async processIncomingWebhook(
   724→    payload: unknown,
   725→    headers: Record<string, string>
   726→  ): Promise<{
   727→    valid: boolean;
   728→    event?: WebhookEvent;
   729→    error?: string;
   730→    errorCode?: ErrorCode;
   731→  }> {
   732→    this.stats.totalEventsReceived++;
   733→    
   734→    const startTime = Date.now();
   735→    
   736→    try {
   737→      // Step 1: Validate payload exists
   738→      if (!payload || typeof payload !== 'object') {
   739→        throw new AppError('Invalid payload: must be an object', ErrorCode.VALIDATION_ERROR);
   740→      }
   741→
   742→      const typedPayload = payload as Record<string, unknown>;
   743→
   744→      // Step 2: Check payload size
   745→      const payloadSize = JSON.stringify(payload).length;
   746→      if (payloadSize > this.config.maxPayloadSizeBytes) {
   747→        throw new AppError(
   748→          `Payload exceeds maximum size of ${this.config.maxPayloadSizeBytes} bytes`,
   749→          ErrorCode.VALIDATION_ERROR
   750→        );
   751→      }
   752→
   753→      // Step 3: Extract and validate required fields
   754→      const eventId = typedPayload.id as string;
   755→      const eventType = typedPayload.type as string;
   756→      const eventCreated = typedPayload.created as string;
   757→      const eventData = typedPayload.data as Record<string, unknown>;
   758→
   759→      if (!eventId || !eventType || !eventData) {
   760→        throw new AppError(
   761→          'Missing required fields: id, type, or data',
   762→          ErrorCode.VALIDATION_ERROR,
   763→          { context: { hasId: !!eventId, hasType: !!eventType, hasData: !!eventData } }
   764→        );
   765→      }
   766→
   767→      // Step 4: Verify signature
   768→      const signatureHeader = headers[WEBHOOK_HEADERS.SIGNATURE.toLowerCase()] ||
   769→                             headers[WEBHOOK_HEADERS.SIGNATURE];
   770→      
   771→      if (!signatureHeader) {
   772→        throw new AppError(
   773→          'Missing signature header',
   774→          ErrorCode.VALIDATION_ERROR,
   775→          { context: { headerName: WEBHOOK_HEADERS.SIGNATURE } }
   776→        );
   777→      }
   778→
   779→      const isValid = await this.verifySignature(
   780→        payload,
   781→        signatureHeader,
   782→        this.config.defaultSigningSecret
   783→      );
   784→
   785→      if (!isValid) {
   786→        throw new AppError(
   787→          'Signature verification failed',
   788→          ErrorCode.PAYMENT_VERIFICATION_FAILED
   789→        );
   790→      }
   791→
   792→      // Step 5: Check idempotency
   793→      if (this.isDuplicateEvent(eventId)) {
   794→        this.stats.duplicateEventsRejected++;
   795→        logger.info('Duplicate webhook event rejected', {
   796→          event: 'webhook.duplicate_rejected',
   797→          metadata: { eventId, eventType },
   798→        });
   799→        return { valid: true }; // Return success but don't reprocess
   800→      }
   801→
   802→      this.markEventAsProcessed(eventId);
   803→
   804→      // Step 6: Construct normalized event
   805→      const event: WebhookEvent = {
   806→        id: eventId,
   807→        type: eventType,
   808→        data: eventData,
   809→        createdAt: eventCreated || new Date().toISOString(),
   810→        version: EVENT_SCHEMA_VERSION,
   811→        source: 'external',
   812→        correlationId: typedPayload.correlationId as string | undefined,
   813→        parentEventId: typedPayload.parentEventId as string | undefined,
   814→      };
   815→
   816→      // Step 7: Route event to subscribers
   817→      await this.routeEventToSubscribers(event);
   818→
   819→      this.stats.totalEventsProcessed++;
   820→      this.stats.uniqueEventsProcessed++;
   821→      this.stats.lastProcessedAt = new Date();
   822→
   823→      const processingTime = Date.now() - startTime;
   824→      logger.info('Incoming webhook processed successfully', {
   825→        event: 'webhook.incoming.processed',
   826→        metadata: {
   827→          eventId: event.id,
   828→          eventType: event.type,
   829→          processingTimeMs: processingTime,
   830→        },
   831→      });
   832→
   833→      return { valid: true, event };
   834→    } catch (error) {
   835→      this.stats.totalEventsFailed++;
   836→      
   837→      const appError = error instanceof AppError 
   838→        ? error 
   839→        : new AppError(
   840→            error instanceof Error ? error.message : 'Unknown error processing webhook',
   841→            ErrorCode.PAYMENT_CALLBACK_FAILED,
   842→            { cause: error instanceof Error ? error : undefined }
   843→          );
   844→
   845→      logger.error('Failed to process incoming webhook', {
   846→        event: 'webhook.incoming.failed',
   847→        error: appError,
   848→        metadata: { processingTimeMs: Date.now() - startTime },
   849→      });
   850→
   851→      return {
   852→        valid: false,
   853→        error: appError.message,
   854→        errorCode: appError.code,
   855→      };
   856→    }
   857→  }
   858→
   859→  // ============== Endpoint Management ==============
   860→
   861→  /**
   862→   * Register a new webhook endpoint
   863→   * 
   864→   * @param config - Endpoint configuration
   865→   * @returns The registered endpoint
   866→   * @throws AppError if configuration is invalid
   867→   * 
   868→   * @example
   869→   * ```typescript
   870→   * const endpoint = await webhookService.registerEndpoint({
   871→   *   url: 'https://api.example.com/webhooks',
   872→   *   secret: 'whsec_my_secret_key',
   873→   *   subscribedEvents: ['payment.*', 'transaction.*'],
   874→   *   description: 'Primary payment webhook handler'
   875→   * });
   876→   * ```
   877→   */
   878→  registerEndpoint(config: {
   879→    url: string;
   880→    secret: string;
   881→    subscribedEvents: string[];
   882→    description?: string;
   883→    metadata?: Record<string, unknown>;
   884→    maxRetries?: number;
   885→    retryConfig?: Partial<RetryConfiguration>;
   886→  }): WebhookEndpoint {
   887→    // Validate URL
   888→    try {
   889→      new URL(config.url);
   890→    } catch {
   891→      throw new AppError(
   892→        `Invalid URL: ${config.url}`,
   893→        ErrorCode.INVALID_CONFIG,
   894→        { context: { url: config.url } }
   895→      );
   896→    }
   897→
   898→    // Validate secret
   899→    if (!config.secret || config.secret.length < 16) {
   900→      throw new AppError(
   901→        'Secret must be at least 16 characters',
   902→        ErrorCode.VALIDATION_ERROR,
   903→        { context: { secretLength: config.secret?.length || 0 } }
   904→      );
   905→    }
   906→
   907→    // Validate subscribed events
   908→    if (!config.subscribedEvents || config.subscribedEvents.length === 0) {
   909→      throw new AppError(
   910→        'At least one event subscription is required',
   911→        ErrorCode.VALIDATION_ERROR
   912→      );
   913→    }
   914→
   915→    const endpointId = `ep_${Date.now()}_${randomUUID().slice(0, 8)}`;
   916→    const now = new Date();
   917→
   918→    const endpoint: WebhookEndpoint = {
   919→      id: endpointId,
   920→      url: config.url,
   921→      secret: config.secret,
   922→      subscribedEvents: config.subscribedEvents,
   923→      isActive: true,
   924→      createdAt: now,
   925→      updatedAt: now,
   926→      consecutiveFailures: 0,
   927→      totalDeliveries: 0,
   928→      successfulDeliveries: 0,
   929→      maxRetries: config.maxRetries ?? this.config.defaultRetryConfig.maxAttempts,
   930→      retryConfig: { ...this.config.defaultRetryConfig, ...config.retryConfig },
   931→      metadata: config.metadata,
   932→      description: config.description,
   933→    };
   934→
   935→    this.endpoints.set(endpointId, endpoint);
   936→    this.updateEndpointStats();
   937→
   938→    logger.info('Webhook endpoint registered', {
   939→      event: 'webhook.endpoint.registered',
   940→      metadata: {
   941→        endpointId,
   942→        url: config.url,
   943→        eventCount: config.subscribedEvents.length,
   944→      },
   945→    });
   946→
   947→    return endpoint;
   948→  }
   949→
   950→  /**
   951→   * Update an existing webhook endpoint
   952→   * 
   953→   * @param endpointId - ID of endpoint to update
   954→   * @param updates - Fields to update
   955→   * @returns Updated endpoint
   956→   * @throws AppError if endpoint not found
   957→   */
   958→  updateEndpoint(
   959→    endpointId: string,
   960→    updates: Partial<{
   961→      url: string;
   962→      secret: string;
   963→      subscribedEvents: string[];
   964→      isActive: boolean;
   965→      description: string;
   966→      metadata: Record<string, unknown>;
   967→      maxRetries: number;
   968→      retryConfig: Partial<RetryConfiguration>;
   969→    }>
   970→  ): WebhookEndpoint {
   971→    const endpoint = this.endpoints.get(endpointId);
   972→    
   973→    if (!endpoint) {
   974→      throw new AppError(
   975→        `Endpoint not found: ${endpointId}`,
   976→        ErrorCode.NOT_FOUND,
   977→        { context: { endpointId } }
   978→      ) as AppError & { code: ErrorCode.NOT_FOUND };
   979→    }
   980→
   981→    // Apply updates
   982→    Object.assign(endpoint, updates, { updatedAt: new Date() });
   983→    
   984→    if (updates.retryConfig) {
   985→      endpoint.retryConfig = { ...endpoint.retryConfig, ...updates.retryConfig };
   986→    }
   987→
   988→    this.endpoints.set(endpointId, endpoint);
   989→    this.updateEndpointStats();
   990→
   991→    logger.info('Webhook endpoint updated', {
   992→      event: 'webhook.endpoint.updated',
   993→      metadata: { endpointId, updatedFields: Object.keys(updates) },
   994→    });
   995→
   996→    return endpoint;
   997→  }
   998→
   999→  /**
  1000→   * Remove a webhook endpoint
  1001→   * 
  1002→   * @param endpointId - ID of endpoint to remove
  1003→   * @returns True if removed, false if not found
  1004→   */
  1005→  removeEndpoint(endpointId: string): boolean {
  1006→    const endpoint = this.endpoints.get(endpointId);
  1007→    
  1008→    if (!endpoint) {
  1009→      return false;
  1010→    }
  1011→
  1012→    // Move pending deliveries to dead letter queue
  1013→    this.movePendingDeliveriesToDLQ(endpointId, 'endpoint_deleted');
  1014→
  1015→    this.endpoints.delete(endpointId);
  1016→    this.updateEndpointStats();
  1017→
  1018→    logger.info('Webhook endpoint removed', {
  1019→      event: 'webhook.endpoint.removed',
  1020→      metadata: { endpointId, url: endpoint.url },
  1021→    });
  1022→
  1023→    return true;
  1024→  }
  1025→
  1026→  /**
  1027→   * Get an endpoint by ID
  1028→   * 
  1029→   * @param endpointId - Endpoint ID
  1030→   * @returns Endpoint or undefined if not found
  1031→   */
  1032→  getEndpoint(endpointId: string): WebhookEndpoint | undefined {
  1033→    return this.endpoints.get(endpointId);
  1034→  }
  1035→
  1036→  /**
  1037→   * Get all registered endpoints
  1038→   * 
  1039→   * @param options - Filter options
  1040→   * @returns Array of endpoints
  1041→   */
  1042→  getEndpoints(options?: {
  1043→    activeOnly?: boolean;
  1044→    eventTypes?: string[];
  1045→  }): WebhookEndpoint[] {
  1046→    let endpoints = Array.from(this.endpoints.values());
  1047→
  1048→    if (options?.activeOnly) {
  1049→      endpoints = endpoints.filter(ep => ep.isActive);
  1050→    }
  1051→
  1052→    if (options?.eventTypes?.length) {
  1053→      endpoints = endpoints.filter(ep =>
  1054→        ep.subscribedEvents.some(pattern =>
  1055→          options.eventTypes!.some(type => this.matchEventPattern(type, pattern))
  1056→        )
  1057→      );
  1058→    }
  1059→
  1060→    return endpoints;
  1061→  }
  1062→
  1063→  /**
  1064→   * Disable an endpoint (soft delete)
  1065→   * 
  1066→   * @param endpointId - Endpoint ID to disable
  1067→   * @returns Updated endpoint
  1068→   */
  1069→  disableEndpoint(endpointId: string): WebhookEndpoint {
  1070→    return this.updateEndpoint(endpointId, { isActive: false });
  1071→  }
  1072→
  1073→  /**
  1074→   * Re-enable a disabled endpoint
  1075→   * 
  1076→   * @param endpointId - Endpoint ID to enable
  1077→   * @returns Updated endpoint
  1078→   */
  1079→  enableEndpoint(endpointId: string): WebhookEndpoint {
  1080→    const endpoint = this.updateEndpoint(endpointId, { isActive: true });
  1081→    // Reset consecutive failures when re-enabling
  1082→    endpoint.consecutiveFailures = 0;
  1083→    return endpoint;
  1084→  }
  1085→
  1086→  // ============== Event Dispatch & Routing ==============
  1087→
  1088→  /**
  1089→   * Create and dispatch a new webhook event
  1090→   * 
  1091→   * @param type - Event type
  1092→   * @param data - Event payload data
  1093→   * @param options - Additional event options
  1094→   * @returns Created event and delivery promises
  1095→   * 
  1096→   * @example
  1097→   * ```typescript
  1098→   * const { event, deliveries } = await webhookService.dispatchEvent(
  1099→   *   'payment.completed',
  1100→   *   { transactionId: 'txn_123', amount: 10000, currency: 'NGN' },
  1101→   *   { correlationId: 'req_456' }
  1102→   * );
  1103→   * ```
  1104→   */
  1105→  async dispatchEvent(
  1106→    type: WebhookEventType,
  1107→    data: Record<string, unknown>,
  1108→    options?: {
  1109→      source?: string;
  1110→      correlationId?: string;
  1111→      parentEventId?: string;
  1112→      priority?: number;
  1113→    }
  1114→  ): Promise<{
  1115→    event: WebhookEvent;
  1116→    deliveries: Promise<WebhookDelivery[]>[];
  1117→  }> {
  1118→    const event: WebhookEvent = {
  1119→      id: `evt_${Date.now()}_${randomUUID().slice(0, 8)}`,
  1120→      type,
  1121→      data,
  1122→      createdAt: new Date().toISOString(),
  1123→      version: EVENT_SCHEMA_VERSION,
  1124→      source: options?.source || 'ssm-pay',
  1125→      correlationId: options?.correlationId,
  1126→      parentEventId: options?.parentEventId,
  1127→    };
  1128→
  1129→    this.stats.totalEventsReceived++;
  1130→
  1131→    logger.info('Dispatching webhook event', {
  1132→      event: 'webhook.event.dispatched',
  1133→      metadata: { eventId: event.id, type: event.type },
  1134→    });
  1135→
  1136→    // Find matching endpoints and create deliveries
  1137→    const targetEndpoints = this.findEndpointsForEventType(event.type);
  1138→    const deliveryPromises: Promise<WebhookDelivery[]>[] = [];
  1139→
  1140→    for (const endpoint of targetEndpoints) {
  1141→      if (!endpoint.isActive) continue;
  1142→
  1143→      const queueItem = this.enqueueDelivery(event, endpoint, options?.priority);
  1144→      deliveryPromises.push(this.processQueueItem(queueItem));
  1145→    }
  1146→
  1147→    // Also trigger local subscriptions
  1148→    await this.routeEventToSubscribers(event);
  1149→
  1150→    this.stats.totalEventsProcessed++;
  1151→
  1152→    return { event, deliveries: deliveryPromises };
  1153→  }
  1154→
  1155→  /**
  1156→   * Find endpoints subscribed to a specific event type
  1157→   * 
  1158→   * @param eventType - Event type to match
  1159→   * @returns Matching endpoints
  1160→   * @private
  1161→   */
  1162→  private findEndpointsForEventType(eventType: string): WebhookEndpoint[] {
  1163→    const matchingEndpoints: WebhookEndpoint[] = [];
  1164→
  1165→    for (const endpoint of this.endpoints.values()) {
  1166→      if (!endpoint.isActive) continue;
  1167→
  1168→      const isMatch = endpoint.subscribedEvents.some(pattern =>
  1169→        this.matchEventPattern(eventType, pattern)
  1170→      );
  1171→
  1172→      if (isMatch) {
  1173→        matchingEndpoints.push(endpoint);
  1174→      }
  1175→    }
  1176→
  1177→    return matchingEndpoints;
  1178→  }
  1179→
  1180→  /**
  1181→   * Match event type against a pattern (supports wildcards)
  1182→   * 
  1183→   * @param eventType - Actual event type
  1184→   * @param pattern - Pattern to match against
  1185→   * @returns True if matches
  1186→   * @private
  1187→   */
  1188→  private matchEventPattern(eventType: string, pattern: string): boolean {
  1189→    // Exact match
  1190→    if (eventType === pattern) return true;
  1191→
  1192→    // Wildcard: payment.* matches payment.completed, payment.failed, etc.
  1193→    if (pattern.endsWith('*')) {
  1194→      const prefix = pattern.slice(0, -1); // Remove *
  1195→      // Support both "event.*" and "event*" patterns
  1196→      if (pattern.endsWith('.*')) {
  1197→        return eventType.startsWith(prefix);
  1198→      }
  1199→      return eventType.startsWith(prefix);
  1200→    }
  1201→
  1202→    // Regex pattern support
  1203→    if (pattern.startsWith('^') || pattern.endsWith('$')) {
  1204→      try {
  1205→        const regex = new RegExp(pattern);
  1206→        return regex.test(eventType);
  1207→      } catch {
  1208→        return false;
  1209→      }
  1210→    }
  1211→
  1212→    return false;
  1213→  }
  1214→
  1215→  // ============== Queue Management ==============
  1216→
  1217→  /**
  1218→   * Add a delivery to the processing queue
  1219→   * 
  1220→   * @param event - Event to deliver
  1221→   * @param endpoint - Target endpoint
  1222→   * @param priority - Queue priority (lower = higher)
  1223→   * @returns Queue item
  1224→   * @private
  1225→   */
  1226→  private enqueueDelivery(
  1227→    event: WebhookEvent,
  1228→    endpoint: WebhookEndpoint,
  1229→    priority: number = 5
  1230→  ): WebhookQueueItem {
  1231→    // Check queue size limit
  1232→    if (this.deliveryQueue.length >= this.config.maxQueueSize) {
  1233→      logger.warn('Webhook queue full, rejecting delivery', {
  1234→        event: 'webhook.queue.full',
  1235→        metadata: {
  1236→          currentSize: this.deliveryQueue.length,
  1237→          maxSize: this.config.maxQueueSize,
  1238→          eventId: event.id,
  1239→        },
  1240→      });
  1241→      throw new AppError(
  1242→        'Webhook queue is full',
  1243→        ErrorCode.API_REQUEST_FAILED,
  1244→        { context: { currentSize: this.deliveryQueue.length, maxSize: this.config.maxQueueSize } }
  1245→      );
  1246→    }
  1247→
  1248→    const queueItem: WebhookQueueItem = {
  1249→      id: `qi_${Date.now()}_${randomUUID().slice(0, 8)}`,
  1250→      event,
  1251→      endpointId: endpoint.id,
  1252→      priority,
  1253→      queuedAt: new Date(),
  1254→      scheduledFor: new Date(), // Schedule immediately
  1255→      attempts: 0,
  1256→      status: 'waiting',
  1257→    };
  1258→
  1259→    // Insert maintaining priority order
  1260→    const insertIndex = this.deliveryQueue.findIndex(
  1261→      item => item.priority > priority
  1262→    );
  1263→    
  1264→    if (insertIndex === -1) {
  1265→      this.deliveryQueue.push(queueItem);
  1266→    } else {
  1267→      this.deliveryQueue.splice(insertIndex, 0, queueItem);
  1268→    }
  1269→
  1270→    this.updateQueueStats();
  1271→
  1272→    logger.debug('Delivery enqueued', {
  1273→      event: 'webhook.queue.enqueued',
  1274→      metadata: {
  1275→        queueItemId: queueItem.id,
  1276→        eventId: event.id,
  1277→        endpointId: endpoint.id,
  1278→        position: this.deliveryQueue.indexOf(queueItem),
  1279→      },
  1280→    });
  1281→
  1282→    return queueItem;
  1283→  }
  1284→
  1285→  /**
  1286→   * Process a queue item (deliver webhook)
  1287→   * 
  1288→   * @param queueItem - Item to process
  1289→   * @returns Array of delivery records
  1290→   * @private
  1291→   */
  1292→  private async processQueueItem(queueItem: WebhookQueueItem): Promise<WebhookDelivery[]> {
  1293→    // Wait until scheduled time
  1294→    const now = new Date();
  1295→    if (queueItem.scheduledFor > now) {
  1296→      const delay = queueItem.scheduledFor.getTime() - now.getTime();
  1297→      await new Promise(resolve => setTimeout(resolve, delay));
  1298→    }
  1299→
  1300→    // Check concurrency limit
  1301→    if (this.currentProcessingCount >= this.config.maxConcurrency) {
  1302→      // Requeue with slight delay
  1303→      queueItem.scheduledFor = new Date(Date.now() + 100);
  1304→      return this.processQueueItem(queueItem);
  1305→    }
  1306→
  1307→    queueItem.status = 'processing';
  1308→    this.currentProcessingCount++;
  1309→
  1310→    try {
  1311→      const endpoint = this.endpoints.get(queueItem.endpointId);
  1312→      
  1313→      if (!endpoint) {
  1314→        throw new Error(`Endpoint not found: ${queueItem.endpointId}`);
  1315→      }
  1316→
  1317→      const delivery = await this.executeDelivery(queueItem.event, endpoint);
  1318→      queueItem.status = 'completed';
  1319→      
  1320→      return [delivery];
  1321→    } catch (error) {
  1322→      queueItem.status = 'failed';
  1323→      throw error;
  1324→    } finally {
  1325→      this.currentProcessingCount--;
  1326→      this.updateQueueStats();
  1327→    }
  1328→  }
  1329→
  1330→  /**
  1331→   * Execute actual HTTP delivery to endpoint
  1332→   * 
  1333→   * @param event - Event to deliver
  1334→   * @param endpoint - Target endpoint
  1335→   * @returns Delivery record
  1336→   * @private
  1337→   */
  1338→  private async executeDelivery(
  1339→    event: WebhookEvent,
  1340→    endpoint: WebhookEndpoint
  1341→  ): Promise<WebhookDelivery> {
  1342→    const deliveryId = `del_${Date.now()}_${randomUUID().slice(0, 8)}`;
  1343→    const startTime = Date.now();
  1344→
  1345→    // Get or create delivery record
  1346→    let delivery = this.deliveries.get(deliveryId);
  1347→    
  1348→    if (!delivery) {
  1349→      delivery = {
  1350→        id: deliveryId,
  1351→        event,
  1352→        endpointId: endpoint.id,
  1353→        status: 'processing',
  1354→        attemptNumber: 1,
  1355→        maxAttempts: endpoint.maxRetries,
  1356→        createdAt: new Date(),
  1357→        updatedAt: new Date(),
  1358→      };
  1359→      this.deliveries.set(deliveryId, delivery);
  1360→    } else {
  1361→      delivery.attemptNumber++;
  1362→      delivery.updatedAt = new Date();
  1363→      delivery.status = 'processing';
  1364→    }
  1365→
  1366→    this.stats.totalDeliveriesAttempted++;
  1367→    endpoint.totalDeliveries++;
  1368→
  1369→    try {
  1370→      // Prepare payload
  1371→      const payload: IncomingWebhookPayload = {
  1372→        id: event.id,
  1373→        type: event.type,
  1374→        created: event.createdAt,
  1375→        data: event.data,
  1376→      };
  1377→
  1378→      // Generate signature
  1379→      const signature = this.generateSignature(payload, endpoint.secret);
  1380→      delivery.signature = signature;
  1381→
  1382→      // Prepare headers
  1383→      const headers: Record<string, string> = {
  1384→        'Content-Type': 'application/json',
  1385→        [WEBHOOK_HEADERS.SIGNATURE]: signature,
  1386→        [WEBHOOK_HEADERS.TIMESTAMP]: Date.now().toString(),
  1387→        [WEBHOOK_HEADERS.EVENT_ID]: event.id,
  1388→        [WEBHOOK_HEADERS.EVENT_TYPE]: event.type,
  1389→        [WEBHOOK_HEADERS.DELIVERY]: deliveryId,
  1390→        [WEBHOOK_HEADERS.ATTEMPT]: delivery.attemptNumber.toString(),
  1391→        'User-Agent': 'SSM-Pay-Webhook/2.0',
  1392→      };
  1393→
  1394→      delivery.requestHeaders = headers;
  1395→
  1396→      // Execute HTTP request with timeout
  1397→      const controller = new AbortController();
  1398→      const timeout = setTimeout(() => controller.abort(), 30000); // 30s timeout
  1399→
  1400→      const response = await fetch(endpoint.url, {
  1401→        method: 'POST',
  1402→        headers,
  1403→        body: JSON.stringify(payload),
  1404→        signal: controller.signal,
  1405→      });
  1406→
  1407→      clearTimeout(timeout);
  1408→
  1409→      const durationMs = Date.now() - startTime;
  1410→      delivery.durationMs = durationMs;
  1411→      delivery.responseStatusCode = response.status;
  1412→      delivery.responseBody = await response.text().catch(() => '');
  1413→
  1414→      if (response.ok) {
  1415→        // Success
  1416→        delivery.status = 'delivered';
  1417→        delivery.deliveredAt = new Date();
  1418→        endpoint.lastSuccessfulDeliveryAt = new Date();
  1419→        endpoint.successfulDeliveries++;
  1420→        endpoint.consecutiveFailures = 0;
  1421→
  1422→        this.stats.totalDeliveriesSucceeded++;
  1423→        this.updateAverageDeliveryTime(durationMs);
  1424→
  1425→        logger.info('Webhook delivered successfully', {
  1426→          event: 'webhook.delivery.success',
  1427→          metadata: {
  1428→            deliveryId,
  1429→            endpointId: endpoint.id,
  1430→            eventId: event.id,
  1431→            statusCode: response.status,
  1432→            attempt: delivery.attemptNumber,
  1433→            durationMs,
  1434→          },
  1435→        });
  1436→      } else if (endpoint.retryConfig.retryableStatusCodes.includes(response.status)) {
  1437→        // Retryable error
  1438→        throw new AppError(
  1439→          `HTTP ${response.status}: ${delivery.responseBody}`,
  1440→          ErrorCode.API_RESPONSE_ERROR,
  1441→          { context: { statusCode: response.status, endpointUrl: endpoint.url } }
  1442→        );
  1443→      } else {
  1444→        // Non-retryable error
  1445→        delivery.status = 'failed';
  1446→        delivery.errorMessage = `HTTP ${response.status}: ${delivery.responseBody}`;
  1447→        
  1448→        this.stats.totalDeliveriesFailed++;
  1449→        endpoint.consecutiveFailures++;
  1450→        endpoint.lastFailureAt = new Date();
  1451→
  1452→        // Move to dead letter if appropriate
  1453→        if (delivery.attemptNumber >= delivery.maxAttempts) {
  1454→          this.moveToDeadLetterQueue(delivery, endpoint, 'max_retries_exceeded');
  1455→        }
  1456→
  1457→        logger.warn('Webhook delivery failed (non-retryable)', {
  1458→          event: 'webhook.delivery.failed',
  1459→          metadata: {
  1460→            deliveryId,
  1461→            endpointId: endpoint.id,
  1462→            statusCode: response.status,
  1463→            attempt: delivery.attemptNumber,
  1464→          },
  1465→        });
  1466→      }
  1467→    } catch (error) {
  1468→      const durationMs = Date.now() - startTime;
  1469→      delivery.durationMs = durationMs;
  1470→      delivery.errorMessage = error instanceof Error ? error.message : String(error);
  1471→      delivery.errorCode = error instanceof AppError ? error.code : 'UNKNOWN';
  1472→
  1473→      this.stats.totalDeliveriesFailed++;
  1474→      endpoint.consecutiveFailures++;
  1475→      endpoint.lastFailureAt = new Date();
  1476→
  1477→      // Determine if we should retry
  1478→      if (delivery.attemptNumber < delivery.maxAttempts) {
  1479→        delivery.status = 'retrying';
  1480→        delivery.nextRetryAt = this.calculateRetryTime(delivery, endpoint);
  1481→        
  1482→        this.stats.totalDeliveriesRetried++;
  1483→
  1484→        logger.warn('Scheduling webhook retry', {
  1485→          event: 'webhook.delivery.retry_scheduled',
  1486→          metadata: {
  1487→            deliveryId,
  1488→            endpointId: endpoint.id,
  1489→            attempt: delivery.attemptNumber,
  1490→            nextRetry: delivery.nextRetryAt?.toISOString(),
  1491→          },
  1492→        });
  1493→
  1494→        // Schedule retry
  1495→        const retryDelay = delivery.nextRetryAt.getTime() - Date.now();
  1496→        setTimeout(async () => {
  1497→          await this.executeDelivery(event, endpoint);
  1498→        }, retryDelay);
  1499→      } else {
  1500→        // Max retries exceeded - move to DLQ
  1501→        delivery.status = 'failed';
  1502→        this.moveToDeadLetterQueue(delivery, endpoint, 'max_retries_exceeded');
  1503→
  1504→        logger.error('Webhook moved to dead letter queue', {
  1505→          event: 'webhook.delivery.dead_lettered',
  1506→          metadata: {
  1507→            deliveryId,
  1508→            endpointId: endpoint.id,
  1509→            totalAttempts: delivery.attemptNumber,
  1510→            lastError: delivery.errorMessage,
  1511→          },
  1512→        });
  1513→      }
  1514→    }
  1515→
  1516→    delivery.updatedAt = new Date();
  1517→    this.deliveries.set(deliveryId, delivery);
  1518→
  1519→    return delivery;
  1520→  }
  1521→
  1522→  /**
  1523→   * Calculate next retry time with exponential backoff and jitter
  1524→   * 
  1525→   * @param delivery - Current delivery record
  1526→   * @param endpoint - Target endpoint
  1527→   * @returns Next retry timestamp
  1528→   * @private
  1529→   */
  1530→  private calculateRetryTime(
  1531→    delivery: WebhookDelivery,
  1532→    endpoint: WebhookEndpoint
  1533→  ): Date {
  1534→    const { retryConfig } = endpoint;
  1535→    
  1536→    // Calculate base delay with exponential backoff
  1537→    const baseDelay = retryConfig.initialDelayMs * 
  1538→      Math.pow(retryConfig.backoffMultiplier, delivery.attemptNumber - 1);
  1539→    
  1540→    // Cap at maximum delay
  1541→    const cappedDelay = Math.min(baseDelay, retryConfig.maxDelayMs);
  1542→    
  1543→    // Add jitter (randomization to prevent thundering herd)
  1544→    const jitterRange = cappedDelay * retryConfig.jitterFactor;
  1545→    const jitter = (Math.random() * 2 - 1) * jitterRange; // -jitter to +jitter
  1546→    
  1547→    const finalDelay = Math.max(0, Math.ceil(cappedDelay + jitter));
  1548→
  1549→    return new Date(Date.now() + finalDelay);
  1550→  }

  1698→  /**
  1699→   * Route event to matching subscribers
  1700→   * 
  1701→   * @param event - Event to route
  1702→   * @private
  1703→   */
  1704→  private async routeEventToSubscribers(event: WebhookEvent): Promise<void> {
  1705→    const subscriptions = this.getSubscriptionsForEvent(event.type);
  1706→
  1707→    for (const subscription of subscriptions) {
  1708→      // Apply filters
  1709→      if (subscription.filters && !this.applyFilters(event.data, subscription.filters)) {
  1710→        continue;
  1711→      }
  1712→
  1713→      // Deliver to each endpoint in subscription
  1714→      for (const endpointId of subscription.endpointIds) {
  1715→        const endpoint = this.endpoints.get(endpointId);
  1716→        if (!endpoint || !endpoint.isActive) continue;
  1717→
  1718→        try {
  1719→          const queueItem = this.enqueueDelivery(event, endpoint);
  1720→          this.processQueueItem(queueItem).catch(error => {
  1721→            logger.error('Subscriber delivery failed', {
  1722→              event: 'webhook.subscriber.error',
  1723→              error: error instanceof Error ? error : new Error(String(error)),
  1724→              metadata: { subscriptionId: subscription.id, endpointId },
  1725→            });
  1726→          });
  1727→        } catch (error) {
  1728→          // Queue full - log and continue
  1729→          logger.warn('Failed to enqueue for subscriber', {
  1730→            event: 'webhook.subscriber.enqueue_failed',
  1731→            error: error instanceof Error ? error : new Error(String(error)),
  1732→            metadata: { subscriptionId: subscription.id, endpointId },
  1733→          });
  1734→        }
  1735→      }
  1736→    }
  1737→  }
  1738→
  1739→  /**
  1740→   * Apply event filters to data
  1741→   * 
  1742→   * @param data - Event data
  1743→   * @param filters - Filters to apply
  1744→   * @returns True if passes all filters
  1745→   * @private
  1746→   */
  1747→  private applyFilters(data: Record<string, unknown>, filters: EventFilter[]): boolean {
  1748→    return every(filter => {
  1749→      const value = this.getNestedValue(data, filter.field);
  1750→      return this.compareValues(value, filter.operator, filter.value);
  1751→    }, filters);
  1752→  }
  1753→
  1754→  /**
  1755→   * Get nested value from object using dot notation
  1756→   * 
  1757→   * @param obj - Source object
  1758→   * @param path - Dot-notation path
  1759→   * @returns Value at path
  1760→   * @private
  1761→   */
  1762→  private getNestedValue(obj: Record<string, unknown>, path: string): unknown {
  1763→    return path.split('.').reduce((current, key) => {
  1764→      return current && typeof current === 'object' 
  1765→        ? (current as Record<string, unknown>)[key] 
  1766→        : undefined;
  1767→    }, obj as unknown);
  1768→  }
  1769→
  1770→  /**
  1771→   * Compare values using operator
  1772→   * 
  1773→   * @param actual - Actual value
  1774→   * @param operator - Comparison operator
  1775→   * @param expected - Expected value
  1776→   * @returns Comparison result
  1777→   * @private
  1778→   */
  1779→  private compareValues(actual: unknown, operator: EventFilter['operator'], expected: unknown): boolean {
  1780→    switch (operator) {
  1781→      case 'eq':
  1782→        return actual === expected;
  1783→      case 'ne':
  1784→        return actual !== expected;
  1785→      case 'gt':
  1786→        return typeof actual === 'number' && typeof expected === 'number' && actual > expected;
  1787→      case 'gte':
  1788→        return typeof actual === 'number' && typeof expected === 'number' && actual >= expected;
  1789→      case 'lt':
  1790→        return typeof actual === 'number' && typeof expected === 'number' && actual < expected;
  1791→      case 'lte':
  1792→        return typeof actual === 'number' && typeof expected === 'number' && actual <= expected;
  1793→      case 'in':
  1794→        return Array.isArray(expected) && expected.includes(actual);
  1795→      case 'contains':
  1796→        return typeof actual === 'string' && typeof expected === 'string' && 
  1797→               actual.includes(expected);
  1798→      case 'regex':
  1799→        return typeof actual === 'string' && typeof expected === 'string' &&
  1800→               new RegExp(expected).test(actual);
  1801→      default:
  1802→        return false;
  1803→    }
  1804→  }

  1806→  // ============== Dead Letter Queue ==============
  1807→
  1808→  /**
  1809→   * Move a failed delivery to the dead letter queue
  1810→   * 
  1811→   * @param delivery - Failed delivery
  1812→   * @param endpoint - Target endpoint
  1813→   * @param reason - Reason for dead lettering
  1814→   * @private
  1815→   */
  1816→  private moveToDeadLetterQueue(
  1817→    delivery: WebhookDelivery,
  1818→    endpoint: WebhookEndpoint,
  1819→    reason: DeadLetterReason
  1820→  ): void {
  1821→    // Check DLQ size limit
  1822→    if (this.deadLetterQueue.size >= this.config.maxDeadLetterQueueSize) {
  1823→      // Evict oldest entry
  1824→      const oldestKey = this.deadLetterQueue.keys().next().value;
  1825→      if (oldestKey) {
  1826→        this.deadLetterQueue.delete(oldestKey);
  1827→      }
  1828→    }
  1829→
  1830→    const dlqEntry: DeadLetterEntry = {
  1831→      id: `dlq_${Date.now()}_${randomUUID().slice(0, 8)}`,
  1832→      event: delivery.event,
  1833→      endpointId: endpoint.id,
  1834→      reason,
  1835→      originalError: delivery.errorMessage || 'Unknown error',
  1836→      deliveryHistory: [{
  1837→        attempt: delivery.attemptNumber,
  1838→        timestamp: delivery.updatedAt,
  1839→        statusCode: delivery.responseStatusCode,
  1840→        error: delivery.errorMessage,
  1841→        durationMs: delivery.durationMs || 0,
  1842→      }],
  1843→      deadLetteredAt: new Date(),
  1844→      totalAttempts: delivery.attemptNumber,
  1845→      priority: 5,
  1846→      isAcknowledged: false,
  1847→    };
  1848→
  1849→    this.deadLetterQueue.set(dlqEntry.id, dlqEntry);
  1850→    delivery.status = 'dead_lettered';
  1851→    this.updateQueueStats();
  1852→
  1853→    logger.warn('Entry added to dead letter queue', {
  1854→      event: 'webhook.dlq.added',
  1855→      metadata: {
  1856→        dlqEntryId: dlqEntry.id,
  1857→        eventId: delivery.event.id,
  1858→        endpointId: endpoint.id,
  1859→        reason,
  1860→        totalAttempts: delivery.attemptNumber,
  1861→      },
  1862→    });
  1863→  }
  1864→
  1865→  /**
  1866→   * Move pending deliveries to DLQ when endpoint is deleted
  1867→   * 
  1868→   * @param endpointId - Deleted endpoint ID
  1869→   * @param reason - Reason for moving
  1870→   * @private
  1871→   */
  1872→  private movePendingDeliveriesToDLQ(endpointId: string, reason: DeadLetterReason): void {
  1873→    const pendingItems = this.deliveryQueue.filter(item => item.endpointId === endpointId);
  1874→    
  1875→    for (const item of pendingItems) {
  1876→      const endpoint = this.endpoints.get(endpointId);
  1877→      if (!endpoint) continue;
  1878→
  1879→      const dlqEntry: DeadLetterEntry = {
  1880→        id: `dlq_${Date.now()}_${randomUUID().slice(0, 8)}`,
  1881→        event: item.event,
  1882→        endpointId,
  1883→        reason,
  1884→        originalError: `Endpoint deleted while in queue`,
  1885→        deliveryHistory: [],
  1886→        deadLetteredAt: new Date(),
  1887→        totalAttempts: item.attempts,
  1888→        priority: item.priority,
  1889→        isAcknowledged: false,
  1890→      };
  1891→
  1892→      this.deadLetterQueue.set(dlqEntry.id, dlqEntry);
  1893→    }
  1894→
  1895→    // Remove from queue
  1896→    this.deliveryQueue = this.deliveryQueue.filter(item => item.endpointId !== endpointId);
  1897→  }
  1898→
  1899→  /**
  1900→   * Get all entries in the dead letter queue
  1901→   * 
  1902→   * @param options - Filter options
  1903→   * @returns DLQ entries
  1904→   */
  1905→  getDeadLetterQueue(options?: {
  1906→    unacknowledgedOnly?: boolean;
  1907→    endpointId?: string;
  1908→    reason?: DeadLetterReason;
  1909→  }): DeadLetterEntry[] {
  1910→    let entries = Array.from(this.deadLetterQueue.values());
  1911→
  1912→    if (options?.unacknowledgedOnly) {
  1913→      entries = entries.filter(e => !e.isAcknowledged);
  1914→    }
  1915→
  1916→    if (options?.endpointId) {
  1917→      entries = entries.filter(e => e.endpointId === options.endpointId);
  1918→    }
  1919→
  1920→    if (options?.reason) {
  1921→      entries = entries.filter(e => e.reason === options.reason);
  1922→    }
  1923→
  1924→    // Sort by dead lettered date (newest first)
  1925→    entries.sort((a, b) => b.deadLetteredAt.getTime() - a.deadLetteredAt.getTime());
  1926→
  1927→    return entries;
  1928→  }
  1929→
  1930→  /**
  1931→   * Retry a dead letter queue entry
  1932→   * 
  1933→   * @param dlqEntryId - DLQ entry ID
  1934→   * @returns New delivery promise or null if not found
  1935→   */
  1936→  async retryDeadLetterEntry(dlqEntryId: string): Promise<WebhookDelivery | null> {
  1937→    const entry = this.deadLetterQueue.get(dlqEntryId);
  1938→    
  1939→    if (!entry) {
  1940→      return null;
  1941→    }
  1942→
