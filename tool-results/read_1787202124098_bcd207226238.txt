   870→
   871→export interface AppError {
   872→  code: string;
   873→  message: string;
   874→  statusCode: number;
   875→  details?: Record<string, unknown>;
   876→  requestId?: string;
   877→  stack?: string;
   878→  cause?: Error;
   879→  suggestions?: string[];
   880→  documentationUrl?: string;
   881→  retryable: boolean;
   882→}
   883→
   884→export class SSMError extends Error implements AppError {
   885→  code: string;
   886→  statusCode: number;
   887→  details?: Record<string, unknown>;
   888→  requestId?: string;
   889→  suggestions?: string[];
   890→  documentationUrl?: string;
   891→  retryable: boolean;
   892→
   893→  constructor(
   894→    message: string,
   895→    options: {
   896→      code: string;
   897→      statusCode?: number;
   898→      details?: Record<string, unknown>;
   899→      cause?: Error;
   900→      suggestions?: string[];
   901→      documentationUrl?: string;
   902→    }
   903→  ) {
   904→    super(message);
   905→    this.name = 'SSMError';
   906→    this.code = options.code;
   907→    this.statusCode = options.statusCode || 500;
   908→    this.message = message;
   909→    this.details = options.details;
   910→    this.cause = options.cause;
   911→    this.suggestions = options.suggestions || [];
   912→    this.documentationUrl = options.documentationUrl;
   913→    this.retryable = false;
   914→
   915→    if (Error.captureStackTrace) {
   916→      Error.captureStackTrace(this, SSMError);
   917→    }
   918→  }
   919→
   920→  toJSON(): AppError {
   921→    return {
   922→      code: this.code,
   923→      message: this.message,
   924→      statusCode: this.statusCode,
   925→      details: this.details,
   926→      requestId: this.requestId,
   927→      stack: this.stack,
   928→      suggestions: this.suggestions,
   929→      documentationUrl: this.documentationUrl,
   930→      retryable: this.retryable,
   931→    };
   932→  }
   933→}
