   485→   485→// ============== Statistical Utility Functions ==============
   486→   486→
   487→   487→/**
   488→   488→ * Calculate mean of numeric array
   489→   489→ * @param values - Array of numbers
   490→   490→ * @returns Mean value
   491→   491→ */
   492→   492→export function calculateMean(values: number[]): number {
   493→   493→  if (values.length === 0) return 0;
   494→   494→  return values.reduce((sum, val) => sum + val, 0) / values.length;
   495→   495→}
   496→   496→
   497→   497→/**
   498→   498→ * Calculate standard deviation
   499→   499→ * @param values - Array of numbers
   500→   500→ * @returns Standard deviation (population)
   501→   501→ */
   502→   502→export function calculateStdDev(values: number[]): number {
   503→   503→  if (values.length === 0) return 0;
   504→   504→  const mean = calculateMean(values);
   505→   505→  const squaredDiffs = values.map(val => Math.pow(val - mean, 2));
   506→   506→  return Math.sqrt(squaredDiffs.reduce((sum, val) => sum + val, 0) / values.length);
   507→   507→}
   508→   508→
   509→   509→/**
   510→   510→ * Calculate variance
   511→   511→ * @param values - Array of numbers
   512→   512→ * @returns Variance
   513→   513→ */
   514→   514→export function calculateVariance(values: number[]): number {
   515→   515→  const stdDev = calculateStdDev(values);
   516→   516→  return stdDev * stdDev;
   517→   517→}
   518→   518→
   519→   519→/**
   520→   520→ * Calculate median value
   521→   521→ * @param values - Array of numbers
   522→   522→ * @returns Median value
   523→   523→ */
   524→   524→export function calculateMedian(values: number[]): number {
   525→   525→  if (values.length === 0) return 0;
   526→   526→  const sorted = [...values].sort((a, b) => a - b);
   527→   527→  const mid = Math.floor(sorted.length / 2);
   528→   528→  return sorted.length % 2 !== 0 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
   529→   529→}
   530→   530→
   531→   531→/**
   532→   532→ * Calculate percentile
   533→   533→ * @param values - Array of numbers
   534→   534→ * @param percentile - Percentile to calculate (0-100)
   535→   535→ * @returns Value at given percentile
   536→   536→ */
   537→   537→export function calculatePercentile(values: number[], percentile: number): number {
   538→   538→  if (values.length === 0) return 0;
   539→   539→  const sorted = [...values].sort((a, b) => a - b);
   540→   540→  const index = (percentile / 100) * (sorted.length - 1);
   541→   541→  const lower = Math.floor(index);
   542→   542→  const upper = Math.ceil(index);
   543→   543→  if (lower === upper) return sorted[lower];
   544→   544→  return sorted[lower] * (upper - index) + sorted[upper] * (index - lower);
   545→   545→}
   546→   546→
   547→   547→/**
   548→   548→ * Calculate correlation coefficient (Pearson)
   549→   549→ * @param x - First array of numbers
   550→   550→ * @param y - Second array of numbers
   551→   551→ * @returns Correlation coefficient (-1 to 1)
   552→   552→ */
   553→   553→export function calculateCorrelation(x: number[], y: number[]): number {
   554→   554→  if (x.length !== y.length || x.length === 0) return 0;
   555→   555→  
   556→   556→  const n = x.length;
   557→   557→  const meanX = calculateMean(x);
   558→   558→  const meanY = calculateMean(y);
   559→   559→  
   560→   560→  let numerator = 0;
   561→   561→  let denomX = 0;
   562→   562→  let denomY = 0;
   563→   563→  
   564→   564→  for (let i = 0; i < n; i++) {
   565→   565→    const dx = x[i] - meanX;
   566→   566→    const dy = y[i] - meanY;
   567→   567→    numerator += dx * dy;
   568→   568→    denomX += dx * dx;
   569→   569→    denomY += dy * dy;
   570→   570→  }
   571→   571→  
   572→   572→  const denominator = Math.sqrt(denomX * denomY);
   573→   573→  return denominator === 0 ? 0 : numerator / denominator;
   574→   574→}
   575→   575→
   576→   576→/**
   577→   577→ * Calculate coefficient of determination (R-squared)
   578→   578→ * @param actual - Actual values
   579→   579→ * @param predicted - Predicted values
   580→   580→ * @returns R-squared value (0-1)
   581→   581→ */
   582→   582→export function calculateRSquared(actual: number[], predicted: number[]): number {
   583→   583→  if (actual.length !== predicted.length || actual.length === 0) return 0;
   584→   584→  
   585→   585→  const meanActual = calculateMean(actual);
   586→   586→  const ssTot = actual.reduce((sum, val) => sum + Math.pow(val - meanActual, 2), 0);
   587→   587→  const ssRes = actual.reduce((sum, val, i) => sum + Math.pow(val - predicted[i], 2), 0);
   588→   588→  
   589→   589→  return ssTot === 0 ? 1 : 1 - (ssRes / ssTot);
   590→   590→}

   597→   597→export function normalizeMinMax(
   598→   598→  values: number[]
   599→   599→): { normalized: number[]; min: number; max: number; scale: number } {
   600→   600→  if (values.length === 0) return { normalized: [], min: 0, max: 0, scale: 1 };
   601→   601→  
   602→   602→  const min = Math.min(...values);
   603→   603→  const max = Math.max(...values);
   604→   604→  const scale = max - min || 1;
   605→   605→  
   606→   606→  return {
   607→   607→    normalized: values.map(val => (val - min) / scale),
   608→   608→    min,
   609→   609→    max,
   610→   610→    scale,
   611→   611→  };
   612→   612→}

   616→   616→ * @param values - Array of numbers
   617→   617→ * @returns Standardized array and statistics

   619→   619→export function standardizeZScore(
   620→   620→  values: number[]
   621→   621→): { standardized: number[]; mean: number; stdDev: number } {
   622→   622→  if (values.length === 0) return { standardized: [], mean: 0, stdDev: 1 };
   623→   623→  
   624→   624→  const mean = calculateMean(values);
   625→   625→  const stdDev = calculateStdDev(values) || 1;
   626→   626→  
   627→   627→  return {
   628→   628→    standardized: values.map(val => (val - mean) / stdDev),
   629→   629→    mean,
   630→   630→    stdDev,
   631→   631→  };
   632→   632→}

   639→   639→export function sigmoid(x: number): number {
   640→   640→  return 1 / (1 + Math.exp(-Math.max(-500, Math.min(500, x))));
   641→   641→}

   646→   646→ * @returns ReLU output

   648→   648→export function relu(x: number): number {
   649→   649→  return Math.max(0, x);

   652→   652→/**
   653→   653→ * Calculate simple moving average
   654→   654→ * @param values - Time series data
   655→   655→ * @param window - Window size
   656→   656→ * @returns Moving averages array
   657→   657→ */
   658→   658→export function simpleMovingAverage(values: number[], window: number): number[] {
   659→   659→  if (window <= 0 || values.length === 0) return [];
   660→   660→  if (window >= values.length) return [calculateMean(values)];
   661→   661→  
   662→   662→  const result: number[] = [];
   663→   663→  for (let i = window - 1; i < values.length; i++) {
   664→   664→    const windowSlice = values.slice(i - window + 1, i + 1);
   665→   665→    result.push(calculateMean(windowSlice));
   666→   666→  }
   667→   667→  
   668→   668→  return result;
   669→   669→}
   670→   670→
   671→   671→/**
   672→   672→ * Calculate weighted moving average
   673→   673→ * @param values - Time series data
   674→   674→ * @param weights - Weights for each position (most recent first or last based on order)
   675→   675→ * @returns Weighted moving averages array
   676→   676→ */
   677→   677→export function weightedMovingAverage(values: number[], weights: number[]): number[] {
   678→   678→  if (weights.length === 0 || values.length === 0) return [];
   679→   679→  const window = weights.length;
   680→   680→  if (values.length < window) return [calculateMean(values)];
   681→   681→  
   682→   682→  const weightSum = weights.reduce((a, b) => a + b, 0);
   683→   683→  const result: number[] = [];
   684→   684→  
   685→   685→  for (let i = window - 1; i < values.length; i++) {
   686→   686→    const windowSlice = values.slice(i - window + 1, i + 1);
   687→   687→    let weightedSum = 0;
   688→   688→    for (let j = 0; j < window; j++) {
   689→   689→      weightedSum += windowSlice[j] * weights[j];
   690→   690→    }
   691→   691→    result.push(weightedSum / weightSum);
   692→   692→  }
   693→   693→  
   694→   694→  return result;
   695→   695→}
   696→   696→
   697→   697→/**
   698→   698→ * Calculate exponential moving average
   699→   699→ * @param values - Time series data
   700→   700→ * @param alpha - Smoothing factor (0-1), higher = more responsive
   701→   701→ * @returns Exponential moving averages array
   702→   702→ */
   703→   703→export function exponentialMovingAverage(values: number[], alpha: number = 0.3): number[] {
   704→   704→  if (values.length === 0) return [];
   705→   705→  if (alpha <= 0 || alpha > 1) alpha = 0.3;
   706→   706→  
   707→   707→  const ema: number[] = [values[0]];
   708→   708→  for (let i = 1; i < values.length; i++) {
   709→   709→    ema.push(alpha * values[i] + (1 - alpha) * ema[i - 1]);
   710→   710→  }
   711→   711→  
   712→   712→  return ema;
   713→   713→}
   714→   714→
   715→   715→/**
   716→   716→ * Apply double exponential smoothing (Holt's method)
   717→   717→ * @param values - Time series data
   718→   718→ * @param alpha - Level smoothing factor
   719→   719→ * @param beta - Trend smoothing factor
   720→   720→ * @returns Smoothed values and trend components
   721→   721→ */
   722→   722→export function doubleExponentialSmoothing(
   723→   723→  values: number[],
   724→   724→  alpha: number = 0.3,
   725→   725→  beta: number = 0.1
   726→   726→): { smoothed: number[]; trend: number[] } {
   727→   727→  if (values.length < 2) {
   728→   728→    return { smoothed: values.slice(), trend: new Array(values.length).fill(0) };
   729→   729→  }
   730→   730→  
   731→   731→  const smoothed: number[] = [values[0]];
   732→   732→  const trend: number[] = [values[1] - values[0]];
   733→   733→  const level: number[] = [values[0]];
   734→   734→  
   735→   735→  for (let i = 1; i < values.length; i++) {
   736→   736→    const prevLevel = level[i - 1];
   737→   737→    const prevTrend = trend[i - 1];
   738→   738→    
   739→   739→    const newLevel = alpha * values[i] + (1 - alpha) * (prevLevel + prevTrend);
   740→   740→    const newTrend = beta * (newLevel - prevLevel) + (1 - beta) * prevTrend;
   741→   741→    
   742→   742→    level.push(newLevel);
   743→   743→    trend.push(newTrend);
   744→   744→    smoothed.push(newLevel + newTrend);
   745→   745→  }
   746→   746→  
   747→   747→  return { smoothed, trend };
   748→   748→}
   749→   749→
   750→   750→/**
   751→   751→ * Apply triple exponential smoothing (Holt-Winters)
   752→   752→ * @param values - Time series data
   753→   753→ * @param seasonLength - Length of seasonal cycle
   754→   754→ * @param alpha - Level smoothing factor
   755→   755→ * @param beta - Trend smoothing factor
   756→   756→ * @param gamma - Seasonal smoothing factor
   757→   757→ * @returns Smoothed values with seasonal components
   758→   758→ */
   759→   759→export function holtWintersSmoothing(
   760→   760→  values: number[],
   761→   761→  seasonLength: number = 7,
   762→   762→  alpha: number = 0.3,
   763→   763→  beta: number = 0.1,
   764→   764→  gamma: number = 0.1
   765→   765→): {
   766→   766→  smoothed: number[];
   767→   767→  level: number[];
   768→   768→  trend: number[];
   769→   769→  seasonal: number[];
   770→   770→} {
   771→   771→  if (values.length < seasonLength * 2) {
   772→   772→    logger.warn('Insufficient data for Holt-Winters smoothing', {
   773→   773→      event: 'ml.holt-warnings',
   774→   774→      metadata: { dataLength: values.length, requiredLength: seasonLength * 2 },
   775→   775→    });
   776→   776→    return {
   777→   777→      smoothed: values.slice(),
   778→   778→      level: values.slice(),
   779→   779→      trend: new Array(values.length).fill(0),
   780→   780→      seasonal: new Array(values.length).fill(0),
   781→   781→    };
   782→   782→  }
   783→   783→  
   784→   784→  // Initialize seasonal indices
   785→   785→  const seasonalIndices: number[] = [];
   786→   786→  for (let s = 0; s < seasonLength; s++) {
   787→   787→    const seasonValues: number[] = [];
   788→   788→    for (let i = s; i < values.length; i += seasonLength) {
   789→   789→      seasonValues.push(values[i]);
   790→   790→    }
   791→   791→    const seasonAvg = calculateMean(seasonValues);
   792→   792→    const globalAvg = calculateMean(values);
   793→   793→    seasonalIndices.push(globalAvg > 0 ? seasonAvg / globalAvg : 1);
   794→   794→  }
   795→   795→  
   796→   796→  // Initialize level and trend using first season
   797→   797→  let level = values[0] / seasonalIndices[0];
   798→   798→  const initialTrend =
   799→   799→    (calculateMean(values.slice(0, seasonLength)) -
   800→   800→      calculateMean(values.slice(seasonLength, seasonLength * 2))) /
   801→   801→    seasonLength;
   802→   802→  let trend = initialTrend;
   803→   803→  
   804→   804→  const levels: number[] = [level];
   805→   805→  const trends: number[] = [trend];
   806→   806→  const seasonal: number[] = seasonalIndices.slice();
   807→   807→  const smoothed: number[] = [];
   808→   808→  
   809→   809→  for (let i = 0; i < values.length; i++) {
   810→   810→    const s = i % seasonLength;
   811→   811→    
   812→   812→    if (i > 0) {
   813→   813→      // Update level, trend, and seasonal
   814→   814→      const newLevel =
   815→   815→        alpha * (values[i] / seasonal[s]) + (1 - alpha) * (level + trend);
   816→   816→      const newTrend =
   817→   817→        beta * (newLevel - level) + (1 - beta) * trend;
   818→   818→      const newSeasonal =
   819→   819→        gamma * (values[i] / newLevel) + (1 - gamma) * seasonal[s];
   820→   820→      
   821→   821→      level = newLevel;
   822→   822→      trend = newTrend;
   823→   823→      seasonal[s] = newSeasonal;
   824→   824→      
   825→   825→      levels.push(level);
   826→   826→      trends.push(trend);
   827→   827→    }
   828→   828→    
   829→   829→    smoothed.push((level + trend) * seasonal[s]);
   830→   830→  }
   831→   831→  
   832→   832→  return { smoothed, level: levels, trend: trends, seasonal };
   833→   833→}
   834→   834→
   835→   835→/**
   836→   836→ * Perform linear regression
   837→   837→ * @param x - Independent variable values
   838→   838→ * @param y - Dependent variable values
   839→   839→ * @returns Regression coefficients and statistics
   840→   840→ */
   841→   841→export function linearRegression(x: number[], y: number[]): {
   842→   842→  slope: number;
   843→   843→  intercept: number;
   844→   844→  rSquared: number;
   845→   845→  standardError: number;
   846→   846→  predict: (x: number) => number;
   847→   847→} {
   848→   848→  if (x.length !== y.length || x.length < 2) {
   849→   849→    return {
   850→   850→      slope: 0,
   851→   851→      intercept: 0,
   852→   852→      rSquared: 0,
   853→   853→      standardError: 0,
   854→   854→      predict: (val: number) => 0,
   855→   855→    };
   856→   856→  }
   857→   857→  
   858→   858→  const n = x.length;
   859→   859→  const meanX = calculateMean(x);
   860→   860→  const meanY = calculateMean(y);
   861→   861→  
   862→   862→  let sumXY = 0;
   863→   863→  let sumXX = 0;
   864→   864→  
   865→   865→  for (let i = 0; i < n; i++) {
   866→   866→    sumXY += (x[i] - meanX) * (y[i] - meanY);
   867→   867→    sumXX += (x[i] - meanX) * (x[i] - meanX);
   868→   868→  }
   869→   869→  
   870→   870→  const slope = sumXX === 0 ? 0 : sumXY / sumXX;
   871→   871→  const intercept = meanY - slope * meanX;
   872→   872→  
   873→   873→  // Calculate R-squared
   874→   874→  const predicted = x.map(xi => slope * xi + intercept);
   875→   875→  const rSquared = calculateRSquared(y, predicted);
   876→   876→  
   877→   877→  // Calculate standard error
   878→   878→  const residuals = y.map((yi, i) => yi - predicted[i]);
   879→   879→  const mse = residuals.reduce((sum, r) => sum + r * r, 0) / (n - 2);
   880→   880→  const standardError = Math.sqrt(mse);
   881→   881→  
   882→   882→  return {
   883→   883→    slope,
   884→   884→    intercept,
   885→   885→    rSquared,
   886→   886→    standardError,
   887→   887→    predict: (val: number) => slope * val + intercept,
   888→   888→  };
   889→   889→}
   890→   890→
   891→   891→/**
   892→   892→ * Perform polynomial regression
   893→   893→ * @param x - Independent variable values
   894→   894→ * @param y - Dependent variable values
   895→   895→ * @param degree - Polynomial degree
   896→   896→ * @returns Coefficients and prediction function
   897→   897→ */
   898→   898→export function polynomialRegression(
   899→   899→  x: number[],
   900→   900→  y: number[],
   901→   901→  degree: number = 2
   902→   902→): {
   903→   903→  coefficients: number[];
   904→   904→  rSquared: number;
   905→   905→  predict: (x: number) => number;
   906→   906→} {
   907→   907→  if (x.length !== y.length || x.length <= degree) {
   908→   908→    return {
   909→   909→      coefficients: new Array(degree + 1).fill(0),
   910→   910→      rSquared: 0,
   911→   911→      predict: (_val: number) => 0,
   912→   912→    };
   913→   913→  }
   914→   914→  
   915→   915→  // Build Vandermonde matrix and solve using normal equations
   916→   916→  const n = x.length;
   917→   917→  const m = degree + 1;
   918→   918→  
   919→   919→  // Create design matrix X
   920→   920→  const X: number[][] = [];
   921→   921→  for (let i = 0; i < n; i++) {
   922→   922→    const row: number[] = [];
   923→   923→    for (let j = 0; j < m; j++) {
   924→   924→      row.push(Math.pow(x[i], j));
   925→   925→    }
   926→   926→    X.push(row);
   927→   927→  }
   928→   928→  
   929→   929→  // Compute X^T * X
   930→   930→  const XtX: number[][] = Array(m).fill(null).map(() => Array(m).fill(0));
   931→   931→  for (let i = 0; i < m; i++) {
   932→   932→    for (let j = 0; j < m; j++) {
   933→   933→      for (let k = 0; k < n; k++) {
   934→   934→        XtX[i][j] += X[k][i] * X[k][j];
   935→   935→      }
   936→   936→    }
   937→   937→  }
   938→   938→  
   939→   939→  // Compute X^T * y
   940→   940→  const Xty: number[] = Array(m).fill(0);
   941→   941→  for (let i = 0; i < m; i++) {
   942→   942→    for (let k = 0; k < n; k++) {
   943→   943→      Xty[i] += X[k][i] * y[k];
   944→   944→    }
   945→   945→  }
   946→   946→  
   947→   947→  // Solve system using Gaussian elimination (simplified)
   948→   948→  const coefficients = gaussianElimination(XtX, Xty);
   949→   949→  
   950→   950→  // Calculate R-squared
   951→   951→  const predicted = x.map(xi =>
   952→   952→    coefficients.reduce((sum, coef, j) => sum + coef * Math.pow(xi, j), 0)
   953→   953→  );
   954→   954→  const rSquared = calculateRSquared(y, predicted);
   955→   955→  
   956→   956→  return {
   957→   957→    coefficients,
   958→   958→    rSquared,
   959→   959→    predict: (val: number) =>
   960→   960→      coefficients.reduce((sum, coef, j) => sum + coef * Math.pow(val, j), 0),
   961→   961→  };
   962→   962→}
   963→   963→
   964→   964→/**
   965→   965→ * Gaussian elimination for solving linear systems
   966→   966→ * @param matrix - Coefficient matrix
   967→   967→ * @param vector - Right-hand side vector
   968→   968→ * @returns Solution vector
   969→   969→ */
   970→   970→function gaussianElimination(matrix: number[][], vector: number[]): number[] {
   971→   971→  const n = vector.length;
   972→   972→  const aug = matrix.map((row, i) => [...row, vector[i]]);
   973→   973→  
   974→   974→  // Forward elimination
   975→   975→  for (let col = 0; col < n; col++) {
   976→   976→    // Find pivot
   977→   977→    let maxRow = col;
   978→   978→    for (let row = col + 1; row < n; row++) {
   979→   979→      if (Math.abs(aug[row][col]) > Math.abs(aug[maxRow][col])) {
   980→   980→        maxRow = row;
   981→   981→      }
   982→   982→    }
   983→   983→    
   984→   984→    [aug[col], aug[maxRow]] = [aug[maxRow], aug[col]];
   985→   985→    
   986→   986→    if (Math.abs(aug[col][col]) < 1e-10) continue;
   987→   987→    
   988→   988→    for (let row = col + 1; row < n; row++) {
   989→   989→      const factor = aug[row][col] / aug[col][col];
   990→   990→      for (let j = col; j <= n; j++) {
   991→   991→        aug[row][j] -= factor * aug[col][j];
   992→   992→      }
   993→   993→    }
   994→   994→  }
   995→   995→  
   996→   996→  // Back substitution
   997→   997→  const solution = new Array(n).fill(0);
   998→   998→  for (let row = n - 1; row >= 0; row--) {
   999→   999→    if (Math.abs(aug[row][row]) < 1e-10) {
  1000→  1000→      solution[row] = 0;
  1001→  1001→      continue;
  1002→  1002→    }
  1003→  1003→    solution[row] = aug[row][n];
  1004→  1004→    for (let col = row + 1; col < n; col++) {
  1005→  1005→      solution[row] -= aug[row][col] * solution[col];
  1006→  1006→    }
  1007→  1007→    solution[row] /= aug[row][row];
  1008→  1008→  }
  1009→  1009→  
  1010→  1010→  return solution;
  1011→  1011→}

  1013→  1013→/**
  1014→  1014→ * Detect anomalies using z-score method
  1015→  1015→ * @param values - Data series
  1016→  1016→ * @param threshold - Number of standard deviations
  1017→  1017→ * @returns Array of anomaly indices and scores
  1018→  1018→ */
  1019→  1019→export function detectAnomaliesZScore(

  1022→  1022→): { index: number; value: number; zScore: number }[] {
  1023→  1023→  const { standardized, mean, stdDev } = standardizeZScore(values);
  1024→  1024→  const anomalies: { index: number; value: number; zScore: number }[] = [];
  1025→  1025→  
  1026→  1026→  for (let i = 0; i < standardized.length; i++) {
  1027→  1027→    if (Math.abs(standardized[i]) > threshold) {
  1028→  1028→      anomalies.push({
  1029→  1029→        index: i,
  1030→  1030→        value: values[i],
  1031→  1031→        zScore: standardized[i],
  1032→  1032→      });
  1033→  1033→    }
  1034→  1034→  }
  1035→  1035→  
  1036→  1036→  return anomalies;

  1038→  1038→
  1039→  1039→/**
  1040→  1040→ * Detect anomalies using IQR method
  1041→  1041→ * @param values - Data series
  1042→  1042→ * @param multiplier - IQR multiplier (typically 1.5)
  1043→  1043→ * @returns Anomaly indices and values
  1044→  1044→ */
  1045→  1045→export function detectAnomaliesIQR(
  1046→  1046→  values: number[],
  1047→  1047→  multiplier: number = 1.5
  1048→  1048→): { index: number; value: number; type: 'outlier_high' | 'outlier_low' }[] {
  1049→  1049→  if (values.length === 0) return [];
  1050→  1050→  
  1051→  1051→  const q1 = calculatePercentile(values, 25);
  1052→  1052→  const q3 = calculatePercentile(values, 75);
  1053→  1053→  const iqr = q3 - q1;
  1054→  1054→  const lowerBound = q1 - multiplier * iqr;
  1055→  1055→  const upperBound = q3 + multiplier * iqr;
  1056→  1056→  
  1057→  1057→  const anomalies: { index: number; value: number; type: 'outlier_high' | 'outlier_low' }[] = [];
  1058→  1058→  
  1059→  1059→  for (let i = 0; i < values.length; i++) {
  1060→  1060→    if (values[i] < lowerBound) {

  1567→  1567→   * @private
  1568→  1568→   */
  1569→  1569→  private calculateLogisticScore(factors: ChurnFactor[]): number {
  1570→  1570→    let weightedSum = 0;
  1571→  1571→    let totalWeight = 0;
  1572→  1572→    
  1573→  1573→    for (const factor of factors) {
  1574→  1574→      // Normalize factor value to 0-1 range based on whether concerning
  1575→  1575→      const normalizedValue = factor.isConcerning ? 1 : 0.3;
  1576→  1576→      weightedSum += factor.weight * normalizedValue;
  1577→  1577→      totalWeight += factor.weight;
  1578→  1578→    }
  1579→  1579→    
  1580→  1580→    const rawScore = totalWeight > 0 ? weightedSum / totalWeight : 0;
  1581→  1581→    
  1582→  1582→    // Apply sigmoid to get probability
  1583→  1583→    // Shift to make moderate scores map to reasonable probabilities
  1584→  1584→    return sigmoid((rawScore - 0.4) * 10);
  1585→  1585→  }
