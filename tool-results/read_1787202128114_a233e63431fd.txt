   198→   198→  /**
   199→   199→   * Validate and normalize pipeline configuration
   200→   200→   */
   201→   201→  private validateConfig(config: PipelineConfig): PipelineConfig {
   202→   202→    // Validate required fields
   203→   203→    if (!config.id || !config.name) {
   204→   204→      throw new Error('Pipeline ID and name are required');
   205→   205→    }
   206→   206→
   207→   207→    if (!config.models || config.models.length === 0) {
   208→   208→      throw new Error('At least one model must be configured');
   209→   209→    }
   210→   210→
   211→   211→    // Validate weights sum to ~1 for weighted methods
   212→   212→    const enabledModels = config.models.filter((m) => m.enabled);
   213→   213→    const totalWeight = enabledModels.reduce((sum, m) => sum + m.weight, 0);
   214→   214→    
   215→   215→    if (
   216→   216→      [AggregationMethod.WEIGHTED_VOTE, AggregationMethod.WEIGHTED_AVERAGE].includes(
   217→   217→        config.aggregation.method
   218→   218→      ) &&
   219→   219→      Math.abs(totalWeight - 1.0) > 0.01
   220→   220→    ) {
   221→   221→      logger.warn(
   222→   222→        `Model weights (${totalWeight}) do not sum to 1.0. Weights will be normalized.`
   223→   223→      );
   224→   224→    }
   225→   225→
   226→   226→    // Set defaults
   227→   227→    return {
   228→   228→      ...config,
   229→   229→      timeout: {
   230→   230→        perModelMs: config.timeout?.perModelMs ?? 5000,
   231→   231→        totalMs: config.timeout?.totalMs ?? 30000,
   232→   232→      },
   233→   233→      retry: {
   234→   234→        enabled: config.retry?.enabled ?? true,
   235→   235→        maxAttempts: config.retry?.maxAttempts ?? 3,
   236→   236→        initialDelayMs: config.retry?.initialDelayMs ?? 100,
   237→   237→        backoffMultiplier: config.retry?.backoffMultiplier ?? 2,
   238→   238→        maxDelayMs: config.retry?.maxDelayMs ?? 5000,
   239→   239→        retryableErrors: config.retry?.retryableErrors ?? ['timeout', 'network_error'],
   240→   240→      },
   241→   241→      caching: {
   242→   242→        enabled: config.caching?.enabled ?? true,
   243→   243→        ttlMs: config.caching?.ttlMs ?? 300000,
   244→   244→        maxSize: config.caching?.maxSize ?? 1000,
   245→   245→        keyStrategy: config.caching?.keyStrategy ?? 'hash',
   246→   246→      },
   247→   247→      logLevel: config.logLevel ?? 'info',
   248→   248→    };
   249→   249→  }
   250→   250→
   251→   251→  /**

   258→   258→  // ============== Model Registration ==============
   259→   259→
   260→   260→  /**
   261→   261→   * Register a model with the pipeline
   262→   262→   * 
   263→   263→   * @param model - The ML predictor to register
   264→   264→   * @throws Error if model ID is already registered or not in config
   265→   265→   */
   266→   266→  registerModel(model: MLPredictor): void {
   267→   267→    const modelConfig = this.config.models.find((m) => m.modelId === model.id);
   268→   268→    
   269→   269→    if (!modelConfig) {
   270→   270→      throw new Error(
   271→   271→        `Model '${model.id}' is not in pipeline configuration. Add it to models array first.`
   272→   272→      );
   273→   273→    }
   274→   274→
   275→   275→    if (this.models.has(model.id)) {
   276→   276→      throw new Error(`Model '${model.id}' is already registered`);
   277→   277→    }
   278→   278→
   279→   279→    this.models.set(model.id, model);
   280→   280→    logger.info(`Model registered: ${model.id} (${model.metadata.name})`);
   281→   281→  }
   282→   282→
   283→   283→  /**
   284→   284→   * Unregister a model from the pipeline

   287→   287→   */
   288→   288→  unregisterModel(modelId: string): void {
   289→   289→    if (this.models.delete(modelId)) {
   290→   290→      logger.info(`Model unregistered: ${modelId}`);
   291→   291→    }
   292→   292→  }
   293→   293→
   294→   294→  /**

   302→   302→   * Get list of all registered model IDs

   306→   306→  }
   307→   307→
   308→   308→  /**
   309→   309→   * Check if a model is registered and available
   310→   310→   */
   311→   311→  isModelAvailable(modelId: string): boolean {

   314→   314→  }
   315→   315→
   316→   316→  // ============== Event System ==============
   317→   317→
   318→   318→  /**
   319→   319→   * Subscribe to pipeline events
   320→   320→   * 
   321→   321→   * @param eventType - Type of event to listen for
   322→   322→   * @param listener - Callback function
   323→   323→   * @returns Unsubscribe function
   324→   324→   */
   325→   325→  on<T = unknown>(
   326→   326→    eventType: PipelineEventType,
   327→   327→    listener: PipelineEventListener<T>
   328→   328→  ): () => void {
   329→   329→    if (!this.listeners.has(eventType)) {
   330→   330→      this.listeners.set(eventType, new Set());
   331→   331→    }
   332→   332→    
   333→   333→    this.listeners.get(eventType)!.add(listener as PipelineEventListener);
   334→   334→    
   335→   335→    // Return unsubscribe function
   336→   336→    return () => {
   337→   337→      this.listeners.get(eventType)?.delete(listener as PipelineEventListener);
   338→   338→    };
   339→   339→  }
   340→   340→
   341→   341→  /**
   342→   342→   * Emit an event to all listeners
   343→   343→   */
   344→   344→  private emit<T>(type: PipelineEventType, data: T, executionId?: string): void {
   345→   345→    const event: PipelineEvent<T> = {
   346→   346→      type,
   347→   347→      timestamp: new Date(),
   348→   348→      data,
   349→   349→      pipelineId: this.config.id,
   350→   350→      executionId,
   351→   351→    };
   352→   352→
   353→   353→    const listeners = this.listeners.get(type);
   354→   354→    if (listeners) {
   355→   355→      for (const listener of listeners) {
   356→   356→        try {
   357→   357→          listener(event as PipelineEvent);
   358→   358→        } catch (error) {
   359→   359→          logger.error(`Event listener error for ${type}:`, error);
   360→   360→        }
   361→   361→      }
   362→   362→    }
   363→   363→  }
   364→   364→
   365→   365→  // ============== Main Execution ==============
   366→   366→
   367→   367→  /**
   368→   368→   * Execute the pipeline on input data
   369→   369→   * 
   370→   370→   * This is the main entry point for running predictions through
   371→   371→   * the configured ML models.
   372→   372→   * 
   373→   373→   * @param input - Input data for prediction
   374→   374→   * @param options - Optional execution options override
   375→   375→   * @returns Complete pipeline result with individual and aggregated results
   376→   376→   * 
   377→   377→   * @example
   378→   378→   * ```typescript
   379→   379→   * const result = await pipeline.execute({
   380→   380→   *   amount: 1000,
   381→   381→   *   userId: 'user_123',
   382→   382→   *   // ... transaction features
   383→   383→   * });
   384→   384→   * 
   385→   385→   * console.log(result.aggregatedResult.finalPrediction);
   386→   386→   * console.log(result.aggregatedResult.confidence);
   387→   387→   * ```
   388→   388→   */
   389→   389→  async execute(
   390→   390→    input: unknown,
   391→   391→    options?: Partial<PipelineConfig>
   392→   392→  ): Promise<PipelineResult> {
   393→   393→    const executionId = this.generateExecutionId();
   394→   394→    const startTime = Date.now();
   395→   395→
   396→   396→    // Merge options with base config if provided
   397→   397→    const effectiveConfig = options
   398→   398→      ? { ...this.config, ...options }
   399→   399→      : this.config;
   400→   400→
   401→   401→    this.emit(PipelineEventType.EXECUTION_STARTED, { input }, executionId);
   402→   402→
   403→   403→    // Check cache first
   404→   404→    if (effectiveConfig.caching.enabled) {
   405→   405→      const cacheKey = this.generateCacheKey(input);
   406→   406→      const cached = this.cache.get(cacheKey);
   407→   407→      
   408→   408→      if (cached) {
   409→   409→        this.emit(PipelineEventType.CACHE_HIT, { cacheKey }, executionId);
   410→   410→        
   411→   411→        return this.createCachedResult(
   412→   412→          cached,
   413→   413→          executionId,
   414→   414→          startTime,
   415→   415→          cacheKey
   416→   416→        );
   417→   417→      }
   418→   418→      
   419→   419→      this.emit(PipelineEventType.CACHE_MISS, { cacheKey }, executionId);
   420→   420→    }
   421→   421→
   422→   422→    try {
   423→   423→      // Handle A/B testing if enabled
   424→   424→      if (effectiveConfig.abTesting?.enabled && this.abTestState) {
   425→   425→        return await this.executeWithABTesting(
   426→   426→          input,
   427→   427→          effectiveConfig,
   428→   428→          executionId,
   429→   429→          startTime
   430→   430→        );
   431→   431→      }
   432→   432→
   433→   433→      // Execute based on mode
   434→   434→      let modelResults: Map<string, BasePredictionResult>;
   435→   435→
   436→   436→      switch (effectiveConfig.mode) {
   437→   437→        case PipelineMode.SEQUENTIAL:
   438→   438→          modelResults = await this.executeSequential(
   439→   439→            input,
   440→   440→            effectiveConfig,
   441→   441→            executionId
   442→   442→          );
   443→   443→          break;
   444→   444→        case PipelineMode.PARALLEL:
   445→   445→          modelResults = await this.executeParallel(
   446→   446→            input,
   447→   447→            effectiveConfig,
   448→   448→            executionId
   449→   449→          );
   450→   450→          break;
   451→   451→        case PipelineMode.DAG:
   452→   452→          modelResults = await this.executeDAG(
   453→   453→            input,
   454→   454→            effectiveConfig,
   455→   455→            executionId
   456→   456→          );
   457→   457→          break;
   458→   458→        case PipelineMode.FALLBACK_CHAIN:
   459→   459→          modelResults = await this.executeFallbackChain(
   460→   460→            input,
   461→   461→            effectiveConfig,
   462→   462→            executionId
   463→   463→          );
   464→   464→          break;
   465→   465→        default:
   466→   466→          throw new Error(`Unknown pipeline mode: ${effectiveConfig.mode}`);
   467→   467→      }
   468→   468→
   469→   469→      // Aggregate results
   470→   470→      const aggregatedResult = this.aggregateResults(
   471→   471→        modelResults,
   472→   472→        effectiveConfig.aggregation
   473→   473→      );
   474→   474→
   475→   475→      // Update metrics
   476→   476→      this.metrics.recordExecution(Date.now() - startTime, true);
   477→   477→
   478→   478→      // Cache result
   479→   479→      if (effectiveConfig.caching.enabled) {
   480→   480→        const cacheKey = this.generateCacheKey(input);
   481→   481→        this.cache.set(cacheKey, aggregatedResult);
   482→   482→      }
   483→   483→
   484→   484→      const result: PipelineResult = {
   485→   485→        executionId,
   486→   486→        pipelineId: effectiveConfig.id,
   487→   487→        startedAt: new Date(startTime),
   488→   488→        completedAt: new Date(),
   489→   489→        totalDurationMs: Date.now() - startTime,
   490→   490→        success: true,
   491→   491→        modelResults,
   492→   492→        aggregatedResult,
   493→   493→        metadata: this.buildMetadata(executionId, modelResults, false, effectiveConfig.mode),
   494→   494→      };
   495→   495→
   496→   496→      this.emit(PipelineEventType.EXECUTION_COMPLETED, result, executionId);
   497→   497→
   498→   498→      return result;
   499→   499→    } catch (error) {
   500→   500→      this.metrics.recordExecution(Date.now() - startTime, false);
   501→   501→      
   502→   502→      const errorResult: PipelineResult = {
   503→   503→        executionId,
   504→   504→        pipelineId: effectiveConfig.id,
   505→   505→        startedAt: new Date(startTime),
   506→   506→        completedAt: new Date(),
   507→   507→        totalDurationMs: Date.now() - startTime,
   508→   508→        success: false,
   509→   509→        modelResults: new Map(),
   510→   510→        aggregatedResult: this.createErrorAggregatedResult(error),
   511→   511→        metadata: this.buildMetadata(executionId, new Map(), true, effectiveConfig.mode),
   512→   512→      };
   513→   513→
   514→   514→      this.emit(PipelineEventType.EXECUTION_FAILED, { error }, executionId);
   515→   515→      
   516→   516→      logger.error(`Pipeline execution failed: ${executionId}`, error);
   517→   517→      
   518→   518→      throw error;
   519→   519→    }
   520→   520→  }
   521→   521→
   522→   522→  // ============== Execution Strategies ==============
   523→   523→
   524→   524→  /**
   525→   525→   * Execute models sequentially in priority order
   526→   526→   */
   527→   527→  private async executeSequential(
   528→   528→    input: unknown,
   529→   529→    config: PipelineConfig,
   530→   530→    executionId: string
   531→   531→  ): Promise<Map<string, BasePredictionResult>> {
   532→   532→    const results = new Map<string, BasePredictionResult>();
   533→   533→    const sortedModels = this.getSortedModels(config);
   534→   534→
   535→   535→    for (const modelConfig of sortedModels) {
   536→   536→      if (!modelConfig.enabled) continue;
   537→   537→
   538→   538→      const result = await this.executeModelWithRetry(
   539→   539→        modelConfig.modelId,
   540→   540→        input,
   541→   541→        config,
   542→   542→        executionId
   543→   543→      );
   544→   544→
   545→   545→      results.set(modelConfig.modelId, result);
   546→   546→
   547→   547→      // In sequential mode, check if we should stop early
   548→   548→      // e.g., if fraud detected with high confidence
   549→   549→      if (this.shouldStopEarly(result, config)) {
   550→   550→        logger.debug(
   551→   551→          `Early stopping after ${modelConfig.modelId} due to high-confidence result`
   552→   552→        );
   553→   553→        break;
   554→   554→      }
   555→   555→    }
   556→   556→
   557→   557→    return results;
   558→   558→  }
   559→   559→
   560→   560→  /**
   561→   561→   * Execute all models in parallel
   562→   562→   */
   563→   563→  private async executeParallel(
   564→   564→    input: unknown,
   565→   565→    config: PipelineConfig,
   566→   566→    executionId: string
   567→   567→  ): Promise<Map<string, BasePredictionResult>> {
   568→   568→    const results = new Map<string, BasePredictionResult>();
   569→   569→    const enabledModels = config.models.filter((m) => m.enabled);
   570→   570→
   571→   571→    // Create promises for all models
   572→   572→    const promises = enabledModels.map(async (modelConfig) => {
   573→   573→      const result = await this.executeModelWithRetry(
   574→   574→        modelConfig.modelId,
   575→   575→        input,
   576→   576→        config,
   577→   577→        executionId
   578→   578→      );
   579→   579→      return { modelId: modelConfig.modelId, result };
   580→   580→    });
   581→   581→
   582→   582→    // Execute with overall timeout
   583→   583→    const timeoutPromise = new Promise<never>((_, reject) => {
   584→   584→      setTimeout(() => reject(new Error('Pipeline execution timeout')), config.timeout.totalMs);
   585→   585→    });
   586→   586→
   587→   587→    const settledResults = await Promise.race([
   588→   588→      Promise.allSettled(promises),
   589→   589→      timeoutPromise,
   590→   590→    ]);
   591→   591→
   592→   592→    // Process results
   593→   593→    if (Array.isArray(settledResults)) {
   594→   594→      for (const settled of settledResults) {
   595→   595→        if (settled.status === 'fulfilled') {
   596→   596→          results.set(settled.value.modelId, settled.value.result);
   597→   597→        } else {
   598→   598→          logger.error(`Model execution failed: ${settled.reason}`);
   599→   599→        }
   600→   600→      }
   601→   601→    }
   602→   602→
   603→   603→    return results;
   604→   604→  }

   606→   606→  /**
   607→   607→   * Execute models respecting dependency graph (DAG mode)
   608→   608→   */
   609→   609→  private async executeDAG(
   610→   610→    input: unknown,
   611→   611→    config: PipelineConfig,
   612→   612→    executionId: string
   613→   613→  ): Promise<Map<string, BasePredictionResult>> {
   614→   614→    const results = new Map<string, BasePredictionResult>();
   615→   615→    const executed = new Set<string>();
   616→   616→    const executing = new Set<string>();
   617→   617→
   618→   618→    const executeWithDependencies = async (
   619→   619→      modelId: string
   620→   620→    ): Promise<void> => {
   621→   621→      if (executed.has(modelId)) return;
   622→   622→      if (executing.has(modelId)) return; // Prevent cycles
   623→   623→
   624→   624→      const modelConfig = config.models.find((m) => m.modelId === modelId);
   625→   625→      if (!modelConfig || !modelConfig.enabled) {
   626→   626→        executed.add(modelId);
   627→   627→        return;
   628→   628→      }
   629→   629→
   630→   630→      executing.add(modelId);
   631→   631→
   632→   632→      // Execute dependencies first
   633→   633→      if (modelConfig.dependencies) {
   634→   634→        for (const dep of modelConfig.dependencies) {
   635→   635→          await executeWithDependencies(dep);
   636→   636→        }
   637→   637→      }
   638→   638→
   639→   639→      // Execute this model
   640→   640→      const result = await this.executeModelWithRetry(
   641→   641→        modelId,
   642→   642→        input,
   643→   643→        config,
   644→   644→        executionId
   645→   645→      );
   646→   646→
   647→   647→      results.set(modelId, result);
   648→   648→      executed.add(modelId);
   649→   649→      executing.delete(modelId);
   650→   650→    };
   651→   651→
   652→   652→    // Execute all enabled models (dependencies will be resolved automatically)
   653→   653→    for (const modelConfig of config.models) {
   654→   654→      if (modelConfig.enabled) {
   655→   655→        await executeWithDependencies(modelConfig.modelId);
   656→   656→      }
   657→   657→    }
   658→   658→
   659→   659→    return results;
   660→   660→  }
   661→   661→
   662→   662→  /**
   663→   663→   * Execute models in fallback chain (try each until one succeeds)
   664→   664→   */
   665→   665→  private async executeFallbackChain(
   666→   666→    input: unknown,
   667→   667→    config: PipelineConfig,
   668→   668→    executionId: string
   669→   669→  ): Promise<Map<string, BasePredictionResult>> {
   670→   670→    const results = new Map<string, BasePredictionResult>();
   671→   671→    const sortedModels = this.getSortedModels(config);
   672→   672→
   673→   673→    for (const modelConfig of sortedModels) {
   674→   674→      if (!modelConfig.enabled) continue;
   675→   675→
   676→   676→      try {
   677→   677→        const result = await this.executeModelWithRetry(
   678→   678→          modelConfig.modelId,
   679→   679→          input,
   680→   680→          config,
   681→   681→          executionId
   682→   682→        );
   683→   683→
   684→   684→        results.set(modelConfig.modelId, result);
   685→   685→
   686→   686→        // If successful, stop the chain
   687→   687→        if (result.success && result.confidence >= 0.5) {
   688→   688→          logger.debug(
   689→   689→            `Fallback chain stopped at ${modelConfig.modelId} (successful prediction)`
   690→   690→          );
   691→   691→          break;
   692→   692→        }
   693→   693→      } catch (error) {
   694→   694→        logger.warn(
   695→   695→          `Model ${modelConfig.modelId} failed in fallback chain, trying next...`,
   696→   696→          error
   697→   697→        );
   698→   698→
   699→   699→        // Try fallback model if specified
   700→   700→        if (modelConfig.fallbackModelId) {
   701→   701→          const fallbackConfig = config.models.find(
   702→   702→            (m) => m.modelId === modelConfig.fallbackModelId
   703→   703→          );
   704→   704→          
   705→   705→          if (fallbackConfig && fallbackConfig.enabled) {
   706→   706→            const fallbackResult = await this.executeModelWithRetry(
   707→   707→              fallbackConfig.modelId,
   708→   708→              input,
   709→   709→              config,
   710→   710→              executionId
   711→   711→            );
   712→   712→            
   713→   713→            results.set(fallbackConfig.modelId, fallbackResult);
   714→   714→            
   715→   715→            if (fallbackResult.success) {
   716→   716→              break;
   717→   717→            }
   718→   718→          }
   719→   719→        }
   720→   720→      }
   721→   721→    }
   722→   722→
   723→   723→    return results;
   724→   724→  }

   728→   728→  /**
   729→   729→   * Execute a single model with retry logic
   730→   730→   */
   731→   731→  private async executeModelWithRetry(
   732→   732→    modelId: string,
   733→   733→    input: unknown,
   734→   734→    config: PipelineConfig,
   735→   735→    executionId: string
   736→   736→  ): Promise<BasePredictionResult> {
   737→   737→    const model = this.models.get(modelId);
   738→   738→    
   739→   739→    if (!model) {
   740→   740→      return this.createModelError(
   741→   741→        modelId,
   742→   742→        `Model '${modelId}' is not registered`,
   743→   743→        'MODEL_NOT_FOUND'
   744→   744→      );
   745→   745→    }
   746→   746→
   747→   747→    if (!model.isAvailable()) {
   748→   748→      return this.createModelError(
   749→   749→        modelId,
   750→   750→        `Model '${modelId}' is not available`,
   751→   751→        'MODEL_UNAVAILABLE'
   752→   752→      );
   753→   753→    }
   754→   754→
   755→   755→    this.emit(PipelineEventType.MODEL_STARTED, { modelId }, executionId);
   756→   756→
   757→   757→    const retryConfig = config.retry;
   758→   758→    let lastError: Error | null = null;
   759→   759→    let attempt = 0;
   760→   760→
   761→   761→    while (attempt <= (retryConfig.enabled ? retryConfig.maxAttempts : 0)) {
   762→   762→      try {
   763→   763→        // Apply timeout
   764→   764→        const result = await Promise.race([
   765→   765→          model.predict(input),
   766→   766→          new Promise<BasePredictionResult>((_, reject) =>
   767→   767→            setTimeout(
   768→   768→              () => reject(new Error('Model prediction timeout')),
   769→   769→              config.timeout.perModelMs
   770→   770→            )
   771→   771→          ),
   772→   772→        ]);
   773→   773→
   774→   774→        this.emit(PipelineEventType.MODEL_COMPLETED, { modelId, result }, executionId);
   775→   775→        this.metrics.recordModelExecution(modelId, Date.now(), true);
   776→   776→
   777→   777→        return result;
   778→   778→      } catch (error) {
   779→   779→        lastError = error as Error;
   780→   780→        attempt++;
   781→   781→
   782→   782→        if (attempt <= retryConfig.maxAttempts && retryConfig.enabled) {
   783→   783→          const delay = Math.min(
   784→   784→            retryConfig.initialDelayMs *
   785→   785→              Math.pow(retryConfig.backoffMultiplier, attempt - 1),
   786→   786→            retryConfig.maxDelayMs
   787→   787→          );
   788→   788→
   789→   789→          this.emit(
   790→   790→            PipelineEventType.MODEL_RETRY,
   791→   791→            { modelId, attempt, delay, error: lastError.message },
   792→   792→            executionId
   793→   793→          );
   794→   794→
   795→   795→          logger.warn(
   796→   796→            `Model ${modelId} attempt ${attempt} failed, retrying in ${delay}ms...`
   797→   797→          );
   798→   798→
   799→   799→          await this.sleep(delay);
   800→   800→        }
   801→   801→      }
   802→   802→    }
   803→   803→
   804→   804→    // All retries exhausted
   805→   805→    this.emit(
   806→   806→      PipelineEventType.MODEL_FAILED,
   807→   807→      { modelId, error: lastError?.message, attempts: attempt },
   808→   808→      executionId
   809→   809→    );
   810→   810→
   811→   811→    this.metrics.recordModelExecution(modelId, Date.now(), false);
   812→   812→
   813→   813→    return this.createModelError(
   814→   814→      modelId,
   815→   815→      lastError?.message || 'Unknown error',
   816→   816→      'MAX_RETRIES_EXCEEDED'
   817→   817→    );
   818→   818→  }

   822→   822→  /**
   823→   823→   * Aggregate results from multiple models into a single prediction
   824→   824→   */
   825→   825→  aggregateResults(
   826→   826→    modelResults: Map<string, BasePredictionResult>,
   827→   827→    aggregationConfig: AggregationConfig
   828→   828→  ): AggregatedResult {
   829→   829→    const resultsArray = Array.from(modelResults.values());
   830→   830→    const successfulResults = resultsArray.filter((r) => r.success);
   831→   831→
   832→   832→    if (successfulResults.length === 0) {
   833→   833→      return {
   834→   834→        finalPrediction: null,
   835→   835→        confidence: 0,
   836→   836→        agreement: 0,
   837→   837→        agreeingModels: 0,
   838→   838→        totalModels: resultsArray.length,
   839→   839→        disagreements: [],
   840→   840→        method: aggregationConfig.method,
   841→   841→      };
   842→   842→    }
   843→   843→
   844→   844→    switch (aggregationConfig.method) {
   845→   845→      case AggregationMethod.MAJORITY_VOTE:
   846→   846→        return this.majorityVoteAggregation(successfulResults, modelResults);
   847→   847→      case AggregationMethod.WEIGHTED_VOTE:
   848→   848→        return this.weightedVoteAggregation(successfulResults, modelResults, aggregationConfig);
   849→   849→      case AggregationMethod.AVERAGE:
   850→   850→        return this.averageAggregation(successfulResults);
   851→   851→      case AggregationMethod.WEIGHTED_AVERAGE:
   852→   852→        return this.weightedAverageAggregation(successfulResults, aggregationConfig);
   853→   853→      case AggregationMethod.MAX:
   854→   854→        return this.maxAggregation(successfulResults);
   855→   855→      case AggregationMethod.MIN:
   856→   856→        return this.minAggregation(successfulResults);
   857→   857→      default:
   858→   858→        return this.averageAggregation(successfulResults);
   859→   859→    }
   860→   860→  }

   862→   862→  /**
   863→   863→   * Majority vote aggregation for classification tasks
   864→   864→   */
   865→   865→  private majorityVoteAggregation(
   866→   866→    results: BasePredictionResult[],
   867→   867→    allResults: Map<string, BasePredictionResult>
   868→   868→  ): AggregatedResult {
   869→   869→    // Extract predictions (assuming they have predictedClass or value)
   870→   870→    const votes = new Map<string, number>();
   871→   871→    
   872→   872→    for (const result of results) {
   873→   873→      const prediction = this.extractPredictionValue(result);
   874→   874→      const key = String(prediction);
   875→   875→      votes.set(key, (votes.get(key) || 0) + 1);
   876→   876→    }
   877→   877→
   878→   878→    // Find majority
   879→   879→    let maxVotes = 0;
   880→   880→    let majorityPrediction: string = '';
   881→   881→    
   882→   882→    for (const [prediction, count] of votes.entries()) {
   883→   883→      if (count > maxVotes) {
   884→   884→        maxVotes = count;
   885→   885→        majorityPrediction = prediction;
   886→   886→      }
   887→   887→    }
   888→   888→
   889→   889→    const confidence = maxVotes / results.length;
   890→   890→    const disagreements = this.findDisagreements(results, majorityPrediction);
   891→   891→
   892→   892→    return {
   893→   893→      finalPrediction: majorityPrediction,
   894→   894→      confidence,
   895→   895→      agreement: confidence,
   896→   896→      agreeingModels: maxVotes,
   897→   897→      totalModels: results.length,
   898→   898→      disagreements: disagreements.length > 0 ? disagreements : undefined,
   899→   899→      method: AggregationMethod.MAJORITY_VOTE,
   900→   900→    };
   901→   901→  }

   903→   903→  /**
   904→   904→   * Weighted vote aggregation
   905→   905→   */
   906→   906→  private weightedVoteAggregation(
   907→   907→    results: BasePredictionResult[],
   908→   908→    allResults: Map<string, BasePredictionResult>,
   909→   909→    config: AggregationConfig
   910→   910→  ): AggregatedResult {
   911→   911→    const votes = new Map<string, number>();
   912→   912→    const weights = this.getModelWeights();
   913→   913→
   914→   914→    for (const result of results) {
   915→   915→      const weight = weights.get(result.modelId) || 1 / results.length;
   916→   916→      const prediction = String(this.extractPredictionValue(result));
   917→   917→      votes.set(prediction, (votes.get(prediction) || 0) + weight);
   918→   918→    }
   919→   919→
   920→   920→    let maxScore = 0;
   921→   921→    let winningPrediction = '';
   922→   922→
   923→   923→    for (const [prediction, score] of votes.entries()) {
   924→   924→      if (score > maxScore) {
   925→   925→        maxScore = score;
   926→   926→        winningPrediction = prediction;
   927→   927→      }
   928→   928→    }
   929→   929→
   930→   930→    const disagreements = this.findDisagreements(results, winningPrediction);
   931→   931→
   932→   932→    return {
   933→   933→      finalPrediction: winningPrediction,
   934→   934→      confidence: Math.min(maxScore, 1),
   935→   935→      agreement: this.calculateAgreement(results),
   936→   936→      agreeingModels: this.countAgreeingModels(results, winningPrediction),
   937→   937→      totalModels: results.length,
   938→   938→      disagreements: disagreements.length > 0 ? disagreements : undefined,
   939→   939→      method: AggregationMethod.WEIGHTED_VOTE,
   940→   940→    };
   941→   941→  }

   943→   943→  /**
   944→   944→   * Simple average aggregation for numeric outputs
   945→   945→   */
   946→   946→  private averageAggregation(results: BasePredictionResult[]): AggregatedResult {
   947→   947→    const values = results.map((r) => this.extractNumericValue(r));
   948→   948→    const avg = values.reduce((a, b) => a + b, 0) / values.length;
   949→   949→    const std = Math.sqrt(
   950→   950→      values.reduce((sum, v) => sum + (v - avg) ** 2, 0) / values.length
   951→   951→    );
   952→   952→
   953→   953→    return {
   954→   954→      finalPrediction: avg,
   955→   955→      confidence: 1 / (1 + std), // Higher confidence when std is lower
   956→   956→      agreement: this.calculateAgreement(results),
   957→   957→      agreeingModels: results.length,
   958→   958→      totalModels: results.length,
   959→   959→      method: AggregationMethod.AVERAGE,
   960→   960→    };
   961→   961→  }

   964→   964→   * Weighted average aggregation
   965→   965→   */
   966→   966→  private weightedAverageAggregation(

   978→   978→      totalWeight += weight;

   980→   980→
   981→   981→    const avg = totalWeight > 0 ? weightedSum / totalWeight : 0;
   982→   982→
   983→   983→    return {
   984→   984→      finalPrediction: avg,
   985→   985→      confidence: this.averageConfidence(results),
   986→   986→      agreement: this.calculateAgreement(results),
   987→   987→      agreeingModels: results.length,
   988→   988→      totalModels: results.length,
   989→   989→      method: AggregationMethod.WEIGHTED_AVERAGE,
   990→   990→    };
   991→   991→  }

  1061→  1061→  /**
  1062→  1062→   * Execute pipeline with A/B testing
  1063→  1063→   */
  1064→  1064→  private async executeWithABTesting(
  1065→  1065→    input: unknown,
  1066→  1066→    config: PipelineConfig,
  1067→  1067→    executionId: string,
  1068→  1068→    startTime: number
  1069→  1069→  ): Promise<PipelineResult> {

  1071→  1071→      throw new Error('A/B testing state not initialized');
  1072→  1072→    }

  1302→  1302→    const uniquePredictions = new Set(predictions);

  1316→  1316→    const maxCount = Math.max(...counts.values());

  1576→  1576→/**
  1577→  1577→ * Internal metrics collector for pipeline monitoring
  1578→  1578→ */
  1579→  1579→class PipelineMetrics {
  1580→  1580→  private totalExecutions = 0;
  1581→  1581→  private successfulExecutions = 0;
  1582→  1582→  private failedExecutions = 0;
  1583→  1583→  private totalExecutionTimeMs = 0;
  1584→  1584→  private modelExecutions = new Map<string, { count: number; successes: number }>();
  1585→  1585→  private totalRetries = 0;
  1586→  1586→  private lastExecutionTime: Date | null = null;
  1587→  1587→
  1588→  1588→  recordExecution(durationMs: number, success: boolean): void {
  1589→  1589→    this.totalExecutions++;
  1590→  1590→    this.totalExecutionTimeMs += durationMs;
  1591→  1591→    this.lastExecutionTime = new Date();
  1592→  1592→
  1593→  1593→    if (success) {
  1594→  1594→      this.successfulExecutions++;
  1595→  1595→    } else {
  1596→  1596→      this.failedExecutions++;
  1597→  1597→    }
  1598→  1598→  }
  1599→  1599→
  1600→  1600→  recordModelExecution(modelId: string, _timestamp: number, success: void | boolean): void {
  1601→  1601→    const current = this.modelExecutions.get(modelId) || { count: 0, successes: 0 };
  1602→  1602→    current.count++;
  1603→  1603→    if (success) current.successes++;
  1604→  1604→    this.modelExecutions.set(modelId, current);
  1605→  1605→  }
  1606→  1606→
  1607→  1607→  recordRetry(): void {
  1608→  1608→    this.totalRetries++;
  1609→  1609→  }
  1610→  1610→
  1611→  1611→  getTotalRetries(): number {
  1612→  1612→    return this.totalRetries;
  1613→  1613→  }
  1614→  1614→
  1615→  1615→  getLastExecutionTime(): Date | null {
  1616→  1616→    return this.lastExecutionTime;
  1617→  1617→  }
  1618→  1618→
  1619→  1619→  getSnapshot(): PipelineMetricsSnapshot {
  1620→  1620→    return {
  1621→  1621→      totalExecutions: this.totalExecutions,
  1622→  1622→      successfulExecutions: this.successfulExecutions,
  1623→  1623→      failedExecutions: this.failedExecutions,
  1624→  1624→      successRate:
  1625→  1625→        this.totalExecutions > 0
  1626→  1626→          ? this.successfulExecutions / this.totalExecutions
  1627→  1627→          : 0,
  1628→  1628→      avgExecutionTimeMs:
  1629→  1629→        this.totalExecutions > 0
  1630→  1630→          ? this.totalExecutionTimeMs / this.totalExecutions
  1631→  1631→          : 0,
  1632→  1632→      totalRetries: this.totalRetries,
  1633→  1633→      modelExecutions: Object.fromEntries(this.modelExecutions),
  1634→  1634→      lastExecutionTime: this.lastExecutionTime,
  1635→  1635→    };
  1636→  1636→  }
  1637→  1637→
  1638→  1638→  reset(): void {
  1639→  1639→    this.totalExecutions = 0;
  1640→  1640→    this.successfulExecutions = 0;
  1641→  1641→    this.failedExecutions = 0;
  1642→  1642→    this.totalExecutionTimeMs = 0;
  1643→  1643→    this.modelExecutions.clear();
  1644→  1644→    this.totalRetries = 0;
  1645→  1645→  }
  1646→  1646→}
