    16→    16→  export function toTitleCase(str: string): string {
    17→    17→    return str
    18→    18→      .toLowerCase()
    19→    19→      .replace(/(?:^|\s|[-_])\w/g, (match) => match.toUpperCase())
    20→    20→      .replace(/[-_]/g, ' ');
    21→    21→  }

    23→    23→  /**
    24→    24→   * Generate a random string of specified length
    25→    25→   */
    26→    26→  export function random(length: number = 16, charset?: string): string {
    27→    27→    const defaultCharset = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
    28→    28→    const chars = charset || defaultCharset;
    29→    29→    let result = '';
    30→    30→    
    31→    31→    for (let i = 0; i < length; i++) {
    32→    32→      result += chars.charAt(Math.floor(Math.random() * chars.length));
    33→    33→    }
    34→    34→    
    35→    35→    return result;
    36→    36→  }

    41→    41→  export function truncate(str: string, maxLength: number, suffix = '...'): string {
    42→    42→    if (str.length <= maxLength) return str;
    43→    43→    return str.slice(0, maxLength - suffix.length) + suffix;
    44→    44→  }

    46→    46→  /**
    47→    47→   * Mask sensitive information (e.g., email, phone)
    48→    48→   */
    49→    49→  export function mask(str: string, visibleChars: number = 4, maskChar = '*'): string {
    50→    50→    if (str.length <= visibleChars) return str;
    51→    51→    
    52→    52→    const start = str.slice(0, Math.ceil(visibleChars / 2));
    53→    53→    const end = str.slice(-Math.floor(visibleChars / 2));
    54→    54→    const maskLength = str.length - visibleChars;
    55→    55→    
    56→    56→    return `${start}${maskChar.repeat(maskLength)}${end}`;
    57→    57→  }

    83→    83→      '<': '&lt;',
    84→    84→      '>': '&gt;',

    87→    87→      '/': '&#x2F;',
    88→    88→    };

    91→    91→  }

   204→   204→  export function formatBytes(bytes: number, decimals: number = 2): string {
   205→   205→    if (bytes === 0) return '0 Bytes';
   206→   206→
   207→   207→    const k = 1024;
   208→   208→    const sizes = ['Bytes', 'KB', 'MB', 'GB', 'TB'];
   209→   209→    const i = Math.floor(Math.log(bytes) / Math.log(k));
   210→   210→
   211→   211→    return parseFloat((bytes / Math.pow(k, i)).toFixed(decimals)) + ' ' + sizes[i];
   212→   212→  }

   235→   235→    inMax: number,
   236→   236→    outMin: number,
   237→   237→    outMax: number

   247→   247→   * Format date to relative time (e.g., "2 hours ago")
   248→   248→   */
   249→   249→  export function timeAgo(date: Date | string): string {
   250→   250→    const now = new Date();
   251→   251→    const past = typeof date === 'string' ? new Date(date) : date;
   252→   252→    const diffMs = now.getTime() - past.getTime();
   253→   253→    const diffSecs = Math.floor(diffMs / 1000);
   254→   254→    const diffMins = Math.floor(diffSecs / 60);
   255→   255→    const diffHours = Math.floor(diffMins / 60);
   256→   256→    const diffDays = Math.floor(diffHours / 24);
   257→   257→    const diffWeeks = Math.floor(diffDays / 7);
   258→   258→    const diffMonths = Math.floor(diffDays / 30);
   259→   259→
   260→   260→    if (diffSecs < 60) return `${diffSecs} seconds ago`;
   261→   261→    if (diffMins < 60) return `${diffMins} minute${diffMins !== 1 ? 's' : ''} ago`;
   262→   262→    if (diffHours < 24) return `${diffHours} hour${diffHours !== 1 ? 's' : ''} ago`;
   263→   263→    if (diffDays < 7) return `${diffDays} day${diffDays !== 1 ? 's' : ''} ago`;
   264→   264→    if (diffWeeks < 4) return `${diffWeeks} week${diffWeeks !== 1 ? 's' : ''} ago`;
   265→   265→    if (diffMonths < 12) return `${diffMonths} month${diffMonths !== 1 ? 's' : ''} ago`;
   266→   266→    
   267→   267→    const diffYears = Math.floor(diffMonths / 12);
   268→   268→    return `${diffYears} year${diffYears !== 1 ? 's' : ''} ago`;
   269→   269→  }

   429→   429→   */
   430→   430→  export function shuffle<T>(array: T[]): T[] {
   431→   431→    const result = [...array];
   432→   432→    for (let i = result.length - 1; i > 0; i--) {
   433→   433→      const j = Math.floor(Math.random() * (i + 1));
   434→   434→      [result[i], result[j]] = [result[j], result[i]];
   435→   435→    }
   436→   436→    return result;
   437→   437→  }

   463→   463→  export function sortBy<T>(array: T[], keyFn: (item: T) => number | string, order: 'asc' | 'desc' = 'asc'): T[] {
   464→   464→    return [...array].sort((a, b) => {
   465→   465→      const valA = keyFn(a);
   466→   466→      const valB = keyFn(b);
   467→   467→      
   468→   468→      if (typeof valA === 'string' && typeof valB === 'string') {
   469→   469→        return order === 'asc' 
   470→   470→          ? valA.localeCompare(valB)
   471→   471→          : valB.localeCompare(valA);
   472→   472→      }
   473→   473→      
   474→   474→      const numA = Number(valA);
   475→   475→      const numB = Number(valB);
   476→   476→      
   477→   477→      return order === 'asc' ? numA - numB : numB - numA;
   478→   478→    });
   479→   479→  }

   802→   802→  export function deepMerge<T extends Record<string, unknown>>(
   803→   803→    target: T,
   804→   804→    source: Partial<T>

   806→   806→    const result = { ...target };
   807→   807→
   808→   808→    for (const key of Object.keys(source)) {
   809→   809→      const sourceVal = source[key as keyof T];
   810→   810→      const targetVal = target[key as keyof T];
   811→   811→
   812→   812→      if (
   813→   813→        sourceVal &&
   814→   814→        typeof sourceVal === 'object' &&
   815→   815→        !Array.isArray(sourceVal) &&
   816→   816→        targetVal &&
   817→   817→        typeof targetVal === 'object' &&
   818→   818→        !Array.isArray(targetVal)
   819→   819→      ) {
   820→   820→        (result as Record<string, unknown>)[key] = deepMerge(
   821→   821→          targetVal as Record<string, unknown>,
   822→   822→          sourceVal as Record<string, unknown>
   823→   823→        );
   824→   824→      } else {
   825→   825→        (result as Record<string, unknown>)[key] = sourceVal;
   826→   826→      }
   827→   827→    }
   828→   828→
   829→   829→    return result;
   830→   830→  }

  1034→  1034→  export function flatten(obj: Record<string, unknown>, separator = '.'): Record<string, unknown> {
  1035→  1035→    const result: Record<string, unknown> = {};
  1036→  1036→
  1037→  1037→    const recurse = (current: any, prefix: string) => {
  1038→  1038→      for (const [key, value] of Object.entries(current)) {
  1039→  1039→        const newKey = prefix ? `${prefix}${separator}${key}` : key;
  1040→  1040→
  1041→  1041→        if (value && typeof value === 'object' && !Array.isArray(value)) {
  1042→  1042→          recurse(value, newKey);
  1043→  1043→        } else {
  1044→  1044→          result[newKey] = value;
  1045→  1045→        }
  1046→  1046→      }
  1047→  1047→    };
  1048→  1048→
  1049→  1049→    recurse(obj, '');
  1050→  1050→    return result;
  1051→  1051→  }

  1057→  1057→    const result: Record<string, unknown> = {};
  1058→  1058→
  1059→  1059→    for (const [key, value] of Object.entries(obj)) {
  1060→  1060→      const keys = key.split(separator);
  1061→  1061→      let current = result;
  1062→  1062→
  1063→  1063→      for (let i = 0; i < keys.length - 1; i++) {
  1064→  1064→        if (!current[keys[i]]) {
  1065→  1065→          current[keys[i]] = {};
  1066→  1066→        }
  1067→  1067→        current = current[keys[i]] as Record<string, unknown>;
  1068→  1068→      }
  1069→  1069→
  1070→  1070→      current[keys[keys.length - 1]] = value;
  1071→  1071→    }
  1072→  1072→
  1073→  1073→    return result;
  1074→  1074→  }
